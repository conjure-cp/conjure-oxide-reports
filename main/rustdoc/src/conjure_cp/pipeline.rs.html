27thread_local! {
28    static ACTIVE_PIPELINE: Cell<*const Pipeline> = const { Cell::new(std::ptr::null()) };
29}
30
31struct ActivePipelineGuard(*const Pipeline);
32
33impl Drop for ActivePipelineGuard {
34    fn drop(&mut self) {
35        ACTIVE_PIPELINE.with(|active| active.set(self.0));
36    }
37}

39/// Modelling and solver options shared by library callers and the CLI.
40#[derive(Clone, Debug)]
41pub struct Config {
42    /// Rewrite engine configuration.
43    pub rewriter: Rewriter,
44    /// Strategy for expanding comprehensions.
45    pub comprehension_expander: QuantifiedExpander,
46    /// Target solver family.
47    pub solver: SolverFamily,
48    /// Additional registered rule sets, alongside the defaults and solver rules.
49    pub extra_rule_sets: Vec<String>,
50    /// Strategy for modelling decisions.
51    pub heuristic: Heuristic,
52    /// Random modelling seed, reset for each operation.
53    pub seed: u64,
54    /// Recorded 1-based interactive answers, reset for each operation.
55    pub responses: Vec<usize>,
56    /// Representation channelling policy.
57    pub channelling: Channelling,
58    /// Optional SAT encoder overrides.
59    pub amo_encoding: Option<AmoEncoding>,
60    /// Optional SAT cardinality encoder override.
61    pub cardinality_encoding: Option<CardinalityEncoding>,
62    /// Optional SAT pseudo-Boolean encoder override.
63    pub pb_encoding: Option<PbEncoding>,
64    /// Optional SAT element-constraint encoder override.
65    pub element_encoding: Option<ElementEncoding>,
66    /// Optional SAT table-constraint encoder override.
67    pub table_encoding: Option<TableEncoding>,
68    /// Optional SAT allDifferent encoder override.
69    pub alldifferent_encoding: Option<AllDifferentEncoding>,
70    /// Optional Minion discrete-domain span threshold.
71    pub minion_discrete_threshold: Option<usize>,
72    /// Optional Minion search orders.
73    pub minion_varorder: Option<MinionVariableOrder>,
74    /// Optional Minion value ordering.
75    pub minion_valorder: Option<MinionValueOrder>,
76    /// Compiled SAT implementation to use.
77    #[cfg(any(
78        feature = "sat-batsat",
79        all(feature = "sat-cadical", not(target_family = "wasm"))
80    ))]
81    pub sat_backend: SatBackend,
82    /// Backend solver seed.
83    pub solver_seed: u32,
84    /// Cumulative wall-clock solver timeout.
85    pub solver_timeout: Option<Duration>,
86    /// Trace events to emit to the caller's tracing subscriber.
87    pub trace: TraceConfig,
88}

122/// Rule trace events enabled for a pipeline; subscribers and output belong to the caller.
123#[derive(Clone, Copy, Debug, Default)]
124pub struct TraceConfig {
125    /// Emit the default rule trace.
126    pub rules: bool,
127    /// Emit rule-attempt events.
128    pub attempts: bool,
129    /// Emit aggregate rule-application events.
130    pub aggregates: bool,
131}

133impl Config {
134    fn apply(&self) {
135        settings::set_current_rewriter(self.rewriter);
136        settings::set_comprehension_expander(self.comprehension_expander);
137        settings::set_current_solver_family(self.solver);
138        settings::set_minion_discrete_threshold(self.minion_discrete_threshold);
139        settings::set_heuristic(self.heuristic);
140        settings::set_heuristic_seed(self.seed);
141        settings::set_heuristic_responses(self.responses.clone());
142        settings::begin_heuristic_all_choices(Vec::new());
143        settings::set_channelling(self.channelling);
144        settings::set_amo_encoding(self.amo_encoding);
145        settings::set_cardinality_encoding(self.cardinality_encoding);
146        settings::set_pb_encoding(self.pb_encoding);
147        settings::set_element_encoding(self.element_encoding);
148        settings::set_table_encoding(self.table_encoding);
149        settings::set_alldifferent_encoding(self.alldifferent_encoding);
150        settings::set_default_rule_trace_enabled(self.trace.rules);
151        settings::set_rule_attempt_trace_enabled(self.trace.attempts);
152        settings::set_rule_trace_aggregates_enabled(self.trace.aggregates);
153        settings::set_rule_trace_enabled(
154            self.trace.rules || self.trace.attempts || self.trace.aggregates,
155        );
156    }
157}

159/// A configured pipeline with a shared model context and accumulated statistics.
160///
161/// Link a rule provider (for example `use conjure_cp_rules as _;`) before constructing one.
162/// Each operation installs its settings on the current thread and restores the caller's settings,
163/// including on panic. A pipeline can be reused for successive fuzzer inputs; construct a new one
164/// when each run needs separate statistics.
165pub struct Pipeline {
166    config: Config,
167    context: Arc<RwLock<Context<'static>>>,
168}
169
170impl Pipeline {
171    /// Resolve registered rules and create a context, returning configuration errors to the caller.
172    pub fn new(config: Config) -> Result<Self> {
173        #[cfg(not(any(
174            feature = "sat-batsat",
175            all(feature = "sat-cadical", not(target_family = "wasm"))
176        )))]
177        if config.solver == SolverFamily::Sat {
178            anyhow::bail!(
179                "SAT solver support was not compiled in (enable `sat-batsat` or `sat-cadical`)."
180            );
181        }
182        #[cfg(not(feature = "z3"))]
183        if config.solver == SolverFamily::Z3 {
184            anyhow::bail!("Z3 solver support was not compiled in (enable the `z3` feature).");
185        }
186        let names: Vec<&str> = DEFAULT_RULE_SETS
187            .iter()
188            .copied()
189            .chain(config.extra_rule_sets.iter().map(String::as_str))
190            .collect();
191        let rule_sets = resolve_rule_sets(config.solver, &names)
192            .context("could not resolve rule sets; link a rule provider such as conjure-cp-rules and check extra rule set names")?;
193        let rules = get_rules(&rule_sets)?.into_iter().collect();
194        tracing::info!(target: "conjure::stage", count = rule_sets.len(), "resolved rule sets");
195        let context = Context::new_ptr(
196            config.solver,
197            names.iter().map(|s| s.to_string()).collect(),
198            rules,
199            rule_sets,
200        );
201        Ok(Self { config, context })
202    }
203
204    /// Access source metadata, enabled rules and accumulated statistics.
205    pub fn context(&self) -> Arc<RwLock<Context<'static>>> {
206        Arc::clone(&self.context)
207    }
208
209    /// Run lower-level library operations with this pipeline's settings in scope.
210    ///
211    /// Use this when loading or running a [`Solver`] directly, including solver-time rewrites.
212    /// Nested operations on this pipeline share the current random stream and interactive answers.
213    /// Keep the closure synchronous; settings are local to its current thread.
214    pub fn with_settings<T>(&self, f: impl FnOnce() -> T) -> T {
215        let pointer = std::ptr::from_ref(self);
216        if ACTIVE_PIPELINE.with(|active| active.get() == pointer) {
217            return f();
218        }
219        settings::with_saved_settings(|| {
220            let previous = ACTIVE_PIPELINE.with(|active| active.replace(pointer));
221            let _active = ActivePipelineGuard(previous);
222            self.config.apply();
223            f()
224        })
225    }
226
227    /// Parse Essence source in memory using the configured parser.
228    pub fn parse(&self, source: &str) -> Result<Model> {
229        self.with_settings(|| self.parse_native(source, None))
230    }
231
232    /// Parse a file using the configured parser without requiring a UTF-8 path.
233    pub fn parse_file(&self, path: impl AsRef<Path>) -> Result<Model> {
234        let path = path.as_ref();
235        tracing::info!(target: "conjure::stage", path = %path.display(), "parsing input model");
236        self.with_settings(|| {
237            let source = std::fs::read_to_string(path)
238                .with_context(|| format!("could not read {}", path.display()))?;
239            self.parse_native(&source, Some(path.to_string_lossy().into_owned()))
240        })
241    }
242
243    fn parse_native(&self, source: &str, filename: Option<String>) -> Result<Model> {
244        let mut errors = Vec::new();
245        match crate::parse::tree_sitter::parse_essence_with_context(
246            source,
247            self.context(),
248            &mut errors,
249        ) {
250            Ok(Some(model)) => {
251                crate::ast::assertions::debug_assert_model_well_formed(&model, "tree-sitter");
252                Ok(model)
253            }
254            Ok(None) => {
255                Err(ParseErrorCollection::multiple(errors, Some(source.into()), filename).into())
256            }
257            Err(error) => Err(ParseErrorCollection::fatal(error).into()),
258        }
259    }
260
261    /// Parse, instantiate optional Essence parameters, and rewrite an in-memory problem.
262    pub fn compile(&self, source: &str, parameters: Option<&str>) -> Result<Model> {
263        self.with_settings(|| {
264            let problem = self.parse(source)?;
265            let model = match parameters {
266                Some(parameters) => instantiate_model(problem, self.parse(parameters)?)?,
267                None => {
268                    let mut problem = problem;
269                    validate_instantiation_conditions(&mut problem)?;
270                    problem
271                }
272            };
273            self.rewrite(model)
274        })
275    }
276
277    /// Rewrite a model parsed with this pipeline, resetting modelling choices for the operation.
278    pub fn rewrite(&self, model: Model) -> Result<Model> {
279        self.check_context(&model)?;
280        for (name, declaration) in model.symbols().iter_local() {
281            ensure!(
282                declaration.as_given().is_none(),
283                "given `{name}` needs a parameter value before rewriting"
284            );
285        }
286        self.with_settings(|| {
287            tracing::debug!(model = %model, "initial model");
288            let Rewriter::Rewrite(config) = self.config.rewriter;
289            let rule_sets = self.context.read().unwrap().rule_sets.clone();
290            tracing::info!(target: "conjure::stage", %config, "rewriting model");
291            let model = rewrite_model(&model, &rule_sets, config)?;
292            tracing::debug!(model = %model, "rewritten model");
293            Ok(model)
294        })
295    }
296
297    /// Create a configured solver; use [`Self::with_settings`] when running it directly.
298    pub fn solver(&self) -> Result<Solver> {
299        let config = &self.config;
300        Ok(match config.solver {
301            SolverFamily::Minion => Solver::new(
302                Minion::with_search_orders(config.minion_varorder, config.minion_valorder)
303                    .with_solver_seed(config.solver_seed)
304                    .with_timeout(config.solver_timeout),
305            ),
306            #[cfg(any(
307                feature = "sat-batsat",
308                all(feature = "sat-cadical", not(target_family = "wasm"))
309            ))]
310            SolverFamily::Sat => Solver::new(
311                Sat::new(config.sat_backend)
312                    .with_solver_seed(config.solver_seed)
313                    .with_timeout(config.solver_timeout),
314            ),
315            #[cfg(not(any(
316                feature = "sat-batsat",
317                all(feature = "sat-cadical", not(target_family = "wasm"))
318            )))]
319            SolverFamily::Sat => anyhow::bail!(
320                "SAT solver support was not compiled in (enable `sat-batsat` or `sat-cadical`)."
321            ),
322            #[cfg(feature = "z3")]
323            SolverFamily::Z3 => {
324                Solver::new(Smt::new(config.solver_timeout).with_solver_seed(config.solver_seed))
325            }
326            #[cfg(not(feature = "z3"))]
327            SolverFamily::Z3 => {
328                anyhow::bail!("Z3 solver support was not compiled in (enable the `z3` feature).")
329            }
330        })
331    }
332
333    /// Solve a rewritten model and reconstruct its high-level Essence values.
334    pub fn solve(
335        &self,
336        model: Model,
337        options: &SolutionOptions,
338    ) -> Result<Vec<BTreeMap<Name, Literal>>> {
339        self.solve_with_timings(model, options)
340            .map(|(solutions, _)| solutions)
341    }
342
343    /// Collect reconstructed solutions and backend, search and reconstruction timings.
344    pub fn solve_with_timings(
345        &self,
346        model: Model,
347        options: &SolutionOptions,
348    ) -> Result<(Vec<BTreeMap<Name, Literal>>, SolutionCollectionTimings)> {
349        self.check_context(&model)?;
350        ensure!(
351            options.number_of_solutions >= 0,
352            "number of solutions must be non-negative (zero requests all)"
353        );
354        self.with_settings(|| {
355            get_solutions_with_timings(
356                self.solver()?,
357                model,
358                options.number_of_solutions,
359                options.keep_intermediate_solutions,
360                &options.solver_input_file,
361                options.rule_trace_cdp,
362            )
363        })
364    }
365
366    fn check_context(&self, model: &Model) -> Result<()> {
367        ensure!(
368            Arc::ptr_eq(&model.context, &self.context),
369            "model belongs to a different pipeline context"
370        );
371        Ok(())
372    }
