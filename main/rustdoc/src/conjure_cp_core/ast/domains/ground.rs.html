31/// Variants use the project-wide type/domain ordering; keep broad matches in the same order.
32pub enum GroundDomain {

34    Empty(ReturnType),
35    /// A boolean value (true / false)
36    Bool,

64/// Counts partitions of `n` labelled elements into unlabelled blocks that are *all* exactly
65/// `block_size` (a `regular` partition), i.e. `n / (block_size! ^ k) / k!` for `k = n /
66/// block_size` -- computed by choosing the `k` same-size groups one at a time (dividing out the
67/// groups' own arbitrary ordering with one final `/ k!`) rather than via raw factorials, so
68/// intermediate values stay as small as the final answer allows.
69fn regular_partition_count(n: u64, block_size: u64) -> Result<u64, DomainOpError> {
70    if n == 0 {
71        return Ok(1);
72    }
73    if block_size == 0 || !n.is_multiple_of(block_size) {
74        return Ok(0);
75    }
76    let num_parts = n / block_size;
77    let mut numerator = 1u64;
78    let mut remaining = n;
79    for _ in 0..num_parts {
80        let choose = count_combinations(remaining, block_size)?;
81        numerator = numerator
82            .checked_mul(choose)
83            .ok_or(DomainOpError::TooLarge)?;
84        remaining -= block_size;
85    }
86    let num_parts_factorial = (1..=num_parts)
87        .try_fold(1u64, |acc, x| acc.checked_mul(x))
88        .ok_or(DomainOpError::TooLarge)?;
89    numerator
90        .checked_div(num_parts_factorial)
91        .ok_or(DomainOpError::TooLarge)
92}

94/// Every unordered partition of `elements` into blocks whose size falls in `[block_min,
95/// block_max]` (both inclusive; `block_min` is floored to `1`, since a block can't be empty).
96///
97/// Built by always rooting the block containing the *first* remaining element (`elements` is
98/// consumed in its given order), then recursing on whatever's left -- the same canonical
99/// construction [`restricted_partition_count`] counts without generating, so the two stay
100/// consistent by construction. Each returned partition is a `Vec` of blocks in the order they
101/// were rooted; each block is a `Vec` of elements in `[root, ..combination order]`.
102fn restricted_partitions(
103    elements: &[Literal],
104    block_min: usize,
105    block_max: usize,
106) -> Vec<Vec<Vec<Literal>>> {
107    let block_min = block_min.max(1);
108    if elements.is_empty() {
109        return vec![vec![]];
110    }
111    let (first, rest) = elements.split_first().expect("checked non-empty above");
112
113    let max_extra = block_max.saturating_sub(1).min(rest.len());
114    if block_min.saturating_sub(1) > max_extra {
115        return vec![];
116    }
117
118    let mut results = vec![];
119    for extra in block_min.saturating_sub(1)..=max_extra {
120        for combo in rest.iter().cloned().combinations(extra) {
121            let mut block = vec![first.clone()];
122            block.extend(combo.iter().cloned());
123
124            let remaining: Vec<Literal> = rest
125                .iter()
126                .filter(|elem| !combo.contains(elem))
127                .cloned()
128                .collect();
129
130            for sub_partition in restricted_partitions(&remaining, block_min, block_max) {
131                let mut whole = vec![block.clone()];
132                whole.extend(sub_partition);
133                results.push(whole);
134            }
135        }
136    }
137    results
138}

140/// Every permutation of `n` in `0..len` (representing positions in `elements`) whose number of
141/// non-fixed positions falls in `[moved_min, moved_max]`, returned as the concrete value each
142/// position maps to (i.e. `result[i]` is what `elements[i]` is sent to).
143fn restricted_permutations(
144    elements: &[Literal],
145    moved_min: usize,
146    moved_max: usize,
147) -> impl Iterator<Item = Vec<Literal>> + '_ {
148    let n = elements.len();
149    (0..n)
150        .permutations(n)
151        .filter(move |perm| {
152            let moved = perm.iter().enumerate().filter(|&(i, p)| i != *p).count();
153            moved >= moved_min && moved <= moved_max
154        })
155        .map(move |perm| perm.into_iter().map(|i| elements[i].clone()).collect())
156}

158/// Converts a full position-to-position permutation mapping (`elements[i]` maps to `mapped[i]`)
159/// into cycle notation, omitting fixed points -- mirrors
160/// `PermutationAsFunction`'s own `up()` (`crates/conjure-cp-rules/src/types/permutation/
161/// as_function/representation.rs`), duplicated here since domain-level enumeration lives in a
162/// lower crate that representation can't depend on.
163fn permutation_mapping_to_cycles(elements: &[Literal], mapped: &[Literal]) -> Vec<Vec<Literal>> {
164    let forward: std::collections::HashMap<&Literal, &Literal> =
165        elements.iter().zip(mapped.iter()).collect();
166    let mut visited: std::collections::HashSet<&Literal> = std::collections::HashSet::new();
167    let mut cycles = vec![];
168    for start in elements {
169        if visited.contains(start) {
170            continue;
171        }
172        let image = forward[start];
173        if image == start {
174            visited.insert(start);
175            continue;
176        }
177        let mut cycle = vec![start.clone()];
178        visited.insert(start);
179        let mut current = image;
180        while current != start {
181            visited.insert(current);
182            cycle.push(current.clone());
183            current = forward[current];
184        }
185        cycles.push(cycle);
186    }
187    cycles
188}

191    pub fn union(&self, other: &GroundDomain) -> Result<GroundDomain, DomainOpError> {
192        // Keep implemented variants before `todo!` variants so mixed-domain unions report a type
193        // error instead of entering an unsupported implementation. Each group uses declaration
194        // order.
195        match (self, other) {
196            (GroundDomain::Empty(ty), dom) | (dom, GroundDomain::Empty(ty)) => {
197                if *ty == dom.return_type() {
198                    Ok(dom.clone())
199                } else {
200                    Err(DomainOpError::WrongType)
201                }
202            }
203            (GroundDomain::Bool, GroundDomain::Bool) => Ok(GroundDomain::Bool),
204            (GroundDomain::Bool, _) | (_, GroundDomain::Bool) => Err(DomainOpError::WrongType),
205            (GroundDomain::Int(r1), GroundDomain::Int(r2)) => {
206                let mut rngs = r1.clone();
207                rngs.extend(r2.clone());
208                Ok(GroundDomain::Int(Range::squeeze(&rngs)))
209            }
210            (GroundDomain::Int(_), _) | (_, GroundDomain::Int(_)) => Err(DomainOpError::WrongType),
211            (GroundDomain::Tuple(in1s), GroundDomain::Tuple(in2s)) if in1s.len() == in2s.len() => {
212                let mut inners = Vec::new();
213                for (in1, in2) in zip(in1s, in2s) {
214                    inners.push(Moo::new(in1.union(in2)?));
215                }
216                Ok(GroundDomain::Tuple(inners))
217            }
218            (GroundDomain::Tuple(_), _) | (_, GroundDomain::Tuple(_)) => {
219                Err(DomainOpError::WrongType)
220            }
221            (GroundDomain::Record(in1s), GroundDomain::Record(in2s))
222                if in1s.len() == in2s.len() =>
223            {
224                let lhs_fields: BTreeMap<&Name, &Moo<GroundDomain>> =
225                    in1s.iter().map(|x| (&x.name, &x.value)).collect();
226                let rhs_fields: BTreeMap<&Name, &Moo<GroundDomain>> =
227                    in2s.iter().map(|x| (&x.name, &x.value)).collect();
228                let mut new_fields = Vec::with_capacity(in1s.len());
229                for (n, d) in lhs_fields {
230                    let d2 = rhs_fields.get(&n).ok_or(DomainOpError::WrongType)?;
231                    let dom = d.union(d2)?;
232                    new_fields.push(Field {
233                        name: n.clone(),
234                        value: dom.into(),
235                    });
236                }
237                Ok(GroundDomain::Record(new_fields))
238            }
239            (GroundDomain::Record(_), _) | (_, GroundDomain::Record(_)) => {
240                Err(DomainOpError::WrongType)
241            }
242            (GroundDomain::Matrix(in1, idx1), GroundDomain::Matrix(in2, idx2)) if idx1 == idx2 => {
243                Ok(GroundDomain::Matrix(
244                    Moo::new(in1.union(in2)?),
245                    idx1.clone(),
246                ))
247            }
248            (GroundDomain::Matrix(_, _), _) | (_, GroundDomain::Matrix(_, _)) => {
249                Err(DomainOpError::WrongType)
250            }
251            (GroundDomain::Set(_, in1), GroundDomain::Set(_, in2)) => Ok(GroundDomain::Set(
252                SetAttr::default(),
253                Moo::new(in1.union(in2)?),
254            )),
255            (GroundDomain::Set(_, _), _) | (_, GroundDomain::Set(_, _)) => {
256                Err(DomainOpError::WrongType)
257            }
258            (GroundDomain::MSet(_, in1), GroundDomain::MSet(_, in2)) => Ok(GroundDomain::MSet(
259                MSetAttr::default(),
260                Moo::new(in1.union(in2)?),
261            )),
262            (GroundDomain::Sequence(_, in1), GroundDomain::Sequence(_, in2)) => Ok(
263                GroundDomain::Sequence(SequenceAttr::default(), Moo::new(in1.union(in2)?)),
264            ),
265            (GroundDomain::Sequence(_, _), _) | (_, GroundDomain::Sequence(_, _)) => {
266                Err(DomainOpError::WrongType)
267            }
268            (GroundDomain::Relation(_, in1s), GroundDomain::Relation(_, in2s)) => {
269                let mut inners = Vec::new();
270                for (in1, in2) in zip(in1s, in2s) {
271                    inners.push(Moo::new(in1.union(in2)?));
272                }
273                Ok(GroundDomain::Relation(RelAttr::default(), inners))
274            }
275            (GroundDomain::Relation(..), _) | (_, GroundDomain::Relation(..)) => {
276                Err(DomainOpError::WrongType)
277            }
278            #[allow(unreachable_patterns)]
279            (GroundDomain::Variant(_), _) | (_, GroundDomain::Variant(_)) => {
280                todo!("union variant domains")
281            }
282            #[allow(unreachable_patterns)]
283            (GroundDomain::Function(..), _) | (_, GroundDomain::Function(..)) => {
284                todo!("union function domains")
285            }
286            #[allow(unreachable_patterns)]
287            (GroundDomain::Partition(..), _) | (_, GroundDomain::Partition(..)) => {
288                todo!("union partition domains")
289            }
290            #[allow(unreachable_patterns)]
291            (GroundDomain::Permutation(..), _) | (_, GroundDomain::Permutation(..)) => {
292                todo!("union permutation domains")
293            }
294        }
295    }

297    /// Calculates the intersection of two domains.
298    ///
299    /// # Errors
300    ///
301    ///  - [`DomainOpError::Unbounded`] if either of the input domains are unbounded.
302    ///  - [`DomainOpError::WrongType`] if the input domains are different types, or are not integer or set domains.
303    pub fn intersect(&self, other: &GroundDomain) -> Result<GroundDomain, DomainOpError> {
304        // TODO: does not consider unbounded domains yet
305        // needs to be tested once comprehension rules are written
306
307        match (self, other) {
308            // one or more arguments is an empty int domain
309            (d @ GroundDomain::Empty(ReturnType::Int), GroundDomain::Int(_)) => Ok(d.clone()),
310            (GroundDomain::Int(_), d @ GroundDomain::Empty(ReturnType::Int)) => Ok(d.clone()),
311            (GroundDomain::Empty(ReturnType::Int), d @ GroundDomain::Empty(ReturnType::Int)) => {
312                Ok(d.clone())
313            }
314
315            // one or more arguments is an empty set(int) domain
316            (GroundDomain::Set(_, inner1), d @ GroundDomain::Empty(ReturnType::Set(inner2)))
317                if matches!(
318                    **inner1,
319                    GroundDomain::Int(_) | GroundDomain::Empty(ReturnType::Int)
320                ) && matches!(**inner2, ReturnType::Int) =>
321            {
322                Ok(d.clone())
323            }
324            (d @ GroundDomain::Empty(ReturnType::Set(inner1)), GroundDomain::Set(_, inner2))
325                if matches!(**inner1, ReturnType::Int)
326                    && matches!(
327                        **inner2,
328                        GroundDomain::Int(_) | GroundDomain::Empty(ReturnType::Int)
329                    ) =>
330            {
331                Ok(d.clone())
332            }
333            (
334                d @ GroundDomain::Empty(ReturnType::Set(inner1)),
335                GroundDomain::Empty(ReturnType::Set(inner2)),
336            ) if matches!(**inner1, ReturnType::Int) && matches!(**inner2, ReturnType::Int) => {
337                Ok(d.clone())
338            }
339
340            // both arguments are non-empy
341            (GroundDomain::Set(_, x), GroundDomain::Set(_, y)) => Ok(GroundDomain::Set(
342                SetAttr::default(),
343                Moo::new((*x).intersect(y)?),
344            )),
345
346            (GroundDomain::Int(_), GroundDomain::Int(_)) => {
347                let mut v: BTreeSet<i32> = BTreeSet::new();
348
349                let v1 = self.values_i32()?;
350                let v2 = other.values_i32()?;
351                for value1 in v1.iter() {
352                    if v2.contains(value1) && !v.contains(value1) {
353                        v.insert(*value1);
354                    }
355                }
356                Ok(GroundDomain::from_set_i32(&v))
357            }
358            (GroundDomain::Relation(_, _), GroundDomain::Relation(_, _)) => {
359                todo!("Relation union not yet supported")
360            }
361            _ => Err(DomainOpError::WrongType),
362        }
363    }

365    pub fn values(&self) -> Result<Box<dyn Iterator<Item = Literal>>, DomainOpError> {
366        match self {
367            GroundDomain::Empty(_) => Ok(Box::new(vec![].into_iter())),
368            GroundDomain::Bool => Ok(Box::new(
369                vec![Literal::from(false), Literal::from(true)].into_iter(),
370            )),
371            GroundDomain::Int(rngs) => {
372                let rng_iters = rngs
373                    .iter()
374                    .map(Range::iter)
375                    .collect::<Option<Vec<_>>>()
376                    .ok_or(DomainOpError::Unbounded)?;
377                Ok(Box::new(
378                    rng_iters.into_iter().flat_map(|ri| ri.map(Literal::from)),
379                ))
380            }
381            GroundDomain::Tuple(elem_doms) => {
382                // Collect the possible values for each element
383                let elem_value_pools: Vec<Vec<Literal>> = elem_doms
384                    .iter()
385                    .map(|d| d.values().map(|it| it.collect()))
386                    .collect::<Result<_, _>>()?;
387
388                // Generate all combinations in lexicographic order
389                let iter = elem_value_pools
390                    .into_iter()
391                    .multi_cartesian_product()
392                    .map(|elems| Literal::AbstractLiteral(AbstractLiteral::Tuple(elems)));
393
394                Ok(Box::new(iter))
395            }
396            GroundDomain::Record(entries) => {
397                // Sort entries by name
398                let mut sorted: Vec<&_> = entries.iter().collect();
399                sorted.sort_by(|a, b| a.name.cmp(&b.name));
400
401                let names: Vec<_> = sorted.iter().map(|e| e.name.clone()).collect();
402                let value_pools: Vec<Vec<Literal>> = sorted
403                    .iter()
404                    .map(|e| e.value.values().map(|it| it.collect()))
405                    .collect::<Result<_, _>>()?;
406
407                // Generate all combinations in lexicographic order
408                let iter = value_pools
409                    .into_iter()
410                    .multi_cartesian_product()
411                    .map(move |vals| {
412                        let record_entries = names
413                            .iter()
414                            .cloned()
415                            .zip(vals)
416                            .map(|(name, value)| Field { name, value })
417                            .collect();
418                        Literal::AbstractLiteral(AbstractLiteral::Record(record_entries))
419                    });
420
421                Ok(Box::new(iter))
422            }
423            GroundDomain::Variant(entries) => {
424                let values = entries
425                    .iter()
426                    .map(|entry| {
427                        let name = entry.name.clone();
428                        entry.value.values().map(|values| {
429                            values.map(move |value| {
430                                Literal::AbstractLiteral(AbstractLiteral::Variant(Moo::new(
431                                    Field {
432                                        name: name.clone(),
433                                        value,
434                                    },
435                                )))
436                            })
437                        })
438                    })
439                    .collect::<Result<Vec<_>, _>>()?;
440                Ok(Box::new(values.into_iter().flatten()))
441            }
442            GroundDomain::Matrix(elem_dom, idx_doms) => {
443                let shape = matrix::shape_of_dom(self)?;
444                let idx_doms = idx_doms.clone();
445
446                // Collect all possible element values
447                let elem_values: Vec<Literal> = elem_dom.values()?.collect();
448
449                // Generate all possible cell assignments in lexicographic order
450                let iter = std::iter::repeat_n(elem_values, shape.size)
451                    .multi_cartesian_product()
452                    .map(move |flat_elems| {
453                        matrix::unflatten_matrix::<Literal>(&flat_elems, &idx_doms, &shape.strides)
454                    });
455
456                Ok(Box::new(iter))
457            }
458            GroundDomain::Sequence(attrs, inner_dom) => {
459                if attrs.jectivity != JectivityAttr::None {
460                    // Injective/surjective/bijective sequence enumeration is not yet needed by any
461                    // in-scope case (the two exhaustive tests using it are deferred pending enum
462                    // support), so it is left unimplemented rather than guessed at untested.
463                    todo!("Enumerating jective sequence domains is not yet supported")
464                }
465
466                let min_sz = attrs.size.low().copied().unwrap_or(0).max(0);
467                let max_sz = attrs.size.high().copied().ok_or(DomainOpError::Unbounded)?;
468
469                let pool = inner_dom.values()?.collect_vec();
470
471                let iter = (min_sz..=max_sz)
472                    .flat_map(move |sz| {
473                        std::iter::repeat_n(pool.clone(), sz.max(0) as usize)
474                            .multi_cartesian_product()
475                    })
476                    .map(|elems| Literal::AbstractLiteral(AbstractLiteral::Sequence(elems)));
477
478                Ok(Box::new(iter))
479            }
480            GroundDomain::Set(attrs, inner_dom) => {
481                let n: Int = inner_dom.len_usize()?.try_into()?;
482                let min_sz = attrs.size.low().copied().unwrap_or(0);
483                let max_sz = attrs.size.high().copied().unwrap_or(n);
484
485                let pool = inner_dom.values()?.collect_vec();
486
487                Ok(Box::new(
488                    (min_sz..=max_sz)
489                        .flat_map(move |sz| pool.clone().into_iter().combinations(sz as usize))
490                        .map(|elems| Literal::AbstractLiteral(AbstractLiteral::Set(elems))),
491                ))
492            }
493            GroundDomain::MSet(..) => todo!("Enumerating multi-set domains is not yet supported"),
494            GroundDomain::Function(..) => {
495                todo!("Enumerating function domains is not yet supported")
496            }
497            GroundDomain::Relation(..) => {
498                todo!("Enumerating relation domains is not yet supported")
499            }
500            GroundDomain::Partition(attr, inner_dom) => {
501                let elements: Vec<Literal> = inner_dom.values()?.collect();
502                let n = elements.len();
503
504                let block_lo = attr.part_len.low().copied().unwrap_or(1).max(1) as usize;
505                let block_hi = attr
506                    .part_len
507                    .high()
508                    .copied()
509                    .map(|h| (h.max(0) as usize).min(n))
510                    .unwrap_or(n);
511                let parts_lo = attr.num_parts.low().copied().unwrap_or(0).max(0) as usize;
512                let parts_hi = attr
513                    .num_parts
514                    .high()
515                    .copied()
516                    .map(|h| (h.max(0) as usize).min(n))
517                    .unwrap_or(n);
518                let is_regular = attr.is_regular;
519
520                let partitions = if block_lo > block_hi {
521                    vec![]
522                } else {
523                    restricted_partitions(&elements, block_lo, block_hi)
524                };
525                let iter = partitions.into_iter().filter(move |parts| {
526                    let k = parts.len();
527                    if k < parts_lo || k > parts_hi {
528                        return false;
529                    }
530                    !is_regular
531                        || parts
532                            .first()
533                            .is_none_or(|first| parts.iter().all(|p| p.len() == first.len()))
534                });
535                Ok(Box::new(iter.map(|parts| {
536                    Literal::AbstractLiteral(AbstractLiteral::Partition(parts))
537                })))
538            }
539            GroundDomain::Permutation(attr, inner_dom) => {
540                let elements: Vec<Literal> = inner_dom.values()?.collect();
541                let n = elements.len();
542
543                let moved_lo = attr.num_moved.low().copied().unwrap_or(0).max(0) as usize;
544                let moved_hi = attr
545                    .num_moved
546                    .high()
547                    .copied()
548                    .map(|h| (h.max(0) as usize).min(n))
549                    .unwrap_or(n);
550
551                if moved_lo > moved_hi {
552                    return Ok(Box::new(std::iter::empty()));
553                }
554                let mappings: Vec<Vec<Literal>> =
555                    restricted_permutations(&elements, moved_lo, moved_hi).collect();
556                let values = mappings.into_iter().map(move |mapped| {
557                    let cycles = permutation_mapping_to_cycles(&elements, &mapped);
558                    Literal::AbstractLiteral(AbstractLiteral::Permutation(cycles))
559                });
560                Ok(Box::new(values.collect_vec().into_iter()))
561            }
562        }
563    }

565    /// Gets the length of this domain.
566    ///
567    /// # Errors
568    ///
569    /// - [`DomainOpError::Unbounded`] if the input domain is of infinite size.
570    pub fn length(&self) -> Result<u64, DomainOpError> {
571        match self {
572            GroundDomain::Empty(_) => Ok(0),
573            GroundDomain::Bool => Ok(2),
574            GroundDomain::Int(ranges) => {
575                if ranges.is_empty() {
576                    return Ok(0);
577                }
578
579                let mut length = 0u64;
580                for range in ranges {
581                    if let Some(range_length) = range.length() {
582                        length += range_length as u64;
583                    } else {
584                        return Err(DomainOpError::Unbounded);
585                    }
586                }
587                Ok(length)
588            }
589            GroundDomain::Tuple(domains) => {
590                let mut ans = 1u64;
591                for domain in domains {
592                    ans = ans
593                        .checked_mul(domain.length()?)
594                        .ok_or(DomainOpError::TooLarge)?;
595                }
596                Ok(ans)
597            }
598            GroundDomain::Record(entries) => {
599                // A record is just a named tuple
600                let mut ans = 1u64;
601                for entry in entries {
602                    let sz = entry.value.length()?;
603                    ans = ans.checked_mul(sz).ok_or(DomainOpError::TooLarge)?;
604                }
605                Ok(ans)
606            }
607            GroundDomain::Variant(entries) => {
608                let mut ans = 0u64;
609                for entry in entries {
610                    let sz = entry.value.length()?;
611                    // Only one field can be in the variant at once
612                    ans = ans.checked_add(sz).ok_or(DomainOpError::TooLarge)?;
613                }
614                Ok(ans)
615            }
616            GroundDomain::Matrix(inner_domain, idx_domains) => {
617                let inner_sz = inner_domain.length()?;
618                let exp = idx_domains.iter().try_fold(1u32, |acc, val| {
619                    let len = val.length()? as u32;
620                    acc.checked_mul(len).ok_or(DomainOpError::TooLarge)
621                })?;
622                inner_sz.checked_pow(exp).ok_or(DomainOpError::TooLarge)
623            }
624            GroundDomain::Sequence(seq_attr, inner_domain) => {
625                if seq_attr.jectivity != JectivityAttr::None {
626                    // See the matching note on `values()` above: not yet needed by any in-scope case.
627                    todo!("Length bound of jective sequences is not yet supported");
628                }
629
630                let inner_len = inner_domain.length()?;
631                let min_sz = seq_attr.size.low().copied().unwrap_or(0).max(0) as u64;
632                let max_sz = seq_attr
633                    .size
634                    .high()
635                    .copied()
636                    .ok_or(DomainOpError::Unbounded)? as u64;
637
638                if min_sz > max_sz {
639                    return Ok(0);
640                }
641
642                let mut ans = 0u64;
643                for sz in min_sz..=max_sz {
644                    let sz: u32 = sz.try_into().map_err(|_| DomainOpError::TooLarge)?;
645                    let c = inner_len.checked_pow(sz).ok_or(DomainOpError::TooLarge)?;
646                    ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
647                }
648                Ok(ans)
649            }
650            GroundDomain::Set(set_attr, inner_domain) => {
651                let inner_len = inner_domain.length()?;
652                let (min_sz, max_sz) = match set_attr.size {
653                    Range::Unbounded => (0, inner_len),
654                    Range::Single(n) => (n as u64, n as u64),
655                    Range::UnboundedR(n) => (n as u64, inner_len),
656                    Range::UnboundedL(n) => (0, n as u64),
657                    Range::Bounded(min, max) => (min as u64, max as u64),
658                };
659                // Attributes may overshoot the inner domain (e.g. maxSize 3 of int(1..2));
660                // only cardinalities that fit can contribute members.
661                let max_sz = max_sz.min(inner_len);
662                if min_sz > max_sz {
663                    return Ok(0);
664                }
665                let mut ans = 0u64;
666                for sz in min_sz..=max_sz {
667                    let c = count_combinations(inner_len, sz)?;
668                    ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
669                }
670                Ok(ans)
671            }
672            GroundDomain::MSet(mset_attr, inner_domain) => {
673                let inner_len = inner_domain.length()?;
674                let (min_sz, max_sz) = match mset_attr.size {
675                    Range::Unbounded => (0, inner_len),
676                    Range::Single(n) => (n as u64, n as u64),
677                    Range::UnboundedR(n) => (n as u64, inner_len),
678                    Range::UnboundedL(n) => (0, n as u64),
679                    Range::Bounded(min, max) => (min as u64, max as u64),
680                };
681                let mut ans = 0u64;
682                for sz in min_sz..=max_sz {
683                    // need  "multichoose", ((n  k)) == (n+k-1  k)
684                    // Where n=inner_len and k=sz
685                    let c = count_combinations(inner_len + sz - 1, sz)?;
686                    ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
687                }
688                Ok(ans)
689            }
690            GroundDomain::Function(attr, domain, codomain) => {
691                let domain_len = domain.length()?;
692                let codomain_len = codomain.length()?;
693
694                match (attr.partiality.clone(), attr.jectivity.clone()) {
695                    (PartialityAttr::Total, JectivityAttr::None) => {
696                        let exp: u32 = domain_len.try_into()?;
697                        codomain_len.checked_pow(exp).ok_or(DomainOpError::TooLarge)
698                    }
699                    (PartialityAttr::Total, JectivityAttr::Injective) => {
700                        if domain_len > codomain_len {
701                            return Ok(0);
702                        }
703                        Ok(count_permutations(codomain_len, domain_len)?)
704                    }
705                    (PartialityAttr::Total, JectivityAttr::Bijective) => {
706                        if domain_len != codomain_len {
707                            return Ok(0);
708                        }
709                        Ok(count_permutations(domain_len, domain_len)?)
710                    }
711                    (PartialityAttr::Total, JectivityAttr::Surjective) => {
712                        let partitions = stirling_second_kind(domain_len, codomain_len)?;
713                        let arrangements = count_permutations(codomain_len, codomain_len)?;
714                        partitions
715                            .checked_mul(arrangements)
716                            .ok_or(DomainOpError::TooLarge)
717                    }
718                    (PartialityAttr::Partial, jectivity) => {
719                        let (min_sz, max_sz) = match attr.size {
720                            Range::Unbounded => (0, domain_len),
721                            Range::Single(n) => (n as u64, n as u64),
722                            Range::UnboundedR(n) => (n as u64, domain_len),
723                            Range::UnboundedL(n) => (0, n as u64),
724                            Range::Bounded(min, max) => (min as u64, max as u64),
725                        };
726                        let max_sz = max_sz.min(domain_len);
727                        if min_sz > max_sz {
728                            return Ok(0);
729                        }
730
731                        let mut ans = 0u64;
732                        for sz in min_sz..=max_sz {
733                            // Choose which `sz` domain elements are defined, then count the ways
734                            // to map exactly those `sz` elements per the jectivity requirement.
735                            let choose = count_combinations(domain_len, sz)?;
736                            let assign = match jectivity {
737                                JectivityAttr::None => codomain_len
738                                    .checked_pow(sz.try_into()?)
739                                    .ok_or(DomainOpError::TooLarge)?,
740                                JectivityAttr::Injective => {
741                                    if sz > codomain_len {
742                                        0
743                                    } else {
744                                        count_permutations(codomain_len, sz)?
745                                    }
746                                }
747                                JectivityAttr::Bijective => {
748                                    if sz != codomain_len {
749                                        0
750                                    } else {
751                                        count_permutations(codomain_len, codomain_len)?
752                                    }
753                                }
754                                JectivityAttr::Surjective => {
755                                    let partitions = stirling_second_kind(sz, codomain_len)?;
756                                    let arrangements =
757                                        count_permutations(codomain_len, codomain_len)?;
758                                    partitions
759                                        .checked_mul(arrangements)
760                                        .ok_or(DomainOpError::TooLarge)?
761                                }
762                            };
763                            let term = choose.checked_mul(assign).ok_or(DomainOpError::TooLarge)?;
764                            ans = ans.checked_add(term).ok_or(DomainOpError::TooLarge)?;
765                        }
766                        Ok(ans)
767                    }
768                }
769            }
770            GroundDomain::Relation(attr, domains) => {
771                let dom_sizes_result: Result<Vec<u64>, DomainOpError> =
772                    domains.iter().map(|x| x.length()).collect();
773                let dom_sizes = dom_sizes_result?;
774                let inner_len: u64 = dom_sizes.iter().product();
775
776                let (min_sz, max_sz) = match attr.size {
777                    Range::Unbounded => (0, inner_len),
778                    Range::Single(n) => (n as u64, n as u64),
779                    Range::UnboundedR(n) => (n as u64, inner_len),
780                    Range::UnboundedL(n) => (0, n as u64),
781                    Range::Bounded(min, max) => (min as u64, max as u64),
782                };
783                let max_sz = max_sz.min(inner_len);
784                if min_sz > max_sz {
785                    return Ok(0);
786                }
787
788                let mut ans = 0u64;
789                for sz in min_sz..=max_sz {
790                    let c = count_combinations(inner_len, sz)?;
791                    ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
792                }
793                Ok(ans)
794            }
795            GroundDomain::Partition(attr, inner_domain) => {
796                let n = inner_domain.length()?;
797                let block_lo = attr.part_len.low().copied().unwrap_or(1).max(1) as u64;
798                let block_hi = attr
799                    .part_len
800                    .high()
801                    .copied()
802                    .map(|h| (h.max(0) as u64).min(n))
803                    .unwrap_or(n);
804                let parts_lo = attr.num_parts.low().copied().unwrap_or(0).max(0) as u64;
805                let parts_hi = attr
806                    .num_parts
807                    .high()
808                    .copied()
809                    .map(|h| (h.max(0) as u64).min(n))
810                    .unwrap_or(n);
811                if block_lo > block_hi || parts_lo > parts_hi {
812                    return Ok(0);
813                }
814
815                if attr.is_regular {
816                    let mut ans = 0u64;
817                    for block_size in block_lo..=block_hi {
818                        if n % block_size != 0 {
819                            continue;
820                        }
821                        let num_parts = n / block_size;
822                        if num_parts < parts_lo || num_parts > parts_hi {
823                            continue;
824                        }
825                        let c = regular_partition_count(n, block_size)?;
826                        ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
827                    }
828                    Ok(ans)
829                } else {
830                    let mut ans = 0u64;
831                    for num_parts in parts_lo..=parts_hi {
832                        let c = restricted_partition_count(n, num_parts, block_lo, block_hi)?;
833                        ans = ans.checked_add(c).ok_or(DomainOpError::TooLarge)?;
834                    }
835                    Ok(ans)
836                }
837            }
838            GroundDomain::Permutation(attr, inner_domain) => {
839                let n = inner_domain.length()?;
840                let moved_lo = attr.num_moved.low().copied().unwrap_or(0).max(0) as u64;
841                let moved_hi = attr
842                    .num_moved
843                    .high()
844                    .copied()
845                    .map(|h| (h.max(0) as u64).min(n))
846                    .unwrap_or(n);
847                if moved_lo > moved_hi {
848                    return Ok(0);
849                }
850
851                let mut ans = 0u64;
852                for moved in moved_lo..=moved_hi {
853                    let choose = count_combinations(n, moved)?;
854                    let derange = derangements(moved)?;
855                    let term = choose.checked_mul(derange).ok_or(DomainOpError::TooLarge)?;
856                    ans = ans.checked_add(term).ok_or(DomainOpError::TooLarge)?;
857                }
858                Ok(ans)
859            }
860        }
861    }

870    pub fn contains(&self, lit: &Literal) -> Result<bool, DomainOpError> {
871        // not adding a generic wildcard condition for all domains, so that this gives a compile
872        // error when a domain is added.
873        match self {
874            // empty domain can't contain anything
875            GroundDomain::Empty(_) => Ok(false),
876            GroundDomain::Bool => match lit {
877                Literal::Bool(_) => Ok(true),
878                _ => Ok(false),
879            },
880            GroundDomain::Int(ranges) => match lit {
881                Literal::Int(x) => {
882                    if ranges.is_empty() {
883                        return Ok(false);
884                    };
885
886                    Ok(ranges.iter().any(|range| range.contains(x)))
887                }
888                _ => Ok(false),
889            },
890            GroundDomain::Tuple(elem_domains) => {
891                match lit {
892                    Literal::AbstractLiteral(AbstractLiteral::Tuple(literal_elems)) => {
893                        if elem_domains.len() != literal_elems.len() {
894                            return Ok(false);
895                        }
896
897                        // for every element in the tuple literal, check if it is in the corresponding domain
898                        for (elem_domain, elem) in itertools::izip!(elem_domains, literal_elems) {
899                            if !elem_domain.contains(elem)? {
900                                return Ok(false);
901                            }
902                        }
903
904                        Ok(true)
905                    }
906                    _ => Ok(false),
907                }
908            }
909            GroundDomain::Record(entries) => match lit {
910                Literal::AbstractLiteral(AbstractLiteral::Record(lit_entries)) => {
911                    if entries.len() != lit_entries.len() {
912                        return Ok(false);
913                    }
914
915                    for (entry, lit_entry) in itertools::izip!(entries, lit_entries) {
916                        if entry.name != lit_entry.name
917                            || !(entry.value.contains(&lit_entry.value)?)
918                        {
919                            return Ok(false);
920                        }
921                    }
922                    Ok(true)
923                }
924                _ => Ok(false),
925            },
926            GroundDomain::Variant(entries) => match lit {
927                Literal::AbstractLiteral(AbstractLiteral::Variant(lit_entry)) => {
928                    let Some(entry) = entries.iter().find(|entry| entry.name == lit_entry.name)
929                    else {
930                        return Ok(false);
931                    };
932                    entry.value.contains(&lit_entry.value)
933                }
934                _ => Ok(false),
935            },
936            GroundDomain::Matrix(elem_domain, index_domains) => {
937                match lit {
938                    Literal::AbstractLiteral(AbstractLiteral::Matrix(elems, idx_domain)) => {
939                        // Matrix literals are represented as nested 1d matrices, so the elements of
940                        // the matrix literal will be the inner dimensions of the matrix.
941
942                        if elems.is_empty()
943                            && index_domains
944                                .iter()
945                                .any(|index_domain| index_domain.length() == Ok(0))
946                        {
947                            return Ok(true);
948                        }
949
950                        let Some((current_index_domain, remaining_index_domains)) =
951                            index_domains.split_first()
952                        else {
953                            panic!("a matrix should have at least one index domain");
954                        };
955
956                        if *current_index_domain != *idx_domain {
957                            return Ok(false);
958                        };
959
960                        let next_elem_domain = if remaining_index_domains.is_empty() {
961                            // Base case - we have a 1D row. Now check if all elements in the
962                            // literal are in this row's element domain.
963                            elem_domain.as_ref().clone()
964                        } else {
965                            // Otherwise, go down a dimension (e.g. 2D matrix inside a 3D tensor)
966                            GroundDomain::Matrix(
967                                elem_domain.clone(),
968                                remaining_index_domains.to_vec(),
969                            )
970                        };
971
972                        for elem in elems {
973                            if !next_elem_domain.contains(elem)? {
974                                return Ok(false);
975                            }
976                        }
977
978                        Ok(true)
979                    }
980                    _ => Ok(false),
981                }
982            }
983            GroundDomain::Sequence(seq_attr, inner_dom) => match lit {
984                Literal::AbstractLiteral(AbstractLiteral::Sequence(elems)) => {
985                    let sz = elems.len().to_i32().ok_or(DomainOpError::TooLarge)?;
986                    if !seq_attr.size.contains(&sz) {
987                        return Ok(false);
988                    }
989
990                    for elem in elems {
991                        if !inner_dom.contains(elem)? {
992                            return Ok(false);
993                        }
994                    }
995                    Ok(true)
996                }
997                _ => Ok(false),
998            },
999            GroundDomain::Set(set_attr, inner_domain) => match lit {
1000                Literal::AbstractLiteral(AbstractLiteral::Set(lit_elems)) => {
1001                    // check if the literal's size is allowed by the set attribute
1002                    let sz = lit_elems.len().to_i32().ok_or(DomainOpError::TooLarge)?;
1003                    if !set_attr.size.contains(&sz) {
1004                        return Ok(false);
1005                    }
1006
1007                    for elem in lit_elems {
1008                        if !inner_domain.contains(elem)? {
1009                            return Ok(false);
1010                        }
1011                    }
1012                    Ok(true)
1013                }
1014                _ => Ok(false),
1015            },
1016            GroundDomain::MSet(mset_attr, inner_domain) => match lit {
1017                Literal::AbstractLiteral(AbstractLiteral::MSet(lit_elems)) => {
1018                    // check if the literal's size is allowed by the mset attribute
1019                    let sz = lit_elems.len().to_i32().ok_or(DomainOpError::TooLarge)?;
1020                    if !mset_attr.size.contains(&sz) {
1021                        return Ok(false);
1022                    }
1023
1024                    for elem in lit_elems {
1025                        if !inner_domain.contains(elem)? {
1026                            return Ok(false);
1027                        }
1028                    }
1029                    Ok(true)
1030                }
1031                _ => Ok(false),
1032            },
1033            GroundDomain::Function(func_attr, domain, codomain) => match lit {
1034                Literal::AbstractLiteral(AbstractLiteral::Function(lit_elems)) => {
1035                    let sz = Int::try_from(lit_elems.len()).expect("Should convert");
1036                    if !func_attr.size.contains(&sz) {
1037                        return Ok(false);
1038                    }
1039                    for lit in lit_elems {
1040                        let domain_element = &lit.0;
1041                        let codomain_element = &lit.1;
1042                        if !domain.contains(domain_element)? {
1043                            return Ok(false);
1044                        }
1045                        if !codomain.contains(codomain_element)? {
1046                            return Ok(false);
1047                        }
1048                    }
1049                    Ok(true)
1050                }
1051                _ => Ok(false),
1052            },
1053            GroundDomain::Relation(rel_attr, inner_domains) => match lit {
1054                Literal::AbstractLiteral(AbstractLiteral::Relation(lit_elems)) => {
1055                    // check if the literal's size is allowed by the attributes
1056                    let sz = lit_elems.len().to_i32().ok_or(DomainOpError::TooLarge)?;
1057                    if !rel_attr.size.contains(&sz) {
1058                        return Ok(false);
1059                    }
1060
1061                    for elem_tuple in lit_elems {
1062                        if elem_tuple.len() == inner_domains.len() {
1063                            for (elem, inner_dom) in elem_tuple.iter().zip(inner_domains.iter()) {
1064                                if !inner_dom.contains(elem)? {
1065                                    return Ok(false);
1066                                }
1067                            }
1068                        } else {
1069                            return Ok(false);
1070                        }
1071                    }
1072                    Ok(true)
1073                }
1074                _ => Ok(false),
1075            },
1076            GroundDomain::Partition(attr, dom) => match lit {
1077                Literal::AbstractLiteral(AbstractLiteral::Partition(lit_elems)) => {
1078                    // let sz = lit_elems.len().to_i32().ok_or(DomainOpError::TooLarge)?;
1079                    let sz: i32 = lit_elems
1080                        .iter()
1081                        .flatten()
1082                        .count()
1083                        .to_i32()
1084                        .ok_or(DomainOpError::TooLarge)?;
1085
1086                    let min: Option<i32> = match (attr.num_parts.low(), attr.part_len.low()) {
1087                        (Some(x), Some(y)) => Some(x * y),
1088                        _ => None,
1089                    };
1090
1091                    let max: Option<i32> = match (attr.num_parts.high(), attr.part_len.high()) {
1092                        (Some(x), Some(y)) => Some(x * y),
1093                        _ => None,
1094                    };
1095
1096                    let rng = Range::new(min, max);
1097                    if !rng.contains(&sz) {
1098                        return Ok(false);
1099                    }
1100
1101                    for elem in lit_elems.iter().flatten() {
1102                        if !dom.contains(elem)? {
1103                            return Ok(false);
1104                        }
1105                    }
1106                    Ok(true)
1107                }
1108                _ => Ok(false),
1109            },
1110            GroundDomain::Permutation(attr, dom) => match lit {
1111                Literal::AbstractLiteral(AbstractLiteral::Permutation(cycles)) => {
1112                    // numMoved is the count of moved points (elements mentioned in some cycle),
1113                    // matching cycle notation's "unmentioned = fixed point" semantics -- not the
1114                    // inner domain's own size.
1115                    let sz: i32 = cycles
1116                        .iter()
1117                        .flatten()
1118                        .count()
1119                        .to_i32()
1120                        .ok_or(DomainOpError::TooLarge)?;
1121                    if !attr.num_moved.contains(&sz) {
1122                        return Ok(false);
1123                    }
1124
1125                    for elem in cycles.iter().flatten() {
1126                        if !dom.contains(elem)? {
1127                            return Ok(false);
1128                        }
1129                    }
1130                    Ok(true)
1131                }
1132                _ => Ok(false),
1133            },
1134        }
1135    }

1137    /// Returns a list of all possible values in an integer domain.
1138    ///
1139    /// # Errors
1140    ///
1141    /// - [`DomainOpError::NotInteger`] if the domain is not an integer domain.
1142    /// - [`DomainOpError::Unbounded`] if the domain is unbounded.
1143    pub fn values_i32(&self) -> Result<Vec<i32>, DomainOpError> {
1144        if let GroundDomain::Empty(ReturnType::Int) = self {
1145            return Ok(vec![]);
1146        }
1147        let GroundDomain::Int(ranges) = self else {
1148            return Err(DomainOpError::NotInteger(self.return_type()));
1149        };
1150
1151        if ranges.is_empty() {
1152            return Ok(vec![]);
1153        }
1154
1155        let mut values = vec![];
1156        for range in ranges {
1157            match range {
1158                Range::Single(i) => {
1159                    values.push(*i);
1160                }
1161                Range::Bounded(i, j) => {
1162                    values.extend(*i..=*j);
1163                }
1164                Range::UnboundedR(_) | Range::UnboundedL(_) | Range::Unbounded => {
1165                    return Err(DomainOpError::Unbounded);
1166                }
1167            }
1168        }
1169
1170        Ok(values)
1171    }

1211    pub fn from_set_i32(elements: &BTreeSet<i32>) -> GroundDomain {
1212        if elements.is_empty() {
1213            return GroundDomain::Empty(ReturnType::Int);
1214        }
1215        if elements.len() == 1 {
1216            return GroundDomain::Int(vec![Range::Single(*elements.first().unwrap())]);
1217        }
1218
1219        let mut elems_iter = elements.iter().copied();
1220
1221        let mut ranges: Vec<Range<i32>> = vec![];
1222
1223        // Loop over the elements in ascending order, turning all sequential runs of
1224        // numbers into ranges.
1225
1226        // the bounds of the current run of numbers.
1227        let mut lower = elems_iter
1228            .next()
1229            .expect("if we get here, elements should have => 2 elements");
1230        let mut upper = lower;
1231
1232        for current in elems_iter {
1233            // As elements is a BTreeSet, current is always strictly larger than lower.
1234
1235            if current == upper + 1 {
1236                // current is part of the current run - we now have the run lower..current
1237                //
1238                upper = current;
1239            } else {
1240                // the run lower..upper has ended.
1241                //
1242                // Add the run lower..upper to the domain, and start a new run.
1243
1244                if lower == upper {
1245                    ranges.push(range!(lower));
1246                } else {
1247                    ranges.push(range!(lower..upper));
1248                }
1249
1250                lower = current;
1251                upper = current;
1252            }
1253        }
1254
1255        // add the final run to the domain
1256        if lower == upper {
1257            ranges.push(range!(lower));
1258        } else {
1259            ranges.push(range!(lower..upper));
1260        }
1261
1262        ranges = Range::squeeze(&ranges);
1263        GroundDomain::Int(ranges)
1264    }

1267    ///
1268    /// The given operator may return `None` if the operation is not defined for its arguments.

1270    ///
1271    /// # Errors
1272    ///
1273    /// - [`DomainOpError::Unbounded`] if either of the input domains are unbounded.

1276        &self,
1277        op: fn(i32, i32) -> Option<i32>,
1278        other: &GroundDomain,
1279    ) -> Result<GroundDomain, DomainOpError> {
1280        let vs1 = self.values_i32()?;
1281        let vs2 = other.values_i32()?;

1283        let mut set = BTreeSet::new();
1284        for (v1, v2) in itertools::iproduct!(vs1, vs2) {

1288        }

1397        let first_literal = literals.first().unwrap();

1416                    Err(DomainOpError::WrongType)
1417                } else {
1418                    Ok(GroundDomain::Bool)
1419                }
1420            }
1421            Literal::AbstractLiteral(AbstractLiteral::Set(_)) => {

1423
1424                for lit in literals {
1425                    let Literal::AbstractLiteral(AbstractLiteral::Set(elems)) = lit else {

1427                    };
1428

1430                }
1431                let elem_domain = GroundDomain::from_literal_vec(&all_elems)?;
1432
1433                Ok(GroundDomain::Set(SetAttr::default(), Moo::new(elem_domain)))

1435            Literal::AbstractLiteral(AbstractLiteral::MSet(_)) => {
1436                let mut all_elems = vec![];
1437
1438                for lit in literals {
1439                    let Literal::AbstractLiteral(AbstractLiteral::MSet(elems)) = lit else {

1441                    };
1442

1446

1457            }

1469                        break;
1470                    };

1472                }
1473
1474                let mut all_elems: Vec<Literal> = vec![];
1475
1476                // check types and index domains

1483

1486                        if index_domain != first_index_domain {
1487                            return Err(DomainOpError::WrongType);

1499                            break;
1500                        };

1506                    }

1871impl Display for GroundDomain {
1872    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
1873        match &self {
1874            GroundDomain::Empty(ty) => write!(f, "empty({ty})"),
1875            GroundDomain::Bool => write!(f, "bool"),
1876            GroundDomain::Int(ranges) => {
1877                if ranges.iter().all(Range::is_lower_or_upper_bounded) {
1878                    let rngs: String = ranges.iter().map(|r| format!("{r}")).join(", ");
1879                    write!(f, "int({})", rngs)
1880                } else {
1881                    write!(f, "int")
1882                }
1883            }
1884            GroundDomain::Tuple(domains) => {
1885                write!(f, "tuple ({})", domains.iter().join(", "))
1886            }
1887            GroundDomain::Record(entries) => {
1888                let inners = entries.iter().map(|t| format!("{}", t)).join(", ");
1889                write!(f, "record {{{inners}}}",)
1890            }
1891            GroundDomain::Variant(entries) => {
1892                let inners = entries.iter().map(|t| format!("{}", t)).join(", ");
1893                write!(f, "variant {{{inners}}}",)
1894            }
1895            GroundDomain::Matrix(value_domain, index_domains) => {
1896                write!(
1897                    f,
1898                    "matrix indexed by {} of {value_domain}",
1899                    pretty_vec(&index_domains.iter().collect_vec())
1900                )
1901            }
1902            GroundDomain::Sequence(attrs, inner_dom) => {
1903                write!(f, "sequence {attrs} of {inner_dom}")
1904            }
1905            GroundDomain::Set(attrs, inner_dom) => {
1906                write!(f, "set")?;
1907                let attrs = attrs.to_string();
1908                if attrs.is_empty() {
1909                    write!(f, " of {inner_dom}")
1910                } else {
1911                    write!(f, " {attrs} of {inner_dom}")
1912                }
1913            }
1914            GroundDomain::MSet(attrs, inner_dom) => {
1915                write!(f, "mset")?;
1916                let attrs = attrs.to_string();
1917                if attrs.is_empty() {
1918                    write!(f, " of {inner_dom}")
1919                } else {
1920                    write!(f, " {attrs} of {inner_dom}")
1921                }
1922            }
1923            GroundDomain::Function(attribute, domain, codomain) => {
1924                write!(f, "function {} {} --> {} ", attribute, domain, codomain)
1925            }
1926            GroundDomain::Relation(attrs, domains) => {
1927                write!(f, "relation {} of ({})", attrs, domains.iter().join(" * "))
1928            }
1929            GroundDomain::Partition(attrs, inner) => {
1930                write!(f, "partition {attrs} from {inner}")
1931            }
1932            GroundDomain::Permutation(attrs, inner) => {
1933                write!(f, "permutation {attrs} of {inner}")
1934            }
1935        }
1936    }
