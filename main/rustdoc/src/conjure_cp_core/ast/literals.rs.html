30/// A literal value, equivalent to constants in Conjure.
31pub enum Literal {
32    Int(i32),
33    Bool(bool),

36    AbstractLiteral(AbstractLiteral<Literal>),

49impl Literal {
50    /// Compare values using Essence-aware value ordering.
51    ///
52    /// Booleans and integers use their natural order, tuple-like values use
53    /// lexicographic order, and sets use lexicographic occurrence order over
54    /// ascending element values (`false < true`).
55    pub fn essence_cmp(&self, other: &Self) -> Ordering {
56        match (self, other) {
57            (Literal::Bool(lhs), Literal::Bool(rhs)) => lhs.cmp(rhs),
58            (Literal::Int(lhs), Literal::Int(rhs)) => lhs.cmp(rhs),
59            (Literal::AbstractLiteral(lhs), Literal::AbstractLiteral(rhs)) => {
60                abstract_literal_essence_cmp(lhs, rhs)
61            }
62            _ => literal_kind(self).cmp(&literal_kind(other)),
63        }
64    }
65}
66
67fn literal_kind(literal: &Literal) -> u8 {
68    match literal {
69        Literal::Bool(_) => 0,
70        Literal::Int(_) => 1,
71        Literal::AbstractLiteral(_) => 2,
72    }
73}

75fn abstract_literal_essence_cmp(
76    lhs: &AbstractLiteral<Literal>,
77    rhs: &AbstractLiteral<Literal>,
78) -> Ordering {
79    match (lhs, rhs) {
80        (AbstractLiteral::Set(lhs), AbstractLiteral::Set(rhs)) => set_essence_cmp(lhs, rhs),
81        (AbstractLiteral::MSet(lhs), AbstractLiteral::MSet(rhs)) => sorted_literals_cmp(lhs, rhs),
82        (AbstractLiteral::Matrix(lhs, _), AbstractLiteral::Matrix(rhs, _))
83        | (AbstractLiteral::Tuple(lhs), AbstractLiteral::Tuple(rhs))
84        | (AbstractLiteral::Sequence(lhs), AbstractLiteral::Sequence(rhs)) => {
85            literal_slice_cmp(lhs, rhs)
86        }
87        (AbstractLiteral::Record(lhs), AbstractLiteral::Record(rhs)) => lhs
88            .iter()
89            .zip(rhs)
90            .find_map(|(lhs, rhs)| {
91                let ordering = lhs.name.to_string().cmp(&rhs.name.to_string());
92                (ordering != Ordering::Equal)
93                    .then_some(ordering)
94                    .or_else(|| {
95                        let ordering = lhs.value.essence_cmp(&rhs.value);
96                        (ordering != Ordering::Equal).then_some(ordering)
97                    })
98            })
99            .unwrap_or_else(|| lhs.len().cmp(&rhs.len())),
100        (AbstractLiteral::Function(lhs), AbstractLiteral::Function(rhs)) => lhs
101            .iter()
102            .zip(rhs)
103            .find_map(|((lhs_from, lhs_to), (rhs_from, rhs_to))| {
104                let ordering = lhs_from.essence_cmp(rhs_from);
105                (ordering != Ordering::Equal)
106                    .then_some(ordering)
107                    .or_else(|| {
108                        let ordering = lhs_to.essence_cmp(rhs_to);
109                        (ordering != Ordering::Equal).then_some(ordering)
110                    })
111            })
112            .unwrap_or_else(|| lhs.len().cmp(&rhs.len())),
113        (AbstractLiteral::Variant(lhs), AbstractLiteral::Variant(rhs)) => lhs
114            .name
115            .to_string()
116            .cmp(&rhs.name.to_string())
117            .then_with(|| lhs.value.essence_cmp(&rhs.value)),
118        (AbstractLiteral::Partition(lhs), AbstractLiteral::Partition(rhs))
119        | (AbstractLiteral::Relation(lhs), AbstractLiteral::Relation(rhs))
120        | (AbstractLiteral::Permutation(lhs), AbstractLiteral::Permutation(rhs)) => lhs
121            .iter()
122            .zip(rhs)
123            .find_map(|(lhs, rhs)| {
124                let ordering = literal_slice_cmp(lhs, rhs);
125                (ordering != Ordering::Equal).then_some(ordering)
126            })
127            .unwrap_or_else(|| lhs.len().cmp(&rhs.len())),
128        _ => abstract_literal_kind(lhs).cmp(&abstract_literal_kind(rhs)),
129    }
130}

132fn abstract_literal_kind(literal: &AbstractLiteral<Literal>) -> u8 {
133    match literal {
134        AbstractLiteral::Set(_) => 0,

148fn literal_slice_cmp(lhs: &[Literal], rhs: &[Literal]) -> Ordering {
149    lhs.iter()
150        .zip(rhs)
151        .find_map(|(lhs, rhs)| {
152            let ordering = lhs.essence_cmp(rhs);
153            (ordering != Ordering::Equal).then_some(ordering)
154        })
155        .unwrap_or_else(|| lhs.len().cmp(&rhs.len()))
156}

158fn sorted_literals_cmp(lhs: &[Literal], rhs: &[Literal]) -> Ordering {
159    let mut lhs = lhs.iter().collect::<Vec<_>>();
160    let mut rhs = rhs.iter().collect::<Vec<_>>();
161    lhs.sort_by(|lhs, rhs| lhs.essence_cmp(rhs));
162    rhs.sort_by(|lhs, rhs| lhs.essence_cmp(rhs));
163    lhs.iter()
164        .zip(&rhs)
165        .find_map(|(lhs, rhs)| {
166            let ordering = lhs.essence_cmp(rhs);
167            (ordering != Ordering::Equal).then_some(ordering)
168        })
169        .unwrap_or_else(|| lhs.len().cmp(&rhs.len()))
170}

172/// Compare sets as occurrence vectors over the ordered union of their elements.
173fn set_essence_cmp(lhs: &[Literal], rhs: &[Literal]) -> Ordering {
174    let mut lhs = lhs.iter().collect::<Vec<_>>();
175    let mut rhs = rhs.iter().collect::<Vec<_>>();
176    lhs.sort_by(|lhs, rhs| lhs.essence_cmp(rhs));
177    rhs.sort_by(|lhs, rhs| lhs.essence_cmp(rhs));
178
179    let (mut lhs_index, mut rhs_index) = (0, 0);
180    while lhs_index < lhs.len() && rhs_index < rhs.len() {
181        match lhs[lhs_index].essence_cmp(rhs[rhs_index]) {
182            Ordering::Equal => {
183                lhs_index += 1;
184                rhs_index += 1;
185            }
186            // `lhs` contains the least differing element and `rhs` does not.
187            Ordering::Less => return Ordering::Greater,
188            // `rhs` contains the least differing element and `lhs` does not.
189            Ordering::Greater => return Ordering::Less,
190        }
191    }
192    match (lhs_index < lhs.len(), rhs_index < rhs.len()) {
193        (false, false) => Ordering::Equal,
194        (true, false) => Ordering::Greater,
195        (false, true) => Ordering::Less,
196        (true, true) => unreachable!(),
197    }
198}

221#[path_prefix(conjure_cp::ast)]
222pub enum AbstractLiteral<T: AbstractLiteralValue> {
223    Set(Vec<T>),
224
225    MSet(Vec<T>),
226
227    /// A 1 dimensional matrix slice with an index domain.
228    Matrix(Vec<T>, T::Dom),
229
230    // a tuple of literals
231    Tuple(Vec<T>),
232
233    Record(Vec<Field<T>>),
234
235    Sequence(Vec<T>),
236
237    Function(Vec<(T, T)>),
238
239    // Variants only contain one of their name-domain pairs
240    Variant(Moo<Field<T>>),
241
242    // A list of partitions, each part has a set of values
243    Partition(Vec<Vec<T>>),
244    Relation(Vec<Vec<T>>),
245
246    /// Cycle notation for a permutation: each inner vec is one cycle. Unlike `Partition`, this is
247    /// *sparse* -- any element of the permutation's domain not mentioned in any cycle is an
248    /// implicit fixed point (maps to itself), rather than every element needing to be covered.
249    Permutation(Vec<Vec<T>>),
250}

252// TODO: use HasDomain instead once Expression::domain_of returns Domain not Option<Domain>
253fn union_item_domains(item_domains: Vec<DomainPtr>, literal_kind: &str) -> Option<DomainPtr> {
254    let mut item_domain_iter = item_domains.into_iter();
255    let first_item = item_domain_iter.next()?;
256
257    Some(
258        item_domain_iter
259            .try_fold(first_item, |x, y| x.union(&y))
260            .unwrap_or_else(|error| {
261                bug!(
262                    "taking the union of all item domains of a {literal_kind} literal should succeed: {error}"
263                )
264            }),
265    )
266}

268impl AbstractLiteral<Expression> {
269    pub fn domain_of(&self) -> Option<DomainPtr> {
270        match self {
271            AbstractLiteral::Set(items) => {
272                // ensure that all items have a domain, or return None
273                let item_domains: Vec<DomainPtr> = items
274                    .iter()
275                    .map(|x| x.domain_of())
276                    .collect::<Option<Vec<DomainPtr>>>()?;
277
278                // union all item domains together
279                let item_domain = union_item_domains(item_domains, "set")?;
280
281                Some(Domain::set(SetAttr::<Int>::default(), item_domain))
282            }
283
284            AbstractLiteral::MSet(items) => {
285                let cardinality = i32::try_from(items.len()).ok()?;
286                // ensure that all items have a domain, or return None
287                let item_domains: Vec<DomainPtr> = items
288                    .iter()
289                    .map(|x| x.domain_of())
290                    .collect::<Option<Vec<DomainPtr>>>()?;
291
292                // union all item domains together
293                let item_domain = union_item_domains(item_domains, "mset")?;
294
295                let occurrence = if cardinality == 0 {
296                    Range::Single(0)
297                } else {
298                    Range::Bounded(1, cardinality)
299                };
300                Some(Domain::mset(
301                    MSetAttr::new(Range::Single(cardinality), occurrence),
302                    item_domain,
303                ))
304            }
305
306            AbstractLiteral::Sequence(elems) => {
307                let item_domains: Vec<DomainPtr> = elems
308                    .iter()
309                    .map(|x| x.domain_of())
310                    .collect::<Option<Vec<DomainPtr>>>()?;
311
312                // Get the union of all domains in the sequence.
313                // i.e. if <(1..3), (1..3), (5), (8..9)> then seq dom is (1..3, 5, 8..9)
314                let item_domain = union_item_domains(item_domains, "sequence")?;
315
316                Some(Domain::sequence(
317                    SequenceAttr::<Int>::default(),
318                    item_domain,
319                ))
320            }
321
322            AbstractLiteral::Partition(items) => {
323                // Flatten the Vec<Vec< into a single vec
324                // ensure that all elemes in each part have a domain, or return None
325
326                let item_domains: Vec<DomainPtr> = items
327                    .iter()
328                    .flatten()
329                    .map(|x| x.domain_of())
330                    .collect::<Option<Vec<DomainPtr>>>()?;
331
332                // union all item domains together
333                let item_domain = union_item_domains(item_domains, "partition")?;
334
335                Some(Domain::partition(
336                    PartitionAttr::<Int>::default(),
337                    item_domain,
338                ))
339            }
340
341            AbstractLiteral::Permutation(cycles) => {
342                // Flatten the Vec<Vec< into a single vec; unlike partition, elements not
343                // mentioned in any cycle are implicit fixed points, so an empty literal (or one
344                // with no domain-bearing elements) has no way to infer an inner domain.
345                let item_domains: Vec<DomainPtr> = cycles
346                    .iter()
347                    .flatten()
348                    .map(|x| x.domain_of())
349                    .collect::<Option<Vec<DomainPtr>>>()?;
350
351                let item_domain = union_item_domains(item_domains, "permutation")?;
352
353                Some(Domain::permutation(
354                    PermutationAttr::<Int>::default(),
355                    item_domain,
356                ))
357            }
358
359            AbstractLiteral::Matrix(items, _) => {
360                // ensure that all items have a domain, or return None
361                let item_domains = items
362                    .iter()
363                    .map(|x| x.domain_of())
364                    .collect::<Option<Vec<DomainPtr>>>()?;
365
366                // union all item domains together
367                let item_domain = union_item_domains(item_domains, "matrix")?;
368
369                let mut new_index_domain = vec![];
370
371                // flatten index domains of n-d matrix into list
372                let mut e = Expression::AbstractLiteral(Metadata::new(), self.clone());
373                while let Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elems, idx)) = e {
374                    bug_assert!(
375                        idx.as_matrix().is_none(),
376                        "n-dimensional matrix literals should be represented as a matrix inside a matrix, got {idx}"
377                    );
378                    new_index_domain.push(idx);
379                    e = elems[0].clone();
380                }
381                Some(Domain::matrix(item_domain, new_index_domain))
382            }
383            AbstractLiteral::Tuple(_) => None,
384            AbstractLiteral::Record(_) => None,
385            AbstractLiteral::Function(_) => None,
386            AbstractLiteral::Variant(_) => None,
387            AbstractLiteral::Relation(_) => None,
388        }
389    }
390}

399impl Typeable for AbstractLiteral<Expression> {
400    fn return_type(&self) -> ReturnType {
401        match self {
402            AbstractLiteral::Set(items) if items.is_empty() => {
403                ReturnType::Set(Box::new(ReturnType::Unknown))
404            }
405            AbstractLiteral::Set(items) => {
406                let item_type = items[0].return_type();
407
408                // if any items do not have a type, return none.
409                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
410
411                bug_assert!(
412                    item_types.iter().all(|x| x == &item_type),
413                    "all items in a set should have the same type"
414                );
415
416                ReturnType::Set(Box::new(item_type))
417            }
418            AbstractLiteral::MSet(items) if items.is_empty() => {
419                ReturnType::MSet(Box::new(ReturnType::Unknown))
420            }
421            AbstractLiteral::MSet(items) => {
422                let item_type = items[0].return_type();
423
424                // if any items do not have a type, return none.
425                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
426
427                bug_assert!(
428                    item_types.iter().all(|x| x == &item_type),
429                    "all items in a set should have the same type"
430                );
431
432                ReturnType::MSet(Box::new(item_type))
433            }
434            AbstractLiteral::Sequence(items) if items.is_empty() => {
435                ReturnType::Sequence(Box::new(ReturnType::Unknown))
436            }
437            AbstractLiteral::Sequence(items) => {
438                let item_type = items[0].return_type();
439
440                // if any items do not have a type, return none.
441                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
442
443                bug_assert!(
444                    item_types.iter().all(|x| x == &item_type),
445                    "all items in a sequence should have the same type"
446                );
447
448                ReturnType::Sequence(Box::new(item_type))
449            }
450            AbstractLiteral::Partition(items) if items.is_empty() || items[0].is_empty() => {
451                ReturnType::Partition(Box::new(ReturnType::Unknown))
452            }
453            AbstractLiteral::Partition(items) => {
454                let item_type = items[0][0].return_type();
455
456                // if any items do not have a type, return none.
457                let item_types: Vec<ReturnType> =
458                    items.iter().flatten().map(|x| x.return_type()).collect();
459
460                bug_assert!(
461                    item_types.iter().all(|x| x == &item_type),
462                    "all items in every part of a partition should have the same type"
463                );
464
465                ReturnType::Partition(Box::new(item_type))
466            }
467            AbstractLiteral::Permutation(items) if items.is_empty() || items[0].is_empty() => {
468                ReturnType::Permutation(Box::new(ReturnType::Unknown))
469            }
470            AbstractLiteral::Permutation(items) => {
471                let item_type = items[0][0].return_type();
472
473                let item_types: Vec<ReturnType> =
474                    items.iter().flatten().map(|x| x.return_type()).collect();
475
476                bug_assert!(
477                    item_types.iter().all(|x| x == &item_type),
478                    "all items in every cycle of a permutation should have the same type"
479                );
480
481                ReturnType::Permutation(Box::new(item_type))
482            }
483            AbstractLiteral::Matrix(items, _) if items.is_empty() => {
484                ReturnType::Matrix(Box::new(ReturnType::Unknown))
485            }
486            AbstractLiteral::Matrix(items, _) => {
487                let item_type = items[0].return_type();
488
489                // if any items do not have a type, return none.
490                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
491
492                bug_assert!(
493                    item_types.iter().all(|x| x == &item_type),
494                    "all items in a matrix should have the same type. items: {items} types: {types:#?}",
495                    items = pretty_vec(items),
496                    types = items
497                        .iter()
498                        .map(|x| x.return_type())
499                        .collect::<Vec<ReturnType>>()
500                );
501
502                ReturnType::Matrix(Box::new(item_type))
503            }
504            AbstractLiteral::Tuple(items) => {
505                let mut item_types = vec![];
506                for item in items {
507                    item_types.push(item.return_type());
508                }
509                ReturnType::Tuple(item_types)
510            }
511            AbstractLiteral::Record(items) => {
512                let mut item_types = vec![];
513                for item in items {
514                    item_types.push(item.clone().func_map(|x| x.return_type()));
515                }
516                ReturnType::Record(item_types)
517            }
518            AbstractLiteral::Function(items) => {
519                if items.is_empty() {
520                    return ReturnType::Function(
521                        Box::new(ReturnType::Unknown),
522                        Box::new(ReturnType::Unknown),
523                    );
524                }
525
526                // Check that all items have the same return type
527                let (x1, y1) = &items[0];
528                let (t1, t2) = (x1.return_type(), y1.return_type());
529                for (x, y) in items {
530                    let (tx, ty) = (x.return_type(), y.return_type());
531                    if tx != t1 {
532                        bug!("Expected {t1}, got {x}: {tx}");
533                    }
534                    if ty != t2 {
535                        bug!("Expected {t2}, got {y}: {ty}");
536                    }
537                }
538
539                ReturnType::Function(Box::new(t1), Box::new(t2))
540            }
541            AbstractLiteral::Variant(item) => {
542                // Variants hold multiple possible types. In the case of a literal we know which type it chose
543                ReturnType::Variant(vec![item.as_ref().clone().func_map(|x| x.return_type())])
544            }
545            AbstractLiteral::Relation(items) => {
546                if items.is_empty() {
547                    return ReturnType::Relation(vec![ReturnType::Unknown]);
548                }
549                let mut item_types = vec![];
550                let x1 = &items[0];
551                let size = x1.len();
552                for item in x1 {
553                    item_types.push(item.return_type());
554                }
555                for x in items {
556                    if x.len() != size {
557                        let strs = item_types.iter().map(|x| format!("{}", x)).join(",");
558                        bug!("Expected ({strs}) with length {size}, got size {}", x.len());
559                    }
560                    for i in 1..size {
561                        if let Some(new_type) = x.get(i)
562                            && let Some(old_type) = item_types.get(i)
563                            && new_type.return_type() != *old_type
564                        {
565                            bug!("Expected {old_type}, got {new_type}");
566                        }
567                    }
568                }
569                ReturnType::Relation(item_types)
570            }
571        }
572    }
573}

575impl<T> AbstractLiteral<T>
576where
577    T: AbstractLiteralValue,
578{
579    /// Creates a matrix with elements `elems`, with domain `int(1..)`.
580    ///
581    /// This acts as a variable sized list.
582    pub fn matrix_implied_indices(elems: Vec<T>) -> Self {
583        AbstractLiteral::Matrix(elems, GroundDomain::Int(vec![Range::UnboundedR(1)]).into())
584    }
585
586    /// If the AbstractLiteral is a list, returns its elements.
587    ///
588    /// A list is any a matrix with the domain `int(1..)`. This includes matrix literals without
589    /// any explicitly specified domain.
590    pub fn unwrap_list(&self) -> Option<&Vec<T>> {
591        let AbstractLiteral::Matrix(elems, domain) = self else {
592            return None;
593        };
594
595        let domain: DomainPtr = domain.clone().into();
596        let Some(GroundDomain::Int(ranges)) = domain.as_ground() else {
597            return None;
598        };
599
600        let [Range::UnboundedR(1)] = ranges[..] else {
601            return None;
602        };
603
604        Some(elems)
605    }
606}

608impl<T> Display for AbstractLiteral<T>
609where
610    T: AbstractLiteralValue,
611{
612    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
613        match self {
614            AbstractLiteral::Set(elems) => {
615                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
616                write!(f, "{{{elems_str}}}")
617            }
618            AbstractLiteral::MSet(elems) => {
619                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
620                write!(f, "mset({elems_str})")
621            }
622            AbstractLiteral::Matrix(elems, index_domain) => {
623                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
624                write!(f, "[{elems_str};{index_domain}]")
625            }
626            AbstractLiteral::Tuple(elems) => {
627                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
628                write!(f, "({elems_str})")
629            }
630            AbstractLiteral::Sequence(elems) => {
631                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
632                write!(f, "sequence({elems_str})")
633            }
634            AbstractLiteral::Partition(parts) => {
635                let elems_str: String = parts
636                    .iter()
637                    .map(|inner| {
638                        let elems_str = inner.iter().map(|x| format!("{x}")).join(",");
639                        format!("{{{}}}", elems_str)
640                    })
641                    .join(", ");
642
643                write!(f, "partition({elems_str})")
644            }
645            AbstractLiteral::Permutation(cycles) => {
646                let cycles_str: String = cycles
647                    .iter()
648                    .map(|cycle| {
649                        let elems_str = cycle.iter().map(|x| format!("{x}")).join(",");
650                        format!("({elems_str})")
651                    })
652                    .join("");
653
654                write!(f, "permutation{cycles_str}")
655            }
656            AbstractLiteral::Record(entries) => {
657                let entries_str: String = entries
658                    .iter()
659                    .map(|entry| format!("{} = {}", entry.name, entry.value))
660                    .join(",");
661                write!(f, "record {{{entries_str}}}")
662            }
663            AbstractLiteral::Function(entries) => {
664                let entries_str: String = entries
665                    .iter()
666                    .map(|entry| format!("{} --> {}", entry.0, entry.1))
667                    .join(",");
668                write!(f, "function({entries_str})")
669            }
670            AbstractLiteral::Variant(entry) => {
671                write!(f, "variant{{{} = {}}}", entry.name, entry.value)
672            }
673            AbstractLiteral::Relation(elems) => {
674                let elems_str: String = elems
675                    .iter()
676                    .map(|x| format!("({})", x.iter().map(|x| format!("{x}")).join(",")))
677                    .join(",");
678                write!(f, "relation({elems_str})")
679            }
680        }
681    }
682}

913    fn children_bi_count(&self) -> usize {
914        // Manual `biplate` (not derive): delegate counts to the plated containers so wide
915        // matrices/lists stay O(1) instead of materialising `children_bi`.
916        if std::any::TypeId::of::<To>() == std::any::TypeId::of::<AbstractLiteral<U>>() {
917            return 1;
918        }
919        match self {
920            AbstractLiteral::Set(v)
921            | AbstractLiteral::MSet(v)
922            | AbstractLiteral::Sequence(v)
923            | AbstractLiteral::Tuple(v)
924            | AbstractLiteral::Matrix(v, _) => <Vec<U> as Biplate<To>>::children_bi_count(v),
925            AbstractLiteral::Record(entries) => {
926                <Vec<Field<U>> as Biplate<To>>::children_bi_count(entries)
927            }
928            AbstractLiteral::Variant(entry) => {
929                <Moo<Field<U>> as Biplate<To>>::children_bi_count(entry)
930            }
931            AbstractLiteral::Relation(elems)
932            | AbstractLiteral::Partition(elems)
933            | AbstractLiteral::Permutation(elems) => {
934                <Vec<Vec<U>> as Biplate<To>>::children_bi_count(elems)
935            }
936            AbstractLiteral::Function(_) => <Self as Biplate<To>>::children_bi(self).len(),
937        }
938    }
939
940    fn try_replace_child_at_bi(&mut self, index: usize, child: To) -> bool {
941        // Same as `children_bi_count`: keep in-place updates for owned vectors (lee-distance).
942        if std::any::TypeId::of::<To>() == std::any::TypeId::of::<AbstractLiteral<U>>() {
943            if index != 0 {
944                return false;
945            }
946            // SAFETY: TypeId equality means To and AbstractLiteral<U> are the same type.
947            unsafe {
948                let child_as_self = std::mem::transmute_copy::<To, AbstractLiteral<U>>(&child);
949                std::mem::forget(child);
950                *self = child_as_self;
951            }
952            return true;
953        }
954        match self {
955            AbstractLiteral::Set(v)
956            | AbstractLiteral::MSet(v)
957            | AbstractLiteral::Sequence(v)
958            | AbstractLiteral::Tuple(v)
959            | AbstractLiteral::Matrix(v, _) => {
960                <Vec<U> as Biplate<To>>::try_replace_child_at_bi(v, index, child)
961            }
962            AbstractLiteral::Record(entries) => {
963                <Vec<Field<U>> as Biplate<To>>::try_replace_child_at_bi(entries, index, child)
964            }
965            AbstractLiteral::Variant(entry) => {
966                <Moo<Field<U>> as Biplate<To>>::try_replace_child_at_bi(entry, index, child)
967            }
968            AbstractLiteral::Relation(elems)
969            | AbstractLiteral::Partition(elems)
970            | AbstractLiteral::Permutation(elems) => {
971                <Vec<Vec<U>> as Biplate<To>>::try_replace_child_at_bi(elems, index, child)
972            }
973            AbstractLiteral::Function(_) => {
974                let mut children = <Self as Biplate<To>>::children_bi(self);
975                if index >= children.len() {
976                    return false;
977                }
978                children[index] = child;
979                *self = self.with_children_bi(children);
980                true
981            }
982        }
983    }

1079impl AbstractLiteral<Expression> {
1080    /// If all the elements are literals, returns this as an AbstractLiteral<Literal>.
1081    /// Otherwise, returns `None`.
1082    pub fn into_literals(self) -> Option<AbstractLiteral<Literal>> {
1083        match self {
1084            AbstractLiteral::Set(elements) => {
1085                let literals = elements
1086                    .into_iter()
1087                    .map(|expr| match expr {
1088                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1089                        Expression::AbstractLiteral(_, abslit) => {
1090                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1091                        }
1092                        _ => None,
1093                    })
1094                    .collect::<Option<Vec<_>>>()?;
1095                Some(AbstractLiteral::Set(literals))
1096            }
1097            AbstractLiteral::MSet(elements) => {
1098                let literals = elements
1099                    .into_iter()
1100                    .map(|expr| match expr {
1101                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1102                        Expression::AbstractLiteral(_, abslit) => {
1103                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1104                        }
1105                        _ => None,
1106                    })
1107                    .collect::<Option<Vec<_>>>()?;
1108                Some(AbstractLiteral::MSet(literals))
1109            }
1110            AbstractLiteral::Partition(elems) => {
1111                // want to ascertain if every elem in Vec<Vec<Expr>> is a literal. If any are not, return none
1112                // otherwise confirm it is an abslit<lit>
1113                let mut partition: Vec<Vec<_>> = Vec::new();
1114
1115                for part in elems {
1116                    let literals = part
1117                        .into_iter()
1118                        .map(|expr| match expr {
1119                            Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1120                            Expression::AbstractLiteral(_, abslit) => {
1121                                Some(Literal::AbstractLiteral(abslit.into_literals()?))
1122                            }
1123                            _ => None,
1124                        })
1125                        .collect::<Option<Vec<_>>>()?;
1126
1127                    partition.push(literals);
1128                }
1129
1130                Some(AbstractLiteral::Partition(partition))
1131            }
1132            AbstractLiteral::Permutation(elems) => {
1133                let mut permutation: Vec<Vec<_>> = Vec::new();
1134
1135                for cycle in elems {
1136                    let literals = cycle
1137                        .into_iter()
1138                        .map(|expr| match expr {
1139                            Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1140                            Expression::AbstractLiteral(_, abslit) => {
1141                                Some(Literal::AbstractLiteral(abslit.into_literals()?))
1142                            }
1143                            _ => None,
1144                        })
1145                        .collect::<Option<Vec<_>>>()?;
1146
1147                    permutation.push(literals);
1148                }
1149
1150                Some(AbstractLiteral::Permutation(permutation))
1151            }
1152            AbstractLiteral::Matrix(items, domain) => {
1153                let mut literals = vec![];
1154                for item in items {
1155                    let literal = match item {
1156                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1157                        Expression::AbstractLiteral(_, abslit) => {
1158                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1159                        }
1160                        _ => None,
1161                    }?;
1162                    literals.push(literal);
1163                }
1164
1165                Some(AbstractLiteral::Matrix(literals, domain.resolve().ok()?))
1166            }
1167            AbstractLiteral::Sequence(elements) => {
1168                let literals = elements
1169                    .into_iter()
1170                    .map(|expr| match expr {
1171                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1172                        Expression::AbstractLiteral(_, abslit) => {
1173                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1174                        }
1175                        _ => None,
1176                    })
1177                    .collect::<Option<Vec<_>>>()?;
1178                Some(AbstractLiteral::Sequence(literals))
1179            }
1180            AbstractLiteral::Tuple(items) => {
1181                let mut literals = vec![];
1182                for item in items {
1183                    let literal = match item {
1184                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1185                        Expression::AbstractLiteral(_, abslit) => {
1186                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1187                        }
1188                        _ => None,
1189                    }?;
1190                    literals.push(literal);
1191                }
1192
1193                Some(AbstractLiteral::Tuple(literals))
1194            }
1195            AbstractLiteral::Record(entries) => {
1196                let mut literals = vec![];
1197                for entry in entries {
1198                    let literal = match entry.value {
1199                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1200                        Expression::AbstractLiteral(_, abslit) => {
1201                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
1202                        }
1203                        _ => None,
1204                    }?;
1205
1206                    literals.push((entry.name, literal));
1207                }
1208                Some(AbstractLiteral::Record(
1209                    literals
1210                        .into_iter()
1211                        .map(|(name, literal)| Field {
1212                            name,
1213                            value: literal,
1214                        })
1215                        .collect(),
1216                ))
1217            }
1218            AbstractLiteral::Function(_) => todo!("Implement into_literals for functions"),
1219            AbstractLiteral::Variant(entry) => {
1220                let literal = match entry.value.clone() {
1221                    Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1222                    Expression::AbstractLiteral(_, abslit) => {
1223                        Some(Literal::AbstractLiteral(abslit.into_literals()?))
1224                    }
1225                    _ => None,
1226                }?;
1227                Some(AbstractLiteral::Variant(Moo::new(Field {
1228                    name: entry.name.clone(),
1229                    value: literal,
1230                })))
1231            }
1232            AbstractLiteral::Relation(tuples) => {
1233                let mut literal_tuples = Vec::with_capacity(tuples.len());
1234                for fields in tuples {
1235                    let literals = fields
1236                        .into_iter()
1237                        .map(|expr| match expr {
1238                            Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
1239                            Expression::AbstractLiteral(_, abslit) => {
1240                                Some(Literal::AbstractLiteral(abslit.into_literals()?))
1241                            }
1242                            _ => None,
1243                        })
1244                        .collect::<Option<Vec<_>>>()?;
1245                    literal_tuples.push(literals);
1246                }
1247                Some(AbstractLiteral::Relation(literal_tuples))
1248            }
1249        }
1250    }
