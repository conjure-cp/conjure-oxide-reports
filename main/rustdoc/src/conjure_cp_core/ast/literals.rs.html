26/// A literal value, equivalent to constants in Conjure.
27pub enum Literal {
28    Int(i32),
29    Bool(bool),
30    //abstract literal variant ends in Literal, but that's ok
31    #[allow(clippy::enum_variant_names)]
32    AbstractLiteral(AbstractLiteral<Literal>),
33}

35impl HasDomain for Literal {
36    fn domain_of(&self) -> DomainPtr {
37        match self {
38            Literal::Int(i) => Domain::int(vec![Range::Single(*i)]),
39            Literal::Bool(_) => Domain::bool(),
40            Literal::AbstractLiteral(abstract_literal) => abstract_literal.domain_of(),
41        }
42    }
43}

45// make possible values of an AbstractLiteral a closed world to make the trait bounds more sane (particularly in Uniplate instances!!)
46pub trait AbstractLiteralValue:
47    Clone + Eq + PartialEq + Display + Uniplate + Biplate<RecordValue<Self>> + 'static
48{
49    type Dom: Clone + Eq + PartialEq + Display + Quine + From<GroundDomain> + Into<DomainPtr>;
50}
51impl AbstractLiteralValue for Expression {
52    type Dom = DomainPtr;
53}
54impl AbstractLiteralValue for Literal {
55    type Dom = Moo<GroundDomain>;

81impl AbstractLiteral<Expression> {
82    pub fn domain_of(&self) -> Option<DomainPtr> {
83        match self {
84            AbstractLiteral::Set(items) => {
85                // ensure that all items have a domain, or return None
86                let item_domains: Vec<DomainPtr> = items
87                    .iter()
88                    .map(|x| x.domain_of())
89                    .collect::<Option<Vec<DomainPtr>>>()?;
90
91                // union all item domains together
92                let mut item_domain_iter = item_domains.iter().cloned();
93                let first_item = item_domain_iter.next()?;
94                let item_domain = item_domains
95                    .iter()
96                    .try_fold(first_item, |x, y| x.union(y))
97                    .expect("taking the union of all item domains of a set literal should succeed");
98
99                Some(Domain::set(SetAttr::<Int>::default(), item_domain))
100            }
101
102            AbstractLiteral::MSet(items) => {
103                // ensure that all items have a domain, or return None
104                let item_domains: Vec<DomainPtr> = items
105                    .iter()
106                    .map(|x| x.domain_of())
107                    .collect::<Option<Vec<DomainPtr>>>()?;
108
109                // union all item domains together
110                let mut item_domain_iter = item_domains.iter().cloned();
111                let first_item = item_domain_iter.next()?;
112                let item_domain = item_domains
113                    .iter()
114                    .try_fold(first_item, |x, y| x.union(y))
115                    .expect("taking the union of all item domains of a set literal should succeed");
116
117                Some(Domain::mset(MSetAttr::<Int>::default(), item_domain))
118            }
119
120            AbstractLiteral::Sequence(elems) => {
121                let item_domains: Vec<DomainPtr> = elems
122                    .iter()
123                    .map(|x| x.domain_of())
124                    .collect::<Option<Vec<DomainPtr>>>()?;
125
126                // Get the union of all domains in the sequence.
127                // i.e. if <(1..3), (1..3), (5), (8..9)> then seq dom is (1..3, 5, 8..9)
128                let mut item_domain_iter = item_domains.iter().cloned();
129                let first_item = item_domain_iter.next()?;
130                let item_domain = item_domains
131                    .iter()
132                    .try_fold(first_item, |x, y| x.union(y))
133                    .expect("taking the union of all item domains of a set literal should succeed");
134
135                Some(Domain::sequence(
136                    SequenceAttr::<Int>::default(),
137                    item_domain,
138                ))
139            }
140
141            AbstractLiteral::Matrix(items, _) => {
142                // ensure that all items have a domain, or return None
143                let item_domains = items
144                    .iter()
145                    .map(|x| x.domain_of())
146                    .collect::<Option<Vec<DomainPtr>>>()?;
147
148                // union all item domains together
149                let mut item_domain_iter = item_domains.iter().cloned();
150
151                let first_item = item_domain_iter.next()?;
152
153                let item_domain = item_domains
154                    .iter()
155                    .try_fold(first_item, |x, y| x.union(y))
156                    .expect(
157                        "taking the union of all item domains of a matrix literal should succeed",
158                    );
159
160                let mut new_index_domain = vec![];
161
162                // flatten index domains of n-d matrix into list
163                let mut e = Expression::AbstractLiteral(Metadata::new(), self.clone());
164                while let Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elems, idx)) = e {
165                    assert!(
166                        idx.as_matrix().is_none(),
167                        "n-dimensional matrix literals should be represented as a matrix inside a matrix, got {idx}"
168                    );
169                    new_index_domain.push(idx);
170                    e = elems[0].clone();
171                }
172                Some(Domain::matrix(item_domain, new_index_domain))
173            }
174            AbstractLiteral::Tuple(_) => None,
175            AbstractLiteral::Record(_) => None,
176            AbstractLiteral::Function(_) => None,
177            AbstractLiteral::Relation(_) => None,
178        }
179    }
180}

189impl Typeable for AbstractLiteral<Expression> {
190    fn return_type(&self) -> ReturnType {
191        match self {
192            AbstractLiteral::Set(items) if items.is_empty() => {
193                ReturnType::Set(Box::new(ReturnType::Unknown))
194            }
195            AbstractLiteral::Set(items) => {
196                let item_type = items[0].return_type();
197
198                // if any items do not have a type, return none.
199                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
200
201                assert!(
202                    item_types.iter().all(|x| x == &item_type),
203                    "all items in a set should have the same type"
204                );
205
206                ReturnType::Set(Box::new(item_type))
207            }
208            AbstractLiteral::MSet(items) if items.is_empty() => {
209                ReturnType::MSet(Box::new(ReturnType::Unknown))
210            }
211            AbstractLiteral::MSet(items) => {
212                let item_type = items[0].return_type();
213
214                // if any items do not have a type, return none.
215                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
216
217                assert!(
218                    item_types.iter().all(|x| x == &item_type),
219                    "all items in a set should have the same type"
220                );
221
222                ReturnType::MSet(Box::new(item_type))
223            }
224            AbstractLiteral::Sequence(items) if items.is_empty() => {
225                ReturnType::Sequence(Box::new(ReturnType::Unknown))
226            }
227            AbstractLiteral::Sequence(items) => {
228                let item_type = items[0].return_type();
229
230                // if any items do not have a type, return none.
231                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
232
233                assert!(
234                    item_types.iter().all(|x| x == &item_type),
235                    "all items in a sequence should have the same type"
236                );
237
238                ReturnType::Sequence(Box::new(item_type))
239            }
240            AbstractLiteral::Matrix(items, _) if items.is_empty() => {
241                ReturnType::Matrix(Box::new(ReturnType::Unknown))
242            }
243            AbstractLiteral::Matrix(items, _) => {
244                let item_type = items[0].return_type();
245
246                // if any items do not have a type, return none.
247                let item_types: Vec<ReturnType> = items.iter().map(|x| x.return_type()).collect();
248
249                assert!(
250                    item_types.iter().all(|x| x == &item_type),
251                    "all items in a matrix should have the same type. items: {items} types: {types:#?}",
252                    items = pretty_vec(items),
253                    types = items
254                        .iter()
255                        .map(|x| x.return_type())
256                        .collect::<Vec<ReturnType>>()
257                );
258
259                ReturnType::Matrix(Box::new(item_type))
260            }
261            AbstractLiteral::Tuple(items) => {
262                let mut item_types = vec![];
263                for item in items {
264                    item_types.push(item.return_type());
265                }
266                ReturnType::Tuple(item_types)
267            }
268            AbstractLiteral::Record(items) => {
269                let mut item_types = vec![];
270                for item in items {
271                    item_types.push(item.value.return_type());
272                }
273                ReturnType::Record(item_types)
274            }
275            AbstractLiteral::Function(items) => {
276                if items.is_empty() {
277                    return ReturnType::Function(
278                        Box::new(ReturnType::Unknown),
279                        Box::new(ReturnType::Unknown),
280                    );
281                }
282
283                // Check that all items have the same return type
284                let (x1, y1) = &items[0];
285                let (t1, t2) = (x1.return_type(), y1.return_type());
286                for (x, y) in items {
287                    let (tx, ty) = (x.return_type(), y.return_type());
288                    if tx != t1 {
289                        bug!("Expected {t1}, got {x}: {tx}");
290                    }
291                    if ty != t2 {
292                        bug!("Expected {t2}, got {y}: {ty}");
293                    }
294                }
295
296                ReturnType::Function(Box::new(t1), Box::new(t2))
297            }
298            AbstractLiteral::Relation(items) => {
299                if items.is_empty() {
300                    return ReturnType::Relation(vec![ReturnType::Unknown]);
301                }
302                let mut item_types = vec![];
303                let x1 = &items[0];
304                let size = x1.len();
305                for item in x1 {
306                    item_types.push(item.return_type());
307                }
308                for x in items {
309                    if x.len() != size {
310                        let strs = item_types.iter().map(|x| format!("{}", x)).join(",");
311                        bug!("Expected ({strs}) with length {size}, got size {}", x.len());
312                    }
313                    for i in 1..size {
314                        if let Some(new_type) = x.get(i)
315                            && let Some(old_type) = item_types.get(i)
316                            && new_type.return_type() != *old_type
317                        {
318                            bug!("Expected {old_type}, got {new_type}");
319                        }
320                    }
321                }
322                ReturnType::Relation(item_types)
323            }
324        }
325    }
326}

328impl<T> AbstractLiteral<T>
329where
330    T: AbstractLiteralValue,
331{
332    /// Creates a matrix with elements `elems`, with domain `int(1..)`.
333    ///
334    /// This acts as a variable sized list.
335    pub fn matrix_implied_indices(elems: Vec<T>) -> Self {
336        AbstractLiteral::Matrix(elems, GroundDomain::Int(vec![Range::UnboundedR(1)]).into())
337    }
338
339    /// If the AbstractLiteral is a list, returns its elements.
340    ///
341    /// A list is any a matrix with the domain `int(1..)`. This includes matrix literals without
342    /// any explicitly specified domain.
343    pub fn unwrap_list(&self) -> Option<&Vec<T>> {
344        let AbstractLiteral::Matrix(elems, domain) = self else {
345            return None;
346        };
347
348        let domain: DomainPtr = domain.clone().into();
349        let Some(GroundDomain::Int(ranges)) = domain.as_ground() else {
350            return None;
351        };
352
353        let [Range::UnboundedR(1)] = ranges[..] else {
354            return None;
355        };
356
357        Some(elems)
358    }
359}

361impl<T> Display for AbstractLiteral<T>
362where
363    T: AbstractLiteralValue,
364{
365    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
366        match self {
367            AbstractLiteral::Set(elems) => {
368                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
369                write!(f, "{{{elems_str}}}")
370            }
371            AbstractLiteral::MSet(elems) => {
372                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
373                write!(f, "mset({elems_str})")
374            }
375            AbstractLiteral::Matrix(elems, index_domain) => {
376                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
377                write!(f, "[{elems_str};{index_domain}]")
378            }
379            AbstractLiteral::Tuple(elems) => {
380                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
381                write!(f, "({elems_str})")
382            }
383            AbstractLiteral::Sequence(elems) => {
384                let elems_str: String = elems.iter().map(|x| format!("{x}")).join(",");
385                write!(f, "sequence({elems_str})")
386            }
387            AbstractLiteral::Record(entries) => {
388                let entries_str: String = entries
389                    .iter()
390                    .map(|entry| format!("{} = {}", entry.name, entry.value))
391                    .join(",");
392                write!(f, "record {{{entries_str}}}")
393            }
394            AbstractLiteral::Function(entries) => {
395                let entries_str: String = entries
396                    .iter()
397                    .map(|entry| format!("{} --> {}", entry.0, entry.1))
398                    .join(",");
399                write!(f, "function({entries_str})")
400            }
401            AbstractLiteral::Relation(elems) => {
402                let elems_str: String = elems
403                    .iter()
404                    .map(|x| format!("({})", x.iter().map(|x| format!("{x}")).join(",")))
405                    .join(",");
406                write!(f, "relation({elems_str})")
407            }
408        }
409    }

412impl<T> Uniplate for AbstractLiteral<T>
413where
414    T: AbstractLiteralValue + Biplate<AbstractLiteral<T>>,
415{
416    fn uniplate(&self) -> (Tree<Self>, Box<dyn Fn(Tree<Self>) -> Self>) {
417        // walking into T
418        match self {
419            AbstractLiteral::Set(vec) => {
420                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(vec);
421                (f1_tree, Box::new(move |x| AbstractLiteral::Set(f1_ctx(x))))
422            }
423            AbstractLiteral::MSet(vec) => {
424                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(vec);
425                (f1_tree, Box::new(move |x| AbstractLiteral::MSet(f1_ctx(x))))
426            }
427            AbstractLiteral::Matrix(elems, index_domain) => {
428                let index_domain = index_domain.clone();
429                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(elems);
430                (
431                    f1_tree,
432                    Box::new(move |x| AbstractLiteral::Matrix(f1_ctx(x), index_domain.clone())),
433                )
434            }
435            AbstractLiteral::Sequence(vec) => {
436                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(vec);
437                (
438                    f1_tree,
439                    Box::new(move |x| AbstractLiteral::Sequence(f1_ctx(x))),
440                )
441            }
442            AbstractLiteral::Tuple(elems) => {
443                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(elems);
444                (
445                    f1_tree,
446                    Box::new(move |x| AbstractLiteral::Tuple(f1_ctx(x))),
447                )
448            }
449            AbstractLiteral::Record(entries) => {
450                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(entries);
451                (
452                    f1_tree,
453                    Box::new(move |x| AbstractLiteral::Record(f1_ctx(x))),
454                )
455            }
456            AbstractLiteral::Function(entries) => {
457                let entry_count = entries.len();
458                let flattened: Vec<T> = entries
459                    .iter()
460                    .flat_map(|(lhs, rhs)| [lhs.clone(), rhs.clone()])
461                    .collect();
462
463                let (f1_tree, f1_ctx) =
464                    <Vec<T> as Biplate<AbstractLiteral<T>>>::biplate(&flattened);
465                (
466                    f1_tree,
467                    Box::new(move |x| {
468                        let rebuilt = f1_ctx(x);
469                        assert_eq!(
470                            rebuilt.len(),
471                            entry_count * 2,
472                            "number of function literal children should remain unchanged"
473                        );
474
475                        let mut iter = rebuilt.into_iter();
476                        let mut pairs = Vec::with_capacity(entry_count);
477                        while let (Some(lhs), Some(rhs)) = (iter.next(), iter.next()) {
478                            pairs.push((lhs, rhs));
479                        }
480
481                        AbstractLiteral::Function(pairs)
482                    }),
483                )
484            }
485            AbstractLiteral::Relation(elems) => {
486                let (f1_tree, f1_ctx) = <_ as Biplate<AbstractLiteral<T>>>::biplate(elems);
487                (
488                    f1_tree,
489                    Box::new(move |x| AbstractLiteral::Relation(f1_ctx(x))),
490                )
491            }
492        }
493    }
494}

660    fn try_from(value: &Literal) -> Result<Self, Self::Error> {
661        match value {
662            Literal::Bool(b) => Ok(*b),
663            _ => Err("Cannot convert non-bool literal to bool"),
664        }
665    }
666}

687impl AbstractLiteral<Expression> {
688    /// If all the elements are literals, returns this as an AbstractLiteral<Literal>.
689    /// Otherwise, returns `None`.
690    pub fn into_literals(self) -> Option<AbstractLiteral<Literal>> {
691        match self {
692            AbstractLiteral::Set(elements) => {
693                let literals = elements
694                    .into_iter()
695                    .map(|expr| match expr {
696                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
697                        Expression::AbstractLiteral(_, abslit) => {
698                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
699                        }
700                        _ => None,
701                    })
702                    .collect::<Option<Vec<_>>>()?;
703                Some(AbstractLiteral::Set(literals))
704            }
705            AbstractLiteral::MSet(elements) => {
706                let literals = elements
707                    .into_iter()
708                    .map(|expr| match expr {
709                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
710                        Expression::AbstractLiteral(_, abslit) => {
711                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
712                        }
713                        _ => None,
714                    })
715                    .collect::<Option<Vec<_>>>()?;
716                Some(AbstractLiteral::MSet(literals))
717            }
718            AbstractLiteral::Matrix(items, domain) => {
719                let mut literals = vec![];
720                for item in items {
721                    let literal = match item {
722                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
723                        Expression::AbstractLiteral(_, abslit) => {
724                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
725                        }
726                        _ => None,
727                    }?;
728                    literals.push(literal);
729                }
730
731                Some(AbstractLiteral::Matrix(literals, domain.resolve()?))
732            }
733            AbstractLiteral::Sequence(elements) => {
734                let literals = elements
735                    .into_iter()
736                    .map(|expr| match expr {
737                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
738                        Expression::AbstractLiteral(_, abslit) => {
739                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
740                        }
741                        _ => None,
742                    })
743                    .collect::<Option<Vec<_>>>()?;
744                Some(AbstractLiteral::Sequence(literals))
745            }
746            AbstractLiteral::Tuple(items) => {
747                let mut literals = vec![];
748                for item in items {
749                    let literal = match item {
750                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
751                        Expression::AbstractLiteral(_, abslit) => {
752                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
753                        }
754                        _ => None,
755                    }?;
756                    literals.push(literal);
757                }
758
759                Some(AbstractLiteral::Tuple(literals))
760            }
761            AbstractLiteral::Record(entries) => {
762                let mut literals = vec![];
763                for entry in entries {
764                    let literal = match entry.value {
765                        Expression::Atomic(_, Atom::Literal(lit)) => Some(lit),
766                        Expression::AbstractLiteral(_, abslit) => {
767                            Some(Literal::AbstractLiteral(abslit.into_literals()?))
768                        }
769                        _ => None,
770                    }?;
771
772                    literals.push((entry.name, literal));
773                }
774                Some(AbstractLiteral::Record(
775                    literals
776                        .into_iter()
777                        .map(|(name, literal)| RecordValue {
778                            name,
779                            value: literal,
780                        })
781                        .collect(),
782                ))
783            }
784            AbstractLiteral::Function(_) => todo!("Implement into_literals for functions"),
785            AbstractLiteral::Relation(_) => todo!("Implement into_literals for relations"),
786        }
787    }
788}
