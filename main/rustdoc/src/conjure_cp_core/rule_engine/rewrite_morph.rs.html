22/// Rewrites a `Model` by applying rule sets using an optimized, tree-morphing rewriter.
23///
24/// This function traverses the expression tree of the model and applies the given rules
25/// to transform it. It operates on the model's internal structure, replacing the root
26/// expression and updating the symbol table based on the transformations performed by the
27/// `morph` function.
28///
29/// # Parameters
30///
31/// - `model`: The `Model` to be rewritten. It is consumed and a new, transformed version is returned.
32/// - `rule_sets`: A vector of `RuleSet` references containing the rules for transformation. These rules are grouped by priority before being applied.
33/// - `prop_multiple_equally_applicable`: A boolean flag to control behavior when multiple rules of the same priority can be applied to the same expression.
34///   - If `true`, the rewriter will use a selection strategy (`select_panic`) that panics.
35///   - If `false`, the rewriter will use a selection strategy (`select_first`) that simply picks the first applicable rule it encounters.
36/// - `variant`: The `MorphVariant` selecting cache and traversal behaviour:
37///   - `NoCache` → no cache, standard traversal
38///   - `Cache` → `HashMapCache`, standard traversal
39///   - `Hashcache` → `CachedHashMapCache`, standard traversal
40///   - `Naive` → no cache, naive traversal
41///
42/// # Returns
43///
44/// The rewritten `Model` after all applicable rules have been applied.
45///
46/// # Panics
47///
48/// This function will panic under two conditions:
49/// - If the internal grouping of rules by priority fails (from `get_rules_grouped`).
50/// - If `prop_multiple_equally_applicable` is set to `true` and more than one rule of the same priority can be applied to the same expression.
51pub fn rewrite_morph<'a>(
52    mut model: Model,
53    rule_sets: &Vec<&'a RuleSet<'a>>,
54    prop_multiple_equally_applicable: bool,
55    config: MorphConfig,
56) -> Model {
57    set_current_rewriter(Rewriter::Morph(config));
58
59    if rule_trace_enabled() {
60        trace!(
61            target: "rule_engine_rule_trace",
62            "Model before rewriting:\n\n{}\n--\n",
63            model
64        );
65    }
66
67    let rules_grouped = get_rules_grouped(rule_sets)
68        .unwrap_or_else(|_| bug!("get_rule_priorities() failed!"))
69        .into_iter()
70        .collect_vec();
71
72    let mut engine = build_engine(&rules_grouped, prop_multiple_equally_applicable, config);
73    let model_ref = &mut model;
74
75    loop {
76        if try_rewrite_value_letting_once(
77            model_ref,
78            &rules_grouped,
79            prop_multiple_equally_applicable,
80        )
81        .is_some()
82        {
83            continue;
84        }
85
86        let initial_state = MorphState {
87            symbols: model_ref.symbols().clone(),
88            clauses: model_ref.clauses().clone(),
89        };
90        let (expr, morph_state) = if config.naive {
91            engine.morph_naive(model_ref.root().clone(), initial_state)
92        } else {
93            engine.morph(model_ref.root().clone(), initial_state)
94        };
95
96        *model_ref.symbols_mut() = morph_state.symbols;
97        model_ref.replace_clauses(morph_state.clauses);
98        model_ref.replace_root(expr);
99
100        if try_rewrite_value_letting_once(
101            model_ref,
102            &rules_grouped,
103            prop_multiple_equally_applicable,
104        )
105        .is_none()
106        {
107            break;
108        }
109    }
110
111    if rule_trace_enabled() {
112        trace!(
113            target: "rule_engine_rule_trace",
114            "Final model:\n\n{}",
115            model
116        );
117    }
118
119    model
120}

122fn build_engine<'a>(
123    rules_grouped: &Vec<(u16, Vec<RuleData<'a>>)>,
124    prop_multiple_equally_applicable: bool,
125    config: MorphConfig,
126) -> Engine<Expression, MorphState, RuleData<'a>, Box<dyn RewriteCache<Expression>>> {
127    let morph_rule_groups = rules_grouped
128        .iter()
129        .map(|(_, rules)| rules.clone())
130        .collect_vec();
131
132    let selector = if prop_multiple_equally_applicable {
133        select_panic
134    } else {
135        select_first
136    };
137
138    let cache: Box<dyn RewriteCache<Expression>> = match config.cache {
139        MorphCachingStrategy::NoCache => Box::new(NoCache),
140        MorphCachingStrategy::Cache => Box::new(HashMapCache::<_, StdHashKey>::new()),
141        MorphCachingStrategy::IncrementalCache => Box::new(CachedHashMapCache::new()),
142    };
143
144    EngineBuilder::new()
145        .set_selector(selector)
146        .append_rule_groups(morph_rule_groups)
147        .add_cacher(cache)
148        .set_discriminant_fn(if config.prefilter {
149            Some(discriminant_from_value)
150        } else {
151            None
152        })
153        .set_parallel(false)
154        .set_fixedpoint(config.fixedpoint)
155        // .add_down_predicate(|node| ! matches!(node, Expression::Comprehension(_, _)))
156        .build()
157}
