24/// Converts a conjure-oxide model to a `minion_sys` model.
25pub fn model_to_minion(model: ConjureModel) -> Result<MinionModel, SolverError> {
26    let mut minion_model = MinionModel::new();
27    load_symbol_table(&model, &mut minion_model)?;
28    load_constraints(&model, &mut minion_model)?;
29    Ok(minion_model)
30}

32/// Loads the symbol table into `minion_model`.
33fn load_symbol_table(
34    conjure_model: &ConjureModel,
35    minion_model: &mut MinionModel,
36) -> Result<(), SolverError> {
37    if let Some(ref vars) = conjure_model.search_order {
38        let search_vars: HashSet<_> = vars.iter().cloned().collect();
39
40        // add search vars in order first
41        for name in vars {
42            let decl = conjure_model
43                .as_submodel()
44                .symbols()
45                .lookup(name)
46                .ok_or_else(|| ModelInvalid(format!("search variable '{name}' does not exist")))?;
47            let var = decl.as_var().ok_or_else(|| {
48                ModelInvalid(format!(
49                    "search variable '{name}' is not a decision variable"
50                ))
51            })?;
52
53            load_var(name, &var, true, minion_model)?;
54        }
55
56        // then add the rest as non-search vars
57        for_each_unrepresented_var(conjure_model, |name, var| {
58            if search_vars.contains(name) {
59                return Ok(());
60            }
61            load_var(name, var, false, minion_model)
62        })?;
63    } else {
64        for_each_unrepresented_var(conjure_model, |name, var| {
65            let is_search_var = !matches!(name, conjure_ast::Name::Machine(_));
66            load_var(name, var, is_search_var, minion_model)
67        })?;
68    }
69    Ok(())
70}

72fn for_each_unrepresented_var(
73    conjure_model: &ConjureModel,
74    mut f: impl FnMut(&conjure_ast::Name, &conjure_ast::DecisionVariable) -> Result<(), SolverError>,
75) -> Result<(), SolverError> {
76    for (name, decl) in conjure_model
77        .as_submodel()
78        .symbols()
79        .clone()
80        .into_iter_local()
81    {
82        let Some(var) = decl.as_var() else {
83            continue;
84        };
85
86        if !conjure_model
87            .as_submodel()
88            .symbols()
89            .representations_for(&name)
90            .is_none_or(|x| x.is_empty())
91        {
92            continue;
93        }
94
95        f(&name, &var)?;
96    }
97
98    Ok(())
99}

101/// Loads a single variable into `minion_model`
102fn load_var(
103    name: &conjure_ast::Name,
104    var: &conjure_ast::DecisionVariable,
105    search_var: bool,
106    minion_model: &mut MinionModel,
107) -> Result<(), SolverError> {
108    let resolved_domain = var.domain_of().resolve();
109    match resolved_domain.as_deref() {
110        Some(conjure_ast::GroundDomain::Int(ranges)) => {
111            load_intdomain_var(name, ranges, search_var, minion_model)
112        }
113        Some(conjure_ast::GroundDomain::Bool) => {
114            load_booldomain_var(name, search_var, minion_model)
115        }
116        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
117    }
118}

120/// Loads a variable with domain IntDomain into `minion_model`
121fn load_intdomain_var(
122    name: &conjure_ast::Name,
123    ranges: &[conjure_ast::Range<i32>],
124    search_var: bool,
125    minion_model: &mut MinionModel,
126) -> Result<(), SolverError> {
127    let str_name = name_to_string(name.to_owned());
128
129    // If the int domain has holes, e.g x: int(1, 3, 5..8)
130    // convert it to a single range x: int(1..8)
131    // and add a w-inset constraint, e.g. w-inset(x, [1, 3, 5, 6, 7, 8])
132    if ranges.len() > 1 {
133        let values = Range::values(ranges)
134            .ok_or_else(|| {
135                ModelFeatureNotSupported(format!(
136                    "non-finite int domain with multiple ranges for variable {name}"
137                ))
138            })?
139            .map(Constant::Integer)
140            .collect_vec();
141        minion_model
142            .constraints
143            .push(Constraint::WInset(Var::NameRef(str_name.clone()), values));
144    }
145    let range = Range::spanning(ranges);
146
147    let (low, high) = match range {
148        Range::Bounded(x, y) => Ok((x.to_owned(), y.to_owned())),
149        Range::Single(x) => Ok((x.to_owned(), x.to_owned())),
150        #[allow(unreachable_patterns)]
151        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
152    }?;
153
154    let domain = minion_ast::VarDomain::Bound(low, high);
155
156    try_add_var(str_name, domain, search_var, minion_model)
157}

159/// Loads a variable with domain BoolDomain into `minion_model`
160fn load_booldomain_var(
161    name: &conjure_ast::Name,
162    search_var: bool,
163    minion_model: &mut MinionModel,
164) -> Result<(), SolverError> {
165    let str_name = name_to_string(name.to_owned());
166    let domain = minion_ast::VarDomain::Bool;
167    try_add_var(str_name, domain, search_var, minion_model)
168}
169
170fn try_add_var(
171    name: minion_ast::VarName,
172    domain: minion_ast::VarDomain,
173    search_var: bool,
174    minion_model: &mut MinionModel,
175) -> Result<(), SolverError> {
176    let added = if search_var {
177        minion_model.named_variables.add_var(name.clone(), domain)
178    } else {
179        minion_model
180            .named_variables
181            .add_aux_var(name.clone(), domain)
182    };
183    added.ok_or(ModelInvalid(format!("variable {name:?} is defined twice")))
184}

186fn name_to_string(name: conjure_ast::Name) -> String {
187    match name {
188        // print machine names in a custom, easier to regex, way.
189        conjure_ast::Name::Machine(x) => format!("__conjure_machine_name_{x}"),
190        conjure_ast::Name::Represented(fields) => {
191            let (name, rule, suffix) = *fields;
192            let name = name_to_string(name);
193            format!("__conjure_represented_name__{name}__{rule}___{suffix}")
194        }
195        x => format!("{x}"),
196    }
197}

199/// Loads the constraints into `minion_model`.
200fn load_constraints(
201    conjure_model: &ConjureModel,
202    minion_model: &mut MinionModel,
203) -> Result<(), SolverError> {
204    for expr in conjure_model.as_submodel().constraints().iter() {
205        // TODO: top level false / trues should not go to the solver to begin with
206        // ... but changing this at this stage would require rewriting the tester
207        use crate::ast::Metadata;
208        use conjure_ast::Atom;
209        use conjure_ast::Expression as Expr;
210        use conjure_ast::Literal;
211
212        match expr {
213            // top level false
214            Expr::Atomic(_, Atom::Literal(Literal::Bool(false))) => {
215                minion_model.constraints.push(minion_ast::Constraint::False);
216            }
217            // top level true
218            Expr::Atomic(_, Atom::Literal(Literal::Bool(true))) => {
219                minion_model.constraints.push(minion_ast::Constraint::True);
220            }
221
222            _ => {
223                load_expr(expr.to_owned(), minion_model)?;
224            }
225        }
226    }
227    Ok(())
228}

230/// Adds `expr` to `minion_model`
231fn load_expr(
232    expr: conjure_ast::Expression,
233    minion_model: &mut MinionModel,
234) -> Result<(), SolverError> {
235    minion_model.constraints.push(parse_expr(expr)?);
236    Ok(())
237}

239/// Parses a Conjure Oxide expression into a `minion_sys` constraint.
240fn parse_expr(expr: conjure_ast::Expression) -> Result<minion_ast::Constraint, SolverError> {
241    match expr {
242        conjure_ast::Expression::Atomic(_metadata, atom) => Ok(minion_ast::Constraint::WLiteral(
243            parse_atom(atom)?,
244            minion_ast::Constant::Integer(1),
245        )),
246
247        // The Minion adaptor currently treats bools as integers anyways, so this is a no-op
248        conjure_ast::Expression::ToInt(_metadata, inner_expr) => {
249            parse_expr(Moo::unwrap_or_clone(inner_expr))
250        }
251
252        conjure_ast::Expression::FlatAllDiff(_metadata, atoms) => {
253            Ok(minion_ast::Constraint::AllDiff(parse_atoms(atoms)?))
254        }
255        conjure_ast::Expression::FlatSumLeq(_metadata, lhs, rhs) => Ok(
256            minion_ast::Constraint::SumLeq(parse_atoms(lhs)?, parse_atom(rhs)?),
257        ),
258        conjure_ast::Expression::FlatSumGeq(_metadata, lhs, rhs) => Ok(
259            minion_ast::Constraint::SumGeq(parse_atoms(lhs)?, parse_atom(rhs)?),
260        ),
261        conjure_ast::Expression::FlatIneq(_metadata, a, b, c) => Ok(minion_ast::Constraint::Ineq(
262            parse_atom(Moo::unwrap_or_clone(a))?,
263            parse_atom(Moo::unwrap_or_clone(b))?,
264            parse_literal(*c)?,
265        )),
266        conjure_ast::Expression::Neq(_metadata, a, b) => Ok(minion_ast::Constraint::DisEq(
267            parse_atomic_expr(Moo::unwrap_or_clone(a))?,
268            parse_atomic_expr(Moo::unwrap_or_clone(b))?,
269        )),
270        conjure_ast::Expression::MinionDivEqUndefZero(_metadata, a, b, c) => {
271            Ok(minion_ast::Constraint::DivUndefZero(
272                (
273                    parse_atom(Moo::unwrap_or_clone(a))?,
274                    parse_atom(Moo::unwrap_or_clone(b))?,
275                ),
276                parse_atom(Moo::unwrap_or_clone(c))?,
277            ))
278        }
279        conjure_ast::Expression::MinionModuloEqUndefZero(_metadata, a, b, c) => {
280            Ok(minion_ast::Constraint::ModuloUndefZero(
281                (
282                    parse_atom(Moo::unwrap_or_clone(a))?,
283                    parse_atom(Moo::unwrap_or_clone(b))?,
284                ),
285                parse_atom(Moo::unwrap_or_clone(c))?,
286            ))
287        }
288        conjure_ast::Expression::MinionWInIntervalSet(_metadata, a, xs) => {
289            Ok(minion_ast::Constraint::WInIntervalSet(
290                parse_atom(a)?,
291                xs.into_iter()
292                    .map(minion_ast::Constant::Integer)
293                    .collect_vec(),
294            ))
295        }
296        conjure_ast::Expression::MinionWInSet(_metadata, a, xs) => {
297            Ok(minion_ast::Constraint::WInset(
298                parse_atom(a)?,
299                xs.into_iter()
300                    .map(minion_ast::Constant::Integer)
301                    .collect_vec(),
302            ))
303        }
304        conjure_ast::Expression::MinionElementOne(_, vec, i, e) => {
305            Ok(minion_ast::Constraint::ElementOne(
306                parse_atoms(vec)?,
307                parse_atom(Moo::unwrap_or_clone(i))?,
308                parse_atom(Moo::unwrap_or_clone(e))?,
309            ))
310        }
311
312        conjure_ast::Expression::Or(_metadata, e) => Ok(minion_ast::Constraint::WatchedOr(
313            Moo::unwrap_or_clone(e)
314                .unwrap_matrix_unchecked()
315                .ok_or_else(|| {
316                    SolverError::ModelFeatureNotSupported(
317                        "The inside of an or expression is not a matrix.".to_string(),
318                    )
319                })?
320                .0
321                .iter()
322                .map(|x| parse_expr(x.to_owned()))
323                .collect::<Result<Vec<minion_ast::Constraint>, SolverError>>()?,
324        )),
325        conjure_ast::Expression::And(_metadata, e) => Ok(minion_ast::Constraint::WatchedAnd(
326            Moo::unwrap_or_clone(e)
327                .unwrap_matrix_unchecked()
328                .ok_or_else(|| {
329                    SolverError::ModelFeatureNotSupported(
330                        "The inside of an and expression is not a matrix.".to_string(),
331                    )
332                })?
333                .0
334                .iter()
335                .map(|x| parse_expr(x.to_owned()))
336                .collect::<Result<Vec<minion_ast::Constraint>, SolverError>>()?,
337        )),
338        conjure_ast::Expression::Eq(_metadata, a, b) => Ok(minion_ast::Constraint::Eq(
339            parse_atomic_expr(Moo::unwrap_or_clone(a))?,
340            parse_atomic_expr(Moo::unwrap_or_clone(b))?,
341        )),
342
343        conjure_ast::Expression::FlatWatchedLiteral(_metadata, reference, k) => {
344            Ok(minion_ast::Constraint::WLiteral(
345                parse_name(reference.ptr().name().clone())?,
346                parse_literal(k)?,
347            ))
348        }
349        conjure_ast::Expression::MinionReify(_metadata, e, v) => Ok(minion_ast::Constraint::Reify(
350            Box::new(parse_expr(Moo::unwrap_or_clone(e))?),
351            parse_atom(v)?,
352        )),
353
354        conjure_ast::Expression::MinionReifyImply(_metadata, e, v) => {
355            Ok(minion_ast::Constraint::ReifyImply(
356                Box::new(parse_expr(Moo::unwrap_or_clone(e))?),
357                parse_atom(v)?,
358            ))
359        }
360
361        conjure_ast::Expression::AuxDeclaration(_metadata, reference, expr) => {
362            Ok(minion_ast::Constraint::Eq(
363                parse_name(reference.ptr().name().clone())?,
364                parse_atomic_expr(Moo::unwrap_or_clone(expr))?,
365            ))
366        }
367
368        conjure_ast::Expression::FlatMinusEq(_metadata, a, b) => {
369            Ok(minion_ast::Constraint::MinusEq(
370                parse_atom(Moo::unwrap_or_clone(a))?,
371                parse_atom(Moo::unwrap_or_clone(b))?,
372            ))
373        }
374
375        conjure_ast::Expression::FlatProductEq(_metadata, a, b, c) => {
376            Ok(minion_ast::Constraint::Product(
377                (
378                    parse_atom(Moo::unwrap_or_clone(a))?,
379                    parse_atom(Moo::unwrap_or_clone(b))?,
380                ),
381                parse_atom(Moo::unwrap_or_clone(c))?,
382            ))
383        }
384        conjure_ast::Expression::FlatWeightedSumLeq(_metadata, coeffs, vars, total) => {
385            Ok(minion_ast::Constraint::WeightedSumLeq(
386                parse_literals(coeffs)?,
387                parse_atoms(vars)?,
388                parse_atom(Moo::unwrap_or_clone(total))?,
389            ))
390        }
391        conjure_ast::Expression::FlatWeightedSumGeq(_metadata, coeffs, vars, total) => {
392            Ok(minion_ast::Constraint::WeightedSumGeq(
393                parse_literals(coeffs)?,
394                parse_atoms(vars)?,
395                parse_atom(Moo::unwrap_or_clone(total))?,
396            ))
397        }
398        conjure_ast::Expression::FlatAbsEq(_metadata, x, y) => Ok(minion_ast::Constraint::Abs(
399            parse_atom(Moo::unwrap_or_clone(x))?,
400            parse_atom(Moo::unwrap_or_clone(y))?,
401        )),
402        conjure_ast::Expression::MinionPow(_, x, y, z) => Ok(minion_ast::Constraint::Pow(
403            (
404                parse_atom(Moo::unwrap_or_clone(x))?,
405                parse_atom(Moo::unwrap_or_clone(y))?,
406            ),
407            parse_atom(Moo::unwrap_or_clone(z))?,
408        )),
409        conjure_ast::Expression::FlatLexLt(_metadata, lhs, rhs) => Ok(
410            minion_ast::Constraint::LexLess(parse_atoms(lhs)?, parse_atoms(rhs)?),
411        ),
412        conjure_ast::Expression::FlatLexLeq(_metadata, lhs, rhs) => Ok(
413            minion_ast::Constraint::LexLeq(parse_atoms(lhs)?, parse_atoms(rhs)?),
414        ),
415        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
416    }
417}

419fn parse_atomic_expr(expr: conjure_ast::Expression) -> Result<minion_ast::Var, SolverError> {
420    match expr {
421        // Minion treats bools as ints anyways, so this is a no-op at this stage
422        conjure_ast::Expression::ToInt(_metadata, inner_expr) => {
423            parse_atomic_expr(Moo::unwrap_or_clone(inner_expr))
424        }
425        conjure_ast::Expression::Atomic(_, atom) => parse_atom(atom),
426        expr => {
427            if let Some(lit) = conjure_ast::eval_constant(&expr) {
428                return Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(lit)?));
429            }
430
431            Err(ModelInvalid(format!(
432                "expected atomic expression, got {expr:?}"
433            )))
434        }
435    }
436}

438fn parse_atoms(exprs: Vec<conjure_ast::Atom>) -> Result<Vec<minion_ast::Var>, SolverError> {
439    exprs.into_iter().map(parse_atom).collect()
440}

442fn parse_atom(atom: conjure_ast::Atom) -> Result<minion_ast::Var, SolverError> {
443    match atom {
444        conjure_ast::Atom::Literal(l) => {
445            Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(l)?))
446        }
447        conjure_ast::Atom::Reference(reference) => parse_reference_atom(reference),
448
449        x => Err(ModelFeatureNotSupported(format!(
450            "expected a literal or a reference but got `{x}`"
451        ))),
452    }
453}
454
455fn parse_reference_atom(reference: conjure_ast::Reference) -> Result<minion_ast::Var, SolverError> {
456    if let Some(lit) = reference.resolve_constant() {
457        return Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(lit)?));
458    }
459
460    if let Some(inner_atom) = reference.resolve_atomic() {
461        return parse_atom(inner_atom);
462    }
463
464    let decl_kind = reference.ptr().kind();
465    match decl_kind.deref() {
466        conjure_ast::DeclarationKind::ValueLetting(expr) => Err(ModelFeatureNotSupported(format!(
467            "value letting '{}' did not resolve to an atomic expression: {expr}",
468            reference.name()
469        ))),
470        conjure_ast::DeclarationKind::DomainLetting(_) => Err(ModelFeatureNotSupported(format!(
471            "domain reference used where atom expected: {}",
472            reference.name()
473        ))),
474        _ => parse_name(reference.name().clone()),
475    }
476}

478fn parse_literal_as_int(k: conjure_ast::Literal) -> Result<i32, SolverError> {
479    match k {
480        conjure_ast::Literal::Int(n) => Ok(n),
481        conjure_ast::Literal::Bool(true) => Ok(1),
482        conjure_ast::Literal::Bool(false) => Ok(0),
483        x => Err(ModelInvalid(format!("expected a literal but got `{x:?}`"))),
484    }
485}

487fn parse_literals(

493fn parse_literal(k: conjure_ast::Literal) -> Result<minion_ast::Constant, SolverError> {
494    Ok(minion_ast::Constant::Integer(parse_literal_as_int(k)?))
