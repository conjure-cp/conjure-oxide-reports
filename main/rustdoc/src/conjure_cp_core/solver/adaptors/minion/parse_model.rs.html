24/// Converts a conjure-oxide model to a `minion_sys` model.
25pub fn model_to_minion(model: ConjureModel) -> Result<MinionModel, SolverError> {
26    let mut minion_model = MinionModel::new();
27    let table_vars = collect_table_variables(&model);
28    load_symbol_table(&model, &table_vars, &mut minion_model)?;
29    load_constraints(&model, &mut minion_model)?;
30    Ok(minion_model)
31}

33/// Loads the symbol table into `minion_model`.
34fn load_symbol_table(
35    conjure_model: &ConjureModel,
36    table_vars: &HashSet<conjure_ast::Name>,
37    minion_model: &mut MinionModel,
38) -> Result<(), SolverError> {
39    if let Some(ref vars) = conjure_model.search_order {
40        let search_vars: HashSet<_> = vars.iter().cloned().collect();
41
42        // add search vars in order first
43        for name in vars {
44            let decl = conjure_model
45                .symbols()
46                .lookup(name)
47                .ok_or_else(|| ModelInvalid(format!("search variable '{name}' does not exist")))?;
48            let var = decl.as_find().ok_or_else(|| {
49                ModelInvalid(format!(
50                    "search variable '{name}' is not a decision variable"
51                ))
52            })?;
53
54            load_var(name, &var, true, table_vars, minion_model)?;
55        }
56
57        // then add the rest as non-search vars
58        for_each_unrepresented_var(conjure_model, |name, var| {
59            if search_vars.contains(name) {
60                return Ok(());
61            }
62            load_var(name, var, false, table_vars, minion_model)
63        })?;
64    } else {
65        for_each_unrepresented_var(conjure_model, |name, var| {
66            let is_search_var = !matches!(name, conjure_ast::Name::Machine(_));
67            load_var(name, var, is_search_var, table_vars, minion_model)
68        })?;
69    }
70    Ok(())
71}

73fn collect_table_variables(conjure_model: &ConjureModel) -> HashSet<conjure_ast::Name> {
74    conjure_model
75        .constraints()
76        .iter()
77        .flat_map(|expr| expr.universe())
78        .filter_map(|expr| match expr {
79            conjure_ast::Expression::Table(_, tuple_expr, _)
80            | conjure_ast::Expression::NegativeTable(_, tuple_expr, _) => {
81                Some(Moo::unwrap_or_clone(tuple_expr))
82            }
83            _ => None,
84        })
85        .flat_map(|tuple_expr| Biplate::<conjure_ast::Reference>::universe_bi(&tuple_expr))
86        .map(|reference| reference.name().clone())
87        .collect()
88}

90fn for_each_unrepresented_var(
91    conjure_model: &ConjureModel,
92    mut f: impl FnMut(&conjure_ast::Name, &conjure_ast::DecisionVariable) -> Result<(), SolverError>,
93) -> Result<(), SolverError> {
94    for (name, decl) in conjure_model.symbols().clone().into_iter_local() {
95        let Some(var) = decl.as_find() else {
96            continue;
97        };
98
99        if !conjure_model
100            .symbols()
101            .representations_for(&name)
102            .is_none_or(|x| x.is_empty())
103        {
104            continue;
105        }
106
107        f(&name, &var)?;
108    }
109
110    Ok(())
111}

113/// Loads a single variable into `minion_model`
114fn load_var(
115    name: &conjure_ast::Name,
116    var: &conjure_ast::DecisionVariable,
117    search_var: bool,
118    table_vars: &HashSet<conjure_ast::Name>,
119    minion_model: &mut MinionModel,
120) -> Result<(), SolverError> {
121    let resolved_domain = var.domain_of().resolve();
122    let force_discrete = table_vars.contains(name);
123    match resolved_domain.as_deref() {
124        Ok(conjure_ast::GroundDomain::Int(ranges)) => {
125            load_intdomain_var(name, ranges, search_var, force_discrete, minion_model)
126        }
127        Ok(conjure_ast::GroundDomain::Bool) => load_booldomain_var(name, search_var, minion_model),
128        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
129    }
130}

132/// Loads a variable with domain IntDomain into `minion_model`
133fn load_intdomain_var(
134    name: &conjure_ast::Name,
135    ranges: &[conjure_ast::Range<i32>],
136    search_var: bool,
137    force_discrete: bool,
138    minion_model: &mut MinionModel,
139) -> Result<(), SolverError> {
140    let str_name = name_to_string(name.to_owned());
141
142    // If the int domain has holes, e.g x: int(1, 3, 5..8)
143    // convert it to a single range x: int(1..8)
144    // and add a w-inset constraint, e.g. w-inset(x, [1, 3, 5, 6, 7, 8])
145    if ranges.len() > 1 {
146        let values = Range::values(ranges)
147            .ok_or_else(|| {
148                ModelFeatureNotSupported(format!(
149                    "non-finite int domain with multiple ranges for variable {name}"
150                ))
151            })?
152            .map(Constant::Integer)
153            .collect_vec();
154        minion_model
155            .constraints
156            .push(Constraint::WInset(Var::NameRef(str_name.clone()), values));
157    }
158    let range = Range::spanning(ranges);
159
160    let (low, high) = match range {
161        Range::Bounded(x, y) => Ok((x.to_owned(), y.to_owned())),
162        Range::Single(x) => Ok((x.to_owned(), x.to_owned())),
163        #[allow(unreachable_patterns)]
164        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
165    }?;
166
167    let size = i64::from(high) - i64::from(low) + 1;
168    let threshold = minion_discrete_threshold() as i64;
169    let use_discrete = force_discrete || size <= threshold;
170
171    let domain = if use_discrete {
172        minion_ast::VarDomain::Discrete(low, high)
173    } else {
174        minion_ast::VarDomain::Bound(low, high)
175    };
176
177    try_add_var(str_name, domain, search_var, minion_model)
178}

180/// Loads a variable with domain BoolDomain into `minion_model`
181fn load_booldomain_var(
182    name: &conjure_ast::Name,
183    search_var: bool,
184    minion_model: &mut MinionModel,
185) -> Result<(), SolverError> {
186    let str_name = name_to_string(name.to_owned());
187    let domain = minion_ast::VarDomain::Bool;
188    try_add_var(str_name, domain, search_var, minion_model)
189}

191fn try_add_var(
192    name: minion_ast::VarName,
193    domain: minion_ast::VarDomain,
194    search_var: bool,
195    minion_model: &mut MinionModel,
196) -> Result<(), SolverError> {
197    let added = if search_var {
198        minion_model.named_variables.add_var(name.clone(), domain)
199    } else {
200        minion_model
201            .named_variables
202            .add_aux_var(name.clone(), domain)
203    };
204    added.ok_or(ModelInvalid(format!("variable {name:?} is defined twice")))
205}

207fn name_to_string(name: conjure_ast::Name) -> String {
208    match name {
209        // print machine names in a custom, easier to regex, way.
210        conjure_ast::Name::Machine(x) => format!("__conjure_machine_name_{x}"),
211        conjure_ast::Name::Represented(fields) => {
212            let (name, rule, suffix) = *fields;
213            let name = name_to_string(name);
214            format!("__conjure_represented_name__{name}__{rule}___{suffix}")
215        }
216        x => format!("{x}"),
217    }
218}

220/// Loads the constraints into `minion_model`.
221fn load_constraints(
222    conjure_model: &ConjureModel,
223    minion_model: &mut MinionModel,
224) -> Result<(), SolverError> {
225    for expr in conjure_model.constraints().iter() {
226        // TODO: top level false / trues should not go to the solver to begin with
227        // ... but changing this at this stage would require rewriting the tester
228        use crate::ast::Metadata;
229        use conjure_ast::Atom;
230        use conjure_ast::Expression as Expr;
231        use conjure_ast::Literal;
232
233        match expr {
234            // top level false
235            Expr::Atomic(_, Atom::Literal(Literal::Bool(false))) => {
236                minion_model.constraints.push(minion_ast::Constraint::False);
237            }
238            // top level true
239            Expr::Atomic(_, Atom::Literal(Literal::Bool(true))) => {
240                minion_model.constraints.push(minion_ast::Constraint::True);
241            }
242
243            _ => {
244                load_expr(expr.to_owned(), minion_model)?;
245            }
246        }
247    }
248    Ok(())
249}

251/// Adds `expr` to `minion_model`
252fn load_expr(
253    expr: conjure_ast::Expression,
254    minion_model: &mut MinionModel,
255) -> Result<(), SolverError> {
256    minion_model.constraints.push(parse_expr(expr)?);
257    Ok(())

260/// Parses a Conjure Oxide expression into a `minion_sys` constraint.
261fn parse_expr(expr: conjure_ast::Expression) -> Result<minion_ast::Constraint, SolverError> {
262    match expr {
263        conjure_ast::Expression::Atomic(_metadata, atom) => Ok(minion_ast::Constraint::WLiteral(
264            parse_atom(atom)?,
265            minion_ast::Constant::Integer(1),
266        )),
267
268        // The Minion adaptor currently treats bools as integers anyways, so this is a no-op
269        conjure_ast::Expression::ToInt(_metadata, inner_expr) => {
270            parse_expr(Moo::unwrap_or_clone(inner_expr))
271        }
272
273        conjure_ast::Expression::FlatAllDiff(_metadata, atoms) => {
274            Ok(minion_ast::Constraint::AllDiff(parse_atoms(atoms)?))
275        }
276        conjure_ast::Expression::FlatSumLeq(_metadata, lhs, rhs) => Ok(
277            minion_ast::Constraint::SumLeq(parse_atoms(lhs)?, parse_atom(rhs)?),
278        ),
279        conjure_ast::Expression::FlatSumGeq(_metadata, lhs, rhs) => Ok(
280            minion_ast::Constraint::SumGeq(parse_atoms(lhs)?, parse_atom(rhs)?),
281        ),
282        conjure_ast::Expression::FlatIneq(_metadata, a, b, c) => Ok(minion_ast::Constraint::Ineq(
283            parse_atom(Moo::unwrap_or_clone(a))?,
284            parse_atom(Moo::unwrap_or_clone(b))?,
285            parse_literal(*c)?,
286        )),
287        conjure_ast::Expression::Not(_metadata, inner) => {
288            let inner_expr = Moo::unwrap_or_clone(inner);
289            match inner_expr {
290                conjure_ast::Expression::Atomic(
291                    _,
292                    conjure_ast::Atom::Literal(conjure_ast::Literal::Bool(v)),
293                ) => Ok(if v {
294                    minion_ast::Constraint::False
295                } else {
296                    minion_ast::Constraint::True
297                }),
298                conjure_ast::Expression::Atomic(_, atom) => Ok(minion_ast::Constraint::WLiteral(
299                    parse_atom(atom)?,
300                    minion_ast::Constant::Integer(0),
301                )),
302                other => {
303                    if let Some(conjure_ast::Literal::Bool(v)) = conjure_ast::eval_constant(&other)
304                    {
305                        Ok(if v {
306                            minion_ast::Constraint::False
307                        } else {
308                            minion_ast::Constraint::True
309                        })
310                    } else {
311                        Err(ModelFeatureNotSupported(format!(
312                            "unsupported negation form for Minion: {other:?}"
313                        )))
314                    }
315                }
316            }
317        }
318        conjure_ast::Expression::Neq(_metadata, a, b) => Ok(minion_ast::Constraint::DisEq(
319            parse_atomic_expr(Moo::unwrap_or_clone(a))?,
320            parse_atomic_expr(Moo::unwrap_or_clone(b))?,
321        )),
322        conjure_ast::Expression::Table(_metadata, tuple_expr, allowed_rows_expr) => {
323            parse_table_constraint(
324                TableConstraintKind::Table,
325                Moo::unwrap_or_clone(tuple_expr),
326                Moo::unwrap_or_clone(allowed_rows_expr),
327            )
328        }
329        conjure_ast::Expression::NegativeTable(_metadata, tuple_expr, forbidden_rows_expr) => {
330            parse_table_constraint(
331                TableConstraintKind::NegativeTable,
332                Moo::unwrap_or_clone(tuple_expr),
333                Moo::unwrap_or_clone(forbidden_rows_expr),
334            )
335        }
336        conjure_ast::Expression::MinionDivEqUndefZero(_metadata, a, b, c) => {
337            Ok(minion_ast::Constraint::DivUndefZero(
338                (
339                    parse_atom(Moo::unwrap_or_clone(a))?,
340                    parse_atom(Moo::unwrap_or_clone(b))?,
341                ),
342                parse_atom(Moo::unwrap_or_clone(c))?,
343            ))
344        }
345        conjure_ast::Expression::MinionModuloEqUndefZero(_metadata, a, b, c) => {
346            Ok(minion_ast::Constraint::ModuloUndefZero(
347                (
348                    parse_atom(Moo::unwrap_or_clone(a))?,
349                    parse_atom(Moo::unwrap_or_clone(b))?,
350                ),
351                parse_atom(Moo::unwrap_or_clone(c))?,
352            ))
353        }
354        conjure_ast::Expression::MinionWInIntervalSet(_metadata, a, xs) => {
355            Ok(minion_ast::Constraint::WInIntervalSet(
356                parse_atom(a)?,
357                xs.into_iter()
358                    .map(minion_ast::Constant::Integer)
359                    .collect_vec(),
360            ))
361        }
362        conjure_ast::Expression::MinionWInSet(_metadata, a, xs) => {
363            Ok(minion_ast::Constraint::WInset(
364                parse_atom(a)?,
365                xs.into_iter()
366                    .map(minion_ast::Constant::Integer)
367                    .collect_vec(),
368            ))
369        }
370        conjure_ast::Expression::MinionElementOne(_, vec, i, e) => {
371            Ok(minion_ast::Constraint::ElementOne(
372                parse_atoms(vec)?,
373                parse_atom(Moo::unwrap_or_clone(i))?,
374                parse_atom(Moo::unwrap_or_clone(e))?,
375            ))
376        }
377
378        conjure_ast::Expression::Or(_metadata, e) => Ok(minion_ast::Constraint::WatchedOr(
379            Moo::unwrap_or_clone(e)
380                .unwrap_matrix_unchecked()
381                .ok_or_else(|| {
382                    SolverError::ModelFeatureNotSupported(
383                        "The inside of an or expression is not a matrix.".to_string(),
384                    )
385                })?
386                .0
387                .iter()
388                .map(|x| parse_expr(x.to_owned()))
389                .collect::<Result<Vec<minion_ast::Constraint>, SolverError>>()?,
390        )),
391        conjure_ast::Expression::And(_metadata, e) => Ok(minion_ast::Constraint::WatchedAnd(
392            Moo::unwrap_or_clone(e)
393                .unwrap_matrix_unchecked()
394                .ok_or_else(|| {
395                    SolverError::ModelFeatureNotSupported(
396                        "The inside of an and expression is not a matrix.".to_string(),
397                    )
398                })?
399                .0
400                .iter()
401                .map(|x| parse_expr(x.to_owned()))
402                .collect::<Result<Vec<minion_ast::Constraint>, SolverError>>()?,
403        )),
404        conjure_ast::Expression::Eq(_metadata, a, b) => Ok(minion_ast::Constraint::Eq(
405            parse_atomic_expr(Moo::unwrap_or_clone(a))?,
406            parse_atomic_expr(Moo::unwrap_or_clone(b))?,
407        )),
408
409        conjure_ast::Expression::FlatWatchedLiteral(_metadata, reference, k) => {
410            Ok(minion_ast::Constraint::WLiteral(
411                parse_name(reference.ptr().name().clone())?,
412                parse_literal(k)?,
413            ))
414        }
415        conjure_ast::Expression::MinionReify(_metadata, e, v) => Ok(minion_ast::Constraint::Reify(
416            Box::new(parse_expr(Moo::unwrap_or_clone(e))?),
417            parse_atom(v)?,
418        )),
419
420        conjure_ast::Expression::MinionReifyImply(_metadata, e, v) => {
421            Ok(minion_ast::Constraint::ReifyImply(
422                Box::new(parse_expr(Moo::unwrap_or_clone(e))?),
423                parse_atom(v)?,
424            ))
425        }
426
427        conjure_ast::Expression::AuxDeclaration(_metadata, reference, expr) => {
428            Ok(minion_ast::Constraint::Eq(
429                parse_name(reference.ptr().name().clone())?,
430                parse_atomic_expr(Moo::unwrap_or_clone(expr))?,
431            ))
432        }
433
434        conjure_ast::Expression::FlatMinusEq(_metadata, a, b) => {
435            Ok(minion_ast::Constraint::MinusEq(
436                parse_atom(Moo::unwrap_or_clone(a))?,
437                parse_atom(Moo::unwrap_or_clone(b))?,
438            ))
439        }
440
441        conjure_ast::Expression::FlatProductEq(_metadata, a, b, c) => {
442            Ok(minion_ast::Constraint::Product(
443                (
444                    parse_atom(Moo::unwrap_or_clone(a))?,
445                    parse_atom(Moo::unwrap_or_clone(b))?,
446                ),
447                parse_atom(Moo::unwrap_or_clone(c))?,
448            ))
449        }
450        conjure_ast::Expression::FlatWeightedSumLeq(_metadata, coeffs, vars, total) => {
451            Ok(minion_ast::Constraint::WeightedSumLeq(
452                parse_literals(coeffs)?,
453                parse_atoms(vars)?,
454                parse_atom(Moo::unwrap_or_clone(total))?,
455            ))
456        }
457        conjure_ast::Expression::FlatWeightedSumGeq(_metadata, coeffs, vars, total) => {
458            Ok(minion_ast::Constraint::WeightedSumGeq(
459                parse_literals(coeffs)?,
460                parse_atoms(vars)?,
461                parse_atom(Moo::unwrap_or_clone(total))?,
462            ))
463        }
464        conjure_ast::Expression::FlatAbsEq(_metadata, x, y) => Ok(minion_ast::Constraint::Abs(
465            parse_atom(Moo::unwrap_or_clone(x))?,
466            parse_atom(Moo::unwrap_or_clone(y))?,
467        )),
468        conjure_ast::Expression::MinionPow(_, x, y, z) => Ok(minion_ast::Constraint::Pow(
469            (
470                parse_atom(Moo::unwrap_or_clone(x))?,
471                parse_atom(Moo::unwrap_or_clone(y))?,
472            ),
473            parse_atom(Moo::unwrap_or_clone(z))?,
474        )),
475        conjure_ast::Expression::FlatLexLt(_metadata, lhs, rhs) => Ok(
476            minion_ast::Constraint::LexLess(parse_atoms(lhs)?, parse_atoms(rhs)?),
477        ),
478        conjure_ast::Expression::FlatLexLeq(_metadata, lhs, rhs) => Ok(
479            minion_ast::Constraint::LexLeq(parse_atoms(lhs)?, parse_atoms(rhs)?),
480        ),
481        x => Err(ModelFeatureNotSupported(format!("{x:?}"))),
482    }
483}

485#[derive(Clone, Copy)]
486enum TableConstraintKind {
487    Table,
488    NegativeTable,
489}
490
491impl TableConstraintKind {
492    fn name(self) -> &'static str {
493        match self {
494            Self::Table => "table",
495            Self::NegativeTable => "negativeTable",
496        }
497    }
498
499    fn into_constraint(
500        self,
501        vars: Vec<minion_ast::Var>,
502        tuples: Vec<Vec<minion_ast::Constant>>,
503    ) -> minion_ast::Constraint {
504        match self {
505            Self::Table => minion_ast::Constraint::Table(vars, tuples),
506            Self::NegativeTable => minion_ast::Constraint::NegativeTable(vars, tuples),
507        }
508    }
509}
510
511fn parse_table_constraint(
512    kind: TableConstraintKind,
513    tuple_expr: conjure_ast::Expression,
514    rows_expr: conjure_ast::Expression,
515) -> Result<minion_ast::Constraint, SolverError> {
516    let (tuple_elems, _) = tuple_expr
517        .unwrap_matrix_unchecked()
518        .ok_or_else(|| ModelInvalid(format!("{} first argument is not a matrix", kind.name())))?;
519    let (rows, _) = rows_expr
520        .unwrap_matrix_unchecked()
521        .ok_or_else(|| ModelInvalid(format!("{} second argument is not a matrix", kind.name())))?;
522
523    let vars = tuple_elems
524        .into_iter()
525        .map(parse_atomic_expr)
526        .collect::<Result<Vec<_>, SolverError>>()?;
527
528    let mut tuples = Vec::with_capacity(rows.len());
529    for row_expr in rows {
530        let (row_elems, _) = row_expr
531            .unwrap_matrix_unchecked()
532            .ok_or_else(|| ModelInvalid(format!("{} row is not a matrix", kind.name())))?;
533
534        if row_elems.len() != vars.len() {
535            return Err(ModelInvalid(format!(
536                "{} row width does not match tuple width",
537                kind.name()
538            )));
539        }
540
541        let tuple = row_elems
542            .into_iter()
543            .map(|row_val_expr| {
544                conjure_ast::eval_constant(&row_val_expr)
545                    .ok_or_else(|| {
546                        ModelInvalid(format!(
547                            "{} row contains a non-constant expression",
548                            kind.name()
549                        ))
550                    })
551                    .and_then(parse_literal)
552            })
553            .collect::<Result<Vec<_>, SolverError>>()?;
554
555        tuples.push(tuple);
556    }
557
558    Ok(kind.into_constraint(vars, tuples))
559}

561fn parse_atomic_expr(expr: conjure_ast::Expression) -> Result<minion_ast::Var, SolverError> {
562    match expr {
563        // Minion treats bools as ints anyways, so this is a no-op at this stage
564        conjure_ast::Expression::ToInt(_metadata, inner_expr) => {
565            parse_atomic_expr(Moo::unwrap_or_clone(inner_expr))
566        }
567        conjure_ast::Expression::Atomic(_, atom) => parse_atom(atom),
568        expr => {
569            if let Some(lit) = conjure_ast::eval_constant(&expr) {
570                return Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(lit)?));
571            }
572
573            Err(ModelInvalid(format!(
574                "expected atomic expression, got {expr:?}"
575            )))
576        }
577    }
578}

580fn parse_atoms(exprs: Vec<conjure_ast::Atom>) -> Result<Vec<minion_ast::Var>, SolverError> {
581    exprs.into_iter().map(parse_atom).collect()
582}

584fn parse_atom(atom: conjure_ast::Atom) -> Result<minion_ast::Var, SolverError> {
585    match atom {
586        conjure_ast::Atom::Literal(l) => {
587            Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(l)?))
588        }
589        conjure_ast::Atom::Reference(reference) => parse_reference_atom(reference),
590
591        x => Err(ModelFeatureNotSupported(format!(
592            "expected a literal or a reference but got `{x}`"
593        ))),
594    }
595}

597fn parse_reference_atom(reference: conjure_ast::Reference) -> Result<minion_ast::Var, SolverError> {
598    if let Some(lit) = reference.resolve_constant() {
599        return Ok(minion_ast::Var::ConstantAsVar(parse_literal_as_int(lit)?));
600    }
601
602    if let Some(inner_atom) = reference.resolve_atomic() {
603        return parse_atom(inner_atom);
604    }
605
606    let decl_kind = reference.ptr().kind();
607    match decl_kind.deref() {
608        conjure_ast::DeclarationKind::ValueLetting(expr, _) => {
609            Err(ModelFeatureNotSupported(format!(
610                "value letting '{}' did not resolve to an atomic expression: {expr}",
611                reference.name()
612            )))
613        }
614        conjure_ast::DeclarationKind::DomainLetting(_) => Err(ModelFeatureNotSupported(format!(
615            "domain reference used where atom expected: {}",
616            reference.name()
617        ))),
618        _ => parse_name(reference.name().clone()),
619    }
620}
621
622fn parse_literal_as_int(k: conjure_ast::Literal) -> Result<i32, SolverError> {
623    match k {
624        conjure_ast::Literal::Int(n) => Ok(n),
625        conjure_ast::Literal::Bool(true) => Ok(1),
626        conjure_ast::Literal::Bool(false) => Ok(0),
627        x => Err(ModelInvalid(format!("expected a literal but got `{x:?}`"))),
628    }
629}
630
631fn parse_literals(
632    literals: Vec<conjure_ast::Literal>,
633) -> Result<Vec<minion_ast::Constant>, SolverError> {
634    literals.into_iter().map(parse_literal).collect()
635}
636
637fn parse_literal(k: conjure_ast::Literal) -> Result<minion_ast::Constant, SolverError> {
638    Ok(minion_ast::Constant::Integer(parse_literal_as_int(k)?))
639}
640
641fn parse_name(name: conjure_ast::Name) -> Result<minion_ast::Var, SolverError> {
642    Ok(minion_ast::Var::NameRef(name_to_string(name)))
643}
