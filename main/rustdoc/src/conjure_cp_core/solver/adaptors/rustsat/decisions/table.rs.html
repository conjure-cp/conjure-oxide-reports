4impl Compiler<'_> {
5    pub(super) fn general_table(
6        &mut self,
7        output: Term,
8        inputs: &[crate::ast::sat_decision::SatIntegerView],
9        rows: &[Vec<crate::ast::sat_decision::SatIntegerView>],
10        negative: bool,
11        encoding: &Option<
12            crate::ast::sat_decision::EncodingSelection<crate::ast::sat_decision::TableEncoding>,
13        >,
14        pb_encoding: &Option<
15            crate::ast::sat_decision::EncodingSelection<crate::ast::sat_decision::PbEncoding>,
16        >,
17    ) -> Result<(), SolverError> {
18        use crate::ast::sat_decision::TableEncoding;
19        let algorithm = encoding
20            .as_ref()
21            .ok_or_else(|| SolverError::ModelInvalid("Unresolved table encoding decision".into()))?
22            .algorithm;
23        if algorithm == TableEncoding::BinarySupport {
24            return Err(SolverError::ModelInvalid(
25                "Binary-support tables require constant rows and two columns".into(),
26            ));
27        }
28        let pb = pb_encoding
29            .as_ref()
30            .ok_or_else(|| {
31                SolverError::ModelInvalid("Unresolved table component PB encoding".into())
32            })?
33            .algorithm;
34        let mut cells = vec![std::collections::BTreeMap::new(); inputs.len()];
35        let mut ids = vec![HashMap::new(); inputs.len()];
36        let mut relation = Vec::new();
37        for row in rows {
38            if row.len() != inputs.len() {
39                return Err(SolverError::ModelInvalid(
40                    "Table row width differs from tuple width".into(),
41                ));
42            }
43            let mut encoded = Vec::new();
44            for (column, (input, cell)) in inputs.iter().zip(row).enumerate() {
45                let equality = self.view_equality(input, cell, pb)?;
46                let next = i64::try_from(ids[column].len())
47                    .map_err(|_| SolverError::ModelInvalid("Too many table cells".into()))?;
48                let id = *ids[column].entry(equality).or_insert(next);
49                cells[column].insert(id, equality);
50                encoded.push(id);
51            }
52            relation.push(encoded);
53        }
54        relation.sort();
55        relation.dedup();
56        let truth = match algorithm {
57            TableEncoding::Tuple => {
58                let matches = relation
59                    .iter()
60                    .map(|row| {
61                        self.combine(
62                            true,
63                            row.iter()
64                                .enumerate()
65                                .map(|(column, id)| cells[column][id])
66                                .collect(),
67                        )
68                    })
69                    .collect();
70                self.combine(false, matches)
71            }
72            TableEncoding::Mdd => self.table_mdd(0, relation, &cells, &mut HashMap::new()),
73            TableEncoding::BinarySupport => unreachable!(),
74        };
75        self.equate(output, if negative { truth.negated() } else { truth });
76        Ok(())
77    }

79    pub(super) fn table(
80        &mut self,
81        output: Term,
82        inputs: &[crate::ast::sat_decision::SatIntegerView],
83        rows: &[Vec<i64>],
84        negative: bool,
85        encoding: &Option<
86            crate::ast::sat_decision::EncodingSelection<crate::ast::sat_decision::TableEncoding>,
87        >,
88        pb_encoding: &Option<
89            crate::ast::sat_decision::EncodingSelection<crate::ast::sat_decision::PbEncoding>,
90        >,
91    ) -> Result<(), SolverError> {
92        use crate::ast::sat_decision::{IntegerRelation, TableEncoding};
93        let algorithm = encoding
94            .as_ref()
95            .ok_or_else(|| SolverError::ModelInvalid("Unresolved table encoding decision".into()))?
96            .algorithm;
97        if algorithm == TableEncoding::BinarySupport && inputs.len() != 2 {
98            return Err(SolverError::ModelInvalid(
99                "Binary-support tables require two columns".into(),
100            ));
101        }
102        if rows.iter().any(|row| row.len() != inputs.len()) {
103            return Err(SolverError::ModelInvalid(
104                "Table row width differs from tuple width".into(),
105            ));
106        }
107        if rows.is_empty() || inputs.is_empty() {
108            let truth = !rows.is_empty();
109            self.equate(output, Term::Constant(truth != negative));
110            return Ok(());
111        }
112        let pb = pb_encoding
113            .as_ref()
114            .ok_or_else(|| {
115                SolverError::ModelInvalid("Unresolved table component PB encoding".into())
116            })?
117            .algorithm;
118        let mut rows = rows.to_vec();
119        rows.sort();
120        rows.dedup();
121        let mut cells = Vec::with_capacity(inputs.len());
122        for (column, input) in inputs.iter().enumerate() {
123            validate_pb_groups(&input.groups, input.terms.len())?;
124            let terms = input
125                .terms
126                .iter()
127                .map(|(weight, expression)| self.encode(expression).map(|term| (*weight, term)))
128                .collect::<Result<Vec<_>, _>>()?;
129            let values = rows
130                .iter()
131                .map(|row| row[column])
132                .collect::<std::collections::BTreeSet<_>>();
133            let mut equalities = std::collections::BTreeMap::new();
134            for value in values {
135                if terms.is_empty() {
136                    equalities.insert(value, Term::Constant(input.constant == value));
137                    continue;
138                }
139                let bound = i64::try_from(i128::from(value) - i128::from(input.constant)).map_err(
140                    |_| {
141                        SolverError::ModelInvalid(
142                            "Table cell difference exceeds the library range".into(),
143                        )
144                    },
145                )?;
146                let truth = Term::Literal(self.instance.new_lit());
147                if !self.choice_relation(
148                    truth,
149                    IntegerRelation::Equal,
150                    bound,
151                    &terms,
152                    &input.groups,
153                ) {
154                    self.integer_relation(
155                        pb,
156                        truth,
157                        IntegerRelation::Equal,
158                        bound,
159                        &terms,
160                        &input.groups,
161                    )?;
162                }
163                equalities.insert(value, truth);
164            }
165            cells.push(equalities);
166        }
167        let truth = match algorithm {
168            TableEncoding::Tuple => {
169                let matches = rows
170                    .iter()
171                    .map(|row| {
172                        self.combine(
173                            true,
174                            row.iter()
175                                .enumerate()
176                                .map(|(column, value)| cells[column][value])
177                                .collect(),
178                        )
179                    })
180                    .collect();
181                self.combine(false, matches)
182            }
183            TableEncoding::Mdd => self.table_mdd(0, rows, &cells, &mut HashMap::new()),
184            TableEncoding::BinarySupport => {
185                let mut conditions = Vec::new();
186                for column in 0..2 {
187                    conditions.push(self.combine(false, cells[column].values().copied().collect()));
188                    for (value, equality) in &cells[column] {
189                        let supports = rows
190                            .iter()
191                            .filter(|row| row[column] == *value)
192                            .map(|row| cells[1 - column][&row[1 - column]])
193                            .collect();
194                        let support = self.combine(false, supports);
195                        conditions.push(self.combine(false, vec![equality.negated(), support]));
196                    }
197                }
198                self.combine(true, conditions)
199            }
200        };
201        self.equate(output, if negative { truth.negated() } else { truth });
202        Ok(())
203    }

205    /// Identical suffix relations share one layered node and its library gates.
206    pub(super) fn table_mdd(
207        &mut self,
208        column: usize,
209        rows: Vec<Vec<i64>>,
210        cells: &[std::collections::BTreeMap<i64, Term>],
211        memo: &mut TableNodeCache,
212    ) -> Term {
213        if column == cells.len() {
214            return Term::Constant(!rows.is_empty());
215        }
216        let key = (column, rows.clone());
217        if let Some(truth) = memo.get(&key) {
218            return *truth;
219        }
220        let mut branches = std::collections::BTreeMap::<i64, Vec<Vec<i64>>>::new();
221        for row in rows {
222            branches.entry(row[0]).or_default().push(row[1..].to_vec());
223        }
224        let mut truths = Vec::new();
225        for (value, mut suffixes) in branches {
226            suffixes.sort();
227            suffixes.dedup();
228            let suffix = self.table_mdd(column + 1, suffixes, cells, memo);
229            truths.push(self.combine(true, vec![cells[column][&value], suffix]));
230        }
231        let truth = self.combine(false, truths);
232        memo.insert(key, truth);
233        truth
234    }
