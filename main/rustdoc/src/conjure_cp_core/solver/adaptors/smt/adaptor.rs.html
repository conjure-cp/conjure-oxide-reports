83fn sub_in_solution_into_dominance_expr(
84    expr: &Expression,
85    solution: &HashMap<Name, Literal>,
86) -> Option<Expression> {
87    match expr {
88        Expression::FromSolution(_, atom_expr) => {
89            if let Atom::Reference(reference) = atom_expr.as_ref() {
90                let var_name = reference.name();
91                let value = solution.get(&var_name)?;
92                let value = if let Some(domain) = reference.resolved_domain() {
93                    if domain.as_ref() == &GroundDomain::Bool {
94                        match value {
95                            Literal::Bool(x) => Literal::Bool(*x),
96                            Literal::Int(1) => Literal::Bool(true),
97                            Literal::Int(0) => Literal::Bool(false),
98                            _ => return None,
99                        }
100                    } else {
101                        value.clone()
102                    }
103                } else {
104                    value.clone()
105                };
106
107                return Some(Expression::Atomic(Metadata::new(), Atom::Literal(value)));
108            }
109            Some(expr.clone())
110        }
111        _ => Some(expr.clone()),
112    }
113}

115fn sub_in_solution_into_current_refs(
116    expr: &Expression,
117    solution: &HashMap<Name, Literal>,
118) -> Option<Expression> {
119    match expr {
120        Expression::Atomic(_, Atom::Reference(reference)) => {
121            let var_name = reference.name();
122            let value = solution.get(&var_name)?;
123            let value = if let Some(domain) = reference.resolved_domain() {
124                if domain.as_ref() == &GroundDomain::Bool {
125                    match value {
126                        Literal::Bool(x) => Literal::Bool(*x),
127                        Literal::Int(1) => Literal::Bool(true),
128                        Literal::Int(0) => Literal::Bool(false),
129                        _ => return None,
130                    }
131                } else {
132                    value.clone()
133                }
134            } else {
135                value.clone()
136            };
137
138            Some(Expression::Atomic(Metadata::new(), Atom::Literal(value)))
139        }
140        _ => Some(expr.clone()),
141    }
142}

144fn swap_from_solution_to_current_ref(expr: &Expression) -> Option<Expression> {

154fn add_represented_decision_values(solution: &mut HashMap<Name, Literal>, model: &Model) {
155    let symbols = model.symbols().clone();
156    let names = symbols.clone().into_iter().map(|x| x.0).collect_vec();
157    let representations = names
158        .into_iter()
159        .filter_map(|name| {
160            symbols
161                .representations_for(&name)
162                .map(|reprs| (name, reprs))
163        })
164        .filter_map(|(name, reprs)| {
165            if reprs.is_empty() {
166                return None;
167            }
168            if reprs.len() > 1 || reprs[0].len() != 1 {
169                return None;
170            }
171            Some((name, reprs[0][0].clone()))
172        })
173        .collect_vec();
174
175    let mut solution_btree = solution
176        .clone()
177        .into_iter()
178        .collect::<BTreeMap<Name, Literal>>();
179    for (name, representation) in representations {
180        let Ok(value) = representation.value_up(&solution_btree) else {
181            continue;
182        };
183        solution.insert(name.clone(), value.clone());
184        solution_btree.insert(name, value);
185    }
186
187    for (_, declaration) in symbols.iter_local() {
188        if declaration.reprs().is_empty() {
189            continue;
190        }
191        if let Ok(value) = try_up(declaration.clone(), solution) {
192            solution.insert(declaration.name().clone(), value);
193        }
194    }
195}

197fn extract_z3_version(full_version: &str) -> Result<&str, SolverError> {
198    match full_version.strip_prefix("Z3 ") {
199        Some(v) => v.split_whitespace().next().ok_or_else(|| {
200            SolverError::Runtime(format!(
201                "could not read Z3 runtime version from '{full_version}'"
202            ))
203        }),
204        None => full_version.split_whitespace().next().ok_or_else(|| {
205            SolverError::Runtime(format!(
206                "could not read Z3 runtime version from '{full_version}'"
207            ))
208        }),
209    }
210}
211
212fn ensure_supported_z3_runtime() -> Result<(), SolverError> {
213    let full_version = z3::full_version();
214    let runtime_version = extract_z3_version(full_version)?;
215    let runtime_version = Versioning::new(runtime_version).ok_or_else(|| {
216        SolverError::Runtime(format!(
217            "could not parse Z3 runtime version from '{full_version}'"
218        ))
219    })?;
220    let minimum_version =
221        Versioning::new(MINIMUM_Z3_VERSION).expect("minimum Z3 version should be valid");
222
223    if runtime_version < minimum_version {
224        return Err(SolverError::Runtime(format!(
225            "unsupported Z3 runtime version '{full_version}' (parsed as {runtime_version}); conjure-oxide requires Z3 >= {MINIMUM_Z3_VERSION}. This usually means an older system or precompiled Z3 was picked up at build time."
226        )));
227    }
228
229    Ok(())
230}

232impl Smt {
233    fn add_dominance_constraints_for_solution(
234        dominance_expression: Option<&Expression>,
235        dominance_model_template: Option<&Model>,
236        solver: &mut Solver,
237        store: &mut SymbolStore,
238        solution: &HashMap<Name, Literal>,
239    ) -> Result<(), SolverError> {
240        let Some(dominance_expression) = dominance_expression else {
241            return Ok(());
242        };
243
244        let Some(model_template) = dominance_model_template else {
245            return Ok(());
246        };
247
248        // Block future solutions dominated by the current solution:
249        // assert NOT dominance(current_solution, future_solution).
250        let rewritten_dominance = Expression::Not(
251            Metadata::new(),
252            Moo::new(
253                dominance_expression
254                    .rewrite(&|e| sub_in_solution_into_current_refs(&e, solution))
255                    .rewrite(&|e| swap_from_solution_to_current_ref(&e)),
256            ),
257        );
258
259        let mut dominance_model = model_template.clone();
260        dominance_model.replace_constraints(vec![]);
261        dominance_model.replace_clauses(vec![]);
262        dominance_model.dominance = None;
263        dominance_model.add_constraint(rewritten_dominance);
264
265        // Prefer the rule sets the model was built with, but fall back to resolving them for this
266        // adaptor's own family: an embedder is not obliged to populate the context, and rewriting
267        // a dominance constraint with no rules at all silently produces a model referring to
268        // variables that representations have since replaced. The Minion adaptor resolves for its
269        // family unconditionally for the same reason.
270        let rule_sets = {
271            let from_context = dominance_model.context.read().unwrap().rule_sets.clone();
272            if from_context.is_empty() {
273                get_rule_sets_for_solver_family(SolverFamily::Z3)
274            } else {
275                from_context
276            }
277        };
278        let rewritten =
279            rewrite_model_with_configured_rewriter(dominance_model, &rule_sets, current_rewriter())
280                .map_err(|e| {
281                    SolverError::Runtime(format!(
282                        "Failed to rewrite dominance constraint for SMT solving: {e}"
283                    ))
284                })?;
285
286        load_model_impl(
287            store,
288            solver,
289            &rewritten.symbols(),
290            rewritten.constraints().as_slice(),
291        )
292    }
293}

295impl SolverAdaptor for Smt {
296    fn solve(
297        &mut self,
298        callback: SolverCallback,
299        _: private::Internal,
300    ) -> Result<SolveSuccess, SolverError> {
301        let solver_send = self.solver_inst.synchronized();
302        let store_send = self.store.synchronized();
303        let dominance_expression = self.dominance_expression.clone();
304        let dominance_model_template = self.dominance_model_template.clone();
305        let solver_seed = self.solver_seed;
306        let budget = SolveTimeBudget::new(self.timeout);
307        let mut stats: SolverStats = Default::default();
308
309        // Apply config when getting solutions
310        let (search_status, final_z3_time) =
311            with_z3_config(&self.solver_cfg, move || -> Result<_, SolverError> {
312                let solver = solver_send.recover();
313                let mut solver_params = Params::new();
314                solver_params.set_u32("random_seed", solver_seed);
315                solver.set_params(&solver_params);
316                let mut final_z3_time: Option<f64> = None;
317                let mut found_solution = false;
318                let mut hook_error: Option<SolverError> = None;
319                let mut solutions = solver.into_solutions_with_statistics(
320                    store_send.recover(),
321                    true,
322                    budget,
323                    |solver, store, instance| {
324                        let mut dominance_solution = match instance.as_literals_map() {
325                            Ok(solution) => solution,
326                            Err(err) => {
327                                hook_error = Some(err);
328                                return false;
329                            }
330                        };
331                        if let Some(model_template) = dominance_model_template.as_ref() {
332                            add_represented_decision_values(
333                                &mut dominance_solution,
334                                model_template,
335                            );
336                        }
337
338                        match Smt::add_dominance_constraints_for_solution(
339                            dominance_expression.as_ref(),
340                            dominance_model_template.as_ref(),
341                            solver,
342                            store,
343                            &dominance_solution,
344                        ) {
345                            Ok(()) => true,
346                            Err(err) => {
347                                hook_error = Some(err);
348                                false
349                            }
350                        }
351                    },
352                );
353
354                let _solution_count = solutions
355                    .by_ref()
356                    .take_while(|(instance, z3_stats)| {
357                        found_solution = true;
358
359                        let time = z3_stats.value("time");
360                        if let Some(z3::StatisticsValue::Double(time)) = time {
361                            final_z3_time = Some(time);
362                        }
363
364                        (callback)(instance.as_literals_map().unwrap())
365                    })
366                    .count();
367
368                let terminal_result = solutions.terminal_result.take();
369                let unknown_reason = solutions.unknown_reason.take();
370                drop(solutions);
371                if let Some(err) = hook_error {
372                    return Err(err);
373                }
374
375                let search_status = search_status_from_terminal_result(
376                    found_solution,
377                    terminal_result,
378                    unknown_reason,
379                )?;
380                Ok((search_status, final_z3_time))
381            })?;
382
383        if let Some(time) = final_z3_time {
384            stats.solver_time_s = time;
385        }
386
387        Ok(SolveSuccess {
388            stats,
389            status: search_status,
390        })
391    }

401    fn load_model(&mut self, model: Model, _: private::Internal) -> Result<(), SolverError> {
402        // Fail fast if an older system or precompiled Z3 was linked in.
403        ensure_supported_z3_runtime()?;
404        self.dominance_expression = model.dominance.as_ref().map(|expr| match expr {
405            Expression::DominanceRelation(_, inner) => inner.as_ref().clone(),
406            _ => expr.clone(),
407        });
408        self.dominance_model_template = self.dominance_expression.as_ref().map(|_| model.clone());
409        load_model_impl(
410            &mut self.store,
411            &mut self.solver_inst,
412            &model.symbols(),
413            model.constraints().as_slice(),
414        )?;
415        Ok(())
416    }

473struct SolverStatsIterator<T, F> {
474    solver: Solver,
475    ast: T,
476    model_completion: bool,
477    budget: SolveTimeBudget,
478    on_solution: F,
479    done: bool,
480    terminal_result: Option<SatResult>,
481    unknown_reason: Option<String>,
482}
483
484impl<T, F> FusedIterator for SolverStatsIterator<T, F>
485where
486    T: Solvable,
487    F: FnMut(&mut Solver, &mut T, &T::ModelInstance) -> bool,
488{
489}
490
491impl<T, F> Iterator for SolverStatsIterator<T, F>
492where
493    T: Solvable,
494    F: FnMut(&mut Solver, &mut T, &T::ModelInstance) -> bool,
495{
496    type Item = (T::ModelInstance, Statistics);
497
498    fn next(&mut self) -> Option<Self::Item> {
499        if self.done {
500            return None;
501        }
502
503        if let Some(remaining) = self.budget.remaining() {
504            if remaining.is_zero() {
505                self.done = true;
506                self.terminal_result = Some(SatResult::Unknown);
507                self.unknown_reason = Some(String::from("cumulative solver timeout"));
508                return None;
509            }
510
511            let timeout_millis = remaining
512                .as_millis()
513                .saturating_add(u128::from(remaining.subsec_nanos() % 1_000_000 != 0));
514            let timeout_millis = u32::try_from(timeout_millis).unwrap_or(u32::MAX).max(1);
515            let mut timeout_params = Params::new();
516            timeout_params.set_u32("timeout", timeout_millis);
517            self.solver.set_params(&timeout_params);
518        }
519
520        match self.solver.check() {
521            SatResult::Sat => {
522                let stats = self.solver.get_statistics();
523                let model = self.solver.get_model()?;
524                let instance = self.ast.read_from_model(&model, self.model_completion)?;
525                let counterexample = self.ast.generate_constraint(&instance);
526
527                if !(self.on_solution)(&mut self.solver, &mut self.ast, &instance) {
528                    self.done = true;
529                    return None;
530                }
531
532                self.solver.assert(counterexample);
533                Some((instance, stats))
534            }
535            SatResult::Unsat => {
536                self.done = true;
537                self.terminal_result = Some(SatResult::Unsat);
538                None
539            }
540            SatResult::Unknown => {
541                self.done = true;
542                self.terminal_result = Some(SatResult::Unknown);
543                self.unknown_reason = self.solver.get_reason_unknown();
544                None
545            }
546        }
547    }
548}

550/// Converts the result that ended Z3's solution iterator into a search status.
551///
552/// `unknown` is not evidence that the model is unsatisfiable. In particular, Z3 returns it when a
553/// configured timeout expires or an in-progress check is interrupted. Treating iterator exhaustion
554/// as `NoSolutions` in that case lets an interrupted integration test pass with an empty solution
555/// file.
556fn search_status_from_terminal_result(
557    found_solution: bool,
558    terminal_result: Option<SatResult>,
559    unknown_reason: Option<String>,
560) -> Result<SearchStatus, SolverError> {
561    match terminal_result {
562        Some(SatResult::Unknown) => {
563            let reason = unknown_reason.unwrap_or_else(|| String::from("no reason given"));
564            let incomplete = if reason.to_ascii_lowercase().contains("timeout") {
565                SearchIncomplete::Timeout
566            } else {
567                SearchIncomplete::UserTerminated
568            };
569            Ok(SearchStatus::Incomplete(incomplete))
570        }
571        Some(SatResult::Unsat) => Ok(SearchStatus::Complete(if found_solution {
572            SearchComplete::HasSolutions
573        } else {
574            SearchComplete::NoSolutions
575        })),
576        // The consumer stopped requesting models, normally because its callback reached a finite
577        // solution limit. It has not established that there are no further solutions.
578        None => Ok(SearchStatus::Incomplete(SearchIncomplete::UserTerminated)),
579        Some(SatResult::Sat) => Err(SolverError::Runtime(String::from(
580            "Z3 solution iteration stopped after a satisfiable check without yielding its model",
581        ))),
582    }
583}

709                Moo::new(Expression::And(
710                    Metadata::new(),
711                    Moo::new(crate::matrix_expr![
712                        Expression::Imply(
713                            Metadata::new(),
714                            Moo::new(x_ref.clone()),
715                            Moo::new(Expression::FromSolution(
716                                Metadata::new(),
717                                Moo::new(Atom::Reference(Reference::new(
718                                    DeclarationPtr::new_find(x.clone(), Domain::bool(),)
719                                ))),
720                            )),
721                        ),
722                        Expression::Imply(
723                            Metadata::new(),
724                            Moo::new(y_ref.clone()),
725                            Moo::new(Expression::FromSolution(
726                                Metadata::new(),
727                                Moo::new(Atom::Reference(Reference::new(
728                                    DeclarationPtr::new_find(y.clone(), Domain::bool(),)
729                                ))),
730                            )),
731                        ),
732                        Expression::Or(
733                            Metadata::new(),
734                            Moo::new(crate::matrix_expr![
735                                Expression::And(
736                                    Metadata::new(),
737                                    Moo::new(crate::matrix_expr![
738                                        Expression::Not(Metadata::new(), Moo::new(x_ref.clone())),
739                                        Expression::FromSolution(
740                                            Metadata::new(),
741                                            Moo::new(Atom::Reference(Reference::new(
742                                                DeclarationPtr::new_find(x.clone(), Domain::bool()),
743                                            ))),
744                                        ),
745                                    ]),
746                                ),
747                                Expression::And(
748                                    Metadata::new(),
749                                    Moo::new(crate::matrix_expr![
750                                        Expression::Not(Metadata::new(), Moo::new(y_ref.clone())),
751                                        Expression::FromSolution(
752                                            Metadata::new(),
753                                            Moo::new(Atom::Reference(Reference::new(
754                                                DeclarationPtr::new_find(y.clone(), Domain::bool()),
755                                            ))),
756                                        ),
757                                    ]),
758                                ),
759                            ]),
760                        ),
761                    ]),
762                )),
763            ));
764
765        let mut smt = Smt::new(None);
766        smt.load_model(model, private::Internal)
767            .expect("SMT model should load");
768
769        let mut prior_solution = HashMap::new();
770        prior_solution.insert(x.clone(), Literal::Bool(true));
771        prior_solution.insert(y.clone(), Literal::Bool(false));
772
773        Smt::add_dominance_constraints_for_solution(
774            smt.dominance_expression.as_ref(),
775            smt.dominance_model_template.as_ref(),
776            &mut smt.solver_inst,
777            &mut smt.store,
778            &prior_solution,
779        )
780        .expect("dominance constraint should be assertable");
781
782        let x_ast = smt
783            .store
784            .get(&x)
785            .expect("x should exist in symbol store")
786            .1
787            .as_bool()
788            .expect("x should be a bool");
789        let y_ast = smt
790            .store
791            .get(&y)
792            .expect("y should exist in symbol store")
793            .1
794            .as_bool()
795            .expect("y should be a bool");
796
797        smt.solver_inst.push();
798        smt.solver_inst
799            .assert(x_ast.eq(z3::ast::Bool::from_bool(true)));
800        smt.solver_inst
801            .assert(y_ast.eq(z3::ast::Bool::from_bool(true)));
802        assert_eq!(smt.solver_inst.check(), SatResult::Unsat);
803        smt.solver_inst.pop(1);
804
805        smt.solver_inst.push();
806        smt.solver_inst
807            .assert(x_ast.eq(z3::ast::Bool::from_bool(false)));
808        smt.solver_inst
809            .assert(y_ast.eq(z3::ast::Bool::from_bool(false)));
810        assert_eq!(smt.solver_inst.check(), SatResult::Sat);
811        smt.solver_inst.pop(1);
812
813        smt.solver_inst.push();
814        smt.solver_inst
815            .assert(x_ast.eq(z3::ast::Bool::from_bool(true)));
816        smt.solver_inst
817            .assert(y_ast.eq(z3::ast::Bool::from_bool(false)));
818        assert_eq!(smt.solver_inst.check(), SatResult::Sat);
819        smt.solver_inst.pop(1);
