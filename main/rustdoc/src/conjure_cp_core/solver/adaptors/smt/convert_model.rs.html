1//! Functions for converting from a Conjure Model to assertions in Z3.
2//!
3//! Conversions are mostly 1-to-1 since any rewriting was done previously using rules.
4//! We recursively transform the AST bottom-up, returning the requested AST type (e.g. Bool, Int)
5//! or an error.
6//!
7//! "Dynamic" or "AST" is used to describe generic Z3 AST values. "Expression" means
8//! a Conjure Oxide [`Expression`] type.
9

24/// Converts the given variables and constraints to assertions by mutating the given model.
25///
26/// SMT does not use bounded domains the same way Conjure Oxide does; for example integers
27/// domains are unbounded. For this reason, additional assertions are made to keep these
28/// variables within their domains.
29pub fn load_model_impl(
30    store: &mut SymbolStore,
31    solver: &mut Solver,
32    symbols: &SymbolTable,
33    model: &[Expression],
34) -> SolverResult<()> {
35    for (name, decl) in symbols.iter_local() {
36        let Some(var) = decl.as_find() else {
37            // Ignore lettings, etc
38            continue;
39        };
40        if !decl.reprs().is_empty() {
41            // This variable has a structured representation; ignore the source variable.
42            continue;
43        }
44        if !symbols
45            .representations_for(name)
46            .is_none_or(|reps| reps.is_empty())
47        {
48            // This variable has representations; ignore it
49            continue;
50        }
51        let (sym, ast, restriction) = var_to_ast(name, &var, int_theory_for(decl))?;
52        store.insert(name.clone(), (decl.resolved_domain().unwrap(), ast, sym));
53        // findAux variables are part of the model, but a different assignment of one is not a
54        // different solution.
55        if !decl.is_find_auxiliary() {
56            store.block_solutions_on(name.clone());
57        }
58        solver.assert(restriction);
59    }
60    let mut lettings = HashSet::new();
61    for expr in model.iter() {
62        let bool: Bool = expr_to_ast(store, &mut lettings, expr)?;
63        solver.assert(bool);
64    }
65    Ok(())
66}

68/// The integer theory a declaration's variable should be created in.
69///
70/// The `lia` and `bv` representations do not decompose the variable -- they leave one variable of
71/// the same domain -- so the choice they record is read back here, off the declaration they were
72/// initialised for. Anything with no such representation gets the default theory.
73fn int_theory_for(decl: &DeclarationPtr) -> IntTheory {
74    let source = decl.source().clone();
75    let Some(source) = source else {
76        return IntTheory::default();
77    };
78    source
79        .reprs()
80        .iter()
81        .find_map(|(_, state)| IntTheory::from_repr_short_name(state.rule().short_name()))
82        .unwrap_or_default()
83}

85/// Returns the AST representation of the variable as well as a boolean assertion which restricts
86/// it to the input variable's domain since most Z3 sorts are unbounded.
87fn var_to_ast(
88    name: &Name,
89    var: &DecisionVariable,
90    ints: IntTheory,
91) -> SolverResult<(Symbol, Dynamic, Bool)> {
92    let sym = name_to_symbol(name)?;
93    let dom = var
94        .domain_of()
95        .resolve()
96        .unwrap_or_else(|e| bug!("could not resolve domain for {name}: {e}"));
97    let (sort, restrict_fn) = domain_to_sort(dom.as_ref(), ints)?;
98    let new_const = Dynamic::new_const(sym.clone(), &sort);
99
100    let restriction = (restrict_fn)(&new_const);
101    Ok((sym, new_const, restriction))
102}

104/// Converts a Conjure Oxide Expression to an AST node for Z3.
105/// The generic type parameter lets us cast the result to a specific return type.
106///
107/// `lettings` are the boolean value lettings currently being expanded. Rewriting leaves a use as
108/// a name, so a constraint can still be a reference whose body is a formula. That formula is
109/// converted in place. A cycle among those names is rejected.
110fn expr_to_ast<Out>(
111    store: &SymbolStore,
112    lettings: &mut HashSet<crate::ast::serde::ObjId>,
113    expr: &Expression,
114) -> SolverResult<Out>
115where
116    Out: TryFrom<Dynamic, Error: std::fmt::Display>,
117{
118    if let Expression::Atomic(_, Atom::Reference(reference)) = expr
119        && let Some(body) = reference.boolean_formula_letting_body()
120    {
121        let id = reference.id();
122        if !lettings.insert(id.clone()) {
123            return Err(SolverError::ModelInvalid(format!(
124                "cyclic value letting '{}' in a Z3 constraint",
125                reference.name()
126            )));
127        }
128        let ast: SolverResult<Dynamic> = expr_to_ast(store, lettings, &body);
129        lettings.remove(&id);
130        return ast?.try_into().map_err(|err| {
131            SolverError::ModelInvalid(format!(
132                "expression has incorrect type for conversion: {err}"
133            ))
134        });
135    }
136
137    let ast = match expr {
138        Expression::Atomic(_, atom) => atom_to_ast(store, atom),
139
140        // Equality is part of the SMT core theory (anything can be compared)
141        // Some types (matrices, sets) must be compared element-wise, since the SMT solver can
142        //  always extend them to make them technically eq/neq in "SMT land", but not in "Oxide land"
143        // This is done during rewriting by rules which unwrap Eq/Neqs over these types
144        //
145        // The two sides can be in different integer theories -- that is what a `lia`/`bv`
146        // channelling constraint looks like -- so they are brought into a common one first.
147        Expression::Eq(_, a, b) => {
148            let [a, b] = aligned_operands(store, lettings, [a.as_ref(), b.as_ref()])?;
149            Ok(a.eq(b).into())
150        }
151        Expression::Neq(_, a, b) => {
152            let [a, b] = aligned_operands(store, lettings, [a.as_ref(), b.as_ref()])?;
153            Ok(a.ne(b).into())
154        }
155
156        // === Boolean Expressions ===
157        Expression::Not(_, a) => unary_op(store, lettings, a, |a: Bool| a.not()),
158        Expression::Imply(_, a, b) => {
159            binary_op(store, lettings, a, b, |a: Bool, b: Bool| a.implies(b))
160        }
161        Expression::Iff(_, a, b) => binary_op(store, lettings, a, b, |a: Bool, b: Bool| a.iff(b)),
162        Expression::Or(_, a) => list_op(store, lettings, a, |asts: &[Bool]| Bool::or(asts)),
163        Expression::And(_, a) => list_op(store, lettings, a, |asts: &[Bool]| Bool::and(asts)),
164
165        // === Expressions over integers ===
166        // Which theory each of these lands in is decided by the operands, not by a setting: a
167        // variable represented as a bit-vector drags the operation into the bit-vector theory,
168        // and everything else in the operation is converted to match.
169        Expression::Neg(_, a) => int_unary_op(
170            store,
171            lettings,
172            a,
173            |a: Int| a.unary_minus(),
174            |a: BV| a.bvneg(),
175        ),
176        Expression::Abs(_, a) => int_unary_op(
177            store,
178            lettings,
179            a,
180            |a: Int| a.lt(Int::from(0)).ite(&a.unary_minus(), &a),
181            |a: BV| a.bvslt(BV::from_i64(0, BV_SIZE)).ite(&a.bvneg(), &a),
182        ),
183        // A Boolean has no theory of its own to follow, so this always produces a mathematical
184        // integer; an enclosing bit-vector operation converts it along with everything else.
185        Expression::ToInt(_, a) => unary_op(store, lettings, a, |a: Bool| {
186            a.ite(&Int::from(1), &Int::from(0))
187        }),
188
189        // Essence division floors, which is what Z3's integer `div` and `mod` already do. The
190        // bit-vector theory offers truncating division instead (`bvsdiv` rounds toward zero, and
191        // `bvsrem` takes the dividend's sign), so the two theories would disagree on negative
192        // operands unless the floored forms are built explicitly.
193        Expression::SafeDiv(_, a, b) => int_binary_op(
194            store,
195            lettings,
196            a,
197            b,
198            |a: Int, b: Int| a.div(b),
199            bv_floor_div,
200        ),
201        Expression::SafeMod(_, a, b) => {
202            // `bvsmod` already takes the divisor's sign, which is the floored remainder.
203            int_binary_op(
204                store,
205                lettings,
206                a,
207                b,
208                |a: Int, b: Int| a.modulo(b),
209                |a: BV, b: BV| a.bvsmod(b),
210            )
211        }
212        Expression::Gt(_, a, b) => int_binary_op(
213            store,
214            lettings,
215            a,
216            b,
217            |a: Int, b: Int| a.gt(b),
218            |a: BV, b: BV| a.bvsgt(b),
219        ),
220        Expression::Lt(_, a, b) => int_binary_op(
221            store,
222            lettings,
223            a,
224            b,
225            |a: Int, b: Int| a.lt(b),
226            |a: BV, b: BV| a.bvslt(b),
227        ),
228        Expression::Geq(_, a, b) => int_binary_op(
229            store,
230            lettings,
231            a,
232            b,
233            |a: Int, b: Int| a.ge(b),
234            |a: BV, b: BV| a.bvsge(b),
235        ),
236        Expression::Leq(_, a, b) => int_binary_op(
237            store,
238            lettings,
239            a,
240            b,
241            |a: Int, b: Int| a.le(b),
242            |a: BV, b: BV| a.bvsle(b),
243        ),
244        Expression::PairwiseSum(_, a, b) => int_binary_op(
245            store,
246            lettings,
247            a,
248            b,
249            |a: Int, b: Int| Int::add(&[a, b]),
250            |a: BV, b: BV| a.bvadd(b),
251        ),
252        Expression::PairwiseProduct(_, a, b) => int_binary_op(
253            store,
254            lettings,
255            a,
256            b,
257            |a: Int, b: Int| Int::mul(&[a, b]),
258            |a: BV, b: BV| a.bvmul(b),
259        ),
260
261        // Bit-vectors have no n-ary addition, so the list is folded pairwise here rather than
262        // being rewritten into `PairwiseSum` chains beforehand.
263        Expression::Sum(_, a) => int_list_op(
264            store,
265            lettings,
266            a,
267            |asts: &[Int]| Int::add(asts),
268            |asts: &[BV]| fold_bvs(asts, |a, b| a.bvadd(b)),
269        ),
270        Expression::Product(_, a) => int_list_op(
271            store,
272            lettings,
273            a,
274            |asts: &[Int]| Int::mul(asts),
275            |asts: &[BV]| fold_bvs(asts, |a, b| a.bvmul(b)),
276        ),
277
278        // Exponentiation has no bit-vector counterpart in Z3, so this one stays integer-only.
279        Expression::SafePow(_, a, b) => {
280            binary_op(store, lettings, a, b, |a: Int, b: Int| a.power(b))
281        }
282
283        // === Expressions involving matrices ===
284        // A matrix literal becomes a Z3 array, which is what makes indexing one by a variable
285        // work. Components-represented matrices are rebuilt as literal lists during rewriting, so
286        // this is the path a variable index into them takes.
287        Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elements, index_domain)) => {
288            matrix_literal_to_array(store, lettings, elements, index_domain)
289        }
290        Expression::SafeIndex(_, m, idxs) => {
291            let arr: Dynamic = expr_to_ast(store, lettings, m)?;
292            slice_op(store, lettings, idxs, move |idxs: &[Dynamic]| {
293                idxs.iter().fold(arr, |cur_arr, idx| {
294                    // Array indices are always in the integer theory, whatever the elements are.
295                    let idx = coerce_to(idx.clone(), Some(IntTheory::Lia));
296                    cur_arr.as_array().unwrap().select(&idx)
297                })
298            })
299        }
300        // `allDifferent` itself never reaches here: the rules turn it into either this native
301        // form or an explicit encoding, and which one is a modelling choice.
302        Expression::SmtDistinct(_, a) => {
303            let elements = list_elements(a)?;
304            let refs: Vec<&Expression> = elements.iter().collect();
305            let asts = aligned_operand_vec(store, lettings, &refs)?;
306            Ok(Dynamic::distinct(&asts).into())
307        }
308
309        // === Expressions involving sets
310        Expression::In(_, x, s) => {
311            binary_op(store, lettings, x, s, |x: Dynamic, s: Set| s.member(&x))
312        }
313
314        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
315            "expression type not implemented: {expr}"
316        ))),
317    }?;
318
319    ast.try_into().map_err(|err| {
320        SolverError::ModelInvalid(format!(
321            "expression has incorrect type for conversion: {err}"
322        ))
323    })
324}

326/// Builds a Z3 array holding the elements of a matrix literal at their index-domain positions.
327///
328/// Entries outside the literal keep the array's default value; the model only ever reads positions
329/// the index domain allows, which `SafeIndex` has already restricted by this point.
330fn matrix_literal_to_array(
331    store: &SymbolStore,
332    lettings: &mut HashSet<crate::ast::serde::ObjId>,
333    elements: &[Expression],
334    index_domain: &DomainPtr,
335) -> SolverResult<Dynamic> {
336    let asts: Vec<Dynamic> = elements
337        .iter()
338        .map(|element| expr_to_ast(store, lettings, element))
339        .collect::<SolverResult<_>>()?;
340
341    let Some(_) = asts.first() else {
342        return Err(SolverError::ModelFeatureNotImplemented(
343            "cannot build an array from an empty matrix literal".to_owned(),
344        ));
345    };
346
347    // Elements can come from declarations represented in different theories; give the array one
348    // element sort and bring them all to it, exactly as an operation over them would.
349    let target = asts
350        .iter()
351        .filter_map(int_theory_of)
352        .fold(None, |acc, theory| match (acc, theory) {
353            (Some(IntTheory::Bv), _) | (_, IntTheory::Bv) => Some(IntTheory::Bv),
354            _ => Some(IntTheory::Lia),
355        });
356    let asts: Vec<Dynamic> = asts.into_iter().map(|ast| coerce_to(ast, target)).collect();
357
358    // A plain list is written with an implied, unbounded index domain (`int(1..)`), which cannot
359    // be enumerated; its positions are simply one per element.
360    let indices = index_domain
361        .resolve()
362        .ok()
363        .and_then(|domain| domain_to_ast_vec(IntTheory::Lia, domain.as_ref()).ok())
364        .filter(|indices| indices.len() == asts.len())
365        .unwrap_or_else(|| {
366            (1..=asts.len())
367                .map(|position| Int::from(position as i64).into())
368                .collect()
369        });
370
371    let index_sort = indices
372        .first()
373        .map(|index| index.get_sort())
374        .unwrap_or_else(Sort::int);
375
376    let mut array = Array::const_array(&index_sort, &asts[0]);
377    for (index, value) in indices.iter().zip(asts.iter()) {
378        array = array.store(index, value);
379    }
380    Ok(array.into())
381}

383/// Floored bit-vector division, matching Essence's `/` and Z3's integer `div`.
384///
385/// `bvsdiv` truncates toward zero, so it is one too high whenever the operands have opposite signs
386/// and the division is inexact; `bvsmod` takes the divisor's sign, so a non-zero remainder is
387/// exactly the case needing the correction.
388fn bv_floor_div(a: BV, b: BV) -> BV {
389    let zero = BV::from_i64(0, BV_SIZE);
390    let one = BV::from_i64(1, BV_SIZE);
391    let truncated = a.bvsdiv(&b);
392    let remainder = a.bvsrem(&b);
393    let signs_differ = a.bvslt(&zero).xor(b.bvslt(&zero));
394    let inexact = remainder.eq(&zero).not();
395    Bool::and(&[signs_differ, inexact]).ite(&truncated.bvsub(&one), &truncated)
396}

398/// Folds a non-empty slice of bit-vectors with a binary operation.
399fn fold_bvs(asts: &[BV], op: impl Fn(BV, BV) -> BV) -> BV {
400    asts.iter()
401        .cloned()
402        .reduce(op)
403        .unwrap_or_else(|| BV::from_i64(0, BV_SIZE))
404}

406/// The integer theory an already-converted AST is in, if it is an integer at all.
407fn int_theory_of(ast: &Dynamic) -> Option<IntTheory> {
408    match ast.sort_kind() {
409        SortKind::Int => Some(IntTheory::Lia),
410        SortKind::Bv => Some(IntTheory::Bv),
411        _ => None,
412    }
413}

415/// Converts an integer AST into `theory`, leaving non-integers and already-matching ASTs alone.
416///
417/// This is what makes a `lia`/`bv` channelling constraint expressible, and what lets an operation
418/// take operands whose declarations were represented in different theories.
419fn coerce_to(ast: Dynamic, theory: Option<IntTheory>) -> Dynamic {
420    match (theory, ast.sort_kind()) {
421        (Some(IntTheory::Bv), SortKind::Int) => {
422            BV::from_int(&ast.as_int().expect("just checked the sort"), BV_SIZE).into()
423        }
424        (Some(IntTheory::Lia), SortKind::Bv) => {
425            Int::from_bv(&ast.as_bv().expect("just checked the sort"), true).into()
426        }
427        _ => ast,
428    }
429}

431/// Converts every operand and brings any integers among them into a common theory.
432///
433/// Bit-vectors win: converting a mathematical integer to a machine word is exact for values that
434/// fit, whereas the reverse would have to pick a signedness for values that may have wrapped.
435fn aligned_operand_vec(
436    store: &SymbolStore,
437    lettings: &mut HashSet<crate::ast::serde::ObjId>,
438    exprs: &[&Expression],
439) -> SolverResult<Vec<Dynamic>> {
440    let asts: Vec<Dynamic> = exprs
441        .iter()
442        .map(|expr| expr_to_ast(store, lettings, expr))
443        .collect::<SolverResult<_>>()?;
444
445    let target = asts
446        .iter()
447        .filter_map(int_theory_of)
448        .fold(None, |acc, theory| match (acc, theory) {
449            (Some(IntTheory::Bv), _) | (_, IntTheory::Bv) => Some(IntTheory::Bv),
450            _ => Some(IntTheory::Lia),
451        });
452
453    Ok(asts.into_iter().map(|ast| coerce_to(ast, target)).collect())
454}

456/// [`aligned_operand_vec`] for a fixed number of operands.
457fn aligned_operands<const N: usize>(
458    store: &SymbolStore,
459    lettings: &mut HashSet<crate::ast::serde::ObjId>,
460    exprs: [&Expression; N],
461) -> SolverResult<[Dynamic; N]> {
462    let refs: Vec<&Expression> = exprs.to_vec();
463    let asts = aligned_operand_vec(store, lettings, &refs)?;
464    asts.try_into().map_err(|_| {
465        SolverError::ModelInvalid("wrong number of operands after alignment".to_owned())
466    })
467}

469/// Applies whichever of the two operations matches the theory the operand landed in.
470fn int_unary_op<OutInt, OutBv>(
471    store: &SymbolStore,
472    lettings: &mut HashSet<crate::ast::serde::ObjId>,
473    a: &Expression,
474    lia: impl FnOnce(Int) -> OutInt,
475    bv: impl FnOnce(BV) -> OutBv,
476) -> SolverResult<Dynamic>
477where
478    OutInt: Into<Dynamic>,
479    OutBv: Into<Dynamic>,
480{
481    let ast: Dynamic = expr_to_ast(store, lettings, a)?;
482    match int_theory_of(&ast) {
483        Some(IntTheory::Bv) => Ok((bv)(ast.as_bv().expect("just checked the sort")).into()),
484        _ => Ok((lia)(ast.as_int().ok_or_else(|| {
485            SolverError::ModelInvalid(format!("expected an integer operand: {a}"))
486        })?)
487        .into()),
488    }
489}

491/// Applies whichever of the two operations matches the theory the operands were aligned into.
492fn int_binary_op<OutInt, OutBv>(
493    store: &SymbolStore,
494    lettings: &mut HashSet<crate::ast::serde::ObjId>,
495    a: &Expression,
496    b: &Expression,
497    lia: impl FnOnce(Int, Int) -> OutInt,
498    bv: impl FnOnce(BV, BV) -> OutBv,
499) -> SolverResult<Dynamic>
500where
501    OutInt: Into<Dynamic>,
502    OutBv: Into<Dynamic>,
503{
504    let [a_ast, b_ast] = aligned_operands(store, lettings, [a, b])?;
505    match int_theory_of(&a_ast) {
506        Some(IntTheory::Bv) => Ok((bv)(
507            a_ast.as_bv().expect("just checked the sort"),
508            b_ast.as_bv().expect("aligned with the first operand"),
509        )
510        .into()),
511        _ => {
512            let cast = |ast: &Dynamic, expr: &Expression| {
513                ast.as_int().ok_or_else(|| {
514                    SolverError::ModelInvalid(format!("expected an integer operand: {expr}"))
515                })
516            };
517            Ok((lia)(cast(&a_ast, a)?, cast(&b_ast, b)?).into())
518        }
519    }
520}

522/// Applies whichever of the two operations matches the theory the list was aligned into.
523fn int_list_op<OutInt, OutBv>(
524    store: &SymbolStore,
525    lettings: &mut HashSet<crate::ast::serde::ObjId>,
526    expr: &Expression,
527    lia: impl FnOnce(&[Int]) -> OutInt,
528    bv: impl FnOnce(&[BV]) -> OutBv,
529) -> SolverResult<Dynamic>
530where
531    OutInt: Into<Dynamic>,
532    OutBv: Into<Dynamic>,
533{
534    let elements = list_elements(expr)?;
535    let refs: Vec<&Expression> = elements.iter().collect();
536    let asts = aligned_operand_vec(store, lettings, &refs)?;
537
538    if asts
539        .iter()
540        .any(|ast| matches!(int_theory_of(ast), Some(IntTheory::Bv)))
541    {
542        let bvs: Vec<BV> = asts
543            .iter()
544            .map(|ast| ast.as_bv().expect("the whole list was aligned"))
545            .collect();
546        return Ok((bv)(&bvs).into());
547    }
548
549    let ints: Vec<Int> = asts
550        .iter()
551        .map(|ast| {
552            ast.as_int().ok_or_else(|| {
553                SolverError::ModelInvalid(format!("expected integer operands: {expr}"))
554            })
555        })
556        .collect::<SolverResult<_>>()?;
557    Ok((lia)(&ints).into())
558}

560/// Interprets an expression as an AST and returns the result of the given operation over it.
561fn unary_op<A, Out>(
562    store: &SymbolStore,
563    lettings: &mut HashSet<crate::ast::serde::ObjId>,
564    a: &Expression,
565    op: impl FnOnce(A) -> Out,
566) -> SolverResult<Dynamic>
567where
568    A: TryFrom<Dynamic, Error: std::fmt::Display>,
569    Out: Into<Dynamic>,
570{
571    let a_ast: A = expr_to_ast(store, lettings, a)?;
572    Ok((op)(a_ast).into())
573}

575/// Interprets two expressions as ASTs and returns the result of the given operation over them.
576fn binary_op<A, B, Out>(
577    store: &SymbolStore,
578    lettings: &mut HashSet<crate::ast::serde::ObjId>,
579    a: &Expression,
580    b: &Expression,
581    op: impl FnOnce(A, B) -> Out,
582) -> SolverResult<Dynamic>
583where
584    A: TryFrom<Dynamic, Error: std::fmt::Display>,
585    B: TryFrom<Dynamic, Error: std::fmt::Display>,
586    Out: Into<Dynamic>,
587{
588    let a_ast: A = expr_to_ast(store, lettings, a)?;
589    let b_ast: B = expr_to_ast(store, lettings, b)?;
590    Ok((op)(a_ast, b_ast).into())
591}

593/// Transforms a list expression into separate ASTs and returns the result of the given operation over them.
594fn list_op<A, Out>(
595    store: &SymbolStore,
596    lettings: &mut HashSet<crate::ast::serde::ObjId>,
597    expr: &Expression,
598    op: impl FnOnce(&[A]) -> Out,
599) -> SolverResult<Dynamic>
600where
601    A: TryFrom<Dynamic, Error: std::fmt::Display>,
602    Out: Into<Dynamic>,
603{
604    let exprs = list_elements(expr)?;
605
606    slice_op(store, lettings, &exprs, op)
607}

609/// Extracts the elements of a list-like expression.
610///
611/// Besides literal lists, this also supports slice expressions such as `x[..]`
612/// by expanding them to explicit indexing expressions.
613/// TODO: Consider moving this out of the smt solver adaptor, it'll be more generally useful.
614fn list_elements(expr: &Expression) -> SolverResult<Vec<Expression>> {
615    if let Some(exprs) = expr.clone().into_list() {
616        return Ok(exprs);
617    }
618
619    // `unwrap_list` only recognises a matrix whose index domain is the implied `int(1..)`. The
620    // operations that reach here -- sums, conjunctions, `distinct` -- care about the elements and
621    // not where they are indexed from, and a matrix rebuilt from its components keeps the index
622    // domain it was declared with, so take those elements too.
623    if let Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elements, _)) = expr {
624        return Ok(elements.clone());
625    }
626
627    let Expression::SafeSlice(_, subject, indices) = expr else {
628        return Err(SolverError::ModelFeatureNotImplemented(format!(
629            "inner expression must be a list: {expr}"
630        )));
631    };
632
633    let subject_domain = subject
634        .domain_of()
635        .and_then(|d| d.resolve().ok())
636        .ok_or_else(|| {
637            SolverError::ModelFeatureNotImplemented(format!(
638                "cannot resolve domain of sliced expression: {expr}"
639            ))
640        })?;
641
642    let GroundDomain::Matrix(_, index_domains, _) = subject_domain.as_ref() else {
643        return Err(SolverError::ModelFeatureNotImplemented(format!(
644            "slice subject must have matrix domain: {expr}"
645        )));
646    };
647
648    if index_domains.len() != indices.len() {
649        return Err(SolverError::ModelInvalid(format!(
650            "slice has wrong number of indices: {expr}"
651        )));
652    }
653
654    let index_options: SolverResult<Vec<Vec<Expression>>> = indices
655        .iter()
656        .zip(index_domains.iter())
657        .map(|(idx, dom)| match idx {
658            Some(idx_expr) => Ok(vec![idx_expr.clone()]),
659            None => {
660                let vals = dom.values().map_err(|_| {
661                    SolverError::ModelFeatureNotImplemented(format!(
662                        "slice index domain is not finite/enumerable: {dom}"
663                    ))
664                })?;
665                Ok(vals
666                    .map(|lit| Expression::Atomic(Metadata::new(), Atom::Literal(lit)))
667                    .collect_vec())
668            }
669        })
670        .collect();
671
672    let index_options = index_options?;
673
674    Ok(index_options
675        .into_iter()
676        .multi_cartesian_product()
677        .map(|concrete_idxs| Expression::SafeIndex(Metadata::new(), subject.clone(), concrete_idxs))
678        .collect())
679}

681/// Transforms a slice of expressions into ASTs and returns the result of the given operation over it.
682fn slice_op<A, Out>(
683    store: &SymbolStore,
684    lettings: &mut HashSet<crate::ast::serde::ObjId>,
685    exprs: &[Expression],
686    op: impl FnOnce(&[A]) -> Out,
687) -> SolverResult<Dynamic>
688where
689    A: TryFrom<Dynamic, Error: std::fmt::Display>,
690    Out: Into<Dynamic>,
691{
692    // Result implements FromIter, collecting into either the full collection or an error
693    let asts_res: SolverResult<Vec<_>> = exprs
694        .iter()
695        .map(|e| expr_to_ast(store, lettings, e))
696        .collect();
697    let asts = asts_res?;
698
699    Ok((op)(asts.as_slice()).into())
700}
