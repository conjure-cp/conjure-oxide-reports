1//! Functions for converting from a Conjure Model to assertions in Z3.
2//!
3//! Conversions are mostly 1-to-1 since any rewriting was done previously using rules.
4//! We recursively transform the AST bottom-up, returning the requested AST type (e.g. Bool, Int)
5//! or an error.
6//!
7//! "Dynamic" or "AST" is used to describe generic Z3 AST values. "Expression" means
8//! a Conjure Oxide [`Expression`] type.
9

24/// Converts the given variables and constraints to assertions by mutating the given model.
25///
26/// SMT does not use bounded domains the same way Conjure Oxide does; for example integers
27/// domains are unbounded. For this reason, additional assertions are made to keep these
28/// variables within their domains.
29pub fn load_model_impl(
30    store: &mut SymbolStore,
31    solver: &mut Solver,
32    symbols: &SymbolTable,
33    model: &[Expression],
34) -> SolverResult<()> {
35    for (name, decl) in symbols.clone().into_iter_local() {
36        let Some(var) = decl.as_find() else {
37            // Ignore lettings, etc
38            continue;
39        };
40        if !decl.reprs().is_empty() {
41            // This variable has a structured representation; ignore the source variable.
42            continue;
43        }
44        if !symbols
45            .representations_for(&name)
46            .is_none_or(|reps| reps.is_empty())
47        {
48            // This variable has representations; ignore it
49            continue;
50        }
51        let (sym, ast, restriction) = var_to_ast(&name, &var, int_theory_for(&decl))?;
52        store.insert(name, (decl.resolved_domain().unwrap(), ast, sym));
53        solver.assert(restriction);
54    }
55    for expr in model.iter() {
56        let bool: Bool = expr_to_ast(store, expr)?;
57        solver.assert(bool);
58    }
59    Ok(())
60}

62/// The integer theory a declaration's variable should be created in.
63///
64/// The `lia` and `bv` representations do not decompose the variable -- they leave one variable of
65/// the same domain -- so the choice they record is read back here, off the declaration they were
66/// initialised for. Anything with no such representation gets the default theory.
67fn int_theory_for(decl: &DeclarationPtr) -> IntTheory {
68    let source = decl.source().clone();
69    let Some(source) = source else {
70        return IntTheory::default();
71    };
72    source
73        .reprs()
74        .iter()
75        .find_map(|(_, state)| IntTheory::from_repr_short_name(state.rule().short_name()))
76        .unwrap_or_default()
77}

79/// Returns the AST representation of the variable as well as a boolean assertion which restricts
80/// it to the input variable's domain since most Z3 sorts are unbounded.
81fn var_to_ast(
82    name: &Name,
83    var: &DecisionVariable,
84    ints: IntTheory,
85) -> SolverResult<(Symbol, Dynamic, Bool)> {
86    let sym = name_to_symbol(name)?;
87    let dom = var
88        .domain_of()
89        .resolve()
90        .unwrap_or_else(|e| bug!("could not resolve domain for {name}: {e}"));
91    let (sort, restrict_fn) = domain_to_sort(dom.as_ref(), ints)?;
92    let new_const = Dynamic::new_const(sym.clone(), &sort);
93
94    let restriction = (restrict_fn)(&new_const);
95    Ok((sym, new_const, restriction))
96}

98/// Converts a Conjure Oxide Expression to an AST node for Z3.
99/// The generic type parameter lets us cast the result to a specific return type.
100fn expr_to_ast<Out>(store: &SymbolStore, expr: &Expression) -> SolverResult<Out>
101where
102    Out: TryFrom<Dynamic, Error: std::fmt::Display>,
103{
104    let ast = match expr {
105        Expression::Atomic(_, atom) => atom_to_ast(store, atom),
106
107        // Equality is part of the SMT core theory (anything can be compared)
108        // Some types (matrices, sets) must be compared element-wise, since the SMT solver can
109        //  always extend them to make them technically eq/neq in "SMT land", but not in "Oxide land"
110        // This is done during rewriting by rules which unwrap Eq/Neqs over these types
111        //
112        // The two sides can be in different integer theories -- that is what a `lia`/`bv`
113        // channelling constraint looks like -- so they are brought into a common one first.
114        Expression::Eq(_, a, b) => {
115            let [a, b] = aligned_operands(store, [a.as_ref(), b.as_ref()])?;
116            Ok(a.eq(b).into())
117        }
118        Expression::Neq(_, a, b) => {
119            let [a, b] = aligned_operands(store, [a.as_ref(), b.as_ref()])?;
120            Ok(a.ne(b).into())
121        }
122
123        // === Boolean Expressions ===
124        Expression::Not(_, a) => unary_op(store, a, |a: Bool| a.not()),
125        Expression::Imply(_, a, b) => binary_op(store, a, b, |a: Bool, b: Bool| a.implies(b)),
126        Expression::Iff(_, a, b) => binary_op(store, a, b, |a: Bool, b: Bool| a.iff(b)),
127        Expression::Or(_, a) => list_op(store, a, |asts: &[Bool]| Bool::or(asts)),
128        Expression::And(_, a) => list_op(store, a, |asts: &[Bool]| Bool::and(asts)),
129
130        // === Expressions over integers ===
131        // Which theory each of these lands in is decided by the operands, not by a setting: a
132        // variable represented as a bit-vector drags the operation into the bit-vector theory,
133        // and everything else in the operation is converted to match.
134        Expression::Neg(_, a) => {
135            int_unary_op(store, a, |a: Int| a.unary_minus(), |a: BV| a.bvneg())
136        }
137        Expression::Abs(_, a) => int_unary_op(
138            store,
139            a,
140            |a: Int| a.lt(Int::from(0)).ite(&a.unary_minus(), &a),
141            |a: BV| a.bvslt(BV::from_i64(0, BV_SIZE)).ite(&a.bvneg(), &a),
142        ),
143        // A Boolean has no theory of its own to follow, so this always produces a mathematical
144        // integer; an enclosing bit-vector operation converts it along with everything else.
145        Expression::ToInt(_, a) => {
146            unary_op(store, a, |a: Bool| a.ite(&Int::from(1), &Int::from(0)))
147        }
148
149        // Essence division floors, which is what Z3's integer `div` and `mod` already do. The
150        // bit-vector theory offers truncating division instead (`bvsdiv` rounds toward zero, and
151        // `bvsrem` takes the dividend's sign), so the two theories would disagree on negative
152        // operands unless the floored forms are built explicitly.
153        Expression::SafeDiv(_, a, b) => {
154            int_binary_op(store, a, b, |a: Int, b: Int| a.div(b), bv_floor_div)
155        }
156        Expression::SafeMod(_, a, b) => {
157            // `bvsmod` already takes the divisor's sign, which is the floored remainder.
158            int_binary_op(
159                store,
160                a,
161                b,
162                |a: Int, b: Int| a.modulo(b),
163                |a: BV, b: BV| a.bvsmod(b),
164            )
165        }
166        Expression::Gt(_, a, b) => int_binary_op(
167            store,
168            a,
169            b,
170            |a: Int, b: Int| a.gt(b),
171            |a: BV, b: BV| a.bvsgt(b),
172        ),
173        Expression::Lt(_, a, b) => int_binary_op(
174            store,
175            a,
176            b,
177            |a: Int, b: Int| a.lt(b),
178            |a: BV, b: BV| a.bvslt(b),
179        ),
180        Expression::Geq(_, a, b) => int_binary_op(
181            store,
182            a,
183            b,
184            |a: Int, b: Int| a.ge(b),
185            |a: BV, b: BV| a.bvsge(b),
186        ),
187        Expression::Leq(_, a, b) => int_binary_op(
188            store,
189            a,
190            b,
191            |a: Int, b: Int| a.le(b),
192            |a: BV, b: BV| a.bvsle(b),
193        ),
194        Expression::PairwiseSum(_, a, b) => int_binary_op(
195            store,
196            a,
197            b,
198            |a: Int, b: Int| Int::add(&[a, b]),
199            |a: BV, b: BV| a.bvadd(b),
200        ),
201        Expression::PairwiseProduct(_, a, b) => int_binary_op(
202            store,
203            a,
204            b,
205            |a: Int, b: Int| Int::mul(&[a, b]),
206            |a: BV, b: BV| a.bvmul(b),
207        ),
208
209        // Bit-vectors have no n-ary addition, so the list is folded pairwise here rather than
210        // being rewritten into `PairwiseSum` chains beforehand.
211        Expression::Sum(_, a) => int_list_op(
212            store,
213            a,
214            |asts: &[Int]| Int::add(asts),
215            |asts: &[BV]| fold_bvs(asts, |a, b| a.bvadd(b)),
216        ),
217        Expression::Product(_, a) => int_list_op(
218            store,
219            a,
220            |asts: &[Int]| Int::mul(asts),
221            |asts: &[BV]| fold_bvs(asts, |a, b| a.bvmul(b)),
222        ),
223
224        // Exponentiation has no bit-vector counterpart in Z3, so this one stays integer-only.
225        Expression::SafePow(_, a, b) => binary_op(store, a, b, |a: Int, b: Int| a.power(b)),
226
227        // === Expressions involving matrices ===
228        // A matrix literal becomes a Z3 array, which is what makes indexing one by a variable
229        // work. Components-represented matrices are rebuilt as literal lists during rewriting, so
230        // this is the path a variable index into them takes.
231        Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elements, index_domain)) => {
232            matrix_literal_to_array(store, elements, index_domain)
233        }
234        Expression::SafeIndex(_, m, idxs) => {
235            let arr: Dynamic = expr_to_ast(store, m)?;
236            slice_op(store, idxs, move |idxs: &[Dynamic]| {
237                idxs.iter().fold(arr, |cur_arr, idx| {
238                    // Array indices are always in the integer theory, whatever the elements are.
239                    let idx = coerce_to(idx.clone(), Some(IntTheory::Lia));
240                    cur_arr.as_array().unwrap().select(&idx)
241                })
242            })
243        }
244        // `allDifferent` itself never reaches here: the rules turn it into either this native
245        // form or an explicit encoding, and which one is a modelling choice.
246        Expression::SmtDistinct(_, a) => {
247            let elements = list_elements(a)?;
248            let refs: Vec<&Expression> = elements.iter().collect();
249            let asts = aligned_operand_vec(store, &refs)?;
250            Ok(Dynamic::distinct(&asts).into())
251        }
252
253        // === Expressions involving sets
254        Expression::In(_, x, s) => binary_op(store, x, s, |x: Dynamic, s: Set| s.member(&x)),
255
256        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
257            "expression type not implemented: {expr}"
258        ))),
259    }?;
260
261    ast.try_into().map_err(|err| {
262        SolverError::ModelInvalid(format!(
263            "expression has incorrect type for conversion: {err}"
264        ))
265    })
266}

268/// Builds a Z3 array holding the elements of a matrix literal at their index-domain positions.
269///
270/// Entries outside the literal keep the array's default value; the model only ever reads positions
271/// the index domain allows, which `SafeIndex` has already restricted by this point.
272fn matrix_literal_to_array(
273    store: &SymbolStore,
274    elements: &[Expression],
275    index_domain: &DomainPtr,
276) -> SolverResult<Dynamic> {
277    let asts: Vec<Dynamic> = elements
278        .iter()
279        .map(|element| expr_to_ast(store, element))
280        .collect::<SolverResult<_>>()?;
281
282    let Some(_) = asts.first() else {
283        return Err(SolverError::ModelFeatureNotImplemented(
284            "cannot build an array from an empty matrix literal".to_owned(),
285        ));
286    };
287
288    // Elements can come from declarations represented in different theories; give the array one
289    // element sort and bring them all to it, exactly as an operation over them would.
290    let target = asts
291        .iter()
292        .filter_map(int_theory_of)
293        .fold(None, |acc, theory| match (acc, theory) {
294            (Some(IntTheory::Bv), _) | (_, IntTheory::Bv) => Some(IntTheory::Bv),
295            _ => Some(IntTheory::Lia),
296        });
297    let asts: Vec<Dynamic> = asts.into_iter().map(|ast| coerce_to(ast, target)).collect();
298
299    // A plain list is written with an implied, unbounded index domain (`int(1..)`), which cannot
300    // be enumerated; its positions are simply one per element.
301    let indices = index_domain
302        .resolve()
303        .ok()
304        .and_then(|domain| domain_to_ast_vec(IntTheory::Lia, domain.as_ref()).ok())
305        .filter(|indices| indices.len() == asts.len())
306        .unwrap_or_else(|| {
307            (1..=asts.len())
308                .map(|position| Int::from(position as i64).into())
309                .collect()
310        });
311
312    let index_sort = indices
313        .first()
314        .map(|index| index.get_sort())
315        .unwrap_or_else(Sort::int);
316
317    let mut array = Array::const_array(&index_sort, &asts[0]);
318    for (index, value) in indices.iter().zip(asts.iter()) {
319        array = array.store(index, value);
320    }
321    Ok(array.into())
322}

324/// Floored bit-vector division, matching Essence's `/` and Z3's integer `div`.
325///
326/// `bvsdiv` truncates toward zero, so it is one too high whenever the operands have opposite signs
327/// and the division is inexact; `bvsmod` takes the divisor's sign, so a non-zero remainder is
328/// exactly the case needing the correction.
329fn bv_floor_div(a: BV, b: BV) -> BV {
330    let zero = BV::from_i64(0, BV_SIZE);
331    let one = BV::from_i64(1, BV_SIZE);
332    let truncated = a.bvsdiv(&b);
333    let remainder = a.bvsrem(&b);
334    let signs_differ = a.bvslt(&zero).xor(b.bvslt(&zero));
335    let inexact = remainder.eq(&zero).not();
336    Bool::and(&[signs_differ, inexact]).ite(&truncated.bvsub(&one), &truncated)
337}

339/// Folds a non-empty slice of bit-vectors with a binary operation.
340fn fold_bvs(asts: &[BV], op: impl Fn(BV, BV) -> BV) -> BV {
341    asts.iter()
342        .cloned()
343        .reduce(op)
344        .unwrap_or_else(|| BV::from_i64(0, BV_SIZE))
345}

347/// The integer theory an already-converted AST is in, if it is an integer at all.
348fn int_theory_of(ast: &Dynamic) -> Option<IntTheory> {
349    match ast.sort_kind() {
350        SortKind::Int => Some(IntTheory::Lia),
351        SortKind::Bv => Some(IntTheory::Bv),
352        _ => None,
353    }
354}

356/// Converts an integer AST into `theory`, leaving non-integers and already-matching ASTs alone.
357///
358/// This is what makes a `lia`/`bv` channelling constraint expressible, and what lets an operation
359/// take operands whose declarations were represented in different theories.
360fn coerce_to(ast: Dynamic, theory: Option<IntTheory>) -> Dynamic {
361    match (theory, ast.sort_kind()) {
362        (Some(IntTheory::Bv), SortKind::Int) => {
363            BV::from_int(&ast.as_int().expect("just checked the sort"), BV_SIZE).into()
364        }
365        (Some(IntTheory::Lia), SortKind::Bv) => {
366            Int::from_bv(&ast.as_bv().expect("just checked the sort"), true).into()
367        }
368        _ => ast,
369    }
370}

372/// Converts every operand and brings any integers among them into a common theory.
373///
374/// Bit-vectors win: converting a mathematical integer to a machine word is exact for values that
375/// fit, whereas the reverse would have to pick a signedness for values that may have wrapped.
376fn aligned_operand_vec(store: &SymbolStore, exprs: &[&Expression]) -> SolverResult<Vec<Dynamic>> {
377    let asts: Vec<Dynamic> = exprs
378        .iter()
379        .map(|expr| expr_to_ast(store, expr))
380        .collect::<SolverResult<_>>()?;
381
382    let target = asts
383        .iter()
384        .filter_map(int_theory_of)
385        .fold(None, |acc, theory| match (acc, theory) {
386            (Some(IntTheory::Bv), _) | (_, IntTheory::Bv) => Some(IntTheory::Bv),
387            _ => Some(IntTheory::Lia),
388        });
389
390    Ok(asts.into_iter().map(|ast| coerce_to(ast, target)).collect())
391}

393/// [`aligned_operand_vec`] for a fixed number of operands.
394fn aligned_operands<const N: usize>(
395    store: &SymbolStore,
396    exprs: [&Expression; N],
397) -> SolverResult<[Dynamic; N]> {
398    let refs: Vec<&Expression> = exprs.to_vec();
399    let asts = aligned_operand_vec(store, &refs)?;
400    asts.try_into().map_err(|_| {
401        SolverError::ModelInvalid("wrong number of operands after alignment".to_owned())
402    })
403}

405/// Applies whichever of the two operations matches the theory the operand landed in.
406fn int_unary_op<OutInt, OutBv>(
407    store: &SymbolStore,
408    a: &Expression,
409    lia: impl FnOnce(Int) -> OutInt,
410    bv: impl FnOnce(BV) -> OutBv,
411) -> SolverResult<Dynamic>
412where
413    OutInt: Into<Dynamic>,
414    OutBv: Into<Dynamic>,
415{
416    let ast: Dynamic = expr_to_ast(store, a)?;
417    match int_theory_of(&ast) {
418        Some(IntTheory::Bv) => Ok((bv)(ast.as_bv().expect("just checked the sort")).into()),
419        _ => Ok((lia)(ast.as_int().ok_or_else(|| {
420            SolverError::ModelInvalid(format!("expected an integer operand: {a}"))
421        })?)
422        .into()),
423    }
424}

426/// Applies whichever of the two operations matches the theory the operands were aligned into.
427fn int_binary_op<OutInt, OutBv>(
428    store: &SymbolStore,
429    a: &Expression,
430    b: &Expression,
431    lia: impl FnOnce(Int, Int) -> OutInt,
432    bv: impl FnOnce(BV, BV) -> OutBv,
433) -> SolverResult<Dynamic>
434where
435    OutInt: Into<Dynamic>,
436    OutBv: Into<Dynamic>,
437{
438    let [a_ast, b_ast] = aligned_operands(store, [a, b])?;
439    match int_theory_of(&a_ast) {
440        Some(IntTheory::Bv) => Ok((bv)(
441            a_ast.as_bv().expect("just checked the sort"),
442            b_ast.as_bv().expect("aligned with the first operand"),
443        )
444        .into()),
445        _ => {
446            let cast = |ast: &Dynamic, expr: &Expression| {
447                ast.as_int().ok_or_else(|| {
448                    SolverError::ModelInvalid(format!("expected an integer operand: {expr}"))
449                })
450            };
451            Ok((lia)(cast(&a_ast, a)?, cast(&b_ast, b)?).into())
452        }
453    }
454}

456/// Applies whichever of the two operations matches the theory the list was aligned into.
457fn int_list_op<OutInt, OutBv>(
458    store: &SymbolStore,
459    expr: &Expression,
460    lia: impl FnOnce(&[Int]) -> OutInt,
461    bv: impl FnOnce(&[BV]) -> OutBv,
462) -> SolverResult<Dynamic>
463where
464    OutInt: Into<Dynamic>,
465    OutBv: Into<Dynamic>,
466{
467    let elements = list_elements(expr)?;
468    let refs: Vec<&Expression> = elements.iter().collect();
469    let asts = aligned_operand_vec(store, &refs)?;
470
471    if asts
472        .iter()
473        .any(|ast| matches!(int_theory_of(ast), Some(IntTheory::Bv)))
474    {
475        let bvs: Vec<BV> = asts
476            .iter()
477            .map(|ast| ast.as_bv().expect("the whole list was aligned"))
478            .collect();
479        return Ok((bv)(&bvs).into());
480    }
481
482    let ints: Vec<Int> = asts
483        .iter()
484        .map(|ast| {
485            ast.as_int().ok_or_else(|| {
486                SolverError::ModelInvalid(format!("expected integer operands: {expr}"))
487            })
488        })
489        .collect::<SolverResult<_>>()?;
490    Ok((lia)(&ints).into())
491}

493/// Interprets an expression as an AST and returns the result of the given operation over it.
494fn unary_op<A, Out>(
495    store: &SymbolStore,
496    a: &Expression,
497    op: impl FnOnce(A) -> Out,
498) -> SolverResult<Dynamic>
499where
500    A: TryFrom<Dynamic, Error: std::fmt::Display>,
501    Out: Into<Dynamic>,
502{
503    let a_ast: A = expr_to_ast(store, a)?;
504    Ok((op)(a_ast).into())
505}

507/// Interprets two expressions as ASTs and returns the result of the given operation over them.
508fn binary_op<A, B, Out>(
509    store: &SymbolStore,
510    a: &Expression,
511    b: &Expression,
512    op: impl FnOnce(A, B) -> Out,
513) -> SolverResult<Dynamic>
514where
515    A: TryFrom<Dynamic, Error: std::fmt::Display>,
516    B: TryFrom<Dynamic, Error: std::fmt::Display>,
517    Out: Into<Dynamic>,
518{
519    let a_ast: A = expr_to_ast(store, a)?;
520    let b_ast: B = expr_to_ast(store, b)?;
521    Ok((op)(a_ast, b_ast).into())
522}

524/// Transforms a list expression into separate ASTs and returns the result of the given operation over them.
525fn list_op<A, Out>(
526    store: &SymbolStore,
527    expr: &Expression,
528    op: impl FnOnce(&[A]) -> Out,
529) -> SolverResult<Dynamic>
530where
531    A: TryFrom<Dynamic, Error: std::fmt::Display>,
532    Out: Into<Dynamic>,
533{
534    let exprs = list_elements(expr)?;
535
536    slice_op(store, &exprs, op)
537}

539/// Extracts the elements of a list-like expression.
540///
541/// Besides literal lists, this also supports slice expressions such as `x[..]`
542/// by expanding them to explicit indexing expressions.
543/// TODO: Consider moving this out of the smt solver adaptor, it'll be more generally useful.
544fn list_elements(expr: &Expression) -> SolverResult<Vec<Expression>> {
545    if let Some(exprs) = expr.clone().unwrap_list() {
546        return Ok(exprs);
547    }
548
549    // `unwrap_list` only recognises a matrix whose index domain is the implied `int(1..)`. The
550    // operations that reach here -- sums, conjunctions, `distinct` -- care about the elements and
551    // not where they are indexed from, and a matrix rebuilt from its components keeps the index
552    // domain it was declared with, so take those elements too.
553    if let Expression::AbstractLiteral(_, AbstractLiteral::Matrix(elements, _)) = expr {
554        return Ok(elements.clone());
555    }
556
557    let Expression::SafeSlice(_, subject, indices) = expr else {
558        return Err(SolverError::ModelFeatureNotImplemented(format!(
559            "inner expression must be a list: {expr}"
560        )));
561    };
562
563    let subject_domain = subject
564        .domain_of()
565        .and_then(|d| d.resolve().ok())
566        .ok_or_else(|| {
567            SolverError::ModelFeatureNotImplemented(format!(
568                "cannot resolve domain of sliced expression: {expr}"
569            ))
570        })?;
571
572    let GroundDomain::Matrix(_, index_domains) = subject_domain.as_ref() else {
573        return Err(SolverError::ModelFeatureNotImplemented(format!(
574            "slice subject must have matrix domain: {expr}"
575        )));
576    };
577
578    if index_domains.len() != indices.len() {
579        return Err(SolverError::ModelInvalid(format!(
580            "slice has wrong number of indices: {expr}"
581        )));
582    }
583
584    let index_options: SolverResult<Vec<Vec<Expression>>> = indices
585        .iter()
586        .zip(index_domains.iter())
587        .map(|(idx, dom)| match idx {
588            Some(idx_expr) => Ok(vec![idx_expr.clone()]),
589            None => {
590                let vals = dom.values().map_err(|_| {
591                    SolverError::ModelFeatureNotImplemented(format!(
592                        "slice index domain is not finite/enumerable: {dom}"
593                    ))
594                })?;
595                Ok(vals
596                    .map(|lit| Expression::Atomic(Metadata::new(), Atom::Literal(lit)))
597                    .collect_vec())
598            }
599        })
600        .collect();
601
602    let index_options = index_options?;
603
604    Ok(index_options
605        .into_iter()
606        .multi_cartesian_product()
607        .map(|concrete_idxs| Expression::SafeIndex(Metadata::new(), subject.clone(), concrete_idxs))
608        .collect())
609}

611/// Transforms a slice of expressions into ASTs and returns the result of the given operation over it.
612fn slice_op<A, Out>(
613    store: &SymbolStore,
614    exprs: &[Expression],
615    op: impl FnOnce(&[A]) -> Out,
616) -> SolverResult<Dynamic>
617where
618    A: TryFrom<Dynamic, Error: std::fmt::Display>,
619    Out: Into<Dynamic>,
620{
621    // Result implements FromIter, collecting into either the full collection or an error
622    let asts_res: SolverResult<Vec<_>> = exprs.iter().map(|e| expr_to_ast(store, e)).collect();
623    let asts = asts_res?;
624
625    Ok((op)(asts.as_slice()).into())
626}
