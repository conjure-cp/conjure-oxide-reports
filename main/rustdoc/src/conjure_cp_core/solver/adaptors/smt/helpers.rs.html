11/// Use 32-bit 2's complement signed bit-vectors
12pub const BV_SIZE: u32 = 32;
13
14/// A function which encodes a restriction for a specific variable. Given an AST of the correct
15/// sort, constructs a boolean assertion which will ensure the variable has the correct domain.
16type RestrictFn = Box<dyn Fn(&Dynamic) -> Bool>;
17
18/// Returns the Oxide domain as a Z3 sort, along with a function to restrict a variable of that sort
19/// to the original domain's restrictions.
20pub fn domain_to_sort(domain: &GroundDomain, ints: IntTheory) -> SolverResult<(Sort, RestrictFn)> {
21    use IntTheory::{Bv, Lia};
22
23    match (ints, domain) {
24        // Booleans of course have the same domain in SMT, so no restriction required
25        (_, GroundDomain::Bool) => Ok((Sort::bool(), Box::new(|_| Bool::from_bool(true)))),
26
27        // Return a disjunction of the restrictions each range of the domain enforces
28        // I.e. `x: int(1, 3..5)` -> `or([x = 1, x >= 3 /\ x <= 5])`
29        (Lia, GroundDomain::Int(ranges)) => {
30            let ranges = ranges.clone();
31            let restrict_fn = move |ast: &Dynamic| {
32                let int = ast.as_int().unwrap();
33                let restrictions: Vec<_> = ranges
34                    .iter()
35                    .map(|range| int_range_to_int_restriction(&int, range))
36                    .collect();
37                Bool::or(restrictions.as_slice())
38            };
39            Ok((Sort::int(), Box::new(restrict_fn)))
40        }
41        (Bv, GroundDomain::Int(ranges)) => {
42            let ranges = ranges.clone();
43            let restrict_fn = move |ast: &Dynamic| {
44                let bv = ast.as_bv().unwrap();
45                let restrictions: Vec<_> = ranges
46                    .iter()
47                    .map(|range| int_range_to_bv_restriction(&bv, range))
48                    .collect();
49                Bool::or(restrictions.as_slice())
50            };
51            Ok((Sort::bitvector(BV_SIZE), Box::new(restrict_fn)))
52        }
53
54        (_, GroundDomain::Matrix(val_domain, idx_domains)) => {
55            // We constrain the inner values of the domain recursively
56            // I.e. every way to index the array must give a value in the correct domain
57
58            let (range_sort, restrict_val) = match idx_domains.as_slice() {
59                [_] => domain_to_sort(val_domain, ints),
60                [_, tail @ ..] => {
61                    // Treat as a matrix containing (n-1)-dimensional matrices
62                    let inner_domain = GroundDomain::Matrix(val_domain.clone(), tail.to_vec());
63                    domain_to_sort(&inner_domain, ints)
64                }
65                [] => Err(SolverError::ModelInvalid(
66                    "empty matrix index domain".into(),
67                )),
68            }?;
69
70            // No need to constrain the indices themselves, that's done through SafeIndex/InDomain
71            //
72            // An index is a position in the matrix, not a value the model ever reasons about, so it
73            // always uses the integer theory whatever the elements are. Otherwise indexing a
74            // bit-vector array would need every index expression converted to a machine word first.
75            let idx_domain = &idx_domains[0];
76            let (domain_sort, _) = domain_to_sort(idx_domain.as_ref(), IntTheory::Lia)?;
77
78            // Use the lower dimension's restricting fn to restrict all indexes in this dimension
79            let idx_asts = domain_to_ast_vec(IntTheory::Lia, idx_domain.as_ref())?;
80            let restrict_fn = move |ast: &Dynamic| {
81                let arr = ast.as_array().unwrap();
82                let restrictions: Vec<_> = idx_asts
83                    .iter()
84                    .map(|idx_ast| (restrict_val)(&arr.select(idx_ast)))
85                    .collect();
86                Bool::and(restrictions.as_slice())
87            };
88            Ok((
89                Sort::array(&domain_sort, &range_sort),
90                Box::new(restrict_fn),
91            ))
92        }
93
94        (_, GroundDomain::Set(attr, elem_domain)) => {
95            let (val_sort, _) = domain_to_sort(elem_domain, ints)?;
96
97            // Restrict the size of the set
98            let member_asts = domain_to_ast_vec(ints, elem_domain)?;
99            let attr_size = attr.size.clone();
100            let restrict_fn = move |ast: &Dynamic| {
101                let set = ast.as_set().unwrap();
102                let is_member: Vec<_> = member_asts
103                    .iter()
104                    .map(|val| set.member(val).ite(&Int::from(1), &Int::from(0)))
105                    .collect();
106                let size = Int::add(&is_member);
107                match attr_size {
108                    Range::Single(n) => size.eq(Int::from(n)),
109                    Range::UnboundedL(r) => size.le(Int::from(r)),
110                    Range::UnboundedR(l) => size.ge(Int::from(l)),
111                    Range::Bounded(l, r) => {
112                        Bool::and(&[size.ge(Int::from(l)), size.le(Int::from(r))])
113                    }
114                    Range::Unbounded => Bool::from_bool(true),
115                }
116            };
117
118            Ok((Sort::set(&val_sort), Box::new(restrict_fn)))
119        }
120
121        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
122            "sort for '{domain}' not implemented"
123        ))),
124    }
125}

127/// Returns a domain as a vector of Z3 AST literals.
128pub fn domain_to_ast_vec(ints: IntTheory, domain: &GroundDomain) -> SolverResult<Vec<Dynamic>> {
129    let lits = domain
130        .values()
131        .map_err(|err| SolverError::Runtime(err.to_string()))?;
132    lits.map(|lit| literal_to_ast(ints, &lit)).collect()
133}

135/// Returns a boolean expression restricting the given integer variable to the given range.
136pub fn int_range_to_int_restriction(var: &Int, range: &Range<i32>) -> Bool {
137    match range {
138        Range::Single(n) => var.eq(Int::from(*n)),
139        Range::UnboundedL(r) => var.le(Int::from(*r)),
140        Range::UnboundedR(l) => var.ge(Int::from(*l)),
141        Range::Bounded(l, r) => Bool::and(&[var.ge(Int::from(*l)), var.le(Int::from(*r))]),
142        _ => bug!("int ranges should not be unbounded"),
143    }
144}

146/// Returns a boolean expression restricting the given bitvector variable to the given integer range.
147pub fn int_range_to_bv_restriction(var: &BV, range: &Range<i32>) -> Bool {
148    match range {
149        Range::Single(n) => var.eq(BV::from_i64(*n as i64, BV_SIZE)),
150        Range::UnboundedL(r) => var.bvsle(BV::from_i64(*r as i64, BV_SIZE)),
151        Range::UnboundedR(l) => var.bvsge(BV::from_i64(*l as i64, BV_SIZE)),
152        Range::Bounded(l, r) => Bool::and(&[
153            var.bvsge(BV::from_i64(*l as i64, BV_SIZE)),
154            var.bvsle(BV::from_i64(*r as i64, BV_SIZE)),
155        ]),
156        _ => bug!("int ranges should not be unbounded"),
157    }
158}

160pub fn name_to_symbol(name: &Name) -> SolverResult<Symbol> {
161    match name {
162        Name::User(ustr) => Ok(Symbol::String((*ustr).into())),
163        Name::Machine(num) => Ok(Symbol::Int(*num as u32)),
164        Name::Represented(parts) => {
165            let (name, rule_str, suffix) = parts.as_ref();
166            Ok(Symbol::String(format!("{name}#{rule_str}_{suffix}")))
167        }
168        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
169            "variable '{name}' name is unsupported"
170        ))),
171    }
172}

174/// Converts an atom (literal or reference) into an AST node.
175pub fn atom_to_ast(store: &SymbolStore, atom: &Atom) -> SolverResult<Dynamic> {
176    match atom {
177        Atom::Reference(reference) => {
178            if let Some((_, ast, _)) = store.get(&reference.name()) {
179                return Ok(ast.clone());
180            }
181
182            if let Some(lit) = reference.resolve_constant() {
183                return literal_to_ast(IntTheory::default(), &lit);
184            }
185
186            if let Some(inner_atom) = reference.resolve_atomic() {
187                return atom_to_ast(store, &inner_atom);
188            }
189
190            let decl_kind = reference.ptr().kind();
191            match decl_kind.deref() {
192                DeclarationKind::ValueLetting(expr, _) => {
193                    Err(SolverError::ModelFeatureNotImplemented(format!(
194                        "value letting '{}' did not resolve to an atomic expression: {expr}",
195                        reference.name()
196                    )))
197                }
198                _ => Err(SolverError::ModelInvalid(format!(
199                    "variable '{}' does not exist",
200                    reference.name()
201                ))),
202            }
203        }
204        Atom::Literal(lit) => literal_to_ast(IntTheory::default(), lit),
205        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
206            "atom sort not implemented: {atom}"
207        ))),
208    }
209}

211/// Converts a CO literal (expression containing no variables) into an AST node.
212///
213/// A constant has no representation of its own to read a theory off, so it is built in whichever
214/// theory the caller is working in; operations mixing theories coerce their operands afterwards.
215pub fn literal_to_ast(ints: IntTheory, lit: &Literal) -> SolverResult<Dynamic> {
216    match lit {
217        Literal::Bool(b) => Ok(Bool::from_bool(*b).into()),
218        Literal::Int(n) => Ok(match ints {
219            IntTheory::Lia => Int::from(*n).into(),
220            IntTheory::Bv => BV::from_i64(*n as i64, BV_SIZE).into(),
221        }),
222        // A constant matrix becomes a Z3 array, so that indexing one by a variable works. The
223        // packed matrix layout decodes through exactly this: a table of the element domain's
224        // values, indexed by a digit computed from the packed integer.
225        Literal::AbstractLiteral(AbstractLiteral::Matrix(elements, index_domain)) => {
226            let values: Vec<Dynamic> = elements
227                .iter()
228                .map(|element| literal_to_ast(ints, element))
229                .collect::<SolverResult<_>>()?;
230            let Some(first) = values.first() else {
231                return Err(SolverError::ModelFeatureNotImplemented(
232                    "cannot build an array from an empty matrix literal".to_owned(),
233                ));
234            };
235
236            // A plain list is written with an implied, unbounded index domain that cannot be
237            // enumerated; its positions are simply one per element.
238            let indices = domain_to_ast_vec(IntTheory::Lia, index_domain)
239                .ok()
240                .filter(|indices| indices.len() == values.len())
241                .unwrap_or_else(|| {
242                    (1..=values.len())
243                        .map(|position| Int::from(position as i64).into())
244                        .collect()
245                });
246            let index_sort = indices
247                .first()
248                .map(|index| index.get_sort())
249                .unwrap_or_else(Sort::int);
250
251            let mut array = Array::const_array(&index_sort, first);
252            for (index, value) in indices.iter().zip(values.iter()) {
253                array = array.store(index, value);
254            }
255            Ok(array.into())
256        }
257        _ => Err(SolverError::ModelFeatureNotImplemented(format!(
258            "literal type not implemented: {lit}"
259        ))),
260    }
