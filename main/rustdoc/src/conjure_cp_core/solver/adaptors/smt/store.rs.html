14/// Maps CO variable names to their CO domains, Z3 symbolic constants, and Z3 symbols.
15#[derive(Clone, Debug)]
16pub struct SymbolStore {
17    map: BTreeMap<Name, (Moo<GroundDomain>, Dynamic, Symbol)>,
18    /// Primary `find` variables. Solution blocking ignores `findAux` variables.
19    blocking_vars: BTreeSet<Name>,
20}
21
22impl SymbolStore {
23    pub fn new() -> Self {
24        SymbolStore {
25            map: BTreeMap::new(),
26            blocking_vars: BTreeSet::new(),
27        }
28    }
29
30    pub fn insert(
31        &mut self,
32        name: Name,
33        val: (Moo<GroundDomain>, Dynamic, Symbol),
34    ) -> Option<(Moo<GroundDomain>, Dynamic, Symbol)> {
35        self.map.insert(name, val)
36    }
37
38    /// Include this variable when blocking solutions. `findAux` variables are omitted.
39    pub fn block_solutions_on(&mut self, name: Name) {
40        self.blocking_vars.insert(name);
41    }
42
43    pub fn get(&self, name: &Name) -> Option<&(Moo<GroundDomain>, Dynamic, Symbol)> {
44        self.map.get(name)
45    }
46}

48/// The integer theory an already-created variable is in, read off its Z3 sort.
49fn int_theory_of_ast(ast: &Dynamic) -> IntTheory {
50    match ast.get_sort().array_domain().map(|sort| sort.kind()) {
51        Some(SortKind::Bv) => IntTheory::Bv,
52        Some(_) => IntTheory::Lia,
53        None => match ast.sort_kind() {
54            SortKind::Bv => IntTheory::Bv,
55            _ => IntTheory::Lia,
56        },
57    }
58}

60/// Builds a disequality over exactly the values admitted by an Essence domain.
61///
62/// SMT arrays and sets are total over their SMT sorts, so comparing either one directly also
63/// observes values outside the finite Essence index/element domains. Z3 can otherwise keep
64/// producing models that differ only there, even though they reconstruct to the same Essence
65/// solution. Recurse through compound domains so those invisible differences are ignored at every
66/// nesting level.
67fn domain_disequality(domain: &GroundDomain, ast: &Dynamic, other: &Dynamic) -> Bool {
68    match domain {
69        GroundDomain::Matrix(value_domain, index_domains, _) => {
70            let neqs: Vec<_> = index_domains
71                .iter()
72                .map(|domain| domain_to_ast_vec(IntTheory::Lia, domain).unwrap())
73                .multi_cartesian_product()
74                .map(|indices| {
75                    indices
76                        .iter()
77                        .fold((ast.clone(), other.clone()), |(left, right), index| {
78                            (
79                                left.as_array().unwrap().select(index),
80                                right.as_array().unwrap().select(index),
81                            )
82                        })
83                })
84                .map(|(left, right)| domain_disequality(value_domain.as_ref(), &left, &right))
85                .collect();
86            Bool::or(&neqs)
87        }
88        GroundDomain::Set(_, element_domain) => {
89            let set = ast.as_set().unwrap();
90            let other_set = other.as_set().unwrap();
91            let neqs: Vec<_> = domain_to_ast_vec(int_theory_of_ast(ast), element_domain)
92                .unwrap()
93                .iter()
94                .map(|element| set.member(element).ne(other_set.member(element)))
95                .collect();
96            Bool::or(&neqs)
97        }
98        _ => ast.ne(other),
99    }
100}

102impl Solvable for SymbolStore {
103    type ModelInstance = LiteralStore;
104
105    fn read_from_model(
106        &self,
107        model: &Model,
108        model_completion: bool,
109    ) -> Option<Self::ModelInstance> {
110        let mut new_store = LiteralStore::new();
111        for (name, (domain, ast, sym)) in self.map.iter() {
112            // Get the interpretation of each constant
113            let (ast, lit) = interpret(model, (domain, ast), model_completion).unwrap();
114            new_store.map.insert(name.clone(), (ast, lit));
115        }
116        Some(new_store)
117    }
118
119    fn generate_constraint(&self, model: &Self::ModelInstance) -> Bool {
120        let bools: Vec<_> = self
121            .map
122            .iter()
123            .filter(|(name, _)| self.blocking_vars.contains(name))
124            .map(|(name, (domain, ast, _))| {
125                let (other, _) = model.map.get(name).unwrap();
126                domain_disequality(domain, ast, other)
127            })
128            .collect();
129        // No primary finds: the auxiliary assignment must not yield another solution.
130        if bools.is_empty() {
131            return Bool::from_bool(false);
132        }
133        Bool::or(bools.as_slice())
134    }
135}
136
137unsafe impl Translate for SymbolStore {
138    fn translate(&self, ctx: &Context) -> Self {
139        let mut new_map = BTreeMap::new();
140        for (name, (domain, ast, sym)) in self.map.iter() {
141            let new_ast = ast.translate(ctx);
142            let new_sym = sym.clone(); // Symbols are not translated
143            new_map.insert(name.clone(), (domain.clone(), new_ast, new_sym));
144        }
145        SymbolStore {
146            map: new_map,
147            blocking_vars: self.blocking_vars.clone(),
148        }
149    }
150}

152/// Maps CO variable names to their literal values in both the Z3 model and Essence.
153#[derive(Clone, Debug)]
154pub struct LiteralStore {
155    map: BTreeMap<Name, (Dynamic, Literal)>,
156}
157
158impl LiteralStore {
159    fn new() -> Self {
160        LiteralStore {
161            map: BTreeMap::new(),
162        }
163    }
164
165    /// Return this store as a mapping of CO names to literals
166    pub fn as_literals_map(&self) -> SolverResult<HashMap<Name, Literal>> {
167        let iter = self
168            .map
169            .iter()
170            .map(|(name, (_, lit))| (name.clone(), lit.clone()));
171        Ok(HashMap::from_iter(iter))
172    }
173}

190#[cfg(test)]
191mod tests {
192    use super::*;
193    use crate::ast::SetAttr;
194    use crate::{domain_int_ground, range};
195    use z3::{SatResult, Solver};
196
197    #[test]
198    fn matrix_of_sets_blocking_ignores_members_outside_the_element_domain() {
199        let element_domain = domain_int_ground!(1..3);
200        let set_domain = Moo::new(GroundDomain::Set(SetAttr::new_size(1), element_domain));
201        let domain = GroundDomain::Matrix(set_domain, vec![domain_int_ground!(1..2)], None);
202        let (sort, _) = domain_to_sort(&domain, IntTheory::Lia).unwrap();
203        let left = Dynamic::new_const("left", &sort);
204        let right = Dynamic::new_const("right", &sort);
205        let solver = Solver::new();
206
207        // The two matrices agree on every observable set member.
208        for index in 1..=2 {
209            let index = Int::from(index);
210            let left_set = left.as_array().unwrap().select(&index).as_set().unwrap();
211            let right_set = right.as_array().unwrap().select(&index).as_set().unwrap();
212            for element in 1..=3 {
213                let element = Int::from(element);
214                solver.assert(left_set.member(&element).eq(right_set.member(&element)));
215            }
216        }
217
218        // They differ only at an integer outside the Essence element domain.
219        let index = Int::from(1);
220        let outside = Int::from(99);
221        let left_set = left.as_array().unwrap().select(&index).as_set().unwrap();
222        let right_set = right.as_array().unwrap().select(&index).as_set().unwrap();
223        solver.assert(left_set.member(&outside).ne(right_set.member(&outside)));
224        solver.assert(domain_disequality(&domain, &left, &right));
225
226        assert_eq!(solver.check(), SatResult::Unsat);
227    }
228}

230/// Interprets the given value within the given model and returns a CO literal.
231/// The value can be any AST including a symbolic constant, as long as it is defined in the model.
232///
233/// This method makes any interpretations necessary to fully evaluate the literal. E.g. for arrays
234/// it must enumerate over all elements in the index domain and evaluate the elements as literals.
235fn interpret(
236    model: &Model,
237    value: (&Moo<GroundDomain>, &Dynamic),
238    model_completion: bool,
239) -> SolverResult<(Dynamic, Literal)> {
240    use IntTheory::{Bv, Lia};
241
242    let (domain, var_ast) = value;
243    // Which theory this variable ended up in is a property of the variable itself, not of the
244    // model, so read it back off the AST's sort rather than off a solver-wide setting.
245    let theory = match var_ast.sort_kind() {
246        SortKind::Bv => Bv,
247        _ => Lia,
248    };
249    let lit_ast = model
250        .eval(var_ast, model_completion)
251        .ok_or(SolverError::Runtime(format!(
252            "could not interpret variable: {var_ast}"
253        )))?;
254
255    let literal = match (theory, domain.as_ref()) {
256        (_, GroundDomain::Bool) => {
257            let bool_ast = lit_ast.as_bool().unwrap();
258            let bool = bool_ast.as_bool().unwrap();
259            Ok(Literal::Bool(bool))
260        }
261        (Lia, GroundDomain::Int(_, _)) => {
262            let int_ast = lit_ast.as_int().unwrap();
263            let int = int_ast
264                .as_i64()
265                .ok_or(SolverError::Runtime(format!(
266                    "could not cast to i64: {lit_ast}"
267                )))?
268                .try_into()
269                .map_err(|err| {
270                    SolverError::Runtime(format!("value {lit_ast} out of range: {err}"))
271                })?;
272            Ok(Literal::Int(int))
273        }
274        (Bv, GroundDomain::Int(_, _)) => {
275            // BVs do not sign-extend when returning u64s (if they are < 64 bits)
276            // To correctly retrieve negative numbers, we downsize to a u32 and then bit-wise
277            // interpret it as an i32, rather than casting.
278            // See https://github.com/prove-rs/z3.rs/issues/458
279            let bv_ast = lit_ast.as_bv().unwrap();
280            let unsigned_64: u64 = bv_ast.as_u64().ok_or(SolverError::Runtime(format!(
281                "could not retrieve u64: {lit_ast}"
282            )))?;
283            let unsigned_32: u32 = unsigned_64.try_into().map_err(|err| {
284                SolverError::Runtime(format!("value {lit_ast} out of range: {err}"))
285            })?;
286            let signed = i32::from_ne_bytes(unsigned_32.to_ne_bytes());
287            Ok(Literal::Int(signed))
288        }
289        (_, GroundDomain::Matrix(val_domain, idx_domains, _)) => {
290            let arr_ast = lit_ast.as_array().unwrap();
291
292            let inner_domain = match idx_domains.as_slice() {
293                [idx_domain] => val_domain.clone(),
294                [idx_domain, tail @ ..] => Moo::new(GroundDomain::Matrix(
295                    val_domain.clone(),
296                    tail.to_vec(),
297                    None,
298                )),
299                [] => return Err(SolverError::Runtime("empty matrix index domain".into())),
300            };
301
302            // Indices are positions, always in the integer theory; see `domain_to_sort`.
303            let indices = domain_to_ast_vec(IntTheory::Lia, &idx_domains[0])?;
304            let elements_res: Result<Vec<_>, _> = indices
305                .iter()
306                .map(|idx| model.eval(&arr_ast.select(idx), model_completion).unwrap())
307                .map(|ast| interpret(model, (&inner_domain, &ast), model_completion))
308                .map(|res| res.map(|(_, lit)| lit))
309                .collect();
310            let elements = elements_res?;
311
312            Ok(Literal::AbstractLiteral(AbstractLiteral::Matrix(
313                elements,
314                domain.clone(),
315            )))
316        }
317        (_, GroundDomain::Set(_, val_domain)) => {
318            let set_ast = lit_ast.as_set().unwrap();
319
320            // Collect every member of the domain that is in the set
321            let dom_iter = val_domain
322                .values()
323                .map_err(|_| SolverError::Runtime("could not construct domain iterator".into()))?;
324            let members: Result<Vec<Literal>, SolverError> = dom_iter
325                .map(|lit| {
326                    // We want to bubble errors up but also only collect set members
327                    let lit_ast = literal_to_ast(theory, &lit)?;
328                    let is_member = model
329                        .eval(&set_ast.member(&lit_ast), model_completion)
330                        .unwrap()
331                        .as_bool()
332                        .unwrap();
333                    Ok(is_member.then_some(lit))
334                })
335                .filter_map(|res: Result<Option<Literal>, SolverError>| res.ok().flatten().map(Ok))
336                .collect();
337            let members = members?;
338
339            Ok(Literal::AbstractLiteral(AbstractLiteral::Set(members)))
340        }
341        _ => Err(SolverError::RuntimeNotImplemented(format!(
342            "conversion from AST to literal of type '{domain}' not implemented: {lit_ast}"
343        ))),
344    }?;
345    Ok((lit_ast, literal))
