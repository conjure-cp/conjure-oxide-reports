24pub fn parse_atom(
25    ctx: &mut ParseContext,
26    node: &Node,
27) -> Result<Option<Expression>, FatalParseError> {
28    match node.kind() {
29        "atom" | "primary_atom" | "sub_atom_expr" => {
30            let Some(inner) = named_child!(recover, ctx, node) else {
31                return Ok(None);
32            };
33            parse_atom(ctx, &inner)
34        }
35        "metavar" => {
36            let Some(ident) = field!(recover, ctx, node, "identifier") else {
37                return Ok(None);
38            };
39            let name_str = &ctx.source_code[ident.start_byte()..ident.end_byte()];
40            Ok(Some(Expression::Metavar(
41                Metadata::new(),
42                Ustr::from(name_str),
43            )))
44        }
45        "identifier" => {
46            let Some(var) = parse_variable(ctx, node)? else {
47                return Ok(None);
48            };
49            Ok(Some(Expression::Atomic(Metadata::new(), var)))
50        }
51        "from_solution" => {
52            if ctx.root.kind() != "dominance_relation" {
53                ctx.record_error(RecoverableParseError::new(
54                    "fromSolution only allowed inside dominance relations".to_string(),
55                    Some(node.range()),
56                ));
57                return Ok(None);
58            }
59
60            let Some(var_node) = field!(recover, ctx, node, "variable") else {
61                return Ok(None);
62            };
63            let Some(inner) = parse_variable(ctx, &var_node)? else {
64                return Ok(None);
65            };
66
67            Ok(Some(Expression::FromSolution(
68                Metadata::new(),
69                Moo::new(inner),
70            )))
71        }
72        "pareto_expression" => parse_pareto_expression(ctx, node),
73        "constant" => {
74            let Some(lit) = parse_constant(ctx, node)? else {
75                return Ok(None);
76            };
77            Ok(Some(Expression::Atomic(
78                Metadata::new(),
79                Atom::Literal(lit),
80            )))
81        }
82        "matrix"
83        | "record"
84        | "variant"
85        | "tuple"
86        | "set_literal"
87        | "mset_literal"
88        | "sequence_literal"
89        | "function_literal"
90        | "relation_literal"
91        | "partition_literal"
92        | "permutation_literal" => {
93            let Some(abs) = parse_abstract(ctx, node)? else {
94                return Ok(None);
95            };
96            Ok(Some(Expression::AbstractLiteral(Metadata::new(), abs)))
97        }
98        "flatten" => parse_flatten(ctx, node),
99        "element_id" => parse_element_id(ctx, node),
100        "table" | "negative_table" => parse_table(ctx, node),
101        "index_or_slice" => parse_index_or_slice(ctx, node),
102        "annotation_expr" => parse_annotation_expression(ctx, node),
103        // for now, assume is binary since powerset isn't implemented
104        // TODO: add powerset support under "set_operation"
105        "set_operation" => parse_binary_expression(ctx, node),
106        "comprehension" => parse_comprehension(ctx, node),
107        "defined_expr" | "range_expr" | "to_set_expr" | "to_mset_expr" | "to_relation_expr"
108        | "perm_inverse_expr" => parse_function_unary_operator(ctx, node),
109        "image_expr" | "image_set_expr" | "pre_image_expr" | "inverse_expr" | "compose_expr" => {
110            parse_function_binary_operator(ctx, node)
111        }
112        "restrict_expr" => parse_restrict_expr(ctx, node),
113        "attribute_as_constraint_expr" => parse_attribute_as_constraint(ctx, node),
114        "parts_expr" | "participants_expr" => parse_partition_unary_operator(ctx, node),
115        "together_expr" | "apart_expr" | "party_expr" => parse_partition_binary_operator(ctx, node),
116        _ => {
117            ctx.record_error(RecoverableParseError::new(
118                format!("Expected atom, got: {}", node.kind()),
119                Some(node.range()),
120            ));
121            Ok(None)
122        }
123    }
124}

126fn parse_flatten(
127    ctx: &mut ParseContext,
128    node: &Node,
129) -> Result<Option<Expression>, FatalParseError> {
130    // add error and return early if we're in a set context, since flatten doesn't produce sets
131    if ctx.typechecking_context == TypecheckingContext::Set {
132        ctx.record_error(RecoverableParseError::new(
133            format!(
134                "Type error: {}\n\tExpected: set\n\tGot: flatten",
135                ctx.source_code[node.start_byte()..node.end_byte()].trim()
136            ),
137            Some(node.range()),
138        ));
139        return Ok(None);
140    }
141
142    let Some(expr_node) = field!(recover, ctx, node, "expression") else {
143        return Ok(None);
144    };
145    // TODO: verify the atom is a matrix
146    let Some(expr) = parse_atom(ctx, &expr_node)? else {
147        return Ok(None);
148    };
149
150    if let Some(depth_node) = node.child_by_field_name("depth") {
151        let Some(depth) = parse_int(ctx, &depth_node) else {
152            return Ok(None);
153        };
154        let depth_expression =
155            Expression::Atomic(Metadata::new(), Atom::Literal(Literal::Int(depth)));
156        Ok(Some(Expression::Flatten(
157            Metadata::new(),
158            Some(Moo::new(depth_expression)),
159            Moo::new(expr),
160        )))
161    } else {
162        Ok(Some(Expression::Flatten(
163            Metadata::new(),
164            None,
165            Moo::new(expr),
166        )))
167    }
168}

170fn parse_table(ctx: &mut ParseContext, node: &Node) -> Result<Option<Expression>, FatalParseError> {
171    // add error and return early if we're in a set context, since tables aren't allowed there
172    if ctx.typechecking_context == TypecheckingContext::Set {
173        ctx.record_error(RecoverableParseError::new(
174            format!(
175                "Type error: {}\n\tExpected: set\n\tGot: table",
176                ctx.source_code[node.start_byte()..node.end_byte()].trim()
177            ),
178            Some(node.range()),
179        ));
180        return Ok(None);
181    }
182
183    // the variables and rows can contain arbitrary expressions, so we temporarily set the context to Unknown to avoid typechecking errors
184    let saved_context = ctx.typechecking_context;
185    ctx.typechecking_context = TypecheckingContext::Unknown;
186
187    let Some(variables_node) = field!(recover, ctx, node, "variables") else {
188        return Ok(None);
189    };
190    let Some(variables) = parse_table_operand(ctx, &variables_node)? else {
191        return Ok(None);
192    };
193
194    let Some(rows_node) = field!(recover, ctx, node, "rows") else {
195        return Ok(None);
196    };
197    let Some(rows) = parse_table_operand(ctx, &rows_node)? else {
198        return Ok(None);
199    };
200
201    ctx.typechecking_context = saved_context;
202
203    match node.kind() {
204        "table" => Ok(Some(Expression::Table(
205            Metadata::new(),
206            Moo::new(variables),
207            Moo::new(rows),
208        ))),
209        "negative_table" => Ok(Some(Expression::NegativeTable(
210            Metadata::new(),
211            Moo::new(variables),
212            Moo::new(rows),
213        ))),
214        _ => {
215            ctx.record_error(RecoverableParseError::new(
216                format!(
217                    "Expected 'table' or 'negative_table', got: '{}'",
218                    node.kind()
219                ),
220                Some(node.range()),
221            ));
222            Ok(None)
223        }
224    }
225}

227pub fn parse_table_operand(
228    ctx: &mut ParseContext,
229    node: &Node,
230) -> Result<Option<Expression>, FatalParseError> {
231    match node.kind() {
232        "identifier" | "constant" | "matrix" | "index_or_slice" | "flatten" => {
233            parse_atom(ctx, node)
234        }
235        "sub_arith_expr" | "negative_expr" | "abs_value" | "factorial_expr" | "arithmetic_expr"
236        | "product_expr" | "sum_expr" | "exponent" | "bool_expr" | "comparison_expr" => {
237            parse_expression(ctx, *node)
238        }
239        _ => parse_expression(ctx, *node),
240    }
241}

243fn parse_element_id(
244    ctx: &mut ParseContext,
245    node: &Node,
246) -> Result<Option<Expression>, FatalParseError> {
247    if ctx.typechecking_context == TypecheckingContext::Set {
248        ctx.record_error(RecoverableParseError::new(
249            format!(
250                "Type error: {}\n\tExpected: set\n\tGot: elementId",
251                ctx.source_code[node.start_byte()..node.end_byte()].trim()
252            ),
253            Some(node.range()),
254        ));
255        return Ok(None);
256    }
257
258    let saved_context = ctx.typechecking_context;
259    ctx.typechecking_context = TypecheckingContext::Unknown;
260
261    let Some(matrix_node) = field!(recover, ctx, node, "matrix") else {
262        return Ok(None);
263    };
264    let Some(matrix) = parse_table_operand(ctx, &matrix_node)? else {
265        ctx.typechecking_context = saved_context;
266        return Ok(None);
267    };
268
269    let Some(value_node) = field!(recover, ctx, node, "value") else {
270        ctx.typechecking_context = saved_context;
271        return Ok(None);
272    };
273    let Some(value) = parse_table_operand(ctx, &value_node)? else {
274        ctx.typechecking_context = saved_context;
275        return Ok(None);
276    };
277
278    ctx.typechecking_context = saved_context;
279
280    let operator_node = node.child(0).unwrap();
281    ctx.add_span_and_doc_hover(&operator_node, ELEMENT_ID, SymbolKind::Function, None, None);
282
283    Ok(Some(Expression::ElementId(
284        Metadata::new(),
285        Moo::new(matrix),
286        Moo::new(value),
287    )))
288}

290/// Parses the single argument of a unary function-call-style operator (`defined`, `range`,
291/// `toSet`, `toMSet`, `toRelation`): keyword immediately followed by a parenthesised argument.
292/// The argument's type varies by operator (function, set, mset, relation), so typechecking
293/// context is relaxed to `Unknown` while parsing it, mirroring `parse_flatten`/`parse_element_id`.
294fn parse_function_unary_operator(
295    ctx: &mut ParseContext,
296    node: &Node,
297) -> Result<Option<Expression>, FatalParseError> {
298    let field_name = if node.kind() == "perm_inverse_expr" {
299        "permutation"
300    } else {
301        "function"
302    };
303
304    let saved_context = ctx.typechecking_context;
305    ctx.typechecking_context = TypecheckingContext::Unknown;
306
307    let Some(function_node) = field!(recover, ctx, node, field_name) else {
308        ctx.typechecking_context = saved_context;
309        return Ok(None);
310    };
311    let Some(function) = parse_expression(ctx, function_node)? else {
312        ctx.typechecking_context = saved_context;
313        return Ok(None);
314    };
315
316    ctx.typechecking_context = saved_context;
317
318    match node.kind() {
319        "defined_expr" => Ok(Some(Expression::Defined(
320            Metadata::new(),
321            Moo::new(function),
322        ))),
323        "range_expr" => Ok(Some(Expression::Range(Metadata::new(), Moo::new(function)))),
324        "to_set_expr" => Ok(Some(Expression::ToSet(Metadata::new(), Moo::new(function)))),
325        "to_mset_expr" => Ok(Some(Expression::ToMSet(
326            Metadata::new(),
327            Moo::new(function),
328        ))),
329        "to_relation_expr" => Ok(Some(Expression::ToRelation(
330            Metadata::new(),
331            Moo::new(function),
332        ))),
333        "perm_inverse_expr" => Ok(Some(Expression::PermInverse(
334            Metadata::new(),
335            Moo::new(function),
336        ))),
337        _ => {
338            ctx.record_error(RecoverableParseError::new(
339                format!("Expected a function operator, got: {}", node.kind()),
340                Some(node.range()),
341            ));
342            Ok(None)
343        }
344    }
345}

347/// Parses an attribute-as-constraint predicate, e.g. `reflexive(r)` or `size(s, 3)`: a keyword
348/// naming the attribute, immediately followed by a parenthesised target and an optional value.
349/// Arity is not checked here (the grammar allows any attribute name with or without a value); a
350/// later pass validates it against the target's actual domain kind.
351fn parse_attribute_as_constraint(
352    ctx: &mut ParseContext,
353    node: &Node,
354) -> Result<Option<Expression>, FatalParseError> {
355    let Some(attr_node) = field!(recover, ctx, node, "attribute") else {
356        return Ok(None);
357    };
358    let attr = Ustr::from(&ctx.source_code[attr_node.start_byte()..attr_node.end_byte()]);
359
360    let saved_context = ctx.typechecking_context;
361    ctx.typechecking_context = TypecheckingContext::Unknown;
362
363    let Some(target_node) = field!(recover, ctx, node, "target") else {
364        ctx.typechecking_context = saved_context;
365        return Ok(None);
366    };
367    let Some(target) = parse_expression(ctx, target_node)? else {
368        ctx.typechecking_context = saved_context;
369        return Ok(None);
370    };
371
372    let value = match node.child_by_field_name("value") {
373        Some(value_node) => match parse_expression(ctx, value_node)? {
374            Some(value_expr) => Some(Moo::new(value_expr)),
375            None => {
376                ctx.typechecking_context = saved_context;
377                return Ok(None);
378            }
379        },
380        None => None,
381    };
382
383    ctx.typechecking_context = saved_context;
384
385    Ok(Some(Expression::AttributeAsConstraint(
386        Metadata::new(),
387        Moo::new(target),
388        attr,
389        value,
390    )))
391}

432/// Parses the two arguments of a binary partition operator (`together`, `apart`, `party`):
433/// keyword immediately followed by a parenthesised, comma-separated argument pair. `party_expr`
434/// uses an `element`/`partition` field pair; `together_expr`/`apart_expr` use `elements`/
435/// `partition`.
436fn parse_partition_binary_operator(
437    ctx: &mut ParseContext,
438    node: &Node,
439) -> Result<Option<Expression>, FatalParseError> {
440    let first_field = if node.kind() == "party_expr" {
441        "element"
442    } else {
443        "elements"
444    };
445
446    let saved_context = ctx.typechecking_context;
447    ctx.typechecking_context = TypecheckingContext::Unknown;
448
449    let Some(first_node) = field!(recover, ctx, node, first_field) else {
450        ctx.typechecking_context = saved_context;
451        return Ok(None);
452    };
453    let Some(first) = parse_expression(ctx, first_node)? else {
454        ctx.typechecking_context = saved_context;
455        return Ok(None);
456    };
457
458    let Some(partition_node) = field!(recover, ctx, node, "partition") else {
459        ctx.typechecking_context = saved_context;
460        return Ok(None);
461    };
462    let Some(partition) = parse_expression(ctx, partition_node)? else {
463        ctx.typechecking_context = saved_context;
464        return Ok(None);
465    };
466
467    ctx.typechecking_context = saved_context;
468
469    match node.kind() {
470        "together_expr" => Ok(Some(Expression::Together(
471            Metadata::new(),
472            Moo::new(first),
473            Moo::new(partition),
474        ))),
475        "apart_expr" => Ok(Some(Expression::Apart(
476            Metadata::new(),
477            Moo::new(first),
478            Moo::new(partition),
479        ))),
480        "party_expr" => Ok(Some(Expression::Party(
481            Metadata::new(),
482            Moo::new(first),
483            Moo::new(partition),
484        ))),
485        _ => {
486            ctx.record_error(RecoverableParseError::new(
487                format!("Expected a partition operator, got: {}", node.kind()),
488                Some(node.range()),
489            ));
490            Ok(None)
491        }
492    }
493}

495/// Parses the two arguments of a binary function-call-style operator (`image`, `imageSet`,
496/// `preImage`, `inverse`): keyword immediately followed by a parenthesised, comma-separated
497/// argument pair. `inverse_expr` uses `left`/`right` fields; the rest use `function`/`argument`.
498fn parse_function_binary_operator(
499    ctx: &mut ParseContext,
500    node: &Node,
501) -> Result<Option<Expression>, FatalParseError> {
502    let (first_field, second_field) = if matches!(node.kind(), "inverse_expr" | "compose_expr") {
503        ("left", "right")
504    } else {
505        ("function", "argument")
506    };
507
508    let saved_context = ctx.typechecking_context;
509    ctx.typechecking_context = TypecheckingContext::Unknown;
510
511    let Some(first_node) = field!(recover, ctx, node, first_field) else {
512        ctx.typechecking_context = saved_context;
513        return Ok(None);
514    };
515    let Some(first) = parse_expression(ctx, first_node)? else {
516        ctx.typechecking_context = saved_context;
517        return Ok(None);
518    };
519
520    let Some(second_node) = field!(recover, ctx, node, second_field) else {
521        ctx.typechecking_context = saved_context;
522        return Ok(None);
523    };
524    let Some(second) = parse_expression(ctx, second_node)? else {
525        ctx.typechecking_context = saved_context;
526        return Ok(None);
527    };
528
529    ctx.typechecking_context = saved_context;
530
531    match node.kind() {
532        "image_expr" => Ok(Some(Expression::Image(
533            Metadata::new(),
534            Moo::new(first),
535            Moo::new(second),
536        ))),
537        "image_set_expr" => Ok(Some(Expression::ImageSet(
538            Metadata::new(),
539            Moo::new(first),
540            Moo::new(second),
541        ))),
542        "pre_image_expr" => Ok(Some(Expression::PreImage(
543            Metadata::new(),
544            Moo::new(first),
545            Moo::new(second),
546        ))),
547        "inverse_expr" => Ok(Some(Expression::Inverse(
548            Metadata::new(),
549            Moo::new(first),
550            Moo::new(second),
551        ))),
552        "compose_expr" => Ok(Some(Expression::Compose(
553            Metadata::new(),
554            Moo::new(first),
555            Moo::new(second),
556        ))),
557        _ => {
558            ctx.record_error(RecoverableParseError::new(
559                format!("Expected a function operator, got: {}", node.kind()),
560                Some(node.range()),
561            ));
562            Ok(None)
563        }
564    }
565}

567/// Parses `restrict(f, dom)`. Unlike the other function operators, the second argument is
568/// syntactically a domain rather than an expression. `Expression::Restrict` still stores it as
569/// an `Expression` though (its `domain_of()` is called directly on that operand): a bare domain
570/// name (e.g. `letting D be domain int(0,2)` used as `restrict(f, D)`) becomes a plain reference
571/// to that declaration, matching the shape the via-conjure/JSON import path produces. An inline
572/// domain literal (not just a name) is wrapped in a `DomainAnnotation` so `domain_of()` still
573/// resolves correctly, though it will not print back out identically (no exhaustive/roundtrip
574/// case in this codebase currently needs that form).
575fn parse_restrict_expr(
576    ctx: &mut ParseContext,
577    node: &Node,
578) -> Result<Option<Expression>, FatalParseError> {
579    let saved_context = ctx.typechecking_context;
580    ctx.typechecking_context = TypecheckingContext::Unknown;
581
582    let Some(function_node) = field!(recover, ctx, node, "function") else {
583        ctx.typechecking_context = saved_context;
584        return Ok(None);
585    };
586    let Some(function) = parse_expression(ctx, function_node)? else {
587        ctx.typechecking_context = saved_context;
588        return Ok(None);
589    };
590
591    let Some(domain_node) = field!(recover, ctx, node, "domain") else {
592        ctx.typechecking_context = saved_context;
593        return Ok(None);
594    };
595    let Some(domain_expr) = parse_restrict_domain_argument(ctx, domain_node)? else {
596        ctx.typechecking_context = saved_context;
597        return Ok(None);
598    };
599
600    ctx.typechecking_context = saved_context;
601
602    Ok(Some(Expression::Restrict(
603        Metadata::new(),
604        Moo::new(function),
605        Moo::new(domain_expr),
606    )))
607}

609fn parse_restrict_domain_argument(
610    ctx: &mut ParseContext,
611    domain_node: Node,
612) -> Result<Option<Expression>, FatalParseError> {
613    // Unwrap the generic "domain"/"annotation_domain" wrapper node(s) to see what's underneath.
614    let mut inner = domain_node;
615    while matches!(inner.kind(), "domain" | "annotation_domain") {
616        let Some(child) = inner.child(0) else {
617            break;
618        };
619        inner = child;
620    }
621
622    if inner.kind() == "identifier" {
623        // A name referring to a domain: keep it as a plain reference expression, so it prints
624        // back out as just the name (e.g. `restrict(f,D)`).
625        return parse_atom(ctx, &inner);
626    }
627
628    let Some(domain) = parse_domain(ctx, domain_node)? else {
629        return Ok(None);
630    };
631    Ok(Some(Expression::DomainAnnotation(
632        Metadata::new(),
633        Moo::new(Expression::Atomic(
634            Metadata::new(),
635            Atom::Literal(Literal::Bool(true)),
636        )),
637        domain,
638    )))
639}

641fn parse_index_or_slice(
642    ctx: &mut ParseContext,
643    node: &Node,
644) -> Result<Option<Expression>, FatalParseError> {
645    // add error and return early if we're in a set context, since indexing/slicing doesn't produce sets
646    if ctx.typechecking_context == TypecheckingContext::Set {
647        ctx.record_error(RecoverableParseError::new(
648            format!(
649                "Type error: {}\n\tExpected: set\n\tGot: index or slice",
650                ctx.source_code[node.start_byte()..node.end_byte()].trim()
651            ),
652            Some(node.range()),
653        ));
654        return Ok(None);
655    }
656
657    // Save current context and temporarily set to Unknown for the collection
658    let saved_context = ctx.typechecking_context;
659    ctx.typechecking_context = TypecheckingContext::Unknown;
660    let mut collection: Expression;
661    match parse_atom(ctx, &field!(node, "collection"))? {
662        Some(expr) => collection = expr,
663        None => return Ok(None),
664    }
665    ctx.typechecking_context = saved_context;
666
667    let indices_field = field!(node, "indices");
668    let mut idx_nodes = named_children(&indices_field).collect::<VecDeque<_>>();
669
670    // If LHS is a record, parse first index as a record field
671    if let ReturnType::Record(ents) | ReturnType::Variant(ents) = collection.return_type() {
672        let idx = idx_nodes
673            .pop_front()
674            .ok_or(FatalParseError::internal_error(
675                "Expected at least one indexing expression".into(),
676                Some(indices_field.range()),
677            ))?;
678        let name = Name::user(ctx.source_code[idx.start_byte()..idx.end_byte()].trim());
679        let has_name = ents.iter().any(|e| e.name == name);
680        if !has_name {
681            return Err(FatalParseError::internal_error(
682                format!("`{name}` is not a valid field name for `{collection}`"),
683                Some(idx.range()),
684            ));
685        }
686        collection = Expression::RecordField(Metadata::new(), Moo::new(collection), name);
687
688        // If there are no more indices, return the record field directly
689        if idx_nodes.is_empty() {
690            return Ok(Some(collection));
691        }
692    }
693
694    // Parse the rest of the indices as normal
695    let mut indices = Vec::new();
696    for idx_node in idx_nodes {
697        indices.push(parse_index(ctx, &idx_node)?);
698    }
699
700    let has_null_idx = indices.iter().any(|idx| idx.is_none());
701    // TODO: We could check whether the slice/index is safe here
702    if has_null_idx {
703        // It's a slice
704        Ok(Some(Expression::UnsafeSlice(
705            Metadata::new(),
706            Moo::new(collection),
707            indices,
708        )))
709    } else {
710        // It's an index
711        let idx_exprs: Vec<Expression> = indices.into_iter().map(|idx| idx.unwrap()).collect();
712        Ok(Some(Expression::UnsafeIndex(
713            Metadata::new(),
714            Moo::new(collection),
715            idx_exprs,
716        )))
717    }
718}

720fn parse_index(ctx: &mut ParseContext, node: &Node) -> Result<Option<Expression>, FatalParseError> {
721    match node.kind() {
722        "arithmetic_expr" | "atom" => {
723            let saved_context = ctx.typechecking_context;
724            ctx.typechecking_context = TypecheckingContext::Unknown;
725
726            // TODO: add collection-aware index typechecking.
727            // For tuple/matrix/set-like indexing, indices should be arithmetic.
728            // For record field access, index atoms should resolve to valid field names.
729            // This requires checking index expression together with the indexed collection type.
730
731            let Some(expr) = parse_expression(ctx, *node)? else {
732                return Ok(None);
733            };
734
735            ctx.typechecking_context = saved_context;
736            Ok(Some(expr))
737        }
738        "null_index" => Ok(None),
739        _ => {
740            ctx.record_error(RecoverableParseError::new(
741                format!("Expected an index, got: '{}'", node.kind()),
742                Some(node.range()),
743            ));
744            Ok(None)
745        }
746    }
747}

749fn parse_variable(ctx: &mut ParseContext, node: &Node) -> Result<Option<Atom>, FatalParseError> {
750    let raw_name = &ctx.source_code[node.start_byte()..node.end_byte()];
751
752    let name = Name::user(raw_name.trim());
753    if let Some(symbols) = &ctx.symbols {
754        let lookup_result = {
755            let symbols_read = symbols.read();
756            symbols_read.lookup(&name)
757        };
758
759        if let Some(decl) = lookup_result {
760            let symbol_kind = match &decl.kind().clone() as &DeclarationKind {
761                DeclarationKind::Find(_) | DeclarationKind::FindAuxiliary(_) => SymbolKind::FindVar,
762                DeclarationKind::Given(_) => SymbolKind::GivenVar,
763                DeclarationKind::ValueLetting(_, _) => SymbolKind::LettingVar,
764                DeclarationKind::TemporaryValueLetting(_) => SymbolKind::LettingVar,
765                DeclarationKind::DomainLetting(_) => SymbolKind::LettingVar,
766                DeclarationKind::Quantified(..) => SymbolKind::FindVar,
767                DeclarationKind::QuantifiedExpr(..) => SymbolKind::FindVar,
768                &_ => todo!(),
769            };
770
771            let hover = HoverInfo {
772                description: format!("Variable: {name}"),
773                doc_key: None,
774                kind: Some(symbol_kind),
775                ty: decl.domain().map(|d| d.to_string()),
776                decl_span: ctx.lookup_decl_span(&name),
777            };
778            span_with_hover(node, ctx.source_code, ctx.source_map, hover);
779
780            // Type check the variable against the expected context
781            if let Some(error_msg) = typecheck_variable(&decl, ctx.typechecking_context, raw_name) {
782                ctx.record_error(RecoverableParseError::new(error_msg, Some(node.range())));
783                return Ok(None);
784            }
785
786            Ok(Some(Atom::Reference(conjure_cp_core::ast::Reference::new(
787                decl,
788            ))))
789        } else {
790            ctx.record_error(RecoverableParseError::new(
791                format!("The identifier '{}' is not defined", raw_name),
792                Some(node.range()),
793            ));
794            Ok(None)
795        }
796    } else {
797        ctx.record_error(RecoverableParseError::new(
798            format!("Symbol table missing when parsing variable '{raw_name}'"),
799            Some(node.range()),
800        ));
801        Ok(None)
802    }
803}

805/// Type check a variable declaration against the expected expression context.
806/// Returns an error message if the variable type doesn't match the context.
807fn typecheck_variable(
808    decl: &DeclarationPtr,
809    context: TypecheckingContext,
810    raw_name: &str,
811) -> Option<String> {
812    // Only type check when context is known
813    if context == TypecheckingContext::Unknown {
814        return None;
815    }
816
817    // Get the variable's domain and resolve it
818    let domain = decl.domain()?;
819    let ground_domain = domain.resolve().ok()?;
820
821    // Determine what type is expected
822    let expected = match context {
823        TypecheckingContext::Boolean => "bool",
824        TypecheckingContext::Arithmetic => "int",
825        TypecheckingContext::Set => "set",
826        TypecheckingContext::SetOrMatrix => "set or matrix",
827        TypecheckingContext::MSet => "mset",
828        TypecheckingContext::Matrix => "matrix",
829        TypecheckingContext::Tuple => "tuple",
830        TypecheckingContext::Record => "record",
831        TypecheckingContext::Partition => "partition",
832        TypecheckingContext::Permutation => "permutation",
833        TypecheckingContext::Sequence => "sequence",
834        TypecheckingContext::Function => "function",
835        TypecheckingContext::Relation => "relation",
836        TypecheckingContext::Unknown => return None, // shouldn't reach here
837    };
838
839    // Determine what type we actually have
840    let actual = match ground_domain.as_ref() {
841        GroundDomain::Empty(_) => "empty",
842        GroundDomain::Bool => "bool",
843        GroundDomain::Int(_) => "int",
844        GroundDomain::Tuple(_) => "tuple",
845        GroundDomain::Record(_) => "record",
846        GroundDomain::Variant(_) => "variant",
847        GroundDomain::Matrix(_, _) => "matrix",
848        GroundDomain::Sequence(_, _) => "sequence",
849        GroundDomain::Set(_, _) => "set",
850        GroundDomain::MSet(_, _) => "mset",
851        GroundDomain::Function(_, _, _) => "function",
852        GroundDomain::Relation(_, _) => "relation",
853        GroundDomain::Partition(_, _) => "partition",
854        GroundDomain::Permutation(_, _) => "permutation",
855    };
856
857    // If types match, no error
858    if expected == actual
859        || (context == TypecheckingContext::SetOrMatrix && matches!(actual, "set" | "matrix"))
860    {
861        return None;
862    }
863
864    // Otherwise, report the type mismatch
865    Some(format!(
866        "Type error: {}\n\tExpected: {}\n\tGot: {}",
867        raw_name, expected, actual
868    ))
869}

871fn parse_constant(ctx: &mut ParseContext, node: &Node) -> Result<Option<Literal>, FatalParseError> {
872    let Some(inner) = named_child!(recover, ctx, node) else {
873        return Ok(None);
874    };
875    let raw_value = &ctx.source_code[inner.start_byte()..inner.end_byte()];
876    let lit = match inner.kind() {
877        "integer" => {
878            let Some(value) = parse_int(ctx, &inner) else {
879                return Ok(None);
880            };
881            Literal::Int(value)
882        }
883        "TRUE" => {
884            let hover = HoverInfo {
885                description: format!("Boolean constant: {raw_value}"),
886                doc_key: None,
887                kind: None,
888                ty: None,
889                decl_span: None,
890            };
891            span_with_hover(&inner, ctx.source_code, ctx.source_map, hover);
892            Literal::Bool(true)
893        }
894        "FALSE" => {
895            let hover = HoverInfo {
896                description: format!("Boolean constant: {raw_value}"),
897                doc_key: None,
898                kind: None,
899                ty: None,
900                decl_span: None,
901            };
902            span_with_hover(&inner, ctx.source_code, ctx.source_map, hover);
903            Literal::Bool(false)
904        }
905        _ => {
906            ctx.record_error(RecoverableParseError::new(
907                format!(
908                    "'{}' (kind: '{}') is not a valid constant",
909                    raw_value,
910                    inner.kind()
911                ),
912                Some(inner.range()),
913            ));
914            return Ok(None);
915        }
916    };
917
918    // Type check the constant against the expected context
919    if ctx.typechecking_context != TypecheckingContext::Unknown {
920        let expected = match ctx.typechecking_context {
921            TypecheckingContext::Boolean => "bool",
922            TypecheckingContext::Arithmetic => "int",
923            TypecheckingContext::Set => "set",
924            TypecheckingContext::SetOrMatrix => "set or matrix",
925            TypecheckingContext::MSet => "mset",
926            TypecheckingContext::Matrix => "matrix",
927            TypecheckingContext::Tuple => "tuple",
928            TypecheckingContext::Record => "record",
929            TypecheckingContext::Partition => "partition",
930            TypecheckingContext::Permutation => "permutation",
931            TypecheckingContext::Sequence => "sequence",
932            TypecheckingContext::Function => "function",
933            TypecheckingContext::Relation => "relation",
934            TypecheckingContext::Unknown => "",
935        };
936
937        let actual = match &lit {
938            Literal::Bool(_) => "bool",
939            Literal::Int(_) => "int",
940            Literal::AbstractLiteral(_) => return Ok(None), // Abstract literals aren't type-checked here
941        };
942
943        if expected != actual {
944            ctx.record_error(RecoverableParseError::new(
945                format!(
946                    "Type error: {}\n\tExpected: {}\n\tGot: {}",
947                    raw_value, expected, actual
948                ),
949                Some(node.range()),
950            ));
951            return Ok(None);
952        }
953    }
954    Ok(Some(lit))
955}

957pub(crate) fn parse_int(ctx: &mut ParseContext, node: &Node) -> Option<i32> {
958    let raw_value = &ctx.source_code[node.start_byte()..node.end_byte()];
959    if let Ok(v) = raw_value.parse::<i32>() {
960        Some(v)
961    } else {
962        ctx.record_error(RecoverableParseError::new(
963            "Expected an integer here".to_string(),
964            Some(node.range()),
965        ));
966        None
967    }
