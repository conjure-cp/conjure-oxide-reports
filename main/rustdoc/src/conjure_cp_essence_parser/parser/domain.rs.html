12/// Parse an Essence variable domain into its Conjure AST representation.
13pub fn parse_domain(
14    domain: Node,
15    source_code: &str,
16    symbols: Option<SymbolTablePtr>,
17    errors: &mut Vec<RecoverableParseError>,
18) -> Result<Option<DomainPtr>, FatalParseError> {
19    match domain.kind() {
20        "domain" => parse_domain(child!(domain, 0, "domain"), source_code, symbols, errors),
21        "bool_domain" => Ok(Some(Domain::bool())),
22        "int_domain" => parse_int_domain(domain, source_code, &symbols, errors),
23        "identifier" => {
24            let Some(decl) =
25                get_declaration_ptr_from_identifier(domain, source_code, &symbols, errors)?
26            else {
27                return Ok(None);
28            };
29            let Some(dom) = Domain::reference(decl) else {
30                errors.push(RecoverableParseError::new(
31                    format!(
32                        "The identifier '{}' is not a valid domain",
33                        &source_code[domain.start_byte()..domain.end_byte()]
34                    ),
35                    Some(domain.range()),
36                ));
37                return Ok(None);
38            };
39            Ok(Some(dom))
40        }
41        "tuple_domain" => parse_tuple_domain(domain, source_code, symbols, errors),
42        "matrix_domain" => parse_matrix_domain(domain, source_code, symbols, errors),
43        "record_domain" => parse_record_domain(domain, source_code, symbols, errors),
44        "set_domain" => parse_set_domain(domain, source_code, symbols, errors),
45        _ => panic!("{} is not a supported domain type", domain.kind()),
46    }
47}

49fn get_declaration_ptr_from_identifier(
50    identifier: Node,
51    source_code: &str,
52    symbols_ptr: &Option<SymbolTablePtr>,
53    errors: &mut Vec<RecoverableParseError>,
54) -> Result<Option<DeclarationPtr>, FatalParseError> {
55    let name = Name::user(&source_code[identifier.start_byte()..identifier.end_byte()]);
56    let decl = symbols_ptr
57        .as_ref()
58        .ok_or(FatalParseError::internal_error(
59            "context needed to resolve identifier".to_string(),
60            Some(identifier.range()),
61        ))?
62        .read()
63        .lookup(&name);
64    match decl {
65        Some(decl) => Ok(Some(decl)),
66        None => {
67            errors.push(RecoverableParseError::new(
68                format!("The identifier '{}' is not defined", name),
69                Some(identifier.range()),
70            ));
71            Ok(None)
72        }
73    }
74}

76/// Parse an integer domain. Can be a single integer or a range.
77fn parse_int_domain(
78    int_domain: Node,
79    source_code: &str,
80    symbols_ptr: &Option<SymbolTablePtr>,
81    errors: &mut Vec<RecoverableParseError>,
82) -> Result<Option<DomainPtr>, FatalParseError> {
83    if int_domain.child_count() == 1 {
84        return Ok(Some(Domain::int(vec![Range::Bounded(i32::MIN, i32::MAX)])));
85    }
86    let mut ranges: Vec<Range<i32>> = Vec::new();
87    let mut ranges_unresolved: Vec<Range<IntVal>> = Vec::new();
88    let range_list = field!(int_domain, "ranges");
89    for domain_component in named_children(&range_list) {
90        match domain_component.kind() {
91            "atom" => {
92                let text = &source_code[domain_component.start_byte()..domain_component.end_byte()];
93                // Try parsing as a literal integer first
94                if let Ok(integer) = text.parse::<i32>() {
95                    ranges.push(Range::Single(integer));
96                    continue;
97                }
98                // Otherwise, treat as a reference
99                let Some(decl) = get_declaration_ptr_from_identifier(
100                    domain_component,
101                    source_code,
102                    symbols_ptr,
103                    errors,
104                )?
105                else {
106                    return Ok(None);
107                };
108                ranges_unresolved.push(Range::Single(IntVal::Reference(Reference::new(decl))));
109            }
110            "int_range" => {
111                let lower_bound: Option<Result<i32, DeclarationPtr>> =
112                    match domain_component.child_by_field_name("lower") {
113                        Some(lower_node) => {
114                            // Try parsing as a literal integer first
115                            let text = &source_code[lower_node.start_byte()..lower_node.end_byte()];
116                            if let Ok(integer) = text.parse::<i32>() {
117                                Some(Ok(integer))
118                            } else {
119                                let Some(decl) = get_declaration_ptr_from_identifier(
120                                    lower_node,
121                                    source_code,
122                                    symbols_ptr,
123                                    errors,
124                                )?
125                                else {
126                                    return Ok(None); // return from function if we can't resolve the identifier
127                                };
128                                Some(Err(decl))
129                            }
130                        }
131                        None => None,
132                    };
133                let upper_bound: Option<Result<i32, DeclarationPtr>> =
134                    match domain_component.child_by_field_name("upper") {
135                        Some(upper_node) => {
136                            // Try parsing as a literal integer first
137                            let text = &source_code[upper_node.start_byte()..upper_node.end_byte()];
138                            if let Ok(integer) = text.parse::<i32>() {
139                                Some(Ok(integer))
140                            } else {
141                                let Some(decl) = get_declaration_ptr_from_identifier(
142                                    upper_node,
143                                    source_code,
144                                    symbols_ptr,
145                                    errors,
146                                )?
147                                else {
148                                    return Ok(None); // return from function if we can't resolve the identifier
149                                };
150                                Some(Err(decl))
151                            }
152                        }
153                        None => None,
154                    };
155
156                match (lower_bound, upper_bound) {
157                    (Some(Ok(lower)), Some(Ok(upper))) => ranges.push(Range::Bounded(lower, upper)),
158                    (Some(Ok(lower)), Some(Err(decl))) => {
159                        ranges_unresolved.push(Range::Bounded(
160                            IntVal::Const(lower),
161                            IntVal::Reference(Reference::new(decl)),
162                        ));
163                    }
164                    (Some(Err(decl)), Some(Ok(upper))) => {
165                        ranges_unresolved.push(Range::Bounded(
166                            IntVal::Reference(Reference::new(decl)),
167                            IntVal::Const(upper),
168                        ));
169                    }
170                    (Some(Err(decl_lower)), Some(Err(decl_upper))) => {
171                        ranges_unresolved.push(Range::Bounded(
172                            IntVal::Reference(Reference::new(decl_lower)),
173                            IntVal::Reference(Reference::new(decl_upper)),
174                        ));
175                    }
176                    (Some(Ok(lower)), None) => {
177                        ranges.push(Range::UnboundedR(lower));
178                    }
179                    (Some(Err(decl)), None) => {
180                        ranges_unresolved
181                            .push(Range::UnboundedR(IntVal::Reference(Reference::new(decl))));
182                    }
183                    (None, Some(Ok(upper))) => {
184                        ranges.push(Range::UnboundedL(upper));
185                    }
186                    (None, Some(Err(decl))) => {
187                        ranges_unresolved
188                            .push(Range::UnboundedL(IntVal::Reference(Reference::new(decl))));
189                    }
190                    (None, None) => {
191                        ranges.push(Range::Unbounded);
192                    }
193                }
194            }
195            _ => panic!("unsupported int range type"),
196        }
197    }
198
199    if !ranges_unresolved.is_empty() {
200        for range in ranges {
201            match range {
202                Range::Single(i) => ranges_unresolved.push(Range::Single(IntVal::Const(i))),
203                Range::Bounded(l, u) => {
204                    ranges_unresolved.push(Range::Bounded(IntVal::Const(l), IntVal::Const(u)))
205                }
206                Range::UnboundedL(l) => ranges_unresolved.push(Range::UnboundedL(IntVal::Const(l))),
207                Range::UnboundedR(u) => ranges_unresolved.push(Range::UnboundedR(IntVal::Const(u))),
208                Range::Unbounded => ranges_unresolved.push(Range::Unbounded),
209            }
210        }
211        return Ok(Some(Domain::int(ranges_unresolved)));
212    }
213
214    Ok(Some(Domain::int(ranges)))
215}

217fn parse_tuple_domain(
218    tuple_domain: Node,
219    source_code: &str,
220    symbols: Option<SymbolTablePtr>,
221    errors: &mut Vec<RecoverableParseError>,
222) -> Result<Option<DomainPtr>, FatalParseError> {
223    let mut domains: Vec<DomainPtr> = Vec::new();
224    for domain in named_children(&tuple_domain) {
225        let Some(parsed_domain) = parse_domain(domain, source_code, symbols.clone(), errors)?
226        else {
227            return Ok(None);
228        };
229        domains.push(parsed_domain);
230    }
231    Ok(Some(Domain::tuple(domains)))
232}

234fn parse_matrix_domain(
235    matrix_domain: Node,
236    source_code: &str,
237    symbols: Option<SymbolTablePtr>,
238    errors: &mut Vec<RecoverableParseError>,
239) -> Result<Option<DomainPtr>, FatalParseError> {
240    let mut domains: Vec<DomainPtr> = Vec::new();
241    let index_domain_list = field!(matrix_domain, "index_domain_list");
242    for domain in named_children(&index_domain_list) {
243        let Some(parsed_domain) = parse_domain(domain, source_code, symbols.clone(), errors)?
244        else {
245            return Ok(None);
246        };
247        domains.push(parsed_domain);
248    }
249    let Some(value_domain) = parse_domain(
250        field!(matrix_domain, "value_domain"),
251        source_code,
252        symbols,
253        errors,
254    )?
255    else {
256        return Ok(None);
257    };
258    Ok(Some(Domain::matrix(value_domain, domains)))
259}

261fn parse_record_domain(
262    record_domain: Node,
263    source_code: &str,
264    symbols: Option<SymbolTablePtr>,
265    errors: &mut Vec<RecoverableParseError>,
266) -> Result<Option<DomainPtr>, FatalParseError> {
267    let mut record_entries: Vec<RecordEntry> = Vec::new();
268    for record_entry in named_children(&record_domain) {
269        let name_node = field!(record_entry, "name");
270        let name = Name::user(&source_code[name_node.start_byte()..name_node.end_byte()]);
271        let domain_node = field!(record_entry, "domain");
272        let Some(domain) = parse_domain(domain_node, source_code, symbols.clone(), errors)? else {
273            return Ok(None);
274        };
275        record_entries.push(RecordEntry { name, domain });
276    }
277    Ok(Some(Domain::record(record_entries)))
278}

280pub fn parse_set_domain(
281    set_domain: Node,
282    source_code: &str,
283    symbols: Option<SymbolTablePtr>,
284    errors: &mut Vec<RecoverableParseError>,
285) -> Result<Option<DomainPtr>, FatalParseError> {
286    let mut set_attribute: Option<SetAttr> = None;
287    let mut value_domain: Option<DomainPtr> = None;
288
289    for child in named_children(&set_domain) {
290        match child.kind() {
291            "set_attributes" => {
292                // Check if we have both minSize and maxSize (minMax case)
293                let min_value_node = child.child_by_field_name("min_value");
294                let max_value_node = child.child_by_field_name("max_value");
295                let size_value_node = child.child_by_field_name("size_value");
296
297                if let (Some(min_node), Some(max_node)) = (min_value_node, max_value_node) {
298                    // MinMax case
299                    let min_val = parse_int(&min_node, source_code, errors)?;
300                    let max_val = parse_int(&max_node, source_code, errors)?;
301
302                    set_attribute = Some(SetAttr::new_min_max_size(min_val, max_val));
303                } else if let Some(size_node) = size_value_node {
304                    // Size case
305                    let size_val = parse_int(&size_node, source_code, errors)?;
306                    set_attribute = Some(SetAttr::new_size(size_val));
307                } else if let Some(min_node) = min_value_node {
308                    // MinSize only case
309                    let min_val = parse_int(&min_node, source_code, errors)?;
310                    set_attribute = Some(SetAttr::new_min_size(min_val));
311                } else if let Some(max_node) = max_value_node {
312                    // MaxSize only case
313                    let max_val = parse_int(&max_node, source_code, errors)?;
314                    set_attribute = Some(SetAttr::new_max_size(max_val));
315                }
316            }
317            "domain" => {
318                let Some(parsed_domain) =
319                    parse_domain(child, source_code, symbols.clone(), errors)?
320                else {
321                    return Ok(None);
322                };
323                value_domain = Some(parsed_domain);
324            }
325            _ => {
326                return Err(FatalParseError::internal_error(
327                    format!("Unrecognized set domain child kind: {}", child.kind()),
328                    Some(child.range()),
329                ));
330            }
331        }
332    }
333
334    if let Some(domain) = value_domain {
335        Ok(Some(Domain::set(set_attribute.unwrap_or_default(), domain)))
336    } else {
337        Err(FatalParseError::internal_error(
338            "Set domain must have a value domain".to_string(),
339            Some(set_domain.range()),
340        ))
341    }
