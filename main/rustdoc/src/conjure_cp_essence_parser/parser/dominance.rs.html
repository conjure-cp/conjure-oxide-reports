26pub fn parse_dominance_relation(
27    ctx: &mut ParseContext,
28    node: &Node,
29) -> Result<Option<Expression>, FatalParseError> {
30    if ctx.root.kind() == "dominance_relation" {
31        ctx.record_error(RecoverableParseError::new(
32            "Nested dominance relations are not allowed".to_string(),
33            Some(node.range()),
34        ));
35        return Ok(None);
36    }
37
38    let Some(inner_node) = field!(recover, ctx, node, "expression") else {
39        return Ok(None);
40    };
41
42    // Create a nested context so downstream parsing knows it is inside a dominance relation.
43    let mut inner_ctx = ParseContext {
44        source_code: ctx.source_code,
45        root: node,
46        symbols: ctx.symbols.clone(),
47        errors: ctx.errors,
48        source_map: &mut *ctx.source_map,
49        decl_spans: ctx.decl_spans,
50        typechecking_context: TypecheckingContext::Unknown,
51        inner_typechecking_context: TypecheckingContext::Unknown,
52    };
53
54    let Some(inner) = parse_expression(&mut inner_ctx, inner_node)? else {
55        return Ok(None);
56    };
57
58    Ok(Some(Expression::DominanceRelation(
59        Metadata::new(),
60        Moo::new(inner),
61    )))
62}

64pub fn parse_pareto_expression(
65    ctx: &mut ParseContext,
66    node: &Node,
67) -> Result<Option<Expression>, FatalParseError> {
68    if ctx.root.kind() != "dominance_relation" {
69        ctx.record_error(RecoverableParseError::new(
70            "pareto(...) only allowed inside dominance relations".to_string(),
71            Some(node.range()),
72        ));
73        return Ok(None);
74    }
75
76    let mut non_worsening = Vec::new();
77    let mut strict_improvements = Vec::new();
78    let components = field!(node, "components");
79
80    if components.kind() != "pareto_items" {
81        return Err(FatalParseError::internal_error(
82            format!("Unexpected pareto component list: '{}'", components.kind()),
83            Some(components.range()),
84        ));
85    }
86
87    for item_node in named_children(&components) {
88        let direction_node = field!(item_node, "direction");
89        let direction_str =
90            &ctx.source_code[direction_node.start_byte()..direction_node.end_byte()];
91        let direction = match direction_str {
92            "minimising" => ParetoDirection::Minimising,
93            "maximising" => ParetoDirection::Maximising,
94            _ => {
95                return Err(FatalParseError::internal_error(
96                    format!("Unexpected pareto direction: '{direction_str}'"),
97                    Some(direction_node.range()),
98                ));
99            }
100        };
101
102        let component_node = field!(item_node, "expression");
103        let Some(component_expr) = parse_pareto_component(ctx, &component_node)? else {
104            return Ok(None);
105        };
106        let Some((non_worse, strict)) =
107            build_pareto_constraints(ctx, &component_node, component_expr, direction)
108        else {
109            return Ok(None);
110        };
111        non_worsening.push(non_worse);
112        strict_improvements.push(strict);
113    }
114
115    let mut conjuncts = non_worsening;
116    conjuncts.push(combine_with_and_or(strict_improvements, true));
117
118    Ok(Some(combine_with_and_or(conjuncts, false)))
119}

121fn parse_pareto_component(
122    ctx: &mut ParseContext,
123    node: &Node,
124) -> Result<Option<Expression>, FatalParseError> {
125    let saved_context = ctx.typechecking_context;
126    let saved_inner_context = ctx.inner_typechecking_context;
127    ctx.typechecking_context = TypecheckingContext::Unknown;
128    ctx.inner_typechecking_context = TypecheckingContext::Unknown;
129
130    let parsed = parse_expression(ctx, *node);
131
132    ctx.typechecking_context = saved_context;
133    ctx.inner_typechecking_context = saved_inner_context;
134    parsed
135}

137fn build_pareto_constraints(
138    ctx: &mut ParseContext,
139    node: &Node,
140    component: Expression,
141    direction: ParetoDirection,
142) -> Option<(Expression, Expression)> {
143    if component
144        .universe()
145        .iter()
146        .any(|expr| matches!(expr, Expression::FromSolution(_, _)))
147    {
148        ctx.record_error(RecoverableParseError::new(
149            "pareto(...) components cannot contain fromSolution(...) explicitly".to_string(),
150            Some(node.range()),
151        ));
152        return None;
153    }
154
155    let current = expand_value_lettings(&component);
156    let previous = lift_to_previous_solution(&current);
157
158    match current.return_type() {
159        ReturnType::Int => Some(match direction {
160            ParetoDirection::Minimising => (
161                Expression::Leq(
162                    Metadata::new(),
163                    Moo::new(current.clone()),
164                    Moo::new(previous.clone()),
165                ),
166                Expression::Lt(Metadata::new(), Moo::new(current), Moo::new(previous)),
167            ),
168            ParetoDirection::Maximising => (
169                Expression::Geq(
170                    Metadata::new(),
171                    Moo::new(current.clone()),
172                    Moo::new(previous.clone()),
173                ),
174                Expression::Gt(Metadata::new(), Moo::new(current), Moo::new(previous)),
175            ),
176        }),
177        ReturnType::Bool => Some(match direction {
178            ParetoDirection::Minimising => (
179                Expression::Imply(
180                    Metadata::new(),
181                    Moo::new(current.clone()),
182                    Moo::new(previous.clone()),
183                ),
184                combine_with_and_or(
185                    vec![
186                        Expression::Not(Metadata::new(), Moo::new(current)),
187                        previous,
188                    ],
189                    false,
190                ),
191            ),
192            ParetoDirection::Maximising => (
193                Expression::Imply(
194                    Metadata::new(),
195                    Moo::new(previous.clone()),
196                    Moo::new(current.clone()),
197                ),
198                combine_with_and_or(
199                    vec![
200                        current,
201                        Expression::Not(Metadata::new(), Moo::new(previous)),
202                    ],
203                    false,
204                ),
205            ),
206        }),
207        found => {
208            ctx.record_error(RecoverableParseError::new(
209                format!(
210                    "pareto(...) only supports int or bool components, found '{}'",
211                    found
212                ),
213                Some(node.range()),
214            ));
215            None
216        }
217    }
218}

220fn expand_value_lettings(expr: &Expression) -> Expression {
221    rewrite_references(expr, false)
222}

224fn lift_to_previous_solution(expr: &Expression) -> Expression {
225    rewrite_references(expr, true)
226}

228fn rewrite_references(expr: &Expression, to_previous_solution: bool) -> Expression {
229    let mut lifted = expr.clone();
230
231    loop {
232        let next = lifted.rewrite(&|subexpr| match subexpr {
233            Expression::Atomic(_, Atom::Reference(ref reference)) => {
234                let action = {
235                    let kind = reference.ptr.kind();
236                    match &*kind {
237                        DeclarationKind::Find(_) if to_previous_solution => {
238                            ReferenceRewriteAction::WrapInFromSolution
239                        }
240                        DeclarationKind::Find(_) => ReferenceRewriteAction::LeaveAsIs,
241                        DeclarationKind::ValueLetting(_, _)
242                        | DeclarationKind::TemporaryValueLetting(_) => {
243                            ReferenceRewriteAction::ExpandValueLetting
244                        }
245                        DeclarationKind::Given(_)
246                        | DeclarationKind::Quantified(_)
247                        | DeclarationKind::QuantifiedExpr(_)
248                        | DeclarationKind::DomainLetting(_)
249                        | _ => ReferenceRewriteAction::LeaveAsIs,
250                    }
251                };
252
253                match action {
254                    ReferenceRewriteAction::LeaveAsIs => Some(subexpr),
255                    ReferenceRewriteAction::ExpandValueLetting => reference.resolve_expression(),
256                    ReferenceRewriteAction::WrapInFromSolution => Some(Expression::FromSolution(
257                        Metadata::new(),
258                        Moo::new(Atom::Reference(reference.clone())),
259                    )),
260                }
261            }
262            _ => Some(subexpr),
263        });
264
265        if next == lifted {
266            return lifted;
267        }
268
269        lifted = next;
270    }
271}

273fn combine_with_and_or(exprs: Vec<Expression>, is_or: bool) -> Expression {
274    match exprs.len() {
275        0 => {
276            if is_or {
277                Expression::Or(Metadata::new(), Moo::new(into_matrix_expr![exprs]))
278            } else {
279                Expression::And(Metadata::new(), Moo::new(into_matrix_expr![exprs]))
280            }
281        }
282        1 => match exprs.into_iter().next() {
283            Some(expr) => expr,
284            None => unreachable!("vector length already checked"),
285        },
286        _ => {
287            if is_or {
288                Expression::Or(Metadata::new(), Moo::new(into_matrix_expr![exprs]))
289            } else {
290                Expression::And(Metadata::new(), Moo::new(into_matrix_expr![exprs]))
291            }
292        }
293    }
