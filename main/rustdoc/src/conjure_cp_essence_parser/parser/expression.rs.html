21pub fn parse_expression(
22    ctx: &mut ParseContext,
23    node: Node,
24) -> Result<Option<Expression>, FatalParseError> {
25    match node.kind() {
26        "atom"
27        | "primary_atom"
28        | "constant"
29        | "identifier"
30        | "metavar"
31        | "matrix"
32        | "record"
33        | "variant"
34        | "tuple"
35        | "set_literal"
36        | "mset_literal"
37        | "sequence_literal"
38        | "function_literal"
39        | "relation_literal"
40        | "partition_literal"
41        | "comprehension"
42        | "index_or_slice"
43        | "flatten"
44        | "element_id"
45        | "table"
46        | "negative_table"
47        | "image_expr"
48        | "image_set_expr"
49        | "pre_image_expr"
50        | "inverse_expr"
51        | "restrict_expr"
52        | "defined_expr"
53        | "range_expr"
54        | "to_set_expr"
55        | "to_mset_expr"
56        | "to_relation_expr"
57        | "attribute_as_constraint_expr"
58        | "together_expr"
59        | "apart_expr"
60        | "party_expr"
61        | "parts_expr"
62        | "participants_expr" => parse_atom(ctx, &node),
63        "bool_expr" => {
64            if ctx.typechecking_context == TypecheckingContext::Arithmetic {
65                ctx.record_error(RecoverableParseError::new(
66                    format!(
67                        "Type error: {}\n\tExepected: int\n\tGot: boolean expression",
68                        &ctx.source_code[node.start_byte()..node.end_byte()]
69                    ),
70                    Some(node.range()),
71                ));
72                return Ok(None);
73            }
74            parse_boolean_expression(ctx, &node)
75        }
76        "arithmetic_expr" => {
77            if ctx.typechecking_context == TypecheckingContext::Boolean {
78                ctx.record_error(RecoverableParseError::new(
79                    format!(
80                        "Type error: {}\n\tExepected: bool\n\tGot: arithmetic expression",
81                        &ctx.source_code[node.start_byte()..node.end_byte()]
82                    ),
83                    Some(node.range()),
84                ));
85                return Ok(None);
86            }
87            parse_arithmetic_expression(ctx, &node)
88        }
89        "comparison_expr" => {
90            if ctx.typechecking_context == TypecheckingContext::Arithmetic {
91                ctx.record_error(RecoverableParseError::new(
92                    format!(
93                        "Type error: {}\n\tExepected: int\n\tGot: comparison expression",
94                        &ctx.source_code[node.start_byte()..node.end_byte()]
95                    ),
96                    Some(node.range()),
97                ));
98                return Ok(None);
99            }
100            parse_comparison_expression(ctx, &node)
101        }
102        "all_diff_comparison" => {
103            if ctx.typechecking_context == TypecheckingContext::Arithmetic {
104                ctx.record_error(RecoverableParseError::new(
105                    format!("Type error: {}\n\tExepected: arithmetic expression\n\tFound: comparison expression", &ctx.source_code[node.start_byte()..node.end_byte()]),
106                    Some(node.range()),
107                ));
108                return Ok(None);
109            }
110            ctx.typechecking_context = TypecheckingContext::Matrix;
111            parse_all_diff_comparison(ctx, &node)
112        }
113        "all_different_except_comparison" => {
114            if ctx.typechecking_context == TypecheckingContext::Arithmetic {
115                ctx.record_error(RecoverableParseError::new(
116                    format!("Type error: {}\n\tExepected: arithmetic expression\n\tFound: comparison expression", &ctx.source_code[node.start_byte()..node.end_byte()]),
117                    Some(node.range()),
118                ));
119                return Ok(None);
120            }
121            ctx.typechecking_context = TypecheckingContext::Matrix;
122            parse_all_different_except_comparison(ctx, &node)
123        }
124        "global_cardinality_comparison" => {
125            if ctx.typechecking_context == TypecheckingContext::Arithmetic {
126                ctx.record_error(RecoverableParseError::new(
127                    format!("Type error: {}\n\tExepected: arithmetic expression\n\tFound: comparison expression", &ctx.source_code[node.start_byte()..node.end_byte()]),
128                    Some(node.range()),
129                ));
130                return Ok(None);
131            }
132            parse_global_cardinality_comparison(ctx, &node)
133        }
134        "annotation_expr" | "type_annotation" | "domain_annotation" => {
135            parse_annotation_expression(ctx, &node)
136        }
137        _ => {
138            ctx.record_error(RecoverableParseError::new(
139                format!("Unexpected expression type: '{}'", node.kind()),
140                Some(node.range()),
141            ));
142            Ok(None)
143        }
144    }
145}

147fn parse_arithmetic_expression(
148    ctx: &mut ParseContext,
149    node: &Node,
150) -> Result<Option<Expression>, FatalParseError> {
151    ctx.typechecking_context = TypecheckingContext::Arithmetic;
152    ctx.inner_typechecking_context = TypecheckingContext::Unknown;
153    let Some(inner) = named_child!(recover, ctx, node) else {
154        return Ok(None);
155    };
156    match inner.kind() {
157        "atom" => parse_atom(ctx, &inner),
158        "negative_expr" | "abs_value" | "sub_arith_expr" | "factorial_expr" => {
159            parse_unary_expression(ctx, &inner)
160        }
161        "toInt_expr" => {
162            // add special handling for toInt, as it is arithmetic but takes a non-arithmetic operand
163            ctx.typechecking_context = TypecheckingContext::Unknown;
164            parse_unary_expression(ctx, &inner)
165        }
166        "exponent" | "product_expr" | "sum_expr" => parse_binary_expression(ctx, &inner),
167        "list_combining_expr_arith" => {
168            // list-combining arithmetic operators accept either set or matrix operands
169            ctx.typechecking_context = TypecheckingContext::SetOrMatrix;
170
171            // set inner context to arithmetic to ensure elements of list are arithmetic expressions
172            ctx.inner_typechecking_context = TypecheckingContext::Arithmetic;
173            parse_list_combining_expression(ctx, &inner)
174        }
175        "aggregate_expr" => {
176            ctx.inner_typechecking_context = TypecheckingContext::Arithmetic;
177            parse_quantifier_or_aggregate_expr(ctx, &inner)
178        }
179        _ => {
180            ctx.record_error(RecoverableParseError::new(
181                format!("Expected arithmetic expression, found: {}", inner.kind()),
182                Some(inner.range()),
183            ));
184            Ok(None)
185        }
186    }
187}

189fn parse_comparison_expression(
190    ctx: &mut ParseContext,
191    node: &Node,
192) -> Result<Option<Expression>, FatalParseError> {
193    let Some(inner) = named_child!(recover, ctx, node) else {
194        return Ok(None);
195    };
196    match inner.kind() {
197        "arithmetic_comparison" => {
198            // <, <=, >, >= work on any orderable type (everything except unnamed types, which
199            // don't support ordering); typechecking of operands is handled within
200            // parse_binary_expression, same as equality.
201            ctx.typechecking_context = TypecheckingContext::Unknown;
202            parse_binary_expression(ctx, &inner)
203        }
204        "lex_comparison" => {
205            // <lex, <=lex, >lex, >=lex are specialised to matrices; other types use the plain
206            // ordering operators above instead.
207            ctx.typechecking_context = TypecheckingContext::Matrix;
208            parse_binary_expression(ctx, &inner)
209        }
210        "equality_comparison" => {
211            // Equality works on any type, typechecking of operands will be handled within parse_binary_expression
212            ctx.typechecking_context = TypecheckingContext::Unknown;
213            parse_binary_expression(ctx, &inner)
214        }
215        "set_comparison" => {
216            // Set comparisons require set operands (except 'in', which is hadled later)
217            ctx.typechecking_context = TypecheckingContext::Set;
218            parse_binary_expression(ctx, &inner)
219        }
220        "sequence_comparison" => {
221            // Sequence comparisons (substring/subsequence) require sequence operands
222            ctx.typechecking_context = TypecheckingContext::Sequence;
223            parse_binary_expression(ctx, &inner)
224        }
225        "all_diff_comparison" => {
226            ctx.typechecking_context = TypecheckingContext::Matrix;
227            parse_all_diff_comparison(ctx, &inner)
228        }
229        "all_different_except_comparison" => {
230            ctx.typechecking_context = TypecheckingContext::Matrix;
231            parse_all_different_except_comparison(ctx, &inner)
232        }
233        "global_cardinality_comparison" => parse_global_cardinality_comparison(ctx, &inner),
234        _ => {
235            ctx.record_error(RecoverableParseError::new(
236                format!("Expected comparison expression, found '{}'", inner.kind()),
237                Some(inner.range()),
238            ));
239            Ok(None)
240        }
241    }
242}

244fn parse_boolean_expression(
245    ctx: &mut ParseContext,
246    node: &Node,
247) -> Result<Option<Expression>, FatalParseError> {
248    ctx.typechecking_context = TypecheckingContext::Boolean;
249    ctx.inner_typechecking_context = TypecheckingContext::Unknown;
250    let Some(inner) = named_child!(recover, ctx, node) else {
251        return Ok(None);
252    };
253    match inner.kind() {
254        "atom" | "table" | "negative_table" => parse_atom(ctx, &inner),
255        "active_expr" => parse_active_expression(ctx, &inner),
256        "not_expr" | "sub_bool_expr" => parse_unary_expression(ctx, &inner),
257        "and_expr" | "or_expr" | "implication" | "iff_expr" => parse_binary_expression(ctx, &inner),
258        "list_combining_expr_bool" => {
259            // list-combining boolean operators accept either set or matrix operands
260            ctx.typechecking_context = TypecheckingContext::SetOrMatrix;
261
262            // set inner context to boolean to ensure elements of list are boolean expressions
263            ctx.inner_typechecking_context = TypecheckingContext::Boolean;
264            parse_list_combining_expression(ctx, &inner)
265        }
266        "quantifier_expr" => parse_quantifier_or_aggregate_expr(ctx, &inner),
267        _ => {
268            ctx.record_error(RecoverableParseError::new(
269                format!("Expected boolean expression, found '{}'", inner.kind()),
270                Some(inner.range()),
271            ));
272            Ok(None)
273        }
274    }
275}

277fn parse_active_expression(
278    ctx: &mut ParseContext,
279    node: &Node,
280) -> Result<Option<Expression>, FatalParseError> {
281    let Some(variant_node) = field!(recover, ctx, node, "variant") else {
282        return Ok(None);
283    };
284    let saved_context = ctx.typechecking_context;
285    ctx.typechecking_context = TypecheckingContext::Unknown;
286    let variant = parse_expression(ctx, variant_node)?;
287    ctx.typechecking_context = saved_context;
288    let Some(variant) = variant else {
289        return Ok(None);
290    };
291    let Some(name_node) = field!(recover, ctx, node, "name") else {
292        return Ok(None);
293    };
294    let name = conjure_cp_core::ast::Name::user(
295        &ctx.source_code[name_node.start_byte()..name_node.end_byte()],
296    );
297    let ReturnType::Variant(entries) = variant.return_type() else {
298        ctx.record_error(RecoverableParseError::new(
299            format!("active expects a variant, got {variant}"),
300            Some(node.range()),
301        ));
302        return Ok(None);
303    };
304    if !entries.iter().any(|entry| entry.name == name) {
305        ctx.record_error(RecoverableParseError::new(
306            format!("unknown variant field `{name}`"),
307            Some(name_node.range()),
308        ));
309        return Ok(None);
310    }
311    Ok(Some(Expression::Active(
312        Metadata::new(),
313        Moo::new(variant),
314        name,
315    )))
316}

318fn parse_list_combining_expression(
319    ctx: &mut ParseContext,
320    node: &Node,
321) -> Result<Option<Expression>, FatalParseError> {
322    let Some(operator_node) = field!(recover, ctx, node, "operator") else {
323        return Ok(None);
324    };
325    let operator_str = &ctx.source_code[operator_node.start_byte()..operator_node.end_byte()];
326
327    let Some(arg_node) = field!(recover, ctx, node, "arg") else {
328        return Ok(None);
329    };
330    // While parsing inner, the typechecking context is SetOrMatrix
331    // The inner context is either Boolean or Arithmetic so the elements of the set/matrix are typechecked correctly.
332    let Some(inner) = parse_expression(ctx, arg_node)? else {
333        return Ok(None);
334    };
335
336    let skip_operator = list_combining_skip_operator(operator_str);
337    let inner = set_comprehension_skip_operator(inner, skip_operator);
338
339    let expr = match operator_str {
340        "and" => Ok(Some(Expression::And(Metadata::new(), Moo::new(inner)))),
341        "or" => Ok(Some(Expression::Or(Metadata::new(), Moo::new(inner)))),
342        "sum" => Ok(Some(Expression::Sum(Metadata::new(), Moo::new(inner)))),
343        "product" => Ok(Some(Expression::Product(Metadata::new(), Moo::new(inner)))),
344        "min" => Ok(Some(Expression::Min(Metadata::new(), Moo::new(inner)))),
345        "max" => Ok(Some(Expression::Max(Metadata::new(), Moo::new(inner)))),
346        _ => {
347            ctx.record_error(RecoverableParseError::new(
348                format!("Invalid operator: '{operator_str}'"),
349                Some(operator_node.range()),
350            ));
351            Ok(None)
352        }
353    };
354
355    if expr.is_ok() {
356        ctx.add_span_and_doc_hover(
357            &operator_node,
358            operator_str,
359            SymbolKind::Function,
360            None,
361            None,
362        );
363    }
364
365    expr
366}

368fn parse_all_diff_comparison(
369    ctx: &mut ParseContext,
370    node: &Node,
371) -> Result<Option<Expression>, FatalParseError> {
372    let Some(arg_node) = field!(recover, ctx, node, "arg") else {
373        return Ok(None);
374    };
375    let Some(inner) = parse_expression(ctx, arg_node)? else {
376        return Ok(None);
377    };
378
379    let operator_node = node
380        .child_by_field_name("operator")
381        .unwrap_or_else(|| node.child(0).unwrap());
382    let operator_str =
383        ctx.source_code[operator_node.start_byte()..operator_node.end_byte()].to_string();
384    if !is_all_different_operator(&operator_str) {
385        ctx.record_error(RecoverableParseError::new(
386            format!("Invalid operator: '{operator_str}'"),
387            Some(operator_node.range()),
388        ));
389        return Ok(None);
390    }
391
392    ctx.add_span_and_doc_hover(
393        &operator_node,
394        ALL_DIFFERENT,
395        SymbolKind::Function,
396        None,
397        None,
398    );
399    Ok(Some(Expression::AllDiff(Metadata::new(), Moo::new(inner))))
400}

402fn parse_all_different_except_comparison(
403    ctx: &mut ParseContext,
404    node: &Node,
405) -> Result<Option<Expression>, FatalParseError> {
406    let operator_node = node
407        .child_by_field_name("operator")
408        .unwrap_or_else(|| node.child(0).unwrap());
409    let operator_str =
410        ctx.source_code[operator_node.start_byte()..operator_node.end_byte()].to_string();
411    if !is_all_different_except_operator(&operator_str) {
412        ctx.record_error(RecoverableParseError::new(
413            format!("Invalid operator: '{operator_str}'"),
414            Some(operator_node.range()),
415        ));
416        return Ok(None);
417    }
418
419    let Some(matrix_node) = field!(recover, ctx, node, "matrix") else {
420        return Ok(None);
421    };
422    let Some(except_node) = field!(recover, ctx, node, "except") else {
423        return Ok(None);
424    };
425
426    let saved_context = ctx.typechecking_context;
427    ctx.typechecking_context = TypecheckingContext::Unknown;
428    let Some(matrix) = parse_table_operand(ctx, &matrix_node)? else {
429        ctx.typechecking_context = saved_context;
430        return Ok(None);
431    };
432    let Some(except) = parse_table_operand(ctx, &except_node)? else {
433        ctx.typechecking_context = saved_context;
434        return Ok(None);
435    };
436    ctx.typechecking_context = saved_context;
437
438    ctx.add_span_and_doc_hover(
439        &operator_node,
440        ALL_DIFFERENT_EXCEPT,
441        SymbolKind::Function,
442        None,
443        None,
444    );
445    Ok(Some(Expression::AllDifferentExcept(
446        Metadata::new(),
447        Moo::new(matrix),
448        Moo::new(except),
449    )))
450}

452fn parse_global_cardinality_comparison(
453    ctx: &mut ParseContext,
454    node: &Node,
455) -> Result<Option<Expression>, FatalParseError> {
456    let Some(operator_node) = field!(recover, ctx, node, "operator") else {
457        return Ok(None);
458    };
459    let operator_str =
460        ctx.source_code[operator_node.start_byte()..operator_node.end_byte()].to_string();
461
462    let Some(variables_node) = field!(recover, ctx, node, "variables") else {
463        return Ok(None);
464    };
465    let Some(arg2_node) = field!(recover, ctx, node, "arg2") else {
466        return Ok(None);
467    };
468    let Some(arg3_node) = field!(recover, ctx, node, "arg3") else {
469        return Ok(None);
470    };
471
472    let saved_context = ctx.typechecking_context;
473    ctx.typechecking_context = TypecheckingContext::Unknown;
474    let Some(variables) = parse_expression(ctx, variables_node)? else {
475        ctx.typechecking_context = saved_context;
476        return Ok(None);
477    };
478    let Some(arg2) = parse_expression(ctx, arg2_node)? else {
479        ctx.typechecking_context = saved_context;
480        return Ok(None);
481    };
482    let Some(arg3) = parse_expression(ctx, arg3_node)? else {
483        ctx.typechecking_context = saved_context;
484        return Ok(None);
485    };
486    ctx.typechecking_context = saved_context;
487
488    ctx.add_span_and_doc_hover(
489        &operator_node,
490        match operator_str.as_str() {
491            op if is_at_least_operator(op) => AT_LEAST,
492            op if is_at_most_operator(op) => AT_MOST,
493            _ => GLOBAL_CARDINALITY,
494        },
495        SymbolKind::Function,
496        None,
497        None,
498    );
499
500    match operator_str.as_str() {
501        op if is_at_least_operator(op) => Ok(Some(Expression::AtLeast(
502            Metadata::new(),
503            Moo::new(variables),
504            Moo::new(arg2),
505            Moo::new(arg3),
506        ))),
507        op if is_at_most_operator(op) => Ok(Some(Expression::AtMost(
508            Metadata::new(),
509            Moo::new(variables),
510            Moo::new(arg2),
511            Moo::new(arg3),
512        ))),
513        op if is_global_cardinality_operator(op) => Ok(Some(Expression::Gcc(
514            Metadata::new(),
515            Moo::new(variables),
516            Moo::new(arg2),
517            Moo::new(arg3),
518        ))),
519        _ => {
520            ctx.record_error(RecoverableParseError::new(
521                format!("Invalid operator: '{operator_str}'"),
522                Some(operator_node.range()),
523            ));
524            Ok(None)
525        }
526    }
527}

529fn parse_unary_expression(
530    ctx: &mut ParseContext,
531    node: &Node,
532) -> Result<Option<Expression>, FatalParseError> {
533    let saved_context = ctx.typechecking_context;
534    if node.kind() == "abs_value" {
535        // Bars are overloaded for numeric absolute value and collection cardinality.
536        ctx.typechecking_context = TypecheckingContext::Unknown;
537    }
538    let Some(expr_node) = field!(recover, ctx, node, "expression") else {
539        ctx.typechecking_context = saved_context;
540        return Ok(None);
541    };
542    let Some(inner) = parse_expression(ctx, expr_node)? else {
543        ctx.typechecking_context = saved_context;
544        return Ok(None);
545    };
546    ctx.typechecking_context = saved_context;
547
548    match node.kind() {
549        "negative_expr" => {
550            if let Expression::Atomic(_, Atom::Literal(Literal::Int(value))) = inner {
551                Ok(Some(Expression::Atomic(
552                    Metadata::new(),
553                    Atom::Literal(Literal::Int(-value)),
554                )))
555            } else {
556                Ok(Some(Expression::Neg(Metadata::new(), Moo::new(inner))))
557            }
558        }
559        "abs_value" => {
560            let constructor = match inner.return_type() {
561                ReturnType::Matrix(_)
562                | ReturnType::Set(_)
563                | ReturnType::MSet(_)
564                | ReturnType::Sequence(_)
565                | ReturnType::Function(_, _)
566                | ReturnType::Relation(_)
567                | ReturnType::Partition(_)
568                | ReturnType::Permutation(_) => Expression::Card,
569                _ => Expression::Abs,
570            };
571            Ok(Some(constructor(Metadata::new(), Moo::new(inner))))
572        }
573        "not_expr" => Ok(Some(Expression::Not(Metadata::new(), Moo::new(inner)))),
574        "toInt_expr" => {
575            let to_int_keyword_node = child!(node, 0, "toInt");
576            ctx.add_span_and_doc_hover(
577                &to_int_keyword_node,
578                "toInt",
579                SymbolKind::Function,
580                None,
581                None,
582            );
583            Ok(Some(Expression::ToInt(Metadata::new(), Moo::new(inner))))
584        }
585        "factorial_expr" => {
586            // looking for the operator node (either '!' at the end or 'factorial' at the start) to add hover info
587            if let Some(op_node) = (0..node.child_count())
588                .filter_map(|i| node.child(i.try_into().unwrap()))
589                .find(|c| matches!(c.kind(), "!" | "factorial"))
590            {
591                ctx.add_span_and_doc_hover(
592                    &op_node,
593                    "post_factorial",
594                    SymbolKind::Function,
595                    None,
596                    None,
597                );
598            }
599
600            Ok(Some(Expression::Factorial(
601                Metadata::new(),
602                Moo::new(inner),
603            )))
604        }
605        "sub_bool_expr" | "sub_arith_expr" => Ok(Some(inner)),
606        _ => {
607            ctx.record_error(RecoverableParseError::new(
608                format!("Unrecognised unary operation: '{}'", node.kind()),
609                Some(node.range()),
610            ));
611            Ok(None)
612        }
613    }
614}

616pub fn parse_binary_expression(
617    ctx: &mut ParseContext,
618    node: &Node,
619) -> Result<Option<Expression>, FatalParseError> {
620    let Some(op_node) = field!(recover, ctx, node, "operator") else {
621        return Ok(None);
622    };
623    let op_str = &ctx.source_code[op_node.start_byte()..op_node.end_byte()];
624
625    let saved_ctx = ctx.typechecking_context;
626
627    // Special handling for 'in' operator, as the left operand doesn't have to be a set
628    if op_str == "in" {
629        ctx.typechecking_context = TypecheckingContext::Unknown
630    }
631
632    // parse left operand
633    let Some(left_node) = field!(recover, ctx, node, "left") else {
634        return Ok(None);
635    };
636    let Some(left) = parse_expression(ctx, left_node)? else {
637        return Ok(None);
638    };
639
640    // reset context, if needed
641    ctx.typechecking_context = saved_ctx;
642
643    // Equality/inequality/ordering: enforce right operand to match left operand type when
644    // inferable. Ordering (<, <=, >, >=) works on any orderable type now, the same as equality,
645    // so it needs the same operand-matching check.
646    if matches!(op_str, "=" | "!=" | "<" | "<=" | ">" | ">=") {
647        ctx.typechecking_context = inferred_context_from_expression(&left);
648    }
649
650    // 'in's right operand is the container, which need not be a set (mset/sequence-via-toSet/
651    // relation/etc all support membership too); the grammar's "set_comparison" node kind sets
652    // `saved_ctx` to `Set` unconditionally, which only the left operand's override above escapes.
653    if op_str == "in" {
654        ctx.typechecking_context = TypecheckingContext::Unknown;
655    }
656
657    // parse right operand
658    let Some(right_node) = field!(recover, ctx, node, "right") else {
659        return Ok(None);
660    };
661    let Some(right) = parse_expression(ctx, right_node)? else {
662        return Ok(None);
663    };
664
665    // restore original contexts for parent expression parsing
666    ctx.typechecking_context = saved_ctx;
667
668    let mut doc_name = "";
669    let expr = match op_str {
670        // NB: We are deliberately setting the index domain to 1.., not 1..2.
671        // Semantically, this means "a list that can grow/shrink arbitrarily".
672        // This is expected by rules which will modify the terms of the sum expression
673        // (e.g. by partially evaluating them).
674        "+" => {
675            doc_name = "L_Plus";
676            Ok(Some(Expression::Sum(
677                Metadata::new(),
678                Moo::new(matrix_expr![left, right; domain_int!(1..)]),
679            )))
680        }
681        "-" => {
682            doc_name = "L_Minus";
683            Ok(Some(Expression::Minus(
684                Metadata::new(),
685                Moo::new(left),
686                Moo::new(right),
687            )))
688        }
689        "*" => {
690            doc_name = "L_Times";
691            Ok(Some(Expression::Product(
692                Metadata::new(),
693                Moo::new(matrix_expr![left, right; domain_int!(1..)]),
694            )))
695        }
696        "/\\" => {
697            doc_name = "and";
698            Ok(Some(Expression::And(
699                Metadata::new(),
700                Moo::new(matrix_expr![left, right; domain_int!(1..)]),
701            )))
702        }
703        "\\/" => {
704            // No documentation for or in Bits yet
705            doc_name = "or";
706            Ok(Some(Expression::Or(
707                Metadata::new(),
708                Moo::new(matrix_expr![left, right; domain_int!(1..)]),
709            )))
710        }
711        "**" => {
712            doc_name = "L_Pow";
713            Ok(Some(Expression::UnsafePow(
714                Metadata::new(),
715                Moo::new(left),
716                Moo::new(right),
717            )))
718        }
719        "/" => {
720            //TODO: add checks for if division is safe or not
721            doc_name = "L_Div";
722            Ok(Some(Expression::UnsafeDiv(
723                Metadata::new(),
724                Moo::new(left),
725                Moo::new(right),
726            )))
727        }
728        "%" => {
729            //TODO: add checks for if mod is safe or not
730            doc_name = "L_Mod";
731            Ok(Some(Expression::UnsafeMod(
732                Metadata::new(),
733                Moo::new(left),
734                Moo::new(right),
735            )))
736        }
737
738        "=" => {
739            doc_name = "L_Eq"; //no docs yet
740            Ok(Some(Expression::Eq(
741                Metadata::new(),
742                Moo::new(left),
743                Moo::new(right),
744            )))
745        }
746        "!=" => {
747            doc_name = "L_Neq"; //no docs yet
748            Ok(Some(Expression::Neq(
749                Metadata::new(),
750                Moo::new(left),
751                Moo::new(right),
752            )))
753        }
754        "<=" => {
755            doc_name = "L_Leq"; //no docs yet
756            Ok(Some(Expression::Leq(
757                Metadata::new(),
758                Moo::new(left),
759                Moo::new(right),
760            )))
761        }
762        ">=" => {
763            doc_name = "L_Geq"; //no docs yet
764            Ok(Some(Expression::Geq(
765                Metadata::new(),
766                Moo::new(left),
767                Moo::new(right),
768            )))
769        }
770        "<" => {
771            doc_name = "L_Lt"; //no docs yet
772            Ok(Some(Expression::Lt(
773                Metadata::new(),
774                Moo::new(left),
775                Moo::new(right),
776            )))
777        }
778        ">" => {
779            doc_name = "L_Gt"; //no docs yet
780            Ok(Some(Expression::Gt(
781                Metadata::new(),
782                Moo::new(left),
783                Moo::new(right),
784            )))
785        }
786
787        "->" => {
788            doc_name = "L_Imply"; //no docs yet
789            Ok(Some(Expression::Imply(
790                Metadata::new(),
791                Moo::new(left),
792                Moo::new(right),
793            )))
794        }
795        "<->" => {
796            doc_name = "L_Iff"; //no docs yet
797            Ok(Some(Expression::Iff(
798                Metadata::new(),
799                Moo::new(left),
800                Moo::new(right),
801            )))
802        }
803        "<lex" => {
804            doc_name = "L_LexLt"; //no docs yet
805            Ok(Some(Expression::LexLt(
806                Metadata::new(),
807                Moo::new(left),
808                Moo::new(right),
809            )))
810        }
811        ">lex" => {
812            doc_name = "L_LexGt"; //no docs yet
813            Ok(Some(Expression::LexGt(
814                Metadata::new(),
815                Moo::new(left),
816                Moo::new(right),
817            )))
818        }
819        "<=lex" => {
820            doc_name = "L_LexLeq"; //no docs yet
821            Ok(Some(Expression::LexLeq(
822                Metadata::new(),
823                Moo::new(left),
824                Moo::new(right),
825            )))
826        }
827        ">=lex" => {
828            doc_name = "L_LexGeq"; //no docs yet
829            Ok(Some(Expression::LexGeq(
830                Metadata::new(),
831                Moo::new(left),
832                Moo::new(right),
833            )))
834        }
835        "in" => {
836            doc_name = "L_in";
837            Ok(Some(Expression::In(
838                Metadata::new(),
839                Moo::new(left),
840                Moo::new(right),
841            )))
842        }
843        "subset" => {
844            doc_name = "L_subset";
845            Ok(Some(Expression::Subset(
846                Metadata::new(),
847                Moo::new(left),
848                Moo::new(right),
849            )))
850        }
851        "subsetEq" => {
852            doc_name = "L_subsetEq";
853            Ok(Some(Expression::SubsetEq(
854                Metadata::new(),
855                Moo::new(left),
856                Moo::new(right),
857            )))
858        }
859        "supset" => {
860            doc_name = "L_supset";
861            Ok(Some(Expression::Supset(
862                Metadata::new(),
863                Moo::new(left),
864                Moo::new(right),
865            )))
866        }
867        "supsetEq" => {
868            doc_name = "L_supsetEq";
869            Ok(Some(Expression::SupsetEq(
870                Metadata::new(),
871                Moo::new(left),
872                Moo::new(right),
873            )))
874        }
875        "substring" => {
876            doc_name = "L_substring";
877            Ok(Some(Expression::Substring(
878                Metadata::new(),
879                Moo::new(left),
880                Moo::new(right),
881            )))
882        }
883        "subsequence" => {
884            doc_name = "L_subsequence";
885            Ok(Some(Expression::Subsequence(
886                Metadata::new(),
887                Moo::new(left),
888                Moo::new(right),
889            )))
890        }
891        "union" => {
892            doc_name = "L_union";
893            Ok(Some(Expression::Union(
894                Metadata::new(),
895                Moo::new(left),
896                Moo::new(right),
897            )))
898        }
899        "intersect" => {
900            doc_name = "L_intersect";
901            Ok(Some(Expression::Intersect(
902                Metadata::new(),
903                Moo::new(left),
904                Moo::new(right),
905            )))
906        }
907        _ => {
908            ctx.record_error(RecoverableParseError::new(
909                format!("Invalid operator: '{op_str}'"),
910                Some(op_node.range()),
911            ));
912            Ok(None)
913        }
914    };
915
916    if expr.is_ok() {
917        ctx.add_span_and_doc_hover(&op_node, doc_name, SymbolKind::Function, None, None);
918    }
919
920    expr
921}

923pub fn parse_annotation_expression(
924    ctx: &mut ParseContext,
925    node: &Node,
926) -> Result<Option<Expression>, FatalParseError> {
927    let annotation_node = if node.kind() == "annotation_expr" {
928        let Some(inner) = named_child!(recover, ctx, node) else {
929            return Ok(None);
930        };
931        inner
932    } else {
933        *node
934    };
935
936    let Some(left_node) = field!(recover, ctx, &annotation_node, "left") else {
937        return Ok(None);
938    };
939    let left_node = if left_node.kind() == "annotation_subject" {
940        let Some(inner) = named_child!(recover, ctx, &left_node) else {
941            return Ok(None);
942        };
943        inner
944    } else {
945        left_node
946    };
947    let left_node = match left_node.kind() {
948        "sub_arith_expr" | "sub_bool_expr" | "sub_atom_expr" => {
949            let Some(inner) = field!(recover, ctx, &left_node, "expression") else {
950                return Ok(None);
951            };
952            inner
953        }
954        _ => left_node,
955    };
956    let saved_context = ctx.typechecking_context;
957    ctx.typechecking_context = TypecheckingContext::Unknown;
958    let left = match left_node.kind() {
959        "constant" | "identifier" | "metavar" => parse_atom(ctx, &left_node)?,
960        _ => parse_expression(ctx, left_node)?,
961    };
962    let Some(left) = left else {
963        ctx.typechecking_context = saved_context;
964        return Ok(None);
965    };
966    ctx.typechecking_context = saved_context;
967
968    let Some(operator_node) = field!(recover, ctx, &annotation_node, "operator") else {
969        return Ok(None);
970    };
971    let operator_str = &ctx.source_code[operator_node.start_byte()..operator_node.end_byte()];
972
973    match annotation_node.kind() {
974        "type_annotation" => {
975            let Some(type_node) = field!(recover, ctx, &annotation_node, "type") else {
976                return Ok(None);
977            };
978            let Some(domain) = parse_domain(ctx, type_node)? else {
979                return Ok(None);
980            };
981            ctx.add_span_and_doc_hover(
982                &operator_node,
983                operator_str,
984                SymbolKind::Function,
985                None,
986                None,
987            );
988            Ok(Some(Expression::TypeAnnotation(
989                Metadata::new(),
990                Moo::new(left),
991                domain,
992            )))
993        }
994        "domain_annotation" => {
995            let Some(domain_node) = field!(recover, ctx, &annotation_node, "domain") else {
996                return Ok(None);
997            };
998            let Some(domain) = parse_domain(ctx, domain_node)? else {
999                return Ok(None);
1000            };
1001            ctx.add_span_and_doc_hover(
1002                &operator_node,
1003                operator_str,
1004                SymbolKind::Function,
1005                None,
1006                None,
1007            );
1008            Ok(Some(Expression::DomainAnnotation(
1009                Metadata::new(),
1010                Moo::new(left),
1011                domain,
1012            )))
1013        }
1014        _ => {
1015            ctx.record_error(RecoverableParseError::new(
1016                format!(
1017                    "Unrecognised annotation expression: '{}'",
1018                    annotation_node.kind()
1019                ),
1020                Some(annotation_node.range()),
1021            ));
1022            Ok(None)
1023        }
1024    }
1025}

1027fn inferred_context_from_expression(expr: &Expression) -> TypecheckingContext {
1028    // TODO: typechecking for index/slice expressions
1029    if matches!(
1030        expr,
1031        Expression::UnsafeIndex(_, _, _) | Expression::UnsafeSlice(_, _, _)
1032    ) {
1033        return TypecheckingContext::Unknown;
1034    }
1035
1036    let Some(domain) = expr.domain_of() else {
1037        return TypecheckingContext::Unknown;
1038    };
1039    let Ok(ground) = domain.resolve() else {
1040        return TypecheckingContext::Unknown;
1041    };
1042
1043    match ground.as_ref() {
1044        GroundDomain::Empty(_) => TypecheckingContext::Unknown,
1045        GroundDomain::Bool => TypecheckingContext::Boolean,
1046        GroundDomain::Int(_) => TypecheckingContext::Arithmetic,
1047        GroundDomain::Tuple(_) => TypecheckingContext::Tuple,
1048        GroundDomain::Record(_) => TypecheckingContext::Record,
1049        GroundDomain::Variant(_) => TypecheckingContext::Unknown,
1050        GroundDomain::Matrix(_, _) => TypecheckingContext::Matrix,
1051        GroundDomain::Sequence(_, _) => TypecheckingContext::Sequence,
1052        GroundDomain::Set(_, _) => TypecheckingContext::Set,
1053        GroundDomain::MSet(_, _) => TypecheckingContext::MSet,
1054        GroundDomain::Function(_, _, _) => TypecheckingContext::Function,
1055        GroundDomain::Relation(_, _) => TypecheckingContext::Relation,
1056        GroundDomain::Partition(_, _) => TypecheckingContext::Partition,
1057        GroundDomain::Permutation(_, _) => TypecheckingContext::Permutation,
1058    }
1059}

1061fn list_combining_skip_operator(operator_str: &str) -> Option<ACOperatorKind> {
1062    match operator_str {
1063        "and" => Some(ACOperatorKind::And),
1064        "or" => Some(ACOperatorKind::Or),
1065        "sum" => Some(ACOperatorKind::Sum),
1066        "product" => Some(ACOperatorKind::Product),
1067        // min/max are not true AC operators (no universal identity element), but still need
1068        // their own skip-operator tag so a symbolic guard inside `min([... | ...])` lowers
1069        // correctly (previously mapped to Sum, which silently substituted 0 for guarded-out
1070        // elements -- wrong for anything but a min/max whose correct answer happens to be <=/>= 0).
1071        "min" => Some(ACOperatorKind::Min),
1072        "max" => Some(ACOperatorKind::Max),
1073        _ => None,
1074    }
1075}

1077fn set_comprehension_skip_operator(
1078    inner: Expression,
1079    skip_operator: Option<ACOperatorKind>,
1080) -> Expression {
1081    let Expression::Comprehension(meta, comprehension) = inner else {
1082        return inner;
1083    };
1084    if let Some(skip_operator) = skip_operator {
1085        let mut comprehension = Moo::unwrap_or_clone(comprehension);
1086        comprehension.skip_operator = Some(skip_operator);
1087        Expression::Comprehension(meta, Moo::new(comprehension))
1088    } else {
1089        Expression::Comprehension(meta, comprehension)
1090    }
