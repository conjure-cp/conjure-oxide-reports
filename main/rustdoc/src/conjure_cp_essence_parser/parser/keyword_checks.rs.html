1use crate::errors::RecoverableParseError;
2use crate::parser::syntax_errors::is_malformed_line_error;
3
4const KEYWORDS: [&str; 21] = [
5    "forall", "exists", "such", "that", "letting", "find", "minimise", "maximise", "subject", "to",
6    "where", "and", "or", "not", "if", "then", "else", "in", "sum", "product", "bool",
7];
8
9pub fn keyword_as_identifier(
10    root: tree_sitter::Node,
11    source: &str,
12    errors: &mut Vec<RecoverableParseError>,
13) {
14    let mut stack = vec![root];
15    while let Some(node) = stack.pop() {
16        if node.is_error() && is_malformed_line_error(&node, source) {
17            return;
18        }
19        if (node.kind() == "variable" || node.kind() == "identifier" || node.kind() == "parameter")
20            && let Ok(text) = node.utf8_text(source.as_bytes())
21        {
22            let ident = text.trim();
23            if KEYWORDS.contains(&ident) {
24                let start_point = node.start_position();
25                let end_point = node.end_position();
26                errors.push(RecoverableParseError::new(
27                    format!("Keyword '{ident}' used as identifier"),
28                    Some(tree_sitter::Range {
29                        start_byte: node.start_byte(),
30                        end_byte: node.end_byte(),
31                        start_point,
32                        end_point,
33                    }),
34                ));
35            }
36        }
37
38        for i in 0..node.child_count() {
39            if let Some(child) = u32::try_from(i).ok().and_then(|i| node.child(i)) {
40                stack.push(child);
41            }
42        }
43    }
44}
