7/// Returns the absolute byte offset of the start of `row` in `source`.
8fn line_start_byte(source: &[u8], row: usize) -> usize {
9    let mut current_row = 0usize;
10    let mut line_start = 0usize;
11    for (idx, b) in source.iter().enumerate() {
12        if current_row == row {
13            break;
14        }
15        if *b == b'\n' {
16            current_row += 1;
17            line_start = idx + 1;
18        }
19    }
20    line_start
21}

23/// This is a reporting-layer fix: even though comments are treated as "extras" by the grammar,
24/// tree-sitter `ERROR` node spans can overlap those bytes during recovery. We clamp to the end of
25/// the non-comment prefix (with trailing whitespace trimmed) so diagnostics don't include comment
26/// contents.
27fn clamp_range_before_line_comment(range: &mut tree_sitter::Range, source: &str) {
28    let Some(line) = source.lines().nth(range.start_point.row) else {
29        return;
30    };
31    let Some(dollar_idx) = line.find('$') else {
32        return;
33    };
34
35    let prefix = &line[..dollar_idx];
36    let clamped_col = prefix.trim_end().len();
37
38    if range.start_point.column > clamped_col {
39        range.start_point.column = clamped_col;
40    }
41    if range.end_point.row == range.start_point.row && range.end_point.column > clamped_col {
42        range.end_point.column = clamped_col;
43    }
44    if range.end_point.row > range.start_point.row {
45        range.end_point.row = range.start_point.row;
46        range.end_point.column = clamped_col;
47    }
48
49    let line_start = line_start_byte(source.as_bytes(), range.start_point.row);
50    range.start_byte = line_start + range.start_point.column;
51    range.end_byte = line_start + range.end_point.column;
52}

54pub fn detect_syntactic_errors(
55    source: &str,
56    tree: &tree_sitter::Tree,
57    errors: &mut Vec<RecoverableParseError>,
58) {
59    let mut malformed_lines_reported = HashSet::new();
60
61    let root_node = tree.root_node();
62    let retract: &dyn Fn(&tree_sitter::Node) -> bool = &|node: &tree_sitter::Node| {
63        node.is_missing() || node.is_error() || node.start_position() == node.end_position()
64    };
65
66    for node in WalkDFS::with_retract(&root_node, &retract) {
67        if node.start_position() == node.end_position() {
68            errors.push(classify_missing_token(node, source));
69            continue;
70        }
71        if node.is_error() {
72            let line = node.start_position().row;
73            // If this line has already been reported as malformed, skip all error nodes on this line
74            if malformed_lines_reported.contains(&line) {
75                continue;
76            }
77            // Ignore error nodes that start inside a single-line comment.
78            if let Some(line_str) = source.lines().nth(line)
79                && let Some(dollar_idx) = line_str.find('$')
80                && node.start_position().column >= dollar_idx
81            {
82                continue;
83            }
84
85            if is_malformed_line_error(&node, source) {
86                malformed_lines_reported.insert(line);
87                let start_byte = node.start_byte();
88                let end_byte = node.end_byte();
89
90                let last_char = source.lines().nth(line).map_or(0, |l| l.len());
91                errors.push(RecoverableParseError::new(
92                    generate_malformed_line_message(line, source),
93                    Some(tree_sitter::Range {
94                        start_byte,
95                        end_byte,
96                        start_point: tree_sitter::Point {
97                            row: line,
98                            column: 0,
99                        },
100                        end_point: tree_sitter::Point {
101                            row: line,
102                            column: last_char,
103                        },
104                    }),
105                ));
106                continue;
107            } else {
108                errors.push(classify_unexpected_token_error(node, source));
109            }
110            continue;
111        }
112    }
113}

115/// Classifies a missing token node and generates a diagnostic with a context-aware message.
116fn classify_missing_token(node: Node, source: &str) -> RecoverableParseError {
117    let mut range = tree_sitter::Range {
118        start_byte: node.start_byte(),
119        end_byte: node.end_byte(),
120        start_point: node.start_position(),
121        end_point: node.end_position(),
122    };
123    clamp_range_before_line_comment(&mut range, source);
124
125    let message = if let Some(parent) = node.parent() {
126        match parent.kind() {
127            "letting_statement" => "Missing Expression or Domain".to_string(),
128            _ => format!("Missing {}", user_friendly_token_name(node.kind(), false)),
129        }
130    } else {
131        format!("Missing {}", user_friendly_token_name(node.kind(), false))
132    };
133
134    RecoverableParseError::new(message, Some(range))
135}

137/// Classifies an unexpected token error node and generates a diagnostic.
138fn classify_unexpected_token_error(node: Node, source_code: &str) -> RecoverableParseError {
139    let mut range = tree_sitter::Range {
140        start_byte: node.start_byte().min(source_code.len()),
141        end_byte: node.end_byte().min(source_code.len()),
142        start_point: node.start_position(),
143        end_point: node.end_position(),
144    };
145    clamp_range_before_line_comment(&mut range, source_code);
146
147    let message = if let Some(parent) = node.parent() {
148        // Extract the unexpected token text, handling out-of-range indices safely.
149        // NOTE: tree-sitter byte offsets can land inside UTF-8 codepoints; decoding lossily avoids panics.
150        let src_token: std::borrow::Cow<'_, str> = source_code
151            .as_bytes()
152            .get(range.start_byte..range.end_byte)
153            .map(String::from_utf8_lossy)
154            .unwrap_or_else(|| std::borrow::Cow::Borrowed("<unknown>"));
155        let src_token = src_token.trim_end();
156
157        if parent.kind() == "program" {
158            format!("Unexpected {}", src_token)
159        } else {
160            format!(
161                "Unexpected {} inside {}",
162                src_token,
163                user_friendly_token_name(parent.kind(), true)
164            )
165        }
166    } else {
167        "Unexpected token".to_string()
168    };
169
170    RecoverableParseError::new(message, Some(range))
171}

173/// Determines if an error node represents a malformed line error.
174pub fn is_malformed_line_error(node: &tree_sitter::Node, source: &str) -> bool {
175    if node.start_position().column == 0 || error_node_out_of_range(node, source) {
176        return true;
177    }
178    let parent = node.parent();
179    let grandparent = parent.and_then(|n| n.parent());
180    let root = grandparent.and_then(|n| n.parent());
181
182    if let (Some(parent), Some(grandparent), Some(root)) = (parent, grandparent, root) {
183        parent.kind() == "set_comparison"
184            && grandparent.kind() == "comparison_expr"
185            && root.kind() == "program"
186    } else {
187        false
188    }
189}

191/// Coverts a token name into a more user-friendly format for error messages.
192/// Removes underscores, replaces certain keywords with more natural language, and adds appropriate articles.
193fn user_friendly_token_name(token: &str, article: bool) -> String {
194    let capitalized = if token.contains("atom") {
195        "Expression".to_string()
196    } else if token == "COLON" {
197        ":".to_string()
198    } else {
199        let friendly_name = token
200            .replace("literal", "")
201            .replace("int", "Integer")
202            .replace("expr", "Expression")
203            .replace('_', " ");
204        friendly_name
205            .split_whitespace()
206            .map(|word| word.capitalize())
207            .collect::<Vec<_>>()
208            .join(" ")
209    };
210
211    if !article {
212        return capitalized;
213    }
214    let first_char = capitalized.chars().next().unwrap();
215    let article = match first_char.to_ascii_lowercase() {
216        'a' | 'e' | 'i' | 'o' | 'u' => "an",
217        _ => "a",
218    };
219    format!("{} {}", article, capitalized)
220}

222// Generates an informative error message for malformed lines
223fn generate_malformed_line_message(line: usize, source: &str) -> String {
224    let got = source.lines().nth(line).unwrap_or("").trim();
225    let got = got.split('$').next().unwrap_or("").trim_end();
226    let got = got.replace('"', "\\\"");
227    let mut words = got.split_whitespace();
228    let first = words.next().unwrap_or("").to_ascii_lowercase();
229    let second = words.next().unwrap_or("").to_ascii_lowercase();
230
231    let expected = match first.as_str() {
232        "find" => "a find declaration statement",
233        "letting" => "a letting declaration statement",
234        "given" => "a given declaration statement",
235        "where" => "an instantiation condition",
236        "minimising" | "maximising" => "an objective statement",
237        "such" => {
238            // Check for invalid constraint statement
239            if second == "that" {
240                "a constraint statement"
241            } else {
242                "a valid top-level statement"
243            }
244        }
245
246        _ => {
247            // Default case for unrecognized starting tokens
248            "a valid top-level statement"
249        }
250    };
251    format!("Expected {}, but got '{}'", expected, got)
252}

254/// Returns true if the node's start or end column is out of range for its line in the source.
255fn error_node_out_of_range(node: &tree_sitter::Node, source: &str) -> bool {
256    let lines: Vec<&str> = source.lines().collect();
257    let start = node.start_position();
258    let end = node.end_position();
259
260    let start_line_len = lines.get(start.row).map_or(0, |l| l.len());
261    let end_line_len = lines.get(end.row).map_or(0, |l| l.len());
262
263    (start.column > start_line_len) || (end.column > end_line_len)
264}
