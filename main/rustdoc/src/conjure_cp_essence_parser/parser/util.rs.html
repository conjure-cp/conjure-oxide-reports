11/// Context for parsing, containing shared state passed through parser functions.
12pub struct ParseContext<'a> {
13    pub source_code: &'a str,
14    pub root: &'a Node<'a>,
15    pub symbols: Option<SymbolTablePtr>,
16    pub errors: &'a mut Vec<RecoverableParseError>,
17    pub source_map: &'a mut SourceMap,
18    pub decl_spans: &'a mut BTreeMap<Name, SpanId>,
19    pub typechecking_context: TypecheckingContext,
20}
21
22impl<'a> ParseContext<'a> {
23    pub fn new(
24        source_code: &'a str,
25        root: &'a Node<'a>,
26        symbols: Option<SymbolTablePtr>,
27        errors: &'a mut Vec<RecoverableParseError>,
28        source_map: &'a mut SourceMap,
29        decl_spans: &'a mut BTreeMap<Name, SpanId>,
30    ) -> Self {
31        Self {
32            source_code,
33            root,
34            symbols,
35            errors,
36            source_map,
37            decl_spans,
38            typechecking_context: TypecheckingContext::Unknown,
39        }
40    }
41
42    pub fn record_error(&mut self, error: RecoverableParseError) {
43        self.errors.push(error);
44    }
45
46    /// Create a new ParseContext with different symbols but sharing source_code, root, errors, and source_map.
47    pub fn with_new_symbols(&mut self, symbols: Option<SymbolTablePtr>) -> ParseContext<'_> {
48        ParseContext {
49            source_code: self.source_code,
50            root: self.root,
51            symbols,
52            errors: self.errors,
53            source_map: self.source_map,
54            decl_spans: self.decl_spans,
55            typechecking_context: self.typechecking_context,
56        }
57    }
58
59    pub fn save_decl_span(&mut self, name: Name, span_id: SpanId) {
60        self.decl_spans.insert(name, span_id);
61    }
62
63    pub fn lookup_decl_span(&self, name: &Name) -> Option<SpanId> {
64        self.decl_spans.get(name).copied()
65    }
66
67    pub fn lookup_decl_line(&self, name: &Name) -> Option<u32> {
68        let span_id = self.lookup_decl_span(name)?;
69        let span = self.source_map.spans.get(span_id as usize)?;
70        Some(span.start_point.line + 1)
71    }
72}

83/// Parse the given source code into a syntax tree using tree-sitter.
84///
85/// If successful, returns a tuple containing the syntax tree and the raw source code.
86/// If the source code is not valid Essence, returns None.
87///
88/// NOTE: The new source code may be different from the original source code.
89///       See implementation for details.
90pub fn get_tree(src: &str) -> Option<(Tree, String)> {
91    let mut parser = Parser::new();
92    parser.set_language(&LANGUAGE.into()).unwrap();
93
94    parser.parse(src, None).and_then(|tree| {
95        let root = tree.root_node();
96        if root.is_error() {
97            return None;
98        }
99
100        let children: Vec<_> = named_children(&root).collect();
101        let first_child = children.first()?;
102
103        // HACK: Tree-sitter can only parse a complete program from top to bottom, not an individual bit of syntax.
104        // See: https://github.com/tree-sitter/tree-sitter/issues/711 and linked issues.
105        // However, we can use a dummy _FRAGMENT_EXPRESSION prefix (which we insert as necessary)
106        // to trick the parser into accepting an isolated expression.
107        // This way we can parse an isolated expression and it is only slightly cursed :)
108        if first_child.is_error() {
109            if src.starts_with("_FRAGMENT_EXPRESSION") {
110                None
111            } else {
112                get_tree(&format!("_FRAGMENT_EXPRESSION {src}"))
113            }
114        } else {
115            Some((tree, src.to_string()))
116        }
117    })
118}

120/// Get the named children of a node
121pub fn named_children<'a>(node: &'a Node<'a>) -> impl Iterator<Item = Node<'a>> + 'a {
122    (0..node.named_child_count())
123        .filter_map(|i| u32::try_from(i).ok().and_then(|i| node.named_child(i)))
124}
125
126pub fn node_is_expression(node: &Node) -> bool {
127    matches!(
128        node.kind(),
129        "bool_expr" | "arithmetic_expr" | "comparison_expr" | "atom"
130    )
131}

133/// Get all top-level nodes that match the given predicate
134pub fn query_toplevel<'a>(
135    node: &'a Node<'a>,
136    predicate: &'a dyn Fn(&Node<'a>) -> bool,
137) -> impl Iterator<Item = Node<'a>> + 'a {
138    WalkDFS::with_retract(node, predicate).filter(|n| n.is_named() && predicate(n))
139}
140
141/// Get all meta-variable names in a node
142pub fn get_metavars<'a>(node: &'a Node<'a>, src: &'a str) -> impl Iterator<Item = String> + 'a {
143    query_toplevel(node, &|n| n.kind() == "metavar").filter_map(|child| {
144        child
145            .named_child(0)
146            .map(|name| src[name.start_byte()..name.end_byte()].to_string())
147    })
148}
