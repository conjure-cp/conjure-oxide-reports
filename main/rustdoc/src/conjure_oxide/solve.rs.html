29pub enum OutputFormat {

37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum NumberOfSolutions {
39    All,
40    Limit(i32),
41}
42
43impl NumberOfSolutions {
44    fn as_solver_limit(self) -> i32 {
45        match self {
46            NumberOfSolutions::All => 0,
47            NumberOfSolutions::Limit(limit) => limit,
48        }
49    }
50}
51
52fn parse_number_of_solutions(input: &str) -> Result<NumberOfSolutions, String> {
53    if input.eq_ignore_ascii_case("all") {
54        return Ok(NumberOfSolutions::All);
55    }
56
57    let limit = input
58        .parse::<i32>()
59        .map_err(|_| "expected a positive integer or 'all'".to_string())?;
60
61    if limit <= 0 {
62        return Err("expected a positive integer or 'all'".to_string());
63    }
64
65    Ok(NumberOfSolutions::Limit(limit))
66}

143fn run_solve(
144    pipeline: &Pipeline,
145    global_args: GlobalArgs,
146    solve_args: Args,
147    rule_application_counts: &RuleApplicationCounter,
148) -> anyhow::Result<()> {
149    let essence_file = solve_args.essence_file.clone();
150    let param_file = solve_args.param_file.clone();
151
152    let context = pipeline.context();
153    {
154        let mut context = context.write().unwrap();
155        context.essence_file_name = Some(essence_file.to_string_lossy().into_owned());
156        context.param_file_name = param_file
157            .as_ref()
158            .map(|path| path.to_string_lossy().into_owned());
159    }
160
161    let translation_started_at = Instant::now();
162    let unified_model = pipeline.with_settings(|| -> anyhow::Result<Model> {
163        let problem_model = pipeline.parse_file(&essence_file)?;
164        match param_file {
165            Some(param_file) => {
166                let param_model = parse_param(pipeline, &param_file, &problem_model)?;
167                Ok(instantiate_model(problem_model, param_model)?)
168            }
169            None => {
170                let mut problem_model = problem_model;
171                validate_instantiation_conditions(&mut problem_model)?;
172                Ok(problem_model)
173            }
174        }
175    })?;
176    let rewritten_model = pipeline.rewrite(unified_model)?;
177
178    // Capture representations and SAT encoding choices
179    let mut model_choices = RecordedRunStats::default();
180
181    record_model_choices(&rewritten_model, &mut model_choices);
182
183    let translation_time = translation_started_at.elapsed().as_secs_f64();
184
185    let solver_started_at = Instant::now();
186
187    let measured = if solve_args.no_run_solver {
188        println!("{}", rewritten_model);
189
190        if let Some(ref path) = global_args.save_solver_input_file {
191            pipeline.with_settings(|| -> anyhow::Result<()> {
192                let solver = pipeline.solver()?.load_model(rewritten_model)?;
193                let mut file: Box<dyn std::io::Write> = Box::new(File::create(path)?);
194                solver.write_solver_input_file(&mut file)?;
195                Ok(())
196            })?;
197        }
198
199        None
200    } else {
201        Some(run_solver(
202            pipeline,
203            &global_args,
204            &solve_args,
205            rewritten_model,
206        )?)
207    };
208
209    let solve_time = measured
210        .as_ref()
211        .map(|_| solver_started_at.elapsed().as_secs_f64());
212
213    let backend_time = measured.as_ref().map(|t| t.0);
214    let search_time = measured.as_ref().map(|t| t.1);
215    let solution_processing_time = measured.as_ref().map(|t| t.2);
216
217    // Still write translation statistics when the solver was not run.
218    if let Some(ref path) = solve_args.stats {
219        let config = RecordedRunConfig {
220            parser: global_args.parser.to_string(),
221            rewriter: global_args.rewriter.to_string(),
222            comprehension_expander: global_args.comprehension_expander.to_string(),
223            heuristic: global_args.heuristic.to_string(),
224            channelling: global_args.channelling.to_string(),
225            seed: global_args.seed,
226            solver_seed: global_args.solver_seed,
227            solver: global_args.solver.as_str(),
228        };
229
230        let case_name = solve_args
231            .essence_file
232            .file_stem()
233            .map(|s| s.to_string_lossy().into_owned())
234            .unwrap_or_else(|| "input".to_owned());
235
236        let mut row = RecordedRunStats {
237            parser: config.parser.clone(),
238            rewriter: config.rewriter.clone(),
239            comprehension_expander: config.comprehension_expander.clone(),
240            heuristic: config.heuristic.clone(),
241            channelling: config.channelling.clone(),
242            seed: config.seed,
243            solver_seed: config.solver_seed,
244            solver: config.solver.clone(),
245            minion_discrete_threshold: global_args.minion_discrete_threshold,
246
247            case_name,
248            model_index: 0,
249            status: "ok".to_owned(),
250
251            solution_limit: (!solve_args.no_run_solver)
252                .then(|| solve_args.number_of_solutions.as_solver_limit()),
253
254            translation_time: Some(translation_time),
255            solve_time,
256            backend_time,
257            search_time,
258            solution_processing_time,
259
260            ..Default::default()
261        };
262
263        row.representations = model_choices.representations;
264
265        row.sat_encoding_amo = model_choices.sat_encoding_amo;
266        row.sat_encoding_cardinality = model_choices.sat_encoding_cardinality;
267        row.sat_encoding_pb = model_choices.sat_encoding_pb;
268        row.sat_encoding_table = model_choices.sat_encoding_table;
269        row.sat_encoding_element = model_choices.sat_encoding_element;
270        row.sat_encoding_alldifferent = model_choices.sat_encoding_alldifferent;
271
272        // Read the same rewrite-attempt counters used by exhaustive testing.
273        let context_json = serde_json::to_value(context.read().unwrap().clone())?;
274
275        let attempt_counts: Vec<u64> = context_json
276            .get("stats")
277            .and_then(|stats| stats.get("rewriterRuns"))
278            .and_then(serde_json::Value::as_array)
279            .into_iter()
280            .flatten()
281            .filter_map(|run| {
282                run.get("rewriterRuleApplicationAttempts")
283                    .and_then(serde_json::Value::as_u64)
284            })
285            .collect();
286
287        if !attempt_counts.is_empty() {
288            row.rule_trace.total_rule_attempts = Some(attempt_counts.iter().sum());
289        }
290
291        let counts = rule_application_counts.snapshot();
292        row.rule_trace.total_rule_applications = Some(counts.values().copied().sum());
293        row.rule_trace.rules = counts;
294
295        // Write the completed statistics record only after all fields are populated.
296        replace_config_run_stats(path, &config, vec![row])?;
297    }
298    Ok(())
299}
300fn record_model_choices(model: &Model, row: &mut RecordedRunStats) {
301    row.representations = model
302        .symbols()
303        .iter_local()
304        .flat_map(|(_, decl)| {
305            decl.reprs()
306                .iter()
307                .map(|(name, _)| name.to_owned())
308                .collect::<Vec<_>>()
309        })
310        .collect::<BTreeSet<_>>()
311        .into_iter()
312        .collect();
313    let mut families: BTreeMap<&str, BTreeSet<String>> = BTreeMap::new();
314    macro_rules! record {
315        ($family:expr, $selection:expr) => {{
316            $selection.iter().for_each(|selection| {
317                families
318                    .entry($family)
319                    .or_default()
320                    .insert(selection.algorithm.to_string());
321            });
322        }};
323    }
324    for decision in model.sat_decisions() {
325        match decision {
326            SatEncodingDecision::AtMostOne { encoding, .. } => {
327                record!("amo", encoding);
328            }
329            SatEncodingDecision::Cardinality { encoding, .. } => {
330                record!("cardinality", encoding);
331            }
332            SatEncodingDecision::CountRelation {
333                encoding,
334                amo_encoding,
335                ..
336            } => {
337                record!("cardinality", encoding);
338                record!("amo", amo_encoding);
339            }
340            SatEncodingDecision::PseudoBoolean { encoding, .. }
341            | SatEncodingDecision::IntegerRelation { encoding, .. }
342            | SatEncodingDecision::Objective { encoding, .. } => record!("pb", encoding),
343            SatEncodingDecision::AllDifferent {
344                encoding,
345                amo_encoding,
346                pb_encoding,
347                ..
348            } => {
349                record!("alldifferent", encoding);
350                record!("amo", amo_encoding);
351                record!("pb", pb_encoding);
352            }
353            SatEncodingDecision::Table {
354                encoding,
355                pb_encoding,
356                ..
357            } => {
358                record!("table", encoding);
359                record!("pb", pb_encoding);
360            }
361            SatEncodingDecision::Element {
362                encoding,
363                pb_encoding,
364                ..
365            } => {
366                record!("element", encoding);
367                record!("pb", pb_encoding);
368            }
369            _ => {}
370        }
371    }
372    row.sat_encoding_amo = families
373        .remove("amo")
374        .unwrap_or_default()
375        .into_iter()
376        .collect();
377    row.sat_encoding_cardinality = families
378        .remove("cardinality")
379        .unwrap_or_default()
380        .into_iter()
381        .collect();
382    row.sat_encoding_pb = families
383        .remove("pb")
384        .unwrap_or_default()
385        .into_iter()
386        .collect();
387    row.sat_encoding_table = families
388        .remove("table")
389        .unwrap_or_default()
390        .into_iter()
391        .collect();
392    row.sat_encoding_element = families
393        .remove("element")
394        .unwrap_or_default()
395        .into_iter()
396        .collect();
397    row.sat_encoding_alldifferent = families
398        .remove("alldifferent")
399        .unwrap_or_default()
400        .into_iter()
401        .collect();
402}
403/// Parse an Essence or simplified-JSON parameter file.
404pub(crate) fn parse_param(
405    pipeline: &Pipeline,
406    file_path: &Path,
407    problem_model: &Model,
408) -> anyhow::Result<Model> {
409    if file_path
410        .extension()
411        .is_some_and(|extension| extension == "json")
412    {
413        tracing::info!(target: "conjure::stage", path = %file_path.display(), "parsing parameter file");
414        let text = std::fs::read_to_string(file_path)?;
415        let given_domains = domains_from_model(problem_model);
416        let params = params_from_simplified_json_str(&text, &given_domains)?;
417        return Ok(param_model_from_assignments(
418            params,
419            &given_domains,
420            pipeline.context(),
421        ));
422    }
423    pipeline.parse_file(file_path)
424}

426fn run_solver(
427    pipeline: &Pipeline,
428    global_args: &GlobalArgs,
429    cmd_args: &Args,
430    model: Model,
431) -> anyhow::Result<(f64, f64, f64)> {
432    let (solutions, measured) = pipeline.solve_with_timings(
433        model,
434        &SolutionOptions {
435            number_of_solutions: cmd_args.number_of_solutions.as_solver_limit(),
436            keep_intermediate_solutions: cmd_args.keep_intermediate_solutions,
437            solver_input_file: global_args.save_solver_input_file.clone(),
438            rule_trace_cdp: global_args.rule_trace_cdp,
439        },
440    )?;
441
442    let output_started_at = Instant::now();
443
444    tracing::debug!(
445        solutions = %solutions_to_json(&solutions),
446        "solver solutions"
447    );
448
449    write_solutions(&solutions, cmd_args)?;
450
451    Ok((
452        measured.backend_time_s,
453        measured.search_time_s,
454        measured.solution_processing_time_s + output_started_at.elapsed().as_secs_f64(),
455    ))
456}

458fn write_solutions(
459    solutions: &[std::collections::BTreeMap<conjure_cp::ast::Name, conjure_cp::ast::Literal>],
460    cmd_args: &Args,
461) -> anyhow::Result<()> {
462    if cmd_args.solutions_in_one_file {
463        let body = match cmd_args.output_format {
464            OutputFormat::Essence => solutions_to_essence(solutions),
465            OutputFormat::Json => solutions_to_simplified_json_string(solutions)?,
466        };
467        match &cmd_args.output {
468            None => print!("{body}"),
469            Some(path) => {
470                std::fs::write(path, body.as_bytes())?;
471                println!("Solutions saved to {:?}", path.canonicalize()?);
472            }
473        }
474        return Ok(());
475    }
476
477    // One file / record per solution.
478    for (index, solution) in solutions.iter().enumerate() {
479        let body = match cmd_args.output_format {
480            OutputFormat::Essence => solutions_to_essence(std::slice::from_ref(solution)),
481            OutputFormat::Json => {
482                let value = solution_to_simplified_json(solution)?;
483                format!("{}\n", serde_json::to_string_pretty(&value)?)
484            }
485        };
486        match &cmd_args.output {
487            None => print!("{body}"),
488            Some(path) => {
489                let file_path = per_solution_output_path(path, index + 1, cmd_args.output_format);
490                std::fs::write(&file_path, body.as_bytes())?;
491                println!("Solution saved to {:?}", file_path.canonicalize()?);
492            }
493        }
494    }
495    Ok(())
496}

498fn per_solution_output_path(base: &Path, index: usize, format: OutputFormat) -> PathBuf {
499    let stem = base
500        .file_stem()
501        .and_then(|s| s.to_str())
502        .unwrap_or("solution");
503    let parent = base.parent().unwrap_or_else(|| std::path::Path::new("."));
504    let extension = match format {
505        OutputFormat::Essence => "solution",
506        OutputFormat::Json => "solution.json",
507    };
508    parent.join(format!("{stem}-{index:06}.{extension}"))
