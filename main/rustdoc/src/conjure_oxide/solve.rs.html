49pub enum OutputFormat {
50    /// Conjure-compatible Essence multi-solution / per-solution files
51    #[default]
52    Essence,
53    /// Conjure `--output-format=json` simplified JSON
54    Json,

57#[derive(Clone, Copy, Debug, PartialEq, Eq)]
58pub enum NumberOfSolutions {
59    All,
60    Limit(i32),
61}
62
63impl NumberOfSolutions {
64    fn as_solver_limit(self) -> i32 {
65        match self {
66            NumberOfSolutions::All => 0,
67            NumberOfSolutions::Limit(limit) => limit,
68        }
69    }
70}

72fn parse_number_of_solutions(input: &str) -> Result<NumberOfSolutions, String> {
73    if input.eq_ignore_ascii_case("all") {
74        return Ok(NumberOfSolutions::All);
75    }
76
77    let limit = input
78        .parse::<i32>()
79        .map_err(|_| "expected a positive integer or 'all'".to_string())?;
80
81    if limit <= 0 {
82        return Err("expected a positive integer or 'all'".to_string());
83    }
84
85    Ok(NumberOfSolutions::Limit(limit))
86}

153pub fn run_solve_command(
154    global_args: GlobalArgs,
155    solve_args: Args,
156    rule_application_counts: &RuleApplicationCounter,
157) -> anyhow::Result<()> {
158    let essence_file = solve_args.essence_file.clone();
159    let param_file = solve_args.param_file.clone();
160
161    // each step is in its own method so that similar commands
162    // (e.g. testsolve) can reuse some of these steps.
163
164    let context = init_context(&global_args, essence_file, param_file)?;
165
166    let ctx_lock = context.read().unwrap();
167    let essence_file_name = ctx_lock
168        .essence_file_name
169        .as_ref()
170        .expect("context should contain the problem input file");
171    let param_file_name = ctx_lock.param_file_name.as_ref();
172
173    let translation_started_at = Instant::now();
174
175    // parse models
176    let problem_model = parse(&global_args, Arc::clone(&context), essence_file_name)?;
177
178    // unify models
179    let unified_model = match param_file_name {
180        Some(param_file_name) => {
181            let param_model = parse_param(
182                &global_args,
183                Arc::clone(&context),
184                param_file_name,
185                &problem_model,
186            )?;
187            instantiate_model(problem_model, param_model)?
188        }
189        None => {
190            let mut problem_model = problem_model;
191            validate_instantiation_conditions(&mut problem_model)?;
192            problem_model
193        }
194    };
195    drop(ctx_lock);
196
197    let rewritten_model = rewrite(unified_model, &global_args, Arc::clone(&context))?;
198
199    // Capture representations and SAT encoding choices
200    let mut model_choices = RecordedRunStats::default();
201
202    record_model_choices(&rewritten_model, &mut model_choices);
203
204    let translation_time = translation_started_at.elapsed().as_secs_f64();
205
206    let solver = init_solver(&global_args)?;
207
208    let solver_started_at = Instant::now();
209
210    let measured = if solve_args.no_run_solver {
211        println!("{}", rewritten_model);
212
213        if let Some(ref path) = global_args.save_solver_input_file {
214            let solver = solver.load_model(rewritten_model)?;
215
216            let mut file: Box<dyn std::io::Write> = Box::new(File::create(path)?);
217
218            solver.write_solver_input_file(&mut file)?;
219        }
220
221        None
222    } else {
223        Some(run_solver(
224            solver,
225            &global_args,
226            &solve_args,
227            rewritten_model,
228        )?)
229    };
230
231    let solve_time = measured
232        .as_ref()
233        .map(|_| solver_started_at.elapsed().as_secs_f64());
234
235    let backend_time = measured.as_ref().map(|t| t.0);
236    let search_time = measured.as_ref().map(|t| t.1);
237    let solution_processing_time = measured.as_ref().map(|t| t.2);
238
239    // Still write translation statistics when the solver was not run.
240    if let Some(ref path) = solve_args.stats {
241        let config = RecordedRunConfig {
242            parser: global_args.parser.to_string(),
243            rewriter: global_args.rewriter.to_string(),
244            comprehension_expander: global_args.comprehension_expander.to_string(),
245            heuristic: global_args.heuristic.to_string(),
246            channelling: global_args.channelling.to_string(),
247            seed: global_args.seed,
248            solver_seed: global_args.solver_seed,
249            solver: global_args.solver.as_str(),
250        };
251
252        let case_name = solve_args
253            .essence_file
254            .file_stem()
255            .map(|s| s.to_string_lossy().into_owned())
256            .unwrap_or_else(|| "input".to_owned());
257
258        let mut row = RecordedRunStats {
259            parser: config.parser.clone(),
260            rewriter: config.rewriter.clone(),
261            comprehension_expander: config.comprehension_expander.clone(),
262            heuristic: config.heuristic.clone(),
263            channelling: config.channelling.clone(),
264            seed: config.seed,
265            solver_seed: config.solver_seed,
266            solver: config.solver.clone(),
267            minion_discrete_threshold: global_args.minion_discrete_threshold,
268
269            case_name,
270            model_index: 0,
271            status: "ok".to_owned(),
272
273            solution_limit: (!solve_args.no_run_solver)
274                .then(|| solve_args.number_of_solutions.as_solver_limit()),
275
276            translation_time: Some(translation_time),
277            solve_time,
278            backend_time,
279            search_time,
280            solution_processing_time,
281
282            ..Default::default()
283        };
284
285        row.representations = model_choices.representations;
286
287        row.sat_encoding_amo = model_choices.sat_encoding_amo;
288        row.sat_encoding_cardinality = model_choices.sat_encoding_cardinality;
289        row.sat_encoding_pb = model_choices.sat_encoding_pb;
290        row.sat_encoding_table = model_choices.sat_encoding_table;
291        row.sat_encoding_element = model_choices.sat_encoding_element;
292        row.sat_encoding_alldifferent = model_choices.sat_encoding_alldifferent;
293
294        // Read the same rewrite-attempt counters used by exhaustive testing.
295        let context_json = serde_json::to_value(context.read().unwrap().clone())?;
296
297        let attempt_counts: Vec<u64> = context_json
298            .get("stats")
299            .and_then(|stats| stats.get("rewriterRuns"))
300            .and_then(serde_json::Value::as_array)
301            .into_iter()
302            .flatten()
303            .filter_map(|run| {
304                run.get("rewriterRuleApplicationAttempts")
305                    .and_then(serde_json::Value::as_u64)
306            })
307            .collect();
308
309        if !attempt_counts.is_empty() {
310            row.rule_trace.total_rule_attempts = Some(attempt_counts.iter().sum());
311        }
312
313        let counts = rule_application_counts.snapshot();
314        row.rule_trace.total_rule_applications = Some(counts.values().copied().sum());
315        row.rule_trace.rules = counts;
316
317        // Write the completed statistics record only after all fields are populated.
318        replace_config_run_stats(path, &config, vec![row])?;
319    }
320    Ok(())
321}
322fn record_model_choices(model: &Model, row: &mut RecordedRunStats) {
323    row.representations = model
324        .symbols()
325        .iter_local()
326        .flat_map(|(_, decl)| {
327            decl.reprs()
328                .iter()
329                .map(|(name, _)| name.to_owned())
330                .collect::<Vec<_>>()
331        })
332        .collect::<BTreeSet<_>>()
333        .into_iter()
334        .collect();
335    let mut families: BTreeMap<&str, BTreeSet<String>> = BTreeMap::new();
336    macro_rules! record {
337        ($family:expr, $selection:expr) => {{
338            $selection.iter().for_each(|selection| {
339                families
340                    .entry($family)
341                    .or_default()
342                    .insert(selection.algorithm.to_string());
343            });
344        }};
345    }
346    for decision in model.sat_decisions() {
347        match decision {
348            SatEncodingDecision::AtMostOne { encoding, .. } => {
349                record!("amo", encoding);
350            }
351            SatEncodingDecision::Cardinality { encoding, .. } => {
352                record!("cardinality", encoding);
353            }
354            SatEncodingDecision::CountRelation {
355                encoding,
356                amo_encoding,
357                ..
358            } => {
359                record!("cardinality", encoding);
360                record!("amo", amo_encoding);
361            }
362            SatEncodingDecision::PseudoBoolean { encoding, .. }
363            | SatEncodingDecision::IntegerRelation { encoding, .. }
364            | SatEncodingDecision::Objective { encoding, .. } => record!("pb", encoding),
365            SatEncodingDecision::AllDifferent {
366                encoding,
367                amo_encoding,
368                pb_encoding,
369                ..
370            } => {
371                record!("alldifferent", encoding);
372                record!("amo", amo_encoding);
373                record!("pb", pb_encoding);
374            }
375            SatEncodingDecision::Table {
376                encoding,
377                pb_encoding,
378                ..
379            } => {
380                record!("table", encoding);
381                record!("pb", pb_encoding);
382            }
383            SatEncodingDecision::Element {
384                encoding,
385                pb_encoding,
386                ..
387            } => {
388                record!("element", encoding);
389                record!("pb", pb_encoding);
390            }
391            _ => {}
392        }
393    }
394    row.sat_encoding_amo = families
395        .remove("amo")
396        .unwrap_or_default()
397        .into_iter()
398        .collect();
399    row.sat_encoding_cardinality = families
400        .remove("cardinality")
401        .unwrap_or_default()
402        .into_iter()
403        .collect();
404    row.sat_encoding_pb = families
405        .remove("pb")
406        .unwrap_or_default()
407        .into_iter()
408        .collect();
409    row.sat_encoding_table = families
410        .remove("table")
411        .unwrap_or_default()
412        .into_iter()
413        .collect();
414    row.sat_encoding_element = families
415        .remove("element")
416        .unwrap_or_default()
417        .into_iter()
418        .collect();
419    row.sat_encoding_alldifferent = families
420        .remove("alldifferent")
421        .unwrap_or_default()
422        .into_iter()
423        .collect();
424}
425/// Returns a new Context and Solver for solving.
426pub(crate) fn init_context(
427    global_args: &GlobalArgs,
428    essence_file: PathBuf,
429    param_file: Option<PathBuf>,
430) -> anyhow::Result<Arc<RwLock<Context<'static>>>> {
431    let default_rule_trace_enabled = global_args.rule_trace.is_some();
432    let rule_attempt_trace_enabled = global_args.rule_attempt_trace.is_some();
433
434    set_current_parser(global_args.parser);
435    set_current_rewriter(global_args.rewriter);
436    set_comprehension_expander(global_args.comprehension_expander);
437    set_current_solver_family(global_args.solver);
438    set_minion_discrete_threshold(global_args.minion_discrete_threshold);
439    // Count every translation's rule applications, regardless of which outputs
440    // were requested. Solver-time CDP tracing remains controlled separately.
441    set_rule_trace_enabled(true);
442    set_default_rule_trace_enabled(default_rule_trace_enabled);
443    set_rule_attempt_trace_enabled(rule_attempt_trace_enabled);
444    set_rule_trace_aggregates_enabled(true);
445
446    let target_family = global_args.solver;
447    let mut extra_rule_sets: Vec<&str> = DEFAULT_RULE_SETS.to_vec();
448    for rs in &global_args.extra_rule_sets {
449        extra_rule_sets.push(rs.as_str());
450    }
451
452    let rule_sets = match resolve_rule_sets(target_family, &extra_rule_sets) {
453        Ok(rs) => rs,
454        Err(e) => {
455            tracing::error!("Error resolving rule sets: {}", e);
456            exit(1);
457        }
458    };
459
460    tracing::info!(
461        target: "conjure::stage",
462        count = rule_sets.len(),
463        "resolved rule sets"
464    );
465    tracing::debug!(
466        rule_sets = %rule_sets
467            .iter()
468            .map(|rule_set| rule_set.name)
469            .collect::<Vec<_>>()
470            .join(", "),
471        "enabled rule sets"
472    );
473
474    let rules = get_rules(&rule_sets)?.into_iter().collect::<Vec<_>>();
475    tracing::debug!(
476        "Rules: {}",
477        rules
478            .iter()
479            .map(|rd| format!("{rd}"))
480            .collect::<Vec<_>>()
481            .join("\n")
482    );
483    let context = Context::new_ptr(
484        target_family,
485        extra_rule_sets.iter().map(|rs| rs.to_string()).collect(),
486        rules,
487        rule_sets.clone(),
488    );
489
490    context.write().unwrap().essence_file_name = Some(essence_file.to_str().expect("").into());
491    if let Some(param_file) = param_file {
492        context.write().unwrap().param_file_name = Some(param_file.to_str().expect("").into());
493    }
494
495    Ok(context)
496}

498pub(crate) fn init_solver(global_args: &GlobalArgs) -> anyhow::Result<Solver> {
499    let family = global_args.solver;
500    let timeout = global_args.solver_timeout.map(Duration::from);
501
502    Ok(match family {
503        SolverFamily::Minion => Solver::new(
504            Minion::with_search_orders(global_args.minion_varorder, global_args.minion_valorder)
505                .with_solver_seed(global_args.solver_seed)
506                .with_timeout(timeout),
507        ),
508        SolverFamily::Sat => Solver::new(
509            Sat::new(global_args.sat_backend)
510                .with_solver_seed(global_args.solver_seed)
511                .with_timeout(timeout),
512        ),
513        #[cfg(feature = "z3")]
514        SolverFamily::Z3 => {
515            Solver::new(Smt::new(timeout).with_solver_seed(global_args.solver_seed))
516        }
517        #[cfg(not(feature = "z3"))]
518        SolverFamily::Z3 => {
519            anyhow::bail!("Z3 solver support was not compiled in (enable the `z3` feature).")
520        }
521    })
522}

524pub(crate) fn parse(
525    global_args: &GlobalArgs,
526    context: Arc<RwLock<Context<'static>>>,
527    file_path: &str,
528) -> anyhow::Result<Model> {
529    tracing::info!(target: "conjure::stage", path = %file_path, "parsing input model");
530
531    match global_args.parser {
532        conjure_cp::settings::Parser::TreeSitter => {
533            parse_essence_file_native(file_path, context.clone()).map_err(|e| e.into())
534        }
535        conjure_cp::settings::Parser::ViaConjure => parse_with_conjure(file_path, context.clone()),
536    }
537}

539/// Parse an Essence or simplified-JSON parameter file.
540pub(crate) fn parse_param(
541    global_args: &GlobalArgs,
542    context: Arc<RwLock<Context<'static>>>,
543    file_path: &str,
544    problem_model: &Model,
545) -> anyhow::Result<Model> {
546    if file_path.ends_with(".json") {
547        tracing::info!(target: "conjure::stage", path = %file_path, "parsing parameter file");
548        let text = std::fs::read_to_string(file_path)?;
549        let given_domains = domains_from_model(problem_model);
550        let params = params_from_simplified_json_str(&text, &given_domains)?;
551        return Ok(param_model_from_assignments(
552            params,
553            &given_domains,
554            context,
555        ));
556    }
557    parse(global_args, context, file_path)
558}

560pub(crate) fn parse_with_conjure(
561    input_file: &str,
562    context: Arc<RwLock<Context<'static>>>,
563) -> anyhow::Result<Model> {
564    conjure_executable().map_err(|e| anyhow!("Could not find correct conjure executable: {e}"))?;
565
566    let mut cmd = std::process::Command::new("conjure");
567    let output = cmd
568        .arg("pretty")
569        .arg("--output-format=astjson")
570        .arg(input_file)
571        .output()?;
572
573    if !output.status.success() {
574        println!("Parsing error: {}", String::from_utf8(output.stderr)?);
575    }
576
577    let astjson = String::from_utf8(output.stdout)?;
578
579    if cfg!(feature = "extra-rule-checks") {
580        tracing::debug!("extra-rule-checks: enabled");
581    } else {
582        tracing::debug!("extra-rule-checks: disabled");
583    }
584
585    model_from_json(&astjson, context.clone()).map_err(|e| anyhow!(e))
586}

588pub(crate) fn rewrite(
589    model: Model,
590    global_args: &GlobalArgs,
591    context: Arc<RwLock<Context<'static>>>,
592) -> anyhow::Result<Model> {
593    tracing::debug!(model = %model, "initial model");
594
595    let rewriter = global_args.rewriter;
596    set_current_rewriter(rewriter);
597
598    let comprehension_expander = global_args.comprehension_expander;
599    set_comprehension_expander(comprehension_expander);
600    tracing::debug!(%comprehension_expander, "configured comprehension expander");
601
602    set_heuristic(global_args.heuristic);
603    set_heuristic_seed(global_args.seed);
604    set_heuristic_responses(global_args.responses.clone());
605    set_channelling(global_args.channelling);
606    conjure_cp::settings::set_amo_encoding(global_args.amo_encoding);
607    conjure_cp::settings::set_pb_encoding(global_args.pb_encoding);
608    conjure_cp::settings::set_element_encoding(global_args.element_encoding);
609    conjure_cp::settings::set_table_encoding(global_args.table_encoding);
610    conjure_cp::settings::set_alldifferent_encoding(global_args.alldifferent_encoding);
611    conjure_cp::settings::set_cardinality_encoding(global_args.cardinality_encoding);
612    tracing::debug!(
613        "Heuristic: {}, seed: {}, responses: {:?}, channelling: {}, solver seed: {}",
614        global_args.heuristic,
615        global_args.seed,
616        global_args.responses,
617        global_args.channelling,
618        global_args.solver_seed
619    );
620
621    let rule_sets = context.read().unwrap().rule_sets.clone();
622
623    let Rewriter::Rewrite(config) = rewriter;
624    tracing::info!(target: "conjure::stage", %config, "rewriting model");
625    let new_model = rewrite_model(&model, &rule_sets, config)?;
626
627    tracing::debug!(model = %new_model, "rewritten model");
628    Ok(new_model)
629}

631fn run_solver(
632    solver: Solver,
633    global_args: &GlobalArgs,
634    cmd_args: &Args,
635    model: Model,
636) -> anyhow::Result<(f64, f64, f64)> {
637    let domains = domains_from_model(&model);
638
639    let (solutions, measured) = get_solutions_with_timings(
640        solver,
641        model,
642        cmd_args.number_of_solutions.as_solver_limit(),
643        cmd_args.keep_intermediate_solutions,
644        &global_args.save_solver_input_file,
645        global_args.rule_trace_cdp,
646    )?;
647
648    let output_started_at = Instant::now();
649
650    tracing::debug!(
651        solutions = %solutions_to_json(&solutions),
652        "solver solutions"
653    );
654
655    let solutions = coerce_bools_in_solutions(&solutions, &domains);
656
657    write_solutions(&solutions, cmd_args)?;
658
659    Ok((
660        measured.backend_time_s,
661        measured.search_time_s,
662        measured.solution_processing_time_s + output_started_at.elapsed().as_secs_f64(),
663    ))
664}

666/// Turn solver `0`/`1` assignments back into booleans when the find domain is bool, so JSON/Essence
667/// output matches Conjure's simplified JSON (`true`/`false`).
668fn coerce_bools_in_solutions(
669    solutions: &[std::collections::BTreeMap<conjure_cp::ast::Name, conjure_cp::ast::Literal>],
670    domains: &std::collections::BTreeMap<conjure_cp::ast::Name, conjure_cp::ast::DomainPtr>,
671) -> Vec<std::collections::BTreeMap<conjure_cp::ast::Name, conjure_cp::ast::Literal>> {
672    use conjure_cp::ast::{GroundDomain, Literal, Name};
673    solutions
674        .iter()
675        .map(|solution| {
676            solution
677                .iter()
678                .map(|(name, value)| {
679                    let value = match domains.get(name).and_then(|domain| domain.as_ground()) {
680                        Some(GroundDomain::Bool) => match value {
681                            Literal::Int(1) => Literal::Bool(true),
682                            Literal::Int(0) => Literal::Bool(false),
683                            other => other.clone(),
684                        },
685                        _ => value.clone(),
686                    };
687                    (name.clone(), value)
688                })
689                .collect::<std::collections::BTreeMap<Name, Literal>>()
690        })
691        .collect()
692}
693
694fn write_solutions(
695    solutions: &[std::collections::BTreeMap<conjure_cp::ast::Name, conjure_cp::ast::Literal>],
696    cmd_args: &Args,
697) -> anyhow::Result<()> {
698    if cmd_args.solutions_in_one_file {
699        let body = match cmd_args.output_format {
700            OutputFormat::Essence => solutions_to_essence(solutions),
701            OutputFormat::Json => solutions_to_simplified_json_string(solutions)?,
702        };
703        match &cmd_args.output {
704            None => print!("{body}"),
705            Some(path) => {
706                std::fs::write(path, body.as_bytes())?;
707                println!("Solutions saved to {:?}", path.canonicalize()?);
708            }
709        }
710        return Ok(());
711    }
712
713    // One file / record per solution.
714    for (index, solution) in solutions.iter().enumerate() {
715        let body = match cmd_args.output_format {
716            OutputFormat::Essence => solutions_to_essence(std::slice::from_ref(solution)),
717            OutputFormat::Json => {
718                let value = solution_to_simplified_json(solution)?;
719                format!("{}\n", serde_json::to_string_pretty(&value)?)
720            }
721        };
722        match &cmd_args.output {
723            None => print!("{body}"),
724            Some(path) => {
725                let file_path = per_solution_output_path(path, index + 1, cmd_args.output_format);
726                std::fs::write(&file_path, body.as_bytes())?;
727                println!("Solution saved to {:?}", file_path.canonicalize()?);
728            }
729        }
730    }
731    Ok(())
732}
733
734fn per_solution_output_path(base: &Path, index: usize, format: OutputFormat) -> PathBuf {
735    let stem = base
736        .file_stem()
737        .and_then(|s| s.to_str())
738        .unwrap_or("solution");
739    let parent = base.parent().unwrap_or_else(|| std::path::Path::new("."));
740    let extension = match format {
741        OutputFormat::Essence => "solution",
742        OutputFormat::Json => "solution.json",
743    };
744    parent.join(format!("{stem}-{index:06}.{extension}"))
745}
