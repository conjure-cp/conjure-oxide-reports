10/// Writes a complete Minion file for this model to `writer`.
11pub fn write_minion_file(writer: &mut impl Write, model: &Model) -> Result<(), std::io::Error> {
12    writeln!(writer, "# Autogenerated by minion-sys")?;
13    writeln!(writer, "MINION 3")?;
14
15    write_variables_section(writer, model)?;
16    write_search_section(writer, model)?;
17    write_constraints_section(writer, model)?;
18    writeln!(writer, "**EOF**")
19}

21/// Writes the `VARIABLES` section of the Minion file to `writer`.
22pub fn write_variables_section(
23    writer: &mut impl Write,
24    model: &Model,
25) -> Result<(), std::io::Error> {
26    writeln!(writer, "**VARIABLES**")?;
27
28    let symtab = &model.named_variables;
29
30    // print variables in declaration order
31    for name in symtab.get_variable_order() {
32        write_variable_declaration(writer, model, name)?;
33    }
34    Ok(())
35}

37/// Writes the `SEARCH` section of the Minion file to `writer`.
38pub fn write_search_section(writer: &mut impl Write, model: &Model) -> Result<(), std::io::Error> {
39    let symtab = &model.named_variables;
40
41    writeln!(writer, "**SEARCH**")?;
42
43    let varorder = symtab.get_search_variable_order();
44
45    writeln!(writer, "VARORDER STATIC [{}]", varorder.join(","))?;
46    writeln!(
47        writer,
48        "PRINT [[{}]]",
49        symtab.get_variable_order().join(",")
50    )?;
51
52    if let Some(Optimise { minimise, var }) = &model.optimise {
53        if *minimise {
54            writeln!(writer, "MINIMISING {var}")?;
55        } else {
56            writeln!(writer, "MAXIMISING {var}")?;
57        }
58    }
59
60    Ok(())
61}

63/// Writes tuple definitions and the `CONSTRAINTS` section to `writer`.
64///
65/// # Panics
66///
67/// If a named tuple table has rows of different lengths.
68#[allow(clippy::panic_in_result_fn)]
69pub fn write_constraints_section(
70    writer: &mut impl Write,
71    model: &Model,
72) -> Result<(), std::io::Error> {
73    let constraints = &model.constraints;
74    if !model.tuple_tables.is_empty() {
75        writeln!(writer, "**TUPLELIST**")?;
76        for (name, tuples) in &model.tuple_tables {
77            let arity = tuples.first().map_or(0, Vec::len);
78            assert!(
79                tuples.iter().all(|tuple| tuple.len() == arity),
80                "ragged tuple table"
81            );
82            writeln!(writer, "{name} {} {arity}", tuples.len())?;
83            for tuple in tuples {
84                writeln!(
85                    writer,
86                    "{}",
87                    tuple
88                        .iter()
89                        .map(ToString::to_string)
90                        .collect::<Vec<_>>()
91                        .join(" ")
92                )?;
93            }
94        }
95    }
96    let mut short_tables = Vec::new();
97    for constraint in constraints {
98        collect_short_tables(constraint, &mut short_tables);
99    }
100    if !short_tables.is_empty() {
101        writeln!(writer, "**SHORTTUPLELIST**")?;
102        for (index, tuples) in short_tables.iter().enumerate() {
103            writeln!(writer, "_minion_sys_short_{index} {}", tuples.len())?;
104            for tuple in *tuples {
105                writeln!(writer, "{}", print_short_tuple(tuple))?;
106            }
107        }
108    }
109    writeln!(writer, "**CONSTRAINTS**")?;
110
111    for constraint in constraints {
112        writeln!(
113            writer,
114            "{}",
115            ConstraintDisplay {
116                constraint,
117                short_tables: &short_tables
118            }
119        )?;
120    }
121
122    Ok(())
123}

125/// Writes the variable declaration of `name` to `writer`.
126///
127/// # Panics
128///
129/// If `name` does not exist.
130pub fn write_variable_declaration(
131    writer: &mut impl Write,
132    model: &Model,
133    name: VarName,
134) -> Result<(), std::io::Error> {
135    let symtab = &model.named_variables;
136
137    #[allow(clippy::expect_used)]
138    match symtab.get_vartype(name.clone()).expect("name should exist") {
139        crate::ast::VarDomain::Bound(i, j) => writeln!(writer, "BOUND {name} {{{i}..{j}}}")?,
140        crate::ast::VarDomain::Discrete(i, j) => writeln!(writer, "DISCRETE {name} {{{i}..{j}}}")?,
141        crate::ast::VarDomain::Bool => writeln!(writer, "BOOL {name}")?,
142        crate::ast::VarDomain::SparseBound(vals) => {
143            let vals_str: Vec<String> = vals.iter().map(|n| n.to_string()).collect();
144            writeln!(writer, "SPARSEBOUND {name} {{ {} }}", vals_str.join(", "))?
145        }
146    };
147
148    Ok(())
149}

151pub(crate) fn print_const_array(array: &[Constant]) -> String {
152    let string_array: Vec<String> = array.iter().map(|x| format!("{x}")).collect();
153    let string = string_array.join(",");
154    format!("[{string}]")
155}

157pub(crate) fn print_var_array(array: &[Var]) -> String {

159    let string = string_array.join(",");

163pub(crate) fn print_tuple_array(array: &[Tuple]) -> String {
164    let string_array: Vec<String> = array
165        .iter()
166        .map(|tup| {
167            format!(
168                "<{}>",
169                tup.iter()
170                    .map(ToString::to_string)
171                    .collect::<Vec<_>>()
172                    .join(",")
173            )
174        })
175        .collect();
176    let string = string_array.join(",");
177    format!("{{{string}}}")
178}

180fn print_short_tuple(tuple: &ShortTuple) -> String {
181    let pairs: Vec<String> = tuple
182        .iter()
183        .map(|(idx, val)| format!("({idx},{val})"))
184        .collect();
185    format!("[{}]", pairs.join(","))
186}

188pub(crate) fn collect_short_tables<'a>(
189    constraint: &'a Constraint,
190    tables: &mut Vec<&'a [ShortTuple]>,
191) {
192    match constraint {
193        Constraint::ShortStr2(_, tuples)
194        | Constraint::HaggisGac(_, tuples)
195        | Constraint::HaggisGacStable(_, tuples)
196        | Constraint::ShortCTupleStr2(_, tuples) => {
197            if !tables.contains(&tuples.as_slice()) {
198                tables.push(tuples);
199            }
200        }
201        Constraint::CheckAssign(child)
202        | Constraint::CheckGsa(child)
203        | Constraint::ForwardChecking(child)
204        | Constraint::Reify(child, _)
205        | Constraint::ReifyImply(child, _)
206        | Constraint::ReifyImplyQuick(child, _) => collect_short_tables(child, tables),
207        Constraint::WatchedAnd(children) | Constraint::WatchedOr(children) => {
208            for child in children {
209                collect_short_tables(child, tables);
210            }
211        }
212        _ => {}
213    }
