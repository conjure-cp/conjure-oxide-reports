/// cond ? b : a
///
/// cond = 1 => b
/// cond = 0 => a
fn tseytin_select_array(
    cond: Expr,
    a: &[Expr],
    b: &[Expr],
    clauses: &mut Vec<CnfClause>,
    symbols: &mut SymbolTable,
) -> Vec<Expr> {
    assert_eq!(
        a.len(),
        b.len(),
        "Input vectors 'a' and 'b' must have the same length"
    );

    let mut out = vec![];

    let bit_count = a.len();

    for i in 0..bit_count {
        out.push(tseytin_mux(
            cond.clone(),
            a[i].clone(),
            b[i].clone(),
            clauses,
            symbols,
        ));
    }

    out
}
