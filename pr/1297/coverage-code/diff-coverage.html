/// Minimizes the length of a log integer by removing redundant bits
pub fn minimize_bits(value: &[Expr]) -> Vec<Expr> {
    let mut result = value.to_owned();

    while result.len() >= 2 && result[result.len() - 1] == result[result.len() - 2] {
        result.pop();
    }

    result
}
