    pub fn length_signed(&self) -> Result<i32, DomainOpError> {
        if let Some(gd) = self.as_ground() {
            return match gd.length() {
                Ok(len) => len.try_into().map_err(|_| DomainOpError::TooLarge),
                Err(error) => Err(error),
            };
        }
        Err(DomainOpError::NotGround)
    }
