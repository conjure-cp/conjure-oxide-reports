impl ExprInfo {
    pub fn create(expr: &Expression) -> ExprInfo {
        let pretty = expr.to_string();
        let domain = expr.domain_of();
        let children = expr.children().iter().map(Self::create).collect();

        ExprInfo {
            pretty,
            domain,
            children,
        }
    }
