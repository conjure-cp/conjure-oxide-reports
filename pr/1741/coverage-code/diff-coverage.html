    let error_ptr = ffi::minion_get_last_error();
    if error_ptr.is_null() {

    }

    let message = CStr::from_ptr(error_ptr).to_string_lossy().into_owned();
    ffi::minion_clear_last_error();

        "unknown Minion FFI error".to_string()
    } else {
        message
    }
}
